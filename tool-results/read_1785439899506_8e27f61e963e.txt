    24→/**
    25→ * GlobalMusicPlayer
    26→ * ─────────────────────────────────────────────────────────────────────────
    27→ * Renders the persistent <audio> element and the bottom player bar at the
    28→ * app root level so that audio playback is NOT destroyed when the user
    29→ * switches tabs (Chats, Status, Calls, Settings, Music).
    30→ *
    31→ * Architecture:
    32→ *   - The Zustand store (`useMusicStore`) is the single source of truth for
    33→ *     playback state (currentTrack, isPlaying, position, volume, queue,
    34→ *     autoplay, shuffle, repeat, activeRoomId).
    35→ *   - This component owns the <audio> element and all audio + Socket.io
    36→ *     side-effects. It reads from the store and writes back to it.
    37→ *   - The MusicView (browse / rooms / queue UI) reads state from the store
    38→ *     and calls the action functions exposed here via `useMusicPlayer()`.
    39→ *   - Socket.io sync events are listened to here. When a remote sync is
    40→ *     applied, the store is updated; the audio element is then loaded /
    41→ *     seeked directly. A ref flag (`applyingRemoteRef`) prevents
    42→ *     `timeupdate` events from fighting with the applied position.
    43→ */
    44→
    45→interface MusicPlayerContextValue {
    46→  /** Play a track (or add it to the queue if addToQueue=true). */
    47→  playTrack: (track: Track, addToQueue?: boolean) => Promise<void>
    48→  /** Advance to the next track (queue → autoplay → repeat → stop). */
    49→  playNext: () => Promise<void>
    50→  /** Toggle play / pause. */
    51→  togglePlay: () => void
    52→  /** Seek to a position (seconds). */
    53→  handleSeek: (pos: number) => void
    54→  /** Stop playback entirely (clears current track). */
    55→  stop: () => void
    56→  /** Add a track to the queue (broadcasts to room). */
    57→  addToQueue: (track: Track) => void
    58→  /** Remove a track from the queue by index (broadcasts to room). */
    59→  removeFromQueue: (index: number) => void
    60→  /** Clear the entire queue (broadcasts to room). */
    61→  clearQueue: () => void
    62→}
    63→
    64→const MusicPlayerContext = createContext<MusicPlayerContextValue | null>(null)
    65→
    66→const NOOP_PLAYER: MusicPlayerContextValue = {
    67→  playTrack: async () => {},
    68→  playNext: async () => {},
    69→  togglePlay: () => {},
    70→  handleSeek: () => {},
    71→  stop: () => {},
    72→  addToQueue: () => {},
    73→  removeFromQueue: () => {},
    74→  clearQueue: () => {},
    75→}
    76→
    77→/**
    78→ * Access the global music player's action functions.
    79→ *
    80→ * Safe to call from any component rendered inside <GlobalMusicPlayer>.
    81→ * Returns no-op functions if no provider is present (defensive).
    82→ */
    83→export function useMusicPlayer(): MusicPlayerContextValue {
    84→  const ctx = useContext(MusicPlayerContext)
    85→  return ctx ?? NOOP_PLAYER
    86→}
    87→
    88→export function GlobalMusicPlayer({ children }: { children: React.ReactNode }) {
    89→  const audioRef = useRef<HTMLAudioElement>(null)
    90→  const { socket } = useSocket()
    91→
    92→  // ─── Store state (subscribe to slices) ────────────────────────────────
    93→  const currentTrack = useMusicStore((s) => s.currentTrack)
    94→  const isPlaying = useMusicStore((s) => s.isPlaying)
    95→  const isLoading = useMusicStore((s) => s.isLoading)
    96→  const position = useMusicStore((s) => s.position)
    97→  const volume = useMusicStore((s) => s.volume)
    98→  const queue = useMusicStore((s) => s.queue)
    99→  const shuffle = useMusicStore((s) => s.shuffle)
   100→  const repeat = useMusicStore((s) => s.repeat)
   101→  const activeRoomId = useMusicStore((s) => s.activeRoomId)
   102→
   103→  // ─── Store actions ────────────────────────────────────────────────────
   104→  const setCurrentTrack = useMusicStore((s) => s.setCurrentTrack)
   105→  const setIsPlaying = useMusicStore((s) => s.setIsPlaying)
   106→  const setPosition = useMusicStore((s) => s.setPosition)
   107→  const setVolume = useMusicStore((s) => s.setVolume)
   108→  const setQueue = useMusicStore((s) => s.setQueue)
   109→  const storeAddToQueue = useMusicStore((s) => s.addToQueue)
   110→  const storeRemoveFromQueue = useMusicStore((s) => s.removeFromQueue)
   111→  const storeClearQueue = useMusicStore((s) => s.clearQueue)
   112→
   113→  // ─── Refs for cross-render coordination ───────────────────────────────
   114→  // The videoId currently loaded into the <audio> element. Used to detect
   115→  // when currentTrack.videoId changes so we know to (re)load the stream.
   116→  const loadedVideoIdRef = useRef<string | null>(null)
   117→  // When applying a remote sync that changes the track, this holds the
   118→  // position we should seek to once the new stream has loaded.
   119→  const pendingSeekRef = useRef<number | null>(null)
   120→  // True while we're applying state that arrived from a remote sync event.
   121→  // Prevents local `timeupdate` from overwriting the freshly-applied
   122→  // position and prevents echoing sync events back to the room.
   123→  const applyingRemoteRef = useRef(false)
   124→
   125→  // ─── Helper: broadcast a sync event to the active room ────────────────
   126→  const broadcastSync = useCallback(
   127→    (payload: {
   128→      state: string
   129→      position: number
   130→      videoId?: string
   131→      trackInfo?: Partial<Track>
   132→      queue?: Track[]
   133→    }) => {
   134→      if (!activeRoomId || !socket) return
   135→      socket.emit('music:sync', {
   136→        roomId: activeRoomId,
   137→        ...payload,
   138→      })
   139→    },
   140→    [activeRoomId, socket],
   141→  )
   142→
   143→  // ─── Action: play a track ─────────────────────────────────────────────
   144→  const playTrack = useCallback(
   145→    async (track: Track, addToQueue = false) => {
   146→      const state = useMusicStore.getState()
   147→
   148→      if (addToQueue && state.currentTrack) {
   149→        state.addToQueue(track)
   150→        toast.success(`Added to queue: ${track.title}`)
   151→        // Broadcast the updated queue so members see it
   152→        broadcastSync({
   153→          state: state.isPlaying ? 'playing' : 'paused',
   154→          position: audioRef.current?.currentTime || 0,
   155→          queue: useMusicStore.getState().queue,
   156→        })
   157→        return
   158→      }
   159→
   160→      // Set state first so the UI updates instantly
   161→      setCurrentTrack(track)
   162→      setIsPlaying(true)
   163→      setPosition(0)
   164→      useMusicStore.getState().setIsLoading(true)
   165→
   166→      // Broadcast to room (include the full queue so members sync)
   167→      broadcastSync({
   168→        state: 'playing',
   169→        position: 0,
   170→        videoId: track.videoId,
   171→        trackInfo: {
   172→          title: track.title,
   173→          artist: track.artist,
   174→          thumbnail: track.thumbnail,
   175→          durationSeconds: track.durationSeconds,
   176→        },
   177→        queue: useMusicStore.getState().queue,
   178→      })
   179→
   180→      // Load + play the audio
   181→      if (audioRef.current) {
   182→        audioRef.current.src = `/api/music/stream/${track.videoId}`
   183→        audioRef.current.volume = useMusicStore.getState().volume
   184→        loadedVideoIdRef.current = track.videoId
   185→        pendingSeekRef.current = null
   186→        try {
   187→          await audioRef.current.play()
   188→          useMusicStore.getState().setIsLoading(false)
   189→        } catch (e: any) {
   190→          useMusicStore.getState().setIsLoading(false)
   191→          if (e?.name === 'NotSupportedError' || e?.name === 'MediaError') {
   192→            try {
   193→              const res = await fetch(`/api/music/stream/${track.videoId}`, {
   194→                method: 'HEAD',
   195→              })
   196→              if (!res.ok) {
   197→                const data = await res.json().catch(() => ({}))
   198→                toast.error(data.error || 'Could not download this track.')
   199→                setCurrentTrack(null)
   200→                setIsPlaying(false)
   201→              }
   202→            } catch {
   203→              toast.error(
   204→                'The server is downloading this track — try again in a moment.',
   205→              )
   206→            }
   207→          }
   208→        }
   209→      }
   210→    },
   211→    [broadcastSync, setCurrentTrack, setIsPlaying, setPosition],
   212→  )
   213→
   214→  // ─── Action: play next track ──────────────────────────────────────────
   215→  const playNext = useCallback(async () => {
   216→    const state = useMusicStore.getState()
   217→    let nextTrack: Track | null = state.popNextFromQueue()
   218→
   219→    if (!nextTrack && state.autoplay && state.currentTrack) {
   220→      try {
   221→        const res = await fetch(
   222→          `/api/music/related/${state.currentTrack.videoId}`,
   223→        )
   224→        if (res.ok) {
   225→          const data = await res.json()
   226→          if (data.tracks?.length > 0) {
   227→            nextTrack = data.tracks[0] as Track
   228→          }
   229→        }
   230→      } catch {
   231→        // Autoplay failed — just stop
   232→      }
   233→    }
   234→
   235→    if (nextTrack) {
   236→      await playTrack(nextTrack)
   237→    } else if (state.repeat && state.currentTrack) {
   238→      await playTrack(state.currentTrack)
   239→    } else {
   240→      setIsPlaying(false)
   241→      setPosition(0)
   242→    }
   243→  }, [playTrack, setIsPlaying, setPosition])
   244→
   245→  // ─── Action: toggle play / pause ──────────────────────────────────────
   246→  const togglePlay = useCallback(() => {
   247→    const audio = audioRef.current
   248→    const state = useMusicStore.getState()
   249→    if (!audio || !state.currentTrack) return
   250→
   251→    if (state.isPlaying) {
   252→      audio.pause()
   253→      setIsPlaying(false)
   254→      broadcastSync({
   255→        state: 'paused',
   256→        position: audio.currentTime,
   257→        queue: state.queue,
   258→      })
   259→    } else {
   260→      audio.play().catch(() => {})
   261→      setIsPlaying(true)
   262→      broadcastSync({
   263→        state: 'playing',
   264→        position: audio.currentTime,
   265→        queue: state.queue,
   266→      })
   267→    }
   268→  }, [broadcastSync, setIsPlaying])
   269→
   270→  // ─── Action: seek ─────────────────────────────────────────────────────
   271→  const handleSeek = useCallback(
   272→    (pos: number) => {
   273→      const audio = audioRef.current
   274→      const state = useMusicStore.getState()
   275→      if (!audio) return
   276→      audio.currentTime = pos
   277→      setPosition(pos)
   278→      broadcastSync({
   279→        state: state.isPlaying ? 'playing' : 'paused',
   280→        position: pos,
   281→        queue: state.queue,
   282→      })
   283→    },
   284→    [broadcastSync, setPosition],
   285→  )
   286→
   287→  // ─── Action: stop playback entirely ───────────────────────────────────
   288→  const stop = useCallback(() => {
   289→    setCurrentTrack(null)
   290→    setIsPlaying(false)
   291→    setPosition(0)
   292→    if (audioRef.current) {
   293→      audioRef.current.pause()
   294→      audioRef.current.src = ''
   295→    }
   296→    loadedVideoIdRef.current = null
   297→  }, [setCurrentTrack, setIsPlaying, setPosition])
   298→
   299→  // ─── Action: add to queue (broadcasts) ────────────────────────────────
   300→  const addToQueue = useCallback(
   301→    (track: Track) => {
   302→      const state = useMusicStore.getState()
   303→      storeAddToQueue(track)
   304→      broadcastSync({
   305→        state: state.isPlaying ? 'playing' : 'paused',
   306→        position: audioRef.current?.currentTime || 0,
   307→        queue: useMusicStore.getState().queue,
   308→      })
   309→    },
   310→    [broadcastSync, storeAddToQueue],
   311→  )
   312→
   313→  // ─── Action: remove from queue (broadcasts) ───────────────────────────
   314→  const removeFromQueue = useCallback(
   315→    (index: number) => {
   316→      const state = useMusicStore.getState()
   317→      storeRemoveFromQueue(index)
   318→      broadcastSync({
   319→        state: state.isPlaying ? 'playing' : 'paused',
   320→        position: audioRef.current?.currentTime || 0,
   321→        queue: useMusicStore.getState().queue,
   322→      })
   323→    },
   324→    [broadcastSync, storeRemoveFromQueue],
   325→  )
   326→
   327→  // ─── Action: clear queue (broadcasts) ─────────────────────────────────
   328→  const clearQueue = useCallback(() => {
   329→    const state = useMusicStore.getState()
   330→    storeClearQueue()
   331→    broadcastSync({
   332→      state: state.isPlaying ? 'playing' : 'paused',
   333→      position: audioRef.current?.currentTime || 0,
   334→      queue: [],
   335→    })
   336→  }, [broadcastSync, storeClearQueue])
   337→
   338→  // ─── Effect: keep <audio> volume in sync with the store ───────────────
   339→  useEffect(() => {
   340→    if (audioRef.current) audioRef.current.volume = volume
   341→  }, [volume])
   342→
   343→  // ─── Effect: wire up <audio> event listeners ──────────────────────────
   344→  // Re-attaches whenever playNext changes (which it does when its deps
   345→  // change). The listeners read fresh state via `useMusicStore.getState()`
   346→  // so they never see stale data.
   347→  useEffect(() => {
   348→    const audio = audioRef.current
   349→    if (!audio) return
   350→
   351→    const onTimeUpdate = () => {
   352→      // Don't fight with a freshly-applied remote position
   353→      if (applyingRemoteRef.current) return
   354→      setPosition(audio.currentTime)
   355→    }
   356→    const onEnded = () => {
   357→      void playNext()
   358→    }
   359→    const onError = () => {
   360→      const state = useMusicStore.getState()
   361→      state.setIsLoading(false)
   362→      if (state.currentTrack) {
   363→        toast.error('Could not play this track — skipping...')
   364→        void playNext()
   365→      }
   366→    }
   367→    const onWaiting = () => {
   368→      useMusicStore.getState().setIsLoading(true)
   369→    }
   370→    const onPlaying = () => {
   371→      useMusicStore.getState().setIsLoading(false)
   372→    }
   373→    const onCanPlay = () => {
   374→      useMusicStore.getState().setIsLoading(false)
   375→    }
   376→
   377→    audio.addEventListener('timeupdate', onTimeUpdate)
   378→    audio.addEventListener('ended', onEnded)
   379→    audio.addEventListener('error', onError)
   380→    audio.addEventListener('waiting', onWaiting)
   381→    audio.addEventListener('playing', onPlaying)
   382→    audio.addEventListener('canplay', onCanPlay)
   383→    return () => {
   384→      audio.removeEventListener('timeupdate', onTimeUpdate)
   385→      audio.removeEventListener('ended', onEnded)
   386→      audio.removeEventListener('error', onError)
   387→      audio.removeEventListener('waiting', onWaiting)
   388→      audio.removeEventListener('playing', onPlaying)
   389→      audio.removeEventListener('canplay', onCanPlay)
   390→    }
   391→  }, [playNext, setPosition])
   392→
   393→  // ─── Effect: Pre-download upcoming queue tracks ───────────────────────
   394→  // When a track starts playing, pre-download the next 2 tracks in the queue
   395→  // so they're ready to play instantly when the user gets to them.
   396→  useEffect(() => {
   397→    if (!currentTrack || queue.length === 0) return
   398→    // Pre-download the next 2 tracks (or all if queue has fewer)
   399→    const toPreDownload = queue.slice(0, 2)
   400→    for (const track of toPreDownload) {
   401→      fetch(`/api/music/predownload/${track.videoId}`, { method: 'POST' }).catch(() => {})
   402→    }
   403→  }, [currentTrack, queue])
   404→
   405→  // ─── Effect: Socket.io room sync ──────────────────────────────────────
   406→  // Joins the music room's socket channel, requests the host's current
   407→  // state, and listens for incoming sync events. Re-subscribes whenever
   408→  // the active room changes.
   409→  useEffect(() => {
   410→    if (!socket || !activeRoomId) return
   411→
   412→    socket.emit('music:join', activeRoomId)
   413→
   414→    // Ask the host for the current state shortly after joining. The small
   415→    // delay gives the join room propagation time to complete.
   416→    const requestTimer = setTimeout(() => {
   417→      socket.emit('music:request-sync', activeRoomId)
   418→    }, 500)
   419→
   420→    // ── Incoming sync from the host (or another member's broadcast) ──
   421→    const onSync = (data: {
   422→      roomId: string
   423→      state: string
   424→      position: number
   425→      videoId?: string
   426→      trackInfo?: Track
   427→      queue?: Track[]
   428→      serverTimestamp: number
   429→    }) => {
   430→      if (data.roomId !== activeRoomId) return
   431→
   432→      applyingRemoteRef.current = true
   433→
   434→      const networkDelay = (Date.now() - data.serverTimestamp) / 1000
   435→      const adjustedPosition =
   436→        data.position + (data.state === 'playing' ? networkDelay : 0)
   437→
   438→      const state = useMusicStore.getState()
   439→      const audio = audioRef.current
   440→
   441→      // Track changed → load the new stream and seek to host's position
   442→      if (
   443→        data.videoId &&
   444→        (!state.currentTrack || state.currentTrack.videoId !== data.videoId)
   445→      ) {
   446→        const track: Track = data.trackInfo || {
   447→          videoId: data.videoId,
   448→          title: 'Now Playing',
   449→          artist: '',
   450→          thumbnail: null,
   451→          durationSeconds: null,
   452→        }
   453→        pendingSeekRef.current = adjustedPosition
   454→        setCurrentTrack(track)
   455→        setIsPlaying(data.state === 'playing')
   456→        setPosition(adjustedPosition)
   457→
   458→        if (audio) {
   459→          audio.src = `/api/music/stream/${data.videoId}`
   460→          audio.volume = state.volume
   461→          loadedVideoIdRef.current = data.videoId
   462→          // Wait a tick for the new src to be accepted before seeking
   463→          setTimeout(() => {
   464→            if (audioRef.current) {
   465→              try {
   466→                audioRef.current.currentTime = adjustedPosition
   467→              } catch {
   468→                // Stream not seekable yet — ignore
   469→              }
   470→              if (data.state === 'playing') {
   471→                audioRef.current.play().catch(() => {})
   472→              }
   473→            }
   474→            applyingRemoteRef.current = false
   475→          }, 200)
   476→        } else {
   477→          applyingRemoteRef.current = false
   478→        }
   479→      } else if (audio && state.currentTrack) {
   480→        // Same track — sync position + play/pause state only
   481→        const drift = Math.abs(audio.currentTime - adjustedPosition)
   482→        if (drift > 1.5) {
   483→          try {
   484→            audio.currentTime = adjustedPosition
   485→          } catch {
   486→            // ignore
   487→          }
   488→          setPosition(adjustedPosition)
   489→        }
   490→        if (data.state === 'playing' && !state.isPlaying) {
   491→          audio.play().catch(() => {})
   492→          setIsPlaying(true)
   493→        } else if (data.state === 'paused' && state.isPlaying) {
   494→          audio.pause()
   495→          setIsPlaying(false)
   496→        }
   497→        applyingRemoteRef.current = false
   498→      } else {
   499→        applyingRemoteRef.current = false
   500→      }
   501→
   502→      // Apply queue update from remote (broadcast by host)
   503→      if (Array.isArray(data.queue)) {
   504→        setQueue(data.queue)
   505→      }
   506→    }
   507→
   508→    // ── A member just joined and is requesting our current state ──
   509→    // Only the host (or anyone with a current track) responds.
   510→    const onRequestSync = (data: {
   511→      roomId: string
   512→      fromUserId: string
   513→    }) => {
   514→      if (data.roomId !== activeRoomId) return
   515→      const state = useMusicStore.getState()
   516→      if (!state.currentTrack || !socket) return
   517→
   518→      socket.emit('music:sync', {
   519→        roomId: activeRoomId,
   520→        state: state.isPlaying ? 'playing' : 'paused',
   521→        position: audioRef.current?.currentTime || 0,
   522→        videoId: state.currentTrack.videoId,
   523→        trackInfo: {
   524→          title: state.currentTrack.title,
   525→          artist: state.currentTrack.artist,
   526→          thumbnail: state.currentTrack.thumbnail,
   527→          durationSeconds: state.currentTrack.durationSeconds,
   528→        },
   529→        queue: state.queue,
   530→      })
   531→    }
   532→
   533→    socket.on('music:sync', onSync)
   534→    socket.on('music:request-sync', onRequestSync)
   535→
   536→    return () => {
   537→      clearTimeout(requestTimer)
   538→      socket.off('music:sync', onSync)
   539→      socket.off('music:request-sync', onRequestSync)
   540→      socket.emit('music:leave', activeRoomId)
   541→      applyingRemoteRef.current = false
   542→    }
   543→  }, [
   544→    socket,
   545→    activeRoomId,
   546→    setCurrentTrack,
   547→    setIsPlaying,
   548→    setPosition,
   549→    setQueue,
   550→  ])
   551→
   552→  // ─── Memoized context value ───────────────────────────────────────────
   553→  const contextValue = useMemo<MusicPlayerContextValue>(
   554→    () => ({
   555→      playTrack,
   556→      playNext,
   557→      togglePlay,
   558→      handleSeek,
   559→      stop,
   560→      addToQueue,
   561→      removeFromQueue,
   562→      clearQueue,
   563→    }),
   564→    [
   565→      playTrack,
   566→      playNext,
   567→      togglePlay,
   568→      handleSeek,
   569→      stop,
   570→      addToQueue,
   571→      removeFromQueue,
   572→      clearQueue,
   573→    ],
   574→  )
   575→
   576→  return (
   577→    <MusicPlayerContext.Provider value={contextValue}>
   578→      {children}
   579→      {/* The hidden audio element — persists across all tabs */}
   580→      <audio ref={audioRef} />
   581→      {/* The persistent floating player — collapses to a mini FAB */}
   582→      <AnimatePresence>
   583→        {currentTrack && (
   584→          <PlayerBar
   585→            currentTrack={currentTrack}
   586→            isPlaying={isPlaying}
   587→            isLoading={isLoading}
   588→            position={position}
   589→            volume={volume}
   590→            queue={queue}
   591→            shuffle={shuffle}
   592→            repeat={repeat}
   593→            onTogglePlay={togglePlay}
   594→            onSeek={handleSeek}
   595→            onNext={playNext}
   596→            onStop={stop}
   597→            onVolumeChange={setVolume}
   598→            onShuffleToggle={() =>
   599→              useMusicStore.getState().setShuffle(!shuffle)
   600→            }
   601→            onRepeatToggle={() =>
   602→              useMusicStore.getState().setRepeat(!repeat)
   603→            }
   604→          />
   605→        )}
   606→      </AnimatePresence>
   607→    </MusicPlayerContext.Provider>
   608→  )
   609→}
   610→
   611→// ─── Player Bar — Collapsible Floating Mini-Player ───────────────────────
   612→// When collapsed: a small floating circle (FAB) with album art + play/pause.
   613→// When expanded: a full player bar with all controls.
   614→// The player collapses automatically when the user switches away from the
   615→// Music tab, so it doesn't block content on other pages.
   616→
   617→function PlayerBar({
   618→  currentTrack,
   619→  isPlaying,
   620→  isLoading,
   621→  position,
   622→  volume,
   623→  queue,
   624→  shuffle,
   625→  repeat,
   626→  onTogglePlay,
   627→  onSeek,
   628→  onNext,
   629→  onStop,
   630→  onVolumeChange,
   631→  onShuffleToggle,
   632→  onRepeatToggle,
   633→}: {
   634→  currentTrack: Track
   635→  isPlaying: boolean
   636→  isLoading: boolean
   637→  position: number
   638→  volume: number
   639→  queue: Track[]
   640→  shuffle: boolean
   641→  repeat: boolean
   642→  onTogglePlay: () => void
   643→  onSeek: (pos: number) => void
   644→  onNext: () => Promise<void>
   645→  onStop: () => void
   646→  onVolumeChange: (v: number) => void
   647→  onShuffleToggle: () => void
   648→  onRepeatToggle: () => void
   649→}) {
   650→  const [expanded, setExpanded] = useState(false)
   651→  const { view } = useAppStore()
   652→  const isDraggingRef = useRef(false)
   653→  const constraintsRef = useRef<HTMLDivElement>(null)
   654→
   655→  // Auto-collapse when not on the Music tab
   656→  useEffect(() => {
   657→    if (view !== 'music') setExpanded(false)
   658→  }, [view])
   659→
   660→  // Prevent clicks from firing after a drag ends
   661→  const handlePlayPauseClick = useCallback(() => {
   662→    if (isDraggingRef.current) return
   663→    onTogglePlay()
   664→  }, [onTogglePlay])
   665→
   666→  // Expand only when the FAB wasn't being dragged
   667→  const handleExpandClick = useCallback(() => {
   668→    if (isDraggingRef.current) return
   669→    setExpanded(true)
   670→  }, [])
   671→
   672→  return (
   673→    <>
   674→      {/* Invisible drag constraint area — keeps FAB on screen */}
   675→      <div ref={constraintsRef} className="fixed inset-0 z-0 pointer-events-none" />
   676→
   677→      {/* ─── Collapsed: Draggable Floating Mini-Player (FAB) ─── */}
   678→      <AnimatePresence>
   679→        {!expanded && (
   680→          <motion.div
   681→            initial={{ scale: 0, opacity: 0 }}
   682→            animate={{ scale: 1, opacity: 1 }}
   683→            exit={{ scale: 0, opacity: 0 }}
   684→            transition={{ type: 'spring', stiffness: 400, damping: 25 }}
   685→            drag
   686→            dragConstraints={constraintsRef}
   687→            dragElastic={0.1}
   688→            dragTransition={{ power: 0.15, timeConstant: 150 }}
   689→            whileDrag={{ scale: 1.08, cursor: 'grabbing' }}
   690→            onDragStart={() => { isDraggingRef.current = true }}
   691→            onDragEnd={() => { setTimeout(() => { isDraggingRef.current = false }, 100) }}
   692→            className="fixed bottom-20 lg:bottom-6 right-4 z-[60] cursor-grab touch-none"
   693→          >
   694→            <div className="flex items-center gap-2.5 glass-dark rounded-full p-1.5 pr-3 shadow-xl pointer-events-auto">
   695→              {/* Album art / Play-pause */}
   696→              <button
   697→                onClick={handlePlayPauseClick}
   698→                onPointerDown={(e) => e.stopPropagation()}
   699→                className="relative w-12 h-12 rounded-full gradient-primary shadow-glow hover:scale-105 active:scale-95 transition-transform flex items-center justify-center shrink-0"
   700→                aria-label={isPlaying ? 'Pause' : 'Play'}
   701→              >
   702→                {currentTrack.thumbnail ? (
   703→                  <img
   704→                    src={currentTrack.thumbnail}
   705→                    alt=""
   706→                    className="absolute inset-0 w-full h-full rounded-full object-cover opacity-80"
   707→                  />
   708→                ) : (
   709→                  <MusicIcon className="w-5 h-5 text-primary-foreground relative z-10" />
   710→                )}
   711→                <div className="absolute inset-0 rounded-full bg-black/40 flex items-center justify-center">
   712→                  {isLoading ? (
   713→                    <Loader2 className="w-5 h-5 text-white animate-spin relative z-10" />
   714→                  ) : isPlaying ? (
   715→                    <Pause className="w-5 h-5 text-white relative z-10" />
   716→                  ) : (
   717→                    <Play className="w-5 h-5 text-white ml-0.5 relative z-10" />
   718→                  )}
   719→                </div>
   720→              </button>
   721→
   722→              {/* Track title — visible on desktop, tap target for expand on mobile */}
   723→              <button
   724→                onClick={handleExpandClick}
   725→                onPointerDown={(e) => e.stopPropagation()}
   726→                className="hidden sm:block min-w-0 max-w-[140px] text-left"
   727→              >
   728→                <div className="text-xs font-medium truncate text-foreground">{currentTrack.title}</div>
   729→                <div className="text-[10px] text-muted-foreground truncate">{currentTrack.artist}</div>
   730→              </button>
   731→
   732→              {/* Expand button */}
   733→              <button
   734→                onClick={handleExpandClick}
   735→                onPointerDown={(e) => e.stopPropagation()}
   736→                className="w-8 h-8 rounded-full bg-white/10 flex items-center justify-center text-foreground hover:bg-white/20 hover:scale-105 active:scale-95 transition-all shrink-0"
   737→                aria-label="Expand player"
   738→              >
   739→                <ChevronUp className="w-4 h-4" />
   740→              </button>
   741→            </div>
   742→          </motion.div>
   743→        )}
   744→      </AnimatePresence>
   745→
   746→      {/* ─── Expanded: Full Player Bar ─── */}
   747→      <AnimatePresence>
   748→        {expanded && (
   749→          <motion.div
   750→            initial={{ y: 120, opacity: 0 }}
   751→            animate={{ y: 0, opacity: 1 }}
   752→            exit={{ y: 120, opacity: 0 }}
   753→            transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   754→            className="fixed bottom-16 lg:bottom-4 left-3 right-3 lg:left-1/2 lg:-translate-x-1/2 lg:right-auto lg:w-[660px] z-[60] glass-dark rounded-2xl shadow-2xl overflow-hidden"
   755→          >
   756→            {/* Mobile layout */}
   757→            <div className="lg:hidden px-4 py-3 space-y-2.5">
   758→              <div className="flex items-center gap-3">
   759→                {currentTrack.thumbnail ? (
   760→                  <img src={currentTrack.thumbnail} alt="" className="w-12 h-12 rounded-xl object-cover shrink-0 ring-1 ring-white/10 shadow-md" />
   761→                ) : (
   762→                  <div className="w-12 h-12 rounded-xl gradient-primary flex items-center justify-center shrink-0 shadow-md">
   763→                    <MusicIcon className="w-5 h-5 text-primary-foreground" />
   764→                  </div>
   765→                )}
   766→                <div className="min-w-0 flex-1">
   767→                  <div className="text-sm font-semibold truncate">{currentTrack.title}</div>
   768→                  <div className="text-xs text-muted-foreground truncate">{currentTrack.artist}</div>
   769→                </div>
   770→                <div className="flex items-center gap-1 shrink-0">
   771→                  <Button onClick={onTogglePlay} size="icon" className="rounded-full h-10 w-10 gradient-primary shadow-glow">
   772→                    {isLoading ? <Loader2 className="w-4 h-4 animate-spin" /> : isPlaying ? <Pause className="w-4 h-4" /> : <Play className="w-4 h-4 ml-0.5" />}
   773→                  </Button>
   774→                  <Button onClick={onNext} variant="ghost" size="icon" className="h-9 w-9 text-foreground">
   775→                    <SkipForward className="w-4 h-4" />
   776→                  </Button>
   777→                  <Button onClick={() => setExpanded(false)} variant="ghost" size="icon" className="h-9 w-9 text-muted-foreground">
   778→                    <ChevronDown className="w-4 h-4" />
   779→                  </Button>
   780→                </div>
   781→              </div>
   782→              {/* Progress bar */}
   783→              <div className="flex items-center gap-2">
   784→                <span className="text-[10px] text-muted-foreground tabular-nums w-8 text-right">{formatTime(position)}</span>
   785→                <input
   786→                  type="range"
   787→                  min={0}
   788→                  max={currentTrack.durationSeconds || 300}
   789→                  value={position}
   790→                  onChange={(e) => onSeek(parseFloat(e.target.value))}
   791→                  className="flex-1 player-slider"
   792→                />
   793→                <span className="text-[10px] text-muted-foreground tabular-nums w-8">{formatTime(currentTrack.durationSeconds || 0)}</span>
   794→              </div>
   795→            </div>
   796→
   797→            {/* Desktop layout */}
   798→            <div className="hidden lg:flex px-5 py-4 pr-7 items-center gap-5">
   799→              {/* Track info */}
   800→              <div className="flex items-center gap-3 min-w-0 w-52 shrink-0">
   801→                {currentTrack.thumbnail ? (
   802→                  <img src={currentTrack.thumbnail} alt="" className="w-12 h-12 rounded-xl object-cover shrink-0 ring-1 ring-white/10 shadow-md" />
   803→                ) : (
   804→                  <div className="w-12 h-12 rounded-xl gradient-primary flex items-center justify-center shrink-0 shadow-md">
   805→                    <MusicIcon className="w-5 h-5 text-primary-foreground" />
   806→                  </div>
   807→                )}
   808→                <div className="min-w-0">
   809→                  <div className="text-sm font-semibold truncate">{currentTrack.title}</div>
   810→                  <div className="text-xs text-muted-foreground truncate">{currentTrack.artist}</div>
   811→                </div>
   812→              </div>
   813→
   814→              {/* Controls + seek bar */}
   815→              <div className="flex-1 flex flex-col items-center gap-2">
   816→                <div className="flex items-center gap-3">
   817→                  <button onClick={onShuffleToggle} className={cn('p-2 rounded-xl transition-all', shuffle ? 'text-primary bg-primary/15 shadow-glow' : 'text-muted-foreground hover:text-foreground hover:bg-white/5')} aria-label="Shuffle">
   818→                    <Shuffle className="w-4 h-4" />
   819→                  </button>
   820→                  <button onClick={onNext} className="p-2 rounded-xl text-foreground hover:bg-white/10 transition-colors" aria-label="Next">
   821→                    <SkipForward className="w-4 h-4" />
   822→                  </button>
   823→                  <button onClick={onTogglePlay} className="w-10 h-10 rounded-full gradient-primary flex items-center justify-center hover:scale-110 active:scale-95 transition-transform shadow-glow" aria-label={isPlaying ? 'Pause' : 'Play'}>
   824→                    {isLoading ? <Loader2 className="w-4 h-4 animate-spin" /> : isPlaying ? <Pause className="w-4 h-4" /> : <Play className="w-4 h-4 ml-0.5" />}
   825→                  </button>
   826→                  <button onClick={onStop} className="p-2 rounded-xl text-foreground hover:bg-white/10 transition-colors" aria-label="Stop">
   827→                    <X className="w-4 h-4" />
   828→                  </button>
   829→                  <button onClick={onRepeatToggle} className={cn('p-2 rounded-xl transition-all', repeat ? 'text-primary bg-primary/15 shadow-glow' : 'text-muted-foreground hover:text-foreground hover:bg-white/5')} aria-label="Repeat">
   830→                    <Repeat className="w-4 h-4" />
   831→                  </button>
   832→                </div>
   833→                <div className="flex items-center gap-2.5 w-full max-w-sm">
   834→                  <span className="text-[10px] text-muted-foreground tabular-nums w-9 text-right">{formatTime(position)}</span>
   835→                  <input
   836→                    type="range"
   837→                    min={0}
   838→                    max={currentTrack.durationSeconds || 300}
   839→                    value={position}
   840→                    onChange={(e) => onSeek(parseFloat(e.target.value))}
   841→                    className="flex-1 player-slider"
   842→                  />
   843→                  <span className="text-[10px] text-muted-foreground tabular-nums w-9">{formatTime(currentTrack.durationSeconds || 0)}</span>
   844→                </div>
   845→              </div>
   846→
   847→              {/* Volume + collapse */}
   848→              <div className="flex items-center gap-2.5 shrink-0">
   849→                <Volume2 className="w-4 h-4 text-muted-foreground shrink-0" />
   850→                <input
   851→                  type="range"
   852→                  min={0}
   853→                  max={1}
   854→                  step={0.01}
   855→                  value={volume}
   856→                  onChange={(e) => onVolumeChange(parseFloat(e.target.value))}
   857→                  className="w-20 volume-slider"
   858→                />
   859→                <button onClick={() => setExpanded(false)} className="p-2 rounded-xl text-muted-foreground hover:text-foreground hover:bg-white/10 transition-colors" aria-label="Collapse player">
   860→                  <ChevronDown className="w-4 h-4" />
   861→                </button>
   862→              </div>
   863→            </div>
   864→          </motion.div>
   865→        )}
   866→      </AnimatePresence>
   867→    </>
   868→  )
   869→}
   870→
   871→function formatTime(seconds: number): string {
   872→  const m = Math.floor(seconds / 60)
   873→  const s = Math.floor(seconds % 60)
   874→  return `${m}:${s.toString().padStart(2, '0')}`
   875→}
