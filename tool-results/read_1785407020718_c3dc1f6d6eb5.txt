    27→    27→export function MessageComposer({ channelId }: MessageComposerProps) {
    28→    28→  const { send, sendTyping, replyTo, setReplyTo } = useChannel(channelId)
    29→    29→  const [text, setText] = useState('')
    30→    30→  const [sending, setSending] = useState(false)
    31→    31→  const [uploading, setUploading] = useState(false)
    32→    32→  const [ttsOpen, setTtsOpen] = useState(false)
    33→    33→  const textareaRef = useRef<HTMLTextAreaElement>(null)
    34→    34→
    35→    35→  useEffect(() => {
    36→    36→    const el = textareaRef.current
    37→    37→    if (!el) return
    38→    38→    el.style.height = 'auto'
    39→    39→    el.style.height = Math.min(el.scrollHeight, 160) + 'px'
    40→    40→  }, [text])
    41→    41→
    42→    42→  useEffect(() => {
    43→    43→    if (replyTo) textareaRef.current?.focus()
    44→    44→  }, [replyTo])
    45→    45→
    46→    46→  const handleSend = async () => {
    47→    47→    const trimmed = text.trim()
    48→    48→    if (!trimmed || sending) return
    49→    49→    setSending(true)
    50→    50→    try {
    51→    51→      await send({ body: trimmed, replyToId: replyTo?.id || null })
    52→    52→      setText('')
    53→    53→      setReplyTo(null)
    54→    54→      sendTyping(false)
    55→    55→    } catch {
    56→    56→      toast.error('Failed to send message')
    57→    57→    } finally {
    58→    58→      setSending(false)
    59→    59→    }
    60→    60→  }
    61→    61→
    62→    62→  const handleKeyDown = (e: React.KeyboardEvent) => {
    63→    63→    if (e.key === 'Enter' && !e.shiftKey) {
    64→    64→      e.preventDefault()
    65→    65→      handleSend()
    66→    66→    }
    67→    67→  }
    68→    68→
    69→    69→  const handleChange = (e: React.ChangeEvent<HTMLTextAreaElement>) => {
    70→    70→    setText(e.target.value)
    71→    71→    if (e.target.value.length > 0) sendTyping(true)
    72→    72→    else sendTyping(false)
    73→    73→  }
    74→    74→
    75→    75→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    76→    76→    const file = e.target.files?.[0]
    77→    77→    if (!file) return
    78→    78→    setUploading(true)
    79→    79→    try {
    80→    80→      const formData = new FormData()
    81→    81→      formData.append('file', file)
    82→    82→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
    83→    83→      if (!res.ok) {
    84→    84→        const text = await res.text()
    85→    85→        throw new Error(text.slice(0, 120) || 'Upload failed')
    86→    86→      }
    87→    87→      const data = await res.json()
    88→    88→      await send({
    89→    89→        body: data.type.startsWith('image') ? '📷' : '📎',
    90→    90→        mediaUrl: data.url,
    91→    91→        mediaType: data.type,
    92→    92→      })
    93→    93→    } catch (e: any) {
    94→    94→      toast.error(e.message || 'Upload failed')
    95→    95→    } finally {
    96→    96→      setUploading(false)
    97→    97→      e.target.value = ''
    98→    98→    }
    99→    99→  }
   100→   100→
   101→   101→  return (
   102→   102→    <div className="border-t border-white/5 bg-background/60 backdrop-blur-3xl px-3 md:px-6 py-4 pb-safe relative z-20">
   103→   103→      {/* Reply banner */}
   104→   104→      {replyTo && (
   105→   105→        <div className="flex items-center gap-3 text-xs bg-black/40 border border-white/10 backdrop-blur-md rounded-xl p-3 mb-3 shadow-lg">
   106→   106→          <Reply className="w-4 h-4 shrink-0 text-primary" />
   107→   107→          <div className="flex-1 min-w-0">
   108→   108→            <div className="font-semibold text-primary">{replyTo.senderName}</div>
   109→   109→            <div className="text-muted-foreground truncate mt-0.5">{replyTo.body}</div>
   110→   110→          </div>
   111→   111→          <button onClick={() => setReplyTo(null)} className="p-1.5 hover:bg-white/10 rounded-full transition-colors">
   112→   112→            <X className="w-4 h-4" />
   113→   113→          </button>
   114→   114→        </div>
   115→   115→      )}
   116→   116→
   117→   117→      <div className="flex items-end gap-2.5">
   118→   118→        {/* Upload button */}
   119→   119→        <label className="cursor-pointer w-10 h-10 flex items-center justify-center hover:bg-white/5 rounded-full transition-colors shrink-0 border border-transparent hover:border-white/10">
   120→   120→          <ImageIcon className="w-[22px] h-[22px] text-muted-foreground hover:text-foreground transition-colors" />
   121→   121→          <input type="file" className="hidden" accept="image/*,video/*,audio/*" onChange={handleUpload} disabled={uploading} />
   122→   122→        </label>
   123→   123→
   124→   124→        {/* TTS voice message button */}
   125→   125→        <button
   126→   126→          onClick={() => setTtsOpen(true)}
   127→   127→          className="w-10 h-10 flex items-center justify-center hover:bg-white/5 rounded-full transition-colors shrink-0 group border border-transparent hover:border-white/10"
   128→   128→          title="Send AI voice message"
   129→   129→        >
   130→   130→          <AudioLines className="w-[22px] h-[22px] text-muted-foreground group-hover:text-primary transition-colors" />
   131→   131→        </button>
   132→   132→
   133→   133→        {/* Text input — grows up to ~4 lines */}
   134→   134→        <div className="flex-1 bg-black/20 backdrop-blur-xl rounded-[24px] px-5 py-3 flex items-end border border-white/10 shadow-inner ring-1 ring-transparent focus-within:ring-primary/30 transition-all">
   135→   135→          <Textarea
   136→   136→            ref={textareaRef}
   137→   137→            value={text}
   138→   138→            onChange={handleChange}
   139→   139→            onKeyDown={handleKeyDown}
   140→   140→            placeholder="iMessage..."
   141→   141→            className="flex-1 resize-none min-h-[24px] max-h-[160px] bg-transparent border-0 focus-visible:ring-0 focus-visible:ring-offset-0 p-0 text-[16px] leading-relaxed shadow-none placeholder:text-muted-foreground/50"

   143→   143→            rows={1}
   144→   144→          />
   145→   145→        </div>
   146→   146→
   147→   147→        {/* Send button */}
   148→   148→        <Button
   149→   149→          onClick={handleSend}
   150→   150→          disabled={!text.trim() || sending}
   151→   151→          size="icon"
   152→   152→          className="rounded-full h-11 w-11 shrink-0 transition-transform hover:scale-105 active:scale-95 gradient-primary shadow-glow disabled:opacity-50 disabled:shadow-none"
   153→   153→        >
   154→   154→          {sending ? <Loader2 className="w-5 h-5 animate-spin" /> : <Send className="w-5 h-5" />}
   155→   155→        </Button>
   156→   156→      </div>
   157→   157→
   158→   158→      {/* TTS Voice Message Dialog */}
   159→   159→      <TtsDialog
   160→   160→        open={ttsOpen}
   161→   161→        onOpenChange={setTtsOpen}
   162→   162→        onSend={async (url) => {
   163→   163→          await send({
   164→   164→            body: '🎙️ AI voice message',
   165→   165→            mediaUrl: url,
   166→   166→            mediaType: 'audio',
   167→   167→          })
   168→   168→          setTtsOpen(false)
   169→   169→          toast.success('Voice message sent')
   170→   170→        }}
   171→   171→      />
   172→   172→    </div>
   173→   173→  )
   174→   174→}

   288→   288→            onClick={() => setTab('voices')}
   289→   289→            className={cn(
   290→   290→              'flex-1 py-1.5 text-sm font-medium rounded-md transition-colors flex items-center justify-center gap-1.5',
   291→   291→              tab === 'voices' ? 'bg-background text-foreground' : 'text-muted-foreground'
   292→   292→            )}
   293→   293→          >
   294→   294→            My Voices
   295→   295→            {customVoices.length > 0 && (
   296→   296→              <span className="text-[10px] bg-primary/20 text-primary px-1.5 py-0.5 rounded-full">

   300→   300→          </button>

   302→   302→
   303→   303→        {tab === 'generate' ? (
   304→   304→          <div className="space-y-4">

   306→   306→              <Label>Message</Label>
   307→   307→              <Textarea
   308→   308→                value={text}
   309→   309→                onChange={(e) => setText(e.target.value)}
   310→   310→                placeholder="Type what the voice should say..."
   311→   311→                rows={3}
   312→   312→                maxLength={500}

   470→   470→function CustomVoicesTab({ onUseVoice }: { onUseVoice: (id: string) => void }) {
   471→   471→  const qc = useQueryClient()
   472→   472→  const [name, setName] = useState('')
   473→   473→  const [audioUrl, setAudioUrl] = useState<string | null>(null)
   474→   474→  const [uploading, setUploading] = useState(false)
   475→   475→  const [recording, setRecording] = useState(false)
   476→   476→  const [mediaRecorder, setMediaRecorder] = useState<MediaRecorder | null>(null)
   477→   477→  const [audioChunks, setAudioChunks] = useState<Blob[]>([])
   478→   478→
   479→   479→  const { data: customVoicesData } = useQuery({
   480→   480→    queryKey: ['tts-voices'],
   481→   481→    queryFn: async () => {
   482→   482→      const res = await fetch('/api/tts/voices')
   483→   483→      if (!res.ok) throw new Error('failed')
   484→   484→      return res.json()
   485→   485→    },
   486→   486→  })
   487→   487→  const customVoices = customVoicesData?.voices || []
   488→   488→
   489→   489→  const deleteVoice = useMutation({
   490→   490→    mutationFn: async (id: string) => {
   491→   491→      await fetch(`/api/tts/voices/${id}`, { method: 'DELETE' })
   492→   492→    },
   493→   493→    onSuccess: () => {
   494→   494→      qc.invalidateQueries({ queryKey: ['tts-voices'] })
   495→   495→      toast.success('Voice deleted')
   496→   496→    },
   497→   497→  })
   498→   498→
   499→   499→  const saveVoice = useMutation({
   500→   500→    mutationFn: async () => {
   501→   501→      const res = await fetch('/api/tts/voices', {
   502→   502→        method: 'POST',
   503→   503→        headers: { 'Content-Type': 'application/json' },
   504→   504→        body: JSON.stringify({ name, audioUrl }),
   505→   505→      })
   506→   506→      if (!res.ok) throw new Error('failed')
   507→   507→      return res.json()
   508→   508→    },
   509→   509→    onSuccess: () => {
   510→   510→      qc.invalidateQueries({ queryKey: ['tts-voices'] })
   511→   511→      toast.success('Custom voice created')
   512→   512→      setName('')
   513→   513→      setAudioUrl(null)
   514→   514→    },
   515→   515→  })
   516→   516→
   517→   517→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   518→   518→    const file = e.target.files?.[0]
   519→   519→    if (!file) return
   520→   520→    setUploading(true)
   521→   521→    try {
   522→   522→      const formData = new FormData()
   523→   523→      formData.append('file', file)
   524→   524→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   525→   525→      if (!res.ok) throw new Error('Upload failed')
   526→   526→      const data = await res.json()
   527→   527→      setAudioUrl(data.url)
   528→   528→    } catch (e: any) {
   529→   529→      toast.error(e.message || 'Upload failed')
   530→   530→    } finally {
   531→   531→      setUploading(false)
   532→   532→      e.target.value = ''
   533→   533→    }
   534→   534→  }
   535→   535→
   536→   536→  const startRecording = async () => {
   537→   537→    try {
   538→   538→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   539→   539→      const recorder = new MediaRecorder(stream)
   540→   540→      const chunks: Blob[] = []
   541→   541→      recorder.ondataavailable = (e) => chunks.push(e.data)
   542→   542→      recorder.onstop = async () => {
   543→   543→        const blob = new Blob(chunks, { type: 'audio/webm' })
   544→   544→        // Upload the recording
   545→   545→        setUploading(true)
   546→   546→        try {
   547→   547→          const formData = new FormData()
   548→   548→          formData.append('file', new File([blob], 'voice.webm', { type: 'audio/webm' }))
   549→   549→          const res = await fetch('/api/upload', { method: 'POST', body: formData })
   550→   550→          if (!res.ok) throw new Error('Upload failed')
   551→   551→          const data = await res.json()
   552→   552→          setAudioUrl(data.url)
   553→   553→        } catch (e: any) {
   554→   554→          toast.error(e.message || 'Upload failed')
   555→   555→        } finally {
   556→   556→          setUploading(false)
   557→   557→        }
   558→   558→        stream.getTracks().forEach((t) => t.stop())
   559→   559→      }
   560→   560→      recorder.start()
   561→   561→      setMediaRecorder(recorder)
   562→   562→      setRecording(true)
   563→   563→      setAudioChunks(chunks)
   564→   564→    } catch {
   565→   565→      toast.error('Microphone access denied')
   566→   566→    }
   567→   567→  }
   568→   568→
   569→   569→  const stopRecording = () => {
   570→   570→    if (mediaRecorder && mediaRecorder.state === 'recording') {
   571→   571→      mediaRecorder.stop()
   572→   572→      setRecording(false)
   573→   573→    }
   574→   574→  }
   575→   575→
   576→   576→  return (
   577→   577→    <div className="space-y-4">
   578→   578→      {/* Create new voice */}
   579→   579→      <div className="space-y-3 p-3 rounded-xl bg-muted/50 border border-border/30">
   580→   580→        <div className="flex items-center gap-2 text-sm font-medium">
   581→   581→          <Mic className="w-4 h-4 text-primary" />
   582→   582→          Create a Custom Voice
   583→   583→        </div>
   584→   584→        <p className="text-xs text-muted-foreground">
   585→   585→          Upload or record a 10-30 second voice clip. We'll clone the voice so you can generate speech in it.
   586→   586→        </p>
   587→   587→        <div className="space-y-2">
   588→   588→          <Input
   589→   589→            value={name}
   590→   590→            onChange={(e) => setName(e.target.value)}
   591→   591→            placeholder="Voice name (e.g. My Voice)"
   592→   592→          />
   593→   593→        </div>
   594→   594→        <div className="flex gap-2">
   595→   595→          {!audioUrl ? (
   596→   596→            <>
   597→   597→              {!recording ? (
   598→   598→                <Button onClick={startRecording} variant="outline" size="sm" className="flex-1">
   599→   599→                  <Mic className="w-4 h-4 mr-1.5" />
   600→   600→                  Record
   601→   601→                </Button>
   602→   602→              ) : (
   603→   603→                <Button onClick={stopRecording} variant="destructive" size="sm" className="flex-1">
   604→   604→                  <span className="w-2 h-2 rounded-full bg-white animate-pulse mr-1.5" />
   605→   605→                  Stop
   606→   606→                </Button>
   607→   607→              )}
   608→   608→              <label className="cursor-pointer flex-1">
   609→   609→                <Button asChild variant="outline" size="sm" className="w-full">
   610→   610→                  <span>
   611→   611→                    <Plus className="w-4 h-4 mr-1.5" />
   612→   612→                    Upload
   613→   613→                  </span>
   614→   614→                </Button>
   615→   615→                <input type="file" className="hidden" accept="audio/*" onChange={handleUpload} disabled={uploading || recording} />
   616→   616→              </label>
   617→   617→            </>
   618→   618→          ) : (
   619→   619→            <div className="flex-1 space-y-2">
   620→   620→              <audio controls src={audioUrl} className="w-full h-8" />
   621→   621→              <div className="flex gap-2">
   622→   622→                <Button
   623→   623→                  onClick={() => saveVoice.mutate()}
   624→   624→                  disabled={!name.trim() || saveVoice.isPending}
   625→   625→                  size="sm"
   626→   626→                  className="flex-1 gradient-primary"
   627→   627→                >
   628→   628→                  {saveVoice.isPending ? (
   629→   629→                    <Loader2 className="w-4 h-4 mr-1.5 animate-spin" />
   630→   630→                  ) : (
   631→   631→                    <Sparkles className="w-4 h-4 mr-1.5" />
   632→   632→                  )}
   633→   633→                  Save voice
   634→   634→                </Button>
   635→   635→                <Button
   636→   636→                  onClick={() => setAudioUrl(null)}
   637→   637→                  variant="outline"
   638→   638→                  size="sm"
   639→   639→                >
   640→   640→                  Discard
   641→   641→                </Button>
   642→   642→              </div>
   643→   643→            </div>
   644→   644→          )}
   645→   645→        </div>
   646→   646→        {uploading && (
   647→   647→          <div className="text-xs text-muted-foreground flex items-center gap-1.5">
   648→   648→            <Loader2 className="w-3 h-3 animate-spin" />
   649→   649→            Uploading...
   650→   650→          </div>
   651→   651→        )}
   652→   652→      </div>
   653→   653→
   654→   654→      {/* Existing custom voices */}
   655→   655→      {customVoices.length > 0 && (
   656→   656→        <div className="space-y-2">
   657→   657→          <Label className="text-xs uppercase tracking-wider text-muted-foreground">
   658→   658→            Saved Voices
   659→   659→          </Label>
   660→   660→          <div className="space-y-1.5 max-h-48 overflow-y-auto">
   661→   661→            {customVoices.map((v: any) => (
   662→   662→              <div
   663→   663→                key={v.id}
   664→   664→                className="flex items-center gap-3 p-2.5 rounded-xl bg-card border border-border/30 hover:border-primary/30 transition-colors group"
   665→   665→              >
   666→   666→                <div className="w-9 h-9 rounded-full bg-primary/15 flex items-center justify-center shrink-0">
   667→   667→                  <User className="w-4 h-4 text-primary" />
   668→   668→                </div>
   669→   669→                <div className="flex-1 min-w-0">
   670→   670→                  <div className="text-sm font-medium truncate">{v.name}</div>
   671→   671→                  <div className="text-xs text-muted-foreground">
   672→   672→                    Created {new Date(v.createdAt).toLocaleDateString()}
   673→   673→                  </div>
   674→   674→                </div>
   675→   675→                <button
   676→   676→                  onClick={() => onUseVoice(v.id)}
   677→   677→                  className="text-xs px-2.5 py-1.5 rounded-lg bg-primary/10 text-primary hover:bg-primary/20 transition-colors font-medium"
   678→   678→                >
   679→   679→                  Use
   680→   680→                </button>
   681→   681→                <button
   682→   682→                  onClick={() => {
   683→   683→                    if (confirm(`Delete "${v.name}"?`)) deleteVoice.mutate(v.id)
   684→   684→                  }}
   685→   685→                  className="p-1.5 rounded-lg text-red-400 hover:bg-red-500/10 transition-colors"
   686→   686→                >
   687→   687→                  <Trash2 className="w-3.5 h-3.5" />
   688→   688→                </button>
   689→   689→              </div>
   690→   690→            ))}
   691→   691→          </div>
   692→   692→        </div>
   693→   693→      )}
   694→   694→
   695→   695→      {customVoices.length === 0 && !audioUrl && (
   696→   696→        <div className="text-center py-6">
   697→   697→          <div className="w-12 h-12 mx-auto rounded-2xl bg-primary/10 flex items-center justify-center mb-2">
   698→   698→            <Mic className="w-5 h-5 text-primary" />
   699→   699→          </div>
   700→   700→          <p className="text-sm text-muted-foreground">No custom voices yet</p>
   701→   701→        </div>
   702→   702→      )}
   703→   703→    </div>
   704→   704→  )
   705→   705→}
