    24→    24→/**
    25→    25→ * GlobalMusicPlayer
    26→    26→ * ─────────────────────────────────────────────────────────────────────────
    27→    27→ * Renders the persistent <audio> element and the bottom player bar at the
    28→    28→ * app root level so that audio playback is NOT destroyed when the user
    29→    29→ * switches tabs (Chats, Status, Calls, Settings, Music).
    30→    30→ *
    31→    31→ * Architecture:
    32→    32→ *   - The Zustand store (`useMusicStore`) is the single source of truth for
    33→    33→ *     playback state (currentTrack, isPlaying, position, volume, queue,
    34→    34→ *     autoplay, shuffle, repeat, activeRoomId).
    35→    35→ *   - This component owns the <audio> element and all audio + Socket.io
    36→    36→ *     side-effects. It reads from the store and writes back to it.
    37→    37→ *   - The MusicView (browse / rooms / queue UI) reads state from the store
    38→    38→ *     and calls the action functions exposed here via `useMusicPlayer()`.
    39→    39→ *   - Socket.io sync events are listened to here. When a remote sync is
    40→    40→ *     applied, the store is updated; the audio element is then loaded /
    41→    41→ *     seeked directly. A ref flag (`applyingRemoteRef`) prevents
    42→    42→ *     `timeupdate` events from fighting with the applied position.
    43→    43→ */
    44→    44→
    45→    45→interface MusicPlayerContextValue {
    46→    46→  /** Play a track (or add it to the queue if addToQueue=true). */
    47→    47→  playTrack: (track: Track, addToQueue?: boolean) => Promise<void>
    48→    48→  /** Advance to the next track (queue → autoplay → repeat → stop). */
    49→    49→  playNext: () => Promise<void>
    50→    50→  /** Toggle play / pause. */
    51→    51→  togglePlay: () => void
    52→    52→  /** Seek to a position (seconds). */
    53→    53→  handleSeek: (pos: number) => void
    54→    54→  /** Stop playback entirely (clears current track). */
    55→    55→  stop: () => void
    56→    56→  /** Add a track to the queue (broadcasts to room). */
    57→    57→  addToQueue: (track: Track) => void
    58→    58→  /** Remove a track from the queue by index (broadcasts to room). */
    59→    59→  removeFromQueue: (index: number) => void
    60→    60→  /** Clear the entire queue (broadcasts to room). */
    61→    61→  clearQueue: () => void
    62→    62→}
    63→    63→
    64→    64→const MusicPlayerContext = createContext<MusicPlayerContextValue | null>(null)
    65→    65→
    66→    66→const NOOP_PLAYER: MusicPlayerContextValue = {
    67→    67→  playTrack: async () => {},
    68→    68→  playNext: async () => {},
    69→    69→  togglePlay: () => {},
    70→    70→  handleSeek: () => {},
    71→    71→  stop: () => {},
    72→    72→  addToQueue: () => {},
    73→    73→  removeFromQueue: () => {},
    74→    74→  clearQueue: () => {},
    75→    75→}
    76→    76→
    77→    77→/**
    78→    78→ * Access the global music player's action functions.
    79→    79→ *
    80→    80→ * Safe to call from any component rendered inside <GlobalMusicPlayer>.
    81→    81→ * Returns no-op functions if no provider is present (defensive).
    82→    82→ */
    83→    83→export function useMusicPlayer(): MusicPlayerContextValue {
    84→    84→  const ctx = useContext(MusicPlayerContext)
    85→    85→  return ctx ?? NOOP_PLAYER
    86→    86→}
    87→    87→
    88→    88→export function GlobalMusicPlayer({ children }: { children: React.ReactNode }) {
    89→    89→  const audioRef = useRef<HTMLAudioElement>(null)
    90→    90→  const { socket } = useSocket()
    91→    91→
    92→    92→  // ─── Store state (subscribe to slices) ────────────────────────────────
    93→    93→  const currentTrack = useMusicStore((s) => s.currentTrack)
    94→    94→  const isPlaying = useMusicStore((s) => s.isPlaying)
    95→    95→  const position = useMusicStore((s) => s.position)
    96→    96→  const volume = useMusicStore((s) => s.volume)
    97→    97→  const queue = useMusicStore((s) => s.queue)
    98→    98→  const shuffle = useMusicStore((s) => s.shuffle)
    99→    99→  const repeat = useMusicStore((s) => s.repeat)
   100→   100→  const activeRoomId = useMusicStore((s) => s.activeRoomId)
   101→   101→
   102→   102→  // ─── Store actions ────────────────────────────────────────────────────
   103→   103→  const setCurrentTrack = useMusicStore((s) => s.setCurrentTrack)
   104→   104→  const setIsPlaying = useMusicStore((s) => s.setIsPlaying)
   105→   105→  const setPosition = useMusicStore((s) => s.setPosition)
   106→   106→  const setVolume = useMusicStore((s) => s.setVolume)
   107→   107→  const setQueue = useMusicStore((s) => s.setQueue)
   108→   108→  const storeAddToQueue = useMusicStore((s) => s.addToQueue)
   109→   109→  const storeRemoveFromQueue = useMusicStore((s) => s.removeFromQueue)
   110→   110→  const storeClearQueue = useMusicStore((s) => s.clearQueue)
   111→   111→
   112→   112→  // ─── Refs for cross-render coordination ───────────────────────────────
   113→   113→  // The videoId currently loaded into the <audio> element. Used to detect
   114→   114→  // when currentTrack.videoId changes so we know to (re)load the stream.
   115→   115→  const loadedVideoIdRef = useRef<string | null>(null)
   116→   116→  // When applying a remote sync that changes the track, this holds the
   117→   117→  // position we should seek to once the new stream has loaded.
   118→   118→  const pendingSeekRef = useRef<number | null>(null)
   119→   119→  // True while we're applying state that arrived from a remote sync event.
   120→   120→  // Prevents local `timeupdate` from overwriting the freshly-applied
   121→   121→  // position and prevents echoing sync events back to the room.
   122→   122→  const applyingRemoteRef = useRef(false)
   123→   123→
   124→   124→  // ─── Helper: broadcast a sync event to the active room ────────────────
   125→   125→  const broadcastSync = useCallback(
   126→   126→    (payload: {
   127→   127→      state: string
   128→   128→      position: number
   129→   129→      videoId?: string
   130→   130→      trackInfo?: Partial<Track>
   131→   131→      queue?: Track[]
   132→   132→    }) => {
   133→   133→      if (!activeRoomId || !socket) return
   134→   134→      socket.emit('music:sync', {
   135→   135→        roomId: activeRoomId,
   136→   136→        ...payload,
   137→   137→      })
   138→   138→    },
   139→   139→    [activeRoomId, socket],
   140→   140→  )
   141→   141→
   142→   142→  // ─── Action: play a track ─────────────────────────────────────────────
   143→   143→  const playTrack = useCallback(
   144→   144→    async (track: Track, addToQueue = false) => {
   145→   145→      const state = useMusicStore.getState()
   146→   146→
   147→   147→      if (addToQueue && state.currentTrack) {
   148→   148→        state.addToQueue(track)
   149→   149→        toast.success(`Added to queue: ${track.title}`)
   150→   150→        // Broadcast the updated queue so members see it
   151→   151→        broadcastSync({
   152→   152→          state: state.isPlaying ? 'playing' : 'paused',
   153→   153→          position: audioRef.current?.currentTime || 0,
   154→   154→          queue: useMusicStore.getState().queue,
   155→   155→        })
   156→   156→        return
   157→   157→      }
   158→   158→
   159→   159→      // Set state first so the UI updates instantly
   160→   160→      setCurrentTrack(track)
   161→   161→      setIsPlaying(true)
   162→   162→      setPosition(0)
   163→   163→
   164→   164→      // Broadcast to room (include the full queue so members sync)
   165→   165→      broadcastSync({
   166→   166→        state: 'playing',
   167→   167→        position: 0,
   168→   168→        videoId: track.videoId,
   169→   169→        trackInfo: {
   170→   170→          title: track.title,
   171→   171→          artist: track.artist,
   172→   172→          thumbnail: track.thumbnail,
   173→   173→          durationSeconds: track.durationSeconds,
   174→   174→        },
   175→   175→        queue: useMusicStore.getState().queue,
   176→   176→      })
   177→   177→
   178→   178→      // Load + play the audio
   179→   179→      if (audioRef.current) {
   180→   180→        audioRef.current.src = `/api/music/stream/${track.videoId}`
   181→   181→        audioRef.current.volume = useMusicStore.getState().volume
   182→   182→        loadedVideoIdRef.current = track.videoId
   183→   183→        pendingSeekRef.current = null
   184→   184→        try {
   185→   185→          await audioRef.current.play()
   186→   186→        } catch (e: any) {
   187→   187→          if (e?.name === 'NotSupportedError' || e?.name === 'MediaError') {
   188→   188→            try {
   189→   189→              const res = await fetch(`/api/music/stream/${track.videoId}`, {
   190→   190→                method: 'HEAD',
   191→   191→              })
   192→   192→              if (!res.ok) {
   193→   193→                const data = await res.json().catch(() => ({}))
   194→   194→                toast.error(data.error || 'Could not download this track.')
   195→   195→                setCurrentTrack(null)
   196→   196→                setIsPlaying(false)
   197→   197→              }
   198→   198→            } catch {
   199→   199→              toast.error(
   200→   200→                'The server is downloading this track — try again in a moment.',
   201→   201→              )
   202→   202→            }
   203→   203→          }
   204→   204→        }
   205→   205→      }
   206→   206→    },
   207→   207→    [broadcastSync, setCurrentTrack, setIsPlaying, setPosition],
   208→   208→  )
   209→   209→
   210→   210→  // ─── Action: play next track ──────────────────────────────────────────
   211→   211→  const playNext = useCallback(async () => {
   212→   212→    const state = useMusicStore.getState()
   213→   213→    let nextTrack: Track | null = state.popNextFromQueue()
   214→   214→
   215→   215→    if (!nextTrack && state.autoplay && state.currentTrack) {
   216→   216→      try {
   217→   217→        const res = await fetch(
   218→   218→          `/api/music/related/${state.currentTrack.videoId}`,
   219→   219→        )
   220→   220→        if (res.ok) {
   221→   221→          const data = await res.json()
   222→   222→          if (data.tracks?.length > 0) {
   223→   223→            nextTrack = data.tracks[0] as Track
   224→   224→          }
   225→   225→        }
   226→   226→      } catch {
   227→   227→        // Autoplay failed — just stop
   228→   228→      }
   229→   229→    }
   230→   230→
   231→   231→    if (nextTrack) {
   232→   232→      await playTrack(nextTrack)
   233→   233→    } else if (state.repeat && state.currentTrack) {
   234→   234→      await playTrack(state.currentTrack)
   235→   235→    } else {
   236→   236→      setIsPlaying(false)
   237→   237→      setPosition(0)
   238→   238→    }
   239→   239→  }, [playTrack, setIsPlaying, setPosition])
   240→   240→
   241→   241→  // ─── Action: toggle play / pause ──────────────────────────────────────
   242→   242→  const togglePlay = useCallback(() => {
   243→   243→    const audio = audioRef.current
   244→   244→    const state = useMusicStore.getState()
   245→   245→    if (!audio || !state.currentTrack) return
   246→   246→
   247→   247→    if (state.isPlaying) {
   248→   248→      audio.pause()
   249→   249→      setIsPlaying(false)
   250→   250→      broadcastSync({
   251→   251→        state: 'paused',
   252→   252→        position: audio.currentTime,
   253→   253→        queue: state.queue,
   254→   254→      })
   255→   255→    } else {
   256→   256→      audio.play().catch(() => {})
   257→   257→      setIsPlaying(true)
   258→   258→      broadcastSync({
   259→   259→        state: 'playing',
   260→   260→        position: audio.currentTime,
   261→   261→        queue: state.queue,
   262→   262→      })
   263→   263→    }
   264→   264→  }, [broadcastSync, setIsPlaying])
   265→   265→
   266→   266→  // ─── Action: seek ─────────────────────────────────────────────────────
   267→   267→  const handleSeek = useCallback(
   268→   268→    (pos: number) => {
   269→   269→      const audio = audioRef.current
   270→   270→      const state = useMusicStore.getState()
   271→   271→      if (!audio) return
   272→   272→      audio.currentTime = pos
   273→   273→      setPosition(pos)
   274→   274→      broadcastSync({
   275→   275→        state: state.isPlaying ? 'playing' : 'paused',
   276→   276→        position: pos,
   277→   277→        queue: state.queue,
   278→   278→      })
   279→   279→    },
   280→   280→    [broadcastSync, setPosition],
   281→   281→  )
   282→   282→
   283→   283→  // ─── Action: stop playback entirely ───────────────────────────────────
   284→   284→  const stop = useCallback(() => {
   285→   285→    setCurrentTrack(null)
   286→   286→    setIsPlaying(false)
   287→   287→    setPosition(0)
   288→   288→    if (audioRef.current) {
   289→   289→      audioRef.current.pause()
   290→   290→      audioRef.current.src = ''
   291→   291→    }
   292→   292→    loadedVideoIdRef.current = null
   293→   293→  }, [setCurrentTrack, setIsPlaying, setPosition])
   294→   294→
   295→   295→  // ─── Action: add to queue (broadcasts) ────────────────────────────────
   296→   296→  const addToQueue = useCallback(
   297→   297→    (track: Track) => {
   298→   298→      const state = useMusicStore.getState()
   299→   299→      storeAddToQueue(track)
   300→   300→      broadcastSync({
   301→   301→        state: state.isPlaying ? 'playing' : 'paused',
   302→   302→        position: audioRef.current?.currentTime || 0,
   303→   303→        queue: useMusicStore.getState().queue,
   304→   304→      })
   305→   305→    },
   306→   306→    [broadcastSync, storeAddToQueue],
   307→   307→  )
   308→   308→
   309→   309→  // ─── Action: remove from queue (broadcasts) ───────────────────────────
   310→   310→  const removeFromQueue = useCallback(
   311→   311→    (index: number) => {
   312→   312→      const state = useMusicStore.getState()
   313→   313→      storeRemoveFromQueue(index)
   314→   314→      broadcastSync({
   315→   315→        state: state.isPlaying ? 'playing' : 'paused',
   316→   316→        position: audioRef.current?.currentTime || 0,
   317→   317→        queue: useMusicStore.getState().queue,
   318→   318→      })
   319→   319→    },
   320→   320→    [broadcastSync, storeRemoveFromQueue],
   321→   321→  )
   322→   322→
   323→   323→  // ─── Action: clear queue (broadcasts) ─────────────────────────────────
   324→   324→  const clearQueue = useCallback(() => {
   325→   325→    const state = useMusicStore.getState()
   326→   326→    storeClearQueue()
   327→   327→    broadcastSync({
   328→   328→      state: state.isPlaying ? 'playing' : 'paused',
   329→   329→      position: audioRef.current?.currentTime || 0,
   330→   330→      queue: [],
   331→   331→    })
   332→   332→  }, [broadcastSync, storeClearQueue])
   333→   333→
   334→   334→  // ─── Effect: keep <audio> volume in sync with the store ───────────────
   335→   335→  useEffect(() => {
   336→   336→    if (audioRef.current) audioRef.current.volume = volume
   337→   337→  }, [volume])
   338→   338→
   339→   339→  // ─── Effect: wire up <audio> event listeners ──────────────────────────
   340→   340→  // Re-attaches whenever playNext changes (which it does when its deps
   341→   341→  // change). The listeners read fresh state via `useMusicStore.getState()`
   342→   342→  // so they never see stale data.
   343→   343→  useEffect(() => {
   344→   344→    const audio = audioRef.current
   345→   345→    if (!audio) return
   346→   346→
   347→   347→    const onTimeUpdate = () => {
   348→   348→      // Don't fight with a freshly-applied remote position
   349→   349→      if (applyingRemoteRef.current) return
   350→   350→      setPosition(audio.currentTime)
   351→   351→    }
   352→   352→    const onEnded = () => {
   353→   353→      void playNext()
   354→   354→    }
   355→   355→    const onError = () => {
   356→   356→      const state = useMusicStore.getState()
   357→   357→      if (state.currentTrack) {
   358→   358→        toast.error('Could not play this track — skipping...')
   359→   359→        void playNext()
   360→   360→      }
   361→   361→    }
   362→   362→
   363→   363→    audio.addEventListener('timeupdate', onTimeUpdate)
   364→   364→    audio.addEventListener('ended', onEnded)
   365→   365→    audio.addEventListener('error', onError)
   366→   366→    return () => {
   367→   367→      audio.removeEventListener('timeupdate', onTimeUpdate)
   368→   368→      audio.removeEventListener('ended', onEnded)
   369→   369→      audio.removeEventListener('error', onError)
   370→   370→    }
   371→   371→  }, [playNext, setPosition])
   372→   372→
   373→   373→  // ─── Effect: Socket.io room sync ──────────────────────────────────────
   374→   374→  // Joins the music room's socket channel, requests the host's current
   375→   375→  // state, and listens for incoming sync events. Re-subscribes whenever
   376→   376→  // the active room changes.
   377→   377→  useEffect(() => {
   378→   378→    if (!socket || !activeRoomId) return
   379→   379→
   380→   380→    socket.emit('music:join', activeRoomId)
   381→   381→
   382→   382→    // Ask the host for the current state shortly after joining. The small
   383→   383→    // delay gives the join room propagation time to complete.
   384→   384→    const requestTimer = setTimeout(() => {
   385→   385→      socket.emit('music:request-sync', activeRoomId)
   386→   386→    }, 500)
   387→   387→
   388→   388→    // ── Incoming sync from the host (or another member's broadcast) ──
   389→   389→    const onSync = (data: {
   390→   390→      roomId: string
   391→   391→      state: string
   392→   392→      position: number
   393→   393→      videoId?: string
   394→   394→      trackInfo?: Track
   395→   395→      queue?: Track[]
   396→   396→      serverTimestamp: number
   397→   397→    }) => {
   398→   398→      if (data.roomId !== activeRoomId) return
   399→   399→
   400→   400→      applyingRemoteRef.current = true
   401→   401→
   402→   402→      const networkDelay = (Date.now() - data.serverTimestamp) / 1000
   403→   403→      const adjustedPosition =
   404→   404→        data.position + (data.state === 'playing' ? networkDelay : 0)
   405→   405→
   406→   406→      const state = useMusicStore.getState()
   407→   407→      const audio = audioRef.current
   408→   408→
   409→   409→      // Track changed → load the new stream and seek to host's position
   410→   410→      if (
   411→   411→        data.videoId &&
   412→   412→        (!state.currentTrack || state.currentTrack.videoId !== data.videoId)
   413→   413→      ) {
   414→   414→        const track: Track = data.trackInfo || {
   415→   415→          videoId: data.videoId,
   416→   416→          title: 'Now Playing',
   417→   417→          artist: '',
   418→   418→          thumbnail: null,
   419→   419→          durationSeconds: null,
   420→   420→        }
   421→   421→        pendingSeekRef.current = adjustedPosition
   422→   422→        setCurrentTrack(track)
   423→   423→        setIsPlaying(data.state === 'playing')
   424→   424→        setPosition(adjustedPosition)
   425→   425→
   426→   426→        if (audio) {
   427→   427→          audio.src = `/api/music/stream/${data.videoId}`
   428→   428→          audio.volume = state.volume
   429→   429→          loadedVideoIdRef.current = data.videoId
   430→   430→          // Wait a tick for the new src to be accepted before seeking
   431→   431→          setTimeout(() => {
   432→   432→            if (audioRef.current) {
   433→   433→              try {
   434→   434→                audioRef.current.currentTime = adjustedPosition
   435→   435→              } catch {
   436→   436→                // Stream not seekable yet — ignore
   437→   437→              }
   438→   438→              if (data.state === 'playing') {
   439→   439→                audioRef.current.play().catch(() => {})
   440→   440→              }
   441→   441→            }
   442→   442→            applyingRemoteRef.current = false
   443→   443→          }, 200)
   444→   444→        } else {
   445→   445→          applyingRemoteRef.current = false
   446→   446→        }
   447→   447→      } else if (audio && state.currentTrack) {
   448→   448→        // Same track — sync position + play/pause state only
   449→   449→        const drift = Math.abs(audio.currentTime - adjustedPosition)
   450→   450→        if (drift > 1.5) {
   451→   451→          try {
   452→   452→            audio.currentTime = adjustedPosition
   453→   453→          } catch {
   454→   454→            // ignore
   455→   455→          }
   456→   456→          setPosition(adjustedPosition)
   457→   457→        }
   458→   458→        if (data.state === 'playing' && !state.isPlaying) {
   459→   459→          audio.play().catch(() => {})
   460→   460→          setIsPlaying(true)
   461→   461→        } else if (data.state === 'paused' && state.isPlaying) {
   462→   462→          audio.pause()
   463→   463→          setIsPlaying(false)
   464→   464→        }
   465→   465→        applyingRemoteRef.current = false
   466→   466→      } else {
   467→   467→        applyingRemoteRef.current = false
   468→   468→      }
   469→   469→
   470→   470→      // Apply queue update from remote (broadcast by host)
   471→   471→      if (Array.isArray(data.queue)) {
   472→   472→        setQueue(data.queue)
   473→   473→      }
   474→   474→    }
   475→   475→
   476→   476→    // ── A member just joined and is requesting our current state ──
   477→   477→    // Only the host (or anyone with a current track) responds.
   478→   478→    const onRequestSync = (data: {
   479→   479→      roomId: string
   480→   480→      fromUserId: string
   481→   481→    }) => {
   482→   482→      if (data.roomId !== activeRoomId) return
   483→   483→      const state = useMusicStore.getState()
   484→   484→      if (!state.currentTrack || !socket) return
   485→   485→
   486→   486→      socket.emit('music:sync', {
   487→   487→        roomId: activeRoomId,
   488→   488→        state: state.isPlaying ? 'playing' : 'paused',
   489→   489→        position: audioRef.current?.currentTime || 0,
   490→   490→        videoId: state.currentTrack.videoId,
   491→   491→        trackInfo: {
