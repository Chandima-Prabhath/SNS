     3→     3→/**
     4→     4→ * Adoo Call Manager — Singleton
     5→     5→ *
     6→     6→ * This is a module-scoped singleton (NOT React state). One instance exists
     7→     7→ * for the entire app lifetime. This avoids the "multiple useVoiceCall instances"
     8→     8→ * bug where each component creates its own manager and they fight over control.
     9→     9→ *
    10→    10→ * Architecture:
    11→    11→ *   - CallManager class: owns the socket, RTCPeerConnections, local streams
    12→    12→ *   - callManager singleton: the one instance, created lazily
    13→    13→ *   - useCall hook: thin React wrapper that reads from the Zustand callStore
    14→    14→ *   - CallController: mounted once at app root, wires manager events to store
    15→    15→ *
    16→    16→ * Mute fix: we toggle sender.track.enabled on the SENDER track (the processed
    17→    17→ * one from AudioWorklet), not the source track. The source track stays enabled
    18→    18→ * so the worklet keeps running. We also gate the worklet output node directly.
    19→    19→ *
    20→    20→ * Call ending: full 6-step cleanup on both sides (stop tracks, close PCs,
    21→    21→ * remove listeners, notify server, detach media, reset state).
    22→    22→ */

    74→    74→// Audio unlock (autoplay policy)
    75→    75→let audioUnlocked = false
    76→    76→const pendingAudioElements: HTMLAudioElement[] = []
    77→    77→
    78→    78→export function unlockAudio() {
    79→    79→  if (audioUnlocked) return
    80→    80→  audioUnlocked = true
    81→    81→  for (const el of pendingAudioElements) {
    82→    82→    el.play().catch(() => {})
    83→    83→  }
    84→    84→  pendingAudioElements.length = 0
    85→    85→  try {
    86→    86→    const AudioContextClass = (window as any).AudioContext || (window as any).webkitAudioContext
    87→    87→    if (AudioContextClass) {
    88→    88→      const ctx = new AudioContextClass()
    89→    89→      if (ctx.state === 'suspended') ctx.resume()
    90→    90→      const buffer = ctx.createBuffer(1, 1, 22050)
    91→    91→      const source = ctx.createBufferSource()
    92→    92→      source.buffer = buffer
    93→    93→      source.connect(ctx.destination)
    94→    94→      source.start(0)
    95→    95→    }
    96→    96→  } catch {}
    97→    97→}

   140→   140→  private registerGlobalListeners() {
   141→   141→    if (!this.socket || this.globalListenersRegistered) return
   142→   142→    this.globalListenersRegistered = true
   143→   143→
   144→   144→    this.socket.on('call:incoming', (payload: any) => {
   145→   145→      console.log('[call] incoming call from', payload.from?.displayName)
   146→   146→      this.callbacks?.onIncomingCall(payload)
   147→   147→    })
   148→   148→
   149→   149→    this.socket.on('call:cancel', (payload: any) => {
   150→   150→      window.dispatchEvent(new CustomEvent('sns:call-cancelled', { detail: payload }))
   151→   151→    })
   152→   152→
   153→   153→    this.socket.on('call:reject', (payload: any) => {
   154→   154→      console.log('[call] call rejected by remote')
   155→   155→      // Stop ringback sound immediately
   156→   156→      import('./call-sounds').then(m => m.CallSounds.stop()).catch(() => {})
   157→   157→      // If we're the caller and the call was rejected, end our side
   158→   158→      if (this.callId && this.status !== 'idle') {
   159→   159→        this.endCallCleanup()
   160→   160→        this.callbacks?.onCallEnded('rejected')
   161→   161→      }
   162→   162→      window.dispatchEvent(new CustomEvent('sns:call-rejected', { detail: payload }))
   163→   163→    })
   164→   164→
   165→   165→    this.socket.on('call:accept', (payload: any) => {
   166→   166→      console.log('[call] call accepted by remote')
   167→   167→      // Stop ringback sound — the call is connecting now
   168→   168→      import('./call-sounds').then(m => m.CallSounds.stop()).catch(() => {})
   169→   169→      window.dispatchEvent(new CustomEvent('sns:call-accepted', { detail: payload }))
   170→   170→    })

   173→   173→      console.log('[call] call ended by server:', payload.reason)
   174→   174→      this.endCallCleanup()

   176→   176→      import('./call-sounds').then(m => m.CallSounds.playEnded()).catch(() => {})
   177→   177→      this.callbacks?.onCallEnded(payload.reason)
   178→   178→    })
   179→   179→  }

   204→   204→  /**
   205→   205→   * Start a call — gets mic (and optionally camera), sets up WebRTC.
   206→   206→   */
   207→   207→  async startCall(params: {
   208→   208→    callId: string
   209→   209→    channelId?: string | null
   210→   210→    dmGroupId?: string | null
   211→   211→    enableVideo?: boolean
   212→   212→  }): Promise<void> {
   213→   213→    if (!this.socket) throw new Error('Socket not connected')
   214→   214→    if (this.iceServers.length === 0) throw new Error('ICE servers not loaded')
   215→   215→
   216→   216→    this.callId = params.callId
   217→   217→    this.isVideoCall = params.enableVideo ?? false
   218→   218→    this.setStatus('connecting')
   219→   219→
   220→   220→    // Acquire wake lock to prevent screen sleep
   221→   221→    try {
   222→   222→      this.wakeLock = await (navigator as any).wakeLock?.request('screen')
   223→   223→    } catch {}
   224→   224→
   225→   225→    // Get audio + video in a SINGLE getUserMedia call
   226→   226→    // Use simple constraints for maximum cross-browser compatibility.
   227→   227→    // Firefox Android is picky about facingMode + resolution combinations.
   228→   228→    const constraints: MediaStreamConstraints = {
   229→   229→      audio: {
   230→   230→        echoCancellation: true,
   231→   231→        autoGainControl: true,
   232→   232→        channelCount: 1,
   233→   233→        noiseSuppression: !this.enableRnnoise,
   234→   234→      } as MediaTrackConstraints,
   235→   235→      video: this.isVideoCall
   236→   236→        ? { width: { ideal: 640 }, height: { ideal: 480 }, frameRate: { ideal: 20 } }
   237→   237→        : false,
   238→   238→    }
   239→   239→
   240→   240→    this.localStream = await navigator.mediaDevices.getUserMedia(constraints)
   241→   241→
   242→   242→    // Store the source audio track (before RNNoise processing)
   243→   243→    this.sourceAudioTrack = this.localStream.getAudioTracks()[0]
   244→   244→
   245→   245→    // Apply RNNoise if enabled
   246→   246→    if (this.enableRnnoise) {
   247→   247→      await this.applyRnnoise()
   248→   248→    }
   249→   249→
   250→   250→    this.processedAudioTrack = this.localStream.getAudioTracks()[0]
   251→   251→
   252→   252→    console.log('[call] local stream ready, video:', this.isVideoCall, 'rnnoise:', !!this.rnnoiseNode)
   253→   253→    this.callbacks?.onLocalStream(this.localStream)
   254→   254→
   255→   255→    // CRITICAL: Register signaling listeners BEFORE joining the call room.
   256→   256→    // The server immediately sends 'call:peers' when we join, so we need
   257→   257→    // the listeners in place BEFORE the emit. Otherwise the event arrives
   258→   258→    // with no handler and the call never connects (stuck on 'connecting').
   259→   259→    this.setupCallSignaling()
   260→   260→
   261→   261→    // Join the call room
   262→   262→    this.socket.emit('call:join', this.callId)
   263→   263→    console.log('[call] joined call room', this.callId)
   264→   264→  }
   265→   265→
   266→   266→  /**
   267→   267→   * Apply RNNoise neural noise suppression.
   268→   268→   * Creates: source → rnnoise worklet → gainNode → destination
   269→   269→   * The gainNode lets us mute without affecting the worklet processing.
   270→   270→   */
   271→   271→  private async applyRnnoise() {
   272→   272→    try {
   273→   273→      const AudioContextClass = (window as any).AudioContext || (window as any).webkitAudioContext
   274→   274→      if (!AudioContextClass) return
   275→   275→
   276→   276→      this.audioContext = new AudioContextClass({ sampleRate: 48000 })
   277→   277→
   278→   278→      // Load WASM
   279→   279→      const wasmResponse = await fetch('/rnnoise.wasm')
   280→   280→      const wasmBuffer = await wasmResponse.arrayBuffer()
   281→   281→      const wasmModule = await WebAssembly.compile(wasmBuffer)
   282→   282→
   283→   283→      await this.audioContext.audioWorklet.addModule('/rnnoise.worklet.js')
   284→   284→      console.log('[call] RNNoise worklet loaded')
   285→   285→
   286→   286→      const source = this.audioContext.createMediaStreamSource(
   287→   287→        new MediaStream([this.sourceAudioTrack!])
   288→   288→      )
   289→   289→      this.rnnoiseNode = new (window as any).AudioWorkletNode(this.audioContext, 'rnnoise', {
   290→   290→        channelCountMode: 'explicit',
   291→   291→        channelCount: 1,
   292→   292→        channelInterpretation: 'speakers',
   293→   293→        numberOfInputs: 1,
   294→   294→        numberOfOutputs: 1,
   295→   295→        outputChannelCount: [1],
   296→   296→        processorOptions: { module: wasmModule },
   297→   297→      })
   298→   298→      // Gain node for muting — set to 0 to mute, 1 to unmute
   299→   299→      this.gainNode = this.audioContext.createGain()
   300→   300→      this.gainNode.gain.value = 1
   301→   301→
   302→   302→      const destination = this.audioContext.createMediaStreamDestination()
   303→   303→
   304→   304→      source.connect(this.rnnoiseNode)
   305→   305→      this.rnnoiseNode.connect(this.gainNode)
   306→   306→      this.gainNode.connect(destination)
   307→   307→
   308→   308→      // Replace the audio track with the processed one
   309→   309→      const processedTrack = destination.stream.getAudioTracks()[0]
   310→   310→      this.localStream!.removeTrack(this.sourceAudioTrack!)
   311→   311→      this.localStream!.addTrack(processedTrack)
   312→   312→
   313→   313→      console.log('[call] RNNoise applied successfully')
   314→   314→    } catch (e) {
   315→   315→      console.warn('[call] RNNoise failed, using browser noiseSuppression:', e)
   316→   316→    }
   317→   317→  }
   318→   318→
   319→   319→  /**
   320→   320→   * Toggle mute — works correctly even with AudioWorklet.
   321→   321→   * We toggle BOTH the sender track AND the gain node.

   323→   323→  setMuted(muted: boolean) {

   332→   332→    if (this.gainNode) {
   333→   333→      this.gainNode.gain.value = muted ? 0 : 1
   334→   334→    }
   335→   335→
   336→   336→    console.log('[call] mute set to', muted)
   337→   337→    this.callbacks?.onMuteChange(muted)
   338→   338→  }

   361→   361→   * Switch camera (front/back).

   363→   363→  async switchCamera(): Promise<boolean> {
   364→   364→    try {
   365→   365→      const videoTrack = this.localStream?.getVideoTracks()[0]
   366→   366→      if (!videoTrack) return false
   367→   367→
   368→   368→      const settings = videoTrack.getSettings()
   369→   369→      const currentFacing = settings.facingMode || 'user'
   370→   370→      const newFacing = currentFacing === 'user' ? 'environment' : 'user'
   371→   371→
   372→   372→      // Use facingMode without 'exact' for better cross-browser support
   373→   373→      const newStream = await navigator.mediaDevices.getUserMedia({
   374→   374→        video: { width: { ideal: 640 }, height: { ideal: 480 }, frameRate: { ideal: 20 }, facingMode: newFacing },
   375→   375→        audio: false,
   376→   376→      })
   377→   377→
   378→   378→      const newTrack = newStream.getVideoTracks()[0]
   379→   379→      for (const peer of this.peers.values()) {
   380→   380→        if (peer.videoSender) {
   381→   381→          await peer.videoSender.replaceTrack(newTrack)
   382→   382→        }
   383→   383→      }
   384→   384→
   385→   385→      this.localStream!.removeTrack(videoTrack)
   386→   386→      this.localStream!.addTrack(newTrack)
   387→   387→      videoTrack.stop()
   388→   388→
   389→   389→      console.log('[call] camera switched to', newFacing)
   390→   390→      return true
   391→   391→    } catch (e) {
   392→   392→      console.error('[call] camera switch failed:', e)
   393→   393→      return false
   394→   394→    }
   395→   395→  }

   410→   410→      this.screenStream = await navigator.mediaDevices.getDisplayMedia({

   423→   423→      }

   425→   425→      // Update the local stream
   426→   426→      this.localStream!.removeTrack(videoTrack)

   438→   438→    } catch (e) {

   441→   441→    }

   444→   444→  /**
   445→   445→   * Stop screen sharing — restores the original camera track.
   446→   446→   */
   447→   447→  async stopScreenShare(): Promise<void> {
   448→   448→    if (!this.isSharingScreen || !this.originalVideoTrack) return
   449→   449→
   450→   450→    const screenTrack = this.screenStream?.getVideoTracks()[0]
   451→   451→    if (screenTrack) {
   452→   452→      screenTrack.stop()
   453→   453→    }
   454→   454→
   455→   455→    // Restore the original camera track
   456→   456→    for (const peer of this.peers.values()) {
   457→   457→      if (peer.videoSender) {
   458→   458→        await peer.videoSender.replaceTrack(this.originalVideoTrack)
   459→   459→      }
   460→   460→    }
   461→   461→
   462→   462→    // Update the local stream
   463→   463→    const currentTrack = this.localStream?.getVideoTracks()[0]
   464→   464→    if (currentTrack) {
   465→   465→      this.localStream!.removeTrack(currentTrack)
   466→   466→    }
   467→   467→    this.localStream!.addTrack(this.originalVideoTrack)
   468→   468→
   469→   469→    this.screenStream = null
   470→   470→    this.isSharingScreen = false
   471→   471→    this.originalVideoTrack = null
   472→   472→    this.callbacks?.onVideoToggle(true) // trigger UI update
   473→   473→    console.log('[call] screen share stopped')
   474→   474→  }

   478→   478→  /**
   479→   479→   * End the call — full cleanup.
   480→   480→   * Called when the user clicks End OR when the server says the call ended.
   481→   481→   */
   482→   482→  async endCall(): Promise<void> {
   483→   483→    if (!this.callId) return
   484→   484→    console.log('[call] ending call', this.callId)
   485→   485→
   486→   486→    // Notify server
   487→   487→    if (this.socket) {
   488→   488→      this.socket.emit('call:leave', this.callId)
   489→   489→    }
   490→   490→
   491→   491→    // Server-side cleanup (mark call as ended)
   492→   492→    if (this.callId) {
   493→   493→      try {
   494→   494→        await fetch(`/api/calls/${this.callId}`, { method: 'DELETE' })
   495→   495→      } catch {}
   496→   496→    }
   497→   497→
   498→   498→    this.endCallCleanup()
   499→   499→    // Play ended sound
   500→   500→    import('./call-sounds').then(m => m.CallSounds.playEnded()).catch(() => {})
   501→   501→    this.callbacks?.onCallEnded('user_ended')
   502→   502→  }
   503→   503→
   504→   504→  /**
   505→   505→   * Full cleanup — stop tracks, close PCs, remove listeners, detach media.
   506→   506→   */
   507→   507→  private endCallCleanup() {
   508→   508→    console.log('[call] running cleanup')
   509→   509→
   510→   510→    // Stop any playing call sounds
   511→   511→    import('./call-sounds').then(m => m.CallSounds.stop()).catch(() => {})
   512→   512→
   513→   513→    // Stop screen share if active
   514→   514→    if (this.screenStream) {
   515→   515→      this.screenStream.getTracks().forEach((t) => t.stop())
   516→   516→      this.screenStream = null
   517→   517→      this.isSharingScreen = false
   518→   518→      this.originalVideoTrack = null
   519→   519→    }
   520→   520→
   521→   521→    // Close all peer connections
   522→   522→    for (const [peerId, peer] of this.peers) {
   523→   523→      try {
   524→   524→        peer.pc.close()
   525→   525→      } catch {}
   526→   526→      if (peer.audioLevelChecker) cancelAnimationFrame(peer.audioLevelChecker)
   527→   527→      if (peer.audioEl) {
   528→   528→        peer.audioEl.srcObject = null
   529→   529→        peer.audioEl.remove()
   530→   530→      }
   531→   531→      if (peer.failedTimer) clearTimeout(peer.failedTimer)
   532→   532→    }
   533→   533→    this.peers.clear()
   534→   534→
   535→   535→    // Stop local tracks
   536→   536→    if (this.localStream) {
   537→   537→      this.localStream.getTracks().forEach((t) => t.stop())
   538→   538→      this.localStream = null
   539→   539→    }
   540→   540→    if (this.sourceAudioTrack) {
   541→   541→      this.sourceAudioTrack.stop()
   542→   542→      this.sourceAudioTrack = null
   543→   543→    }
   544→   544→    this.processedAudioTrack = null
   545→   545→
   546→   546→    // Close audio context
   547→   547→    if (this.audioContext) {
   548→   548→      try { this.audioContext.close() } catch {}
   549→   549→      this.audioContext = null
   550→   550→      this.rnnoiseNode = null
   551→   551→      this.gainNode = null
   552→   552→    }
   553→   553→
   554→   554→    // Release wake lock
   555→   555→    if (this.wakeLock) {
   556→   556→      try { this.wakeLock.release() } catch {}
   557→   557→      this.wakeLock = null
   558→   558→    }
   559→   559→
   560→   560→    // Remove socket listeners for this call
   561→   561→    if (this.socket) {
   562→   562→      this.socket.removeAllListeners('call:peer-joined')
   563→   563→      this.socket.removeAllListeners('call:peers')
   564→   564→      this.socket.removeAllListeners('call:offer')
   565→   565→      this.socket.removeAllListeners('call:answer')
   566→   566→      this.socket.removeAllListeners('call:ice-candidate')
   567→   567→      this.socket.removeAllListeners('call:peer-left')
   568→   568→    }
   569→   569→    this.signalingSetup = false // allow setupCallSignaling() on next call
   570→   570→
   571→   571→    // Reset state
   572→   572→    this.callId = null
   573→   573→    this.userMuted = false
   574→   574→    this.videoEnabled = true
   575→   575→    this.isVideoCall = false
   576→   576→    this.setStatus('idle')
   577→   577→
   578→   578→    // Notify participants cleared
   579→   579→    this.callbacks?.onLocalStream(null)
   580→   580→    this.callbacks?.onParticipantsChange([])
   581→   581→  }

   583→   583→  // ─── WebRTC signaling ──────────────────────────────────────────────────
   584→   584→
   585→   585→  private signalingSetup = false
   586→   586→
   587→   587→  private setupCallSignaling() {
   588→   588→    if (!this.socket || this.signalingSetup) return
   589→   589→    this.signalingSetup = true
   590→   590→    console.log('[call] setting up signaling listeners')
   591→   591→
   592→   592→    this.socket.on('call:peer-joined', async (payload: { peerId: string; userId: string; username: string }) => {
   593→   593→      await this.ensurePeer(payload.peerId, payload.userId, payload.username)
   594→   594→    })
   595→   595→
   596→   596→    this.socket.on('call:peers', async (payload: { peers: Array<{ peerId: string; userId: string; username: string }> }) => {
   597→   597→      for (const peer of payload.peers) {
   598→   598→        await this.ensurePeer(peer.peerId, peer.userId, peer.username)
   599→   599→      }
   600→   600→    })
   601→   601→
   602→   602→    this.socket.on('call:offer', async (payload: { from: string; sdp: any }) => {
   603→   603→      const peer = this.peers.get(payload.from)
   604→   604→      if (!peer) return
   605→   605→      const offerCollision = peer.makingOffer || peer.pc.signalingState !== 'stable'
   606→   606→      peer.ignoreOffer = !peer.isPolite && offerCollision
   607→   607→      if (peer.ignoreOffer) return
   608→   608→      try {
   609→   609→        await peer.pc.setRemoteDescription(new RTCSessionDescription(payload.sdp))
   610→   610→        await peer.pc.setLocalDescription()
   611→   611→        this.socket!.emit('call:answer', { to: payload.from, sdp: peer.pc.localDescription })
   612→   612→      } catch (e) {
   613→   613→        console.error('[call] error handling offer:', e)
   614→   614→      }
   615→   615→    })
   616→   616→
   617→   617→    this.socket.on('call:answer', async (payload: { from: string; sdp: any }) => {
   618→   618→      const peer = this.peers.get(payload.from)
   619→   619→      if (!peer) return
   620→   620→      try {
   621→   621→        await peer.pc.setRemoteDescription(new RTCSessionDescription(payload.sdp))
   622→   622→      } catch (e) {
   623→   623→        console.warn('[call] setRemoteDescription(answer) failed:', (e as Error).message)
   624→   624→      }
   625→   625→    })
   626→   626→
   627→   627→    this.socket.on('call:ice-candidate', async (payload: { from: string; candidate: any }) => {
   628→   628→      const peer = this.peers.get(payload.from)
   629→   629→      if (!peer) return
   630→   630→      try {
   631→   631→        await peer.pc.addIceCandidate(new RTCIceCandidate(payload.candidate))
   632→   632→      } catch (e: any) {
   633→   633→        if (!/ufrag/i.test(e.message)) {
   634→   634→          console.warn('[call] addIceCandidate error:', e.message)
   635→   635→        }
   636→   636→      }
   637→   637→    })
   638→   638→
   639→   639→    this.socket.on('call:peer-left', (payload: { peerId: string; userId: string }) => {
   640→   640→      const peer = this.peers.get(payload.peerId)
   641→   641→      if (peer) {
   642→   642→        peer.pc.close()
   643→   643→        if (peer.audioLevelChecker) cancelAnimationFrame(peer.audioLevelChecker)
   644→   644→        if (peer.audioEl) {
   645→   645→          peer.audioEl.srcObject = null
   646→   646→          peer.audioEl.remove()
   647→   647→        }
   648→   648→        this.peers.delete(payload.peerId)
   649→   649→        this.callbacks?.onParticipantsChange(this.getParticipants())
   650→   650→      }
   651→   651→    })
   652→   652→  }
   653→   653→
   654→   654→  private async ensurePeer(peerId: string, userId: string, username: string) {
   655→   655→    if (this.peers.has(peerId)) return
   656→   656→
   657→   657→    const myId = this.socket!.id || ''
   658→   658→    const isInitiator = myId < peerId
   659→   659→    const isPolite = myId > peerId
   660→   660→
   661→   661→    const pc = new RTCPeerConnection({
   662→   662→      iceServers: this.iceServers,
   663→   663→      iceTransportPolicy: 'all',
   664→   664→    })
   665→   665→
   666→   666→    const peer: PeerEntry = {
   667→   667→      pc,
   668→   668→      userId,
   669→   669→      username,
   670→   670→      makingOffer: false,
   671→   671→      ignoreOffer: false,
   672→   672→      isInitiator,
   673→   673→      isPolite,
   674→   674→      remoteStream: new MediaStream(),
   675→   675→    }
   676→   676→    this.peers.set(peerId, peer)
   677→   677→
   678→   678→    // Add local tracks
   679→   679→    if (this.localStream) {
   680→   680→      for (const track of this.localStream.getTracks()) {
   681→   681→        const sender = pc.addTrack(track, this.localStream)
   682→   682→        if (track.kind === 'audio') {
   683→   683→          peer.audioSender = sender
   684→   684→          this.capSenderBitrate(sender, 32_000)
   685→   685→        }
   686→   686→        if (track.kind === 'video') {
   687→   687→          peer.videoSender = sender
   688→   688→          this.capSenderBitrate(sender, 1_000_000)
   689→   689→        }
   690→   690→      }
   691→   691→      // Prefer VP8 video codec — Firefox Android doesn't support H.264 reliably.
   692→   692→      // Must be called AFTER tracks are added (transceivers are created by addTrack).
   693→   693→      this.preferVp8(pc)
   694→   694→    }
   695→   695→
   696→   696→    pc.ontrack = (event) => {
   697→   697→      console.log('[call] ontrack, kind:', event.track.kind)
   698→   698→      peer.remoteStream.addTrack(event.track)
   699→   699→
   700→   700→      if (event.track.kind === 'audio') {
   701→   701→        if (!peer.audioEl) {
   702→   702→          const el = document.createElement('audio')
   703→   703→          el.autoplay = true
   704→   704→          el.setAttribute('playsinline', '')
   705→   705→          el.style.display = 'none'
   706→   706→          document.body.appendChild(el)
   707→   707→          peer.audioEl = el
   708→   708→        }
   709→   709→        peer.audioEl.srcObject = peer.remoteStream
   710→   710→        peer.audioEl.play().catch((e) => {
   711→   711→          if (!pendingAudioElements.includes(peer.audioEl!)) {
   712→   712→            pendingAudioElements.push(peer.audioEl!)
   713→   713→          }
   714→   714→        })
   715→   715→        this.startRemoteAudioMonitoring(peerId, peer.remoteStream)
   716→   716→      }
   717→   717→
   718→   718→      this.callbacks?.onParticipantsChange(this.getParticipants())
   719→   719→    }
   720→   720→
   721→   721→    pc.onnegotiationneeded = async () => {
   722→   722→      if (!peer.isInitiator) return
   723→   723→      try {
   724→   724→        peer.makingOffer = true
   725→   725→        await pc.setLocalDescription()
   726→   726→        this.socket!.emit('call:offer', { to: peerId, sdp: pc.localDescription })
   727→   727→      } catch (e) {
   728→   728→        console.error('[call] negotiationneeded error:', e)
   729→   729→      } finally {
   730→   730→        peer.makingOffer = false
   731→   731→      }
   732→   732→    }
   733→   733→
   734→   734→    pc.onicecandidate = (event) => {
   735→   735→      if (event.candidate) {
   736→   736→        this.socket!.emit('call:ice-candidate', { to: peerId, candidate: event.candidate })
   737→   737→      }
   738→   738→    }
   739→   739→
   740→   740→    pc.oniceconnectionstatechange = () => {
   741→   741→      const state = pc.iceConnectionState
   742→   742→      console.log(`[call] ICE state (${username}): ${state}`)
   743→   743→      if (state === 'connected' || state === 'completed') {
   744→   744→        if (this.status !== 'connected') {
   745→   745→          this.setStatus('connected')
   746→   746→          // Play connected sound
   747→   747→          import('./call-sounds').then(m => { m.CallSounds.stop(); m.CallSounds.playConnected() }).catch(() => {})
   748→   748→        }
   749→   749→        if (peer.failedTimer) { clearTimeout(peer.failedTimer); peer.failedTimer = undefined }
   750→   750→        setTimeout(() => this.checkConnectionType(peerId), 2000)
   751→   751→      } else if (state === 'failed') {
   752→   752→        if (!peer.failedTimer) {
   753→   753→          peer.failedTimer = window.setTimeout(() => {
   754→   754→            pc.restartIce()
   755→   755→            peer.failedTimer = undefined
   756→   756→          }, 2000)
   757→   757→        }
   758→   758→      }
   759→   759→    }
   760→   760→  }
   761→   761→
   762→   762→  /**
   763→   763→   * Prefer VP8 video codec for cross-browser compatibility.
   764→   764→   * Firefox Android doesn't support H.264 reliably — VP8 is the only
   765→   765→   * universally supported WebRTC video codec (RFC 7742).
   766→   766→   *
   767→   767→   * IMPORTANT: We must keep RTX, RED, and ULPFEC in the preference list.
   768→   768→   * Stripping them out causes "Failed to parse codecs correctly" errors
   769→   769→   * when setRemoteDescription receives the answer — the SDP parser rejects
   770→   770→   * an offer/answer that references codecs not in the local preference list.
   771→   771→   */
   772→   772→  private preferVp8(pc: RTCPeerConnection) {
   773→   773→    try {
   774→   774→      const transceivers = pc.getTransceivers()
   775→   775→      for (const t of transceivers) {
   776→   776→        if (t.receiver && 'track' in t.receiver && t.receiver.track?.kind === 'video') {
   777→   777→          const codecs = RTCRtpReceiver.getCapabilities('video')?.codecs || []
   778→   778→          if (codecs.length === 0) continue
   779→   779→          const vp8 = codecs.filter((c) => c.mimeType === 'video/VP8')
   780→   780→          if (vp8.length === 0) continue
   781→   781→          // Keep everything else (including RTX, RED, ULPFEC) — just put VP8 first.
   782→   782→          const rest = codecs.filter((c) => c.mimeType !== 'video/VP8')
   783→   783→          t.setCodecPreferences([...vp8, ...rest])
   784→   784→        }
   785→   785→      }
   786→   786→    } catch {
   787→   787→      // setCodecPreferences not supported on older browsers — ignore
   788→   788→    }
   789→   789→  }
   790→   790→
   791→   791→  private async capSenderBitrate(sender: RTCRtpSender, maxBitrate: number) {
   792→   792→    try {
   793→   793→      const params = sender.getParameters()
   794→   794→      if (!params.encodings || params.encodings.length === 0) {
   795→   795→        params.encodings = [{}]
   796→   796→      }
   797→   797→      params.encodings[0].maxBitrate = maxBitrate
   798→   798→      await sender.setParameters(params)
   799→   799→    } catch {}
   800→   800→  }
   801→   801→
   802→   802→  private async checkConnectionType(peerId: string) {
   803→   803→    const peer = this.peers.get(peerId)
   804→   804→    if (!peer) return
   805→   805→    try {
   806→   806→      const stats = await peer.pc.getStats()
   807→   807→      let foundType: 'p2p' | 'turn' | 'unknown' = 'unknown'
   808→   808→      stats.forEach((report) => {
   809→   809→        if (report.type === 'candidate-pair') {
   810→   810→          const cp = report as any
   811→   811→          if (cp.nominated || cp.state === 'succeeded') {
   812→   812→            const local = stats.get(cp.localCandidateId) as any
   813→   813→            if (local) {
   814→   814→              foundType = local.candidateType === 'relay' ? 'turn' : 'p2p'
   815→   815→            }
   816→   816→          }
   817→   817→        }
   818→   818→      })
   819→   819→      if (foundType !== peer.lastStatsType) {
   820→   820→        peer.lastStatsType = foundType
   821→   821→        this.callbacks?.onConnectionType(peerId, foundType)
   822→   822→      }
   823→   823→    } catch {}
   824→   824→  }
   825→   825→
   826→   826→  private startRemoteAudioMonitoring(peerId: string, stream: MediaStream) {
   827→   827→    try {
   828→   828→      const AudioContextClass = (window as any).AudioContext || (window as any).webkitAudioContext
   829→   829→      if (!AudioContextClass) return
   830→   830→      const peer = this.peers.get(peerId)
   831→   831→      if (!peer) return
   832→   832→      const ctx = new AudioContextClass()
   833→   833→      const source = ctx.createMediaStreamSource(stream)
   834→   834→      const analyser = ctx.createAnalyser()
   835→   835→      analyser.fftSize = 256
   836→   836→      source.connect(analyser)
   837→   837→      const data = new Uint8Array(analyser.frequencyBinCount)
   838→   838→      const check = () => {
   839→   839→        if (!this.peers.has(peerId)) { ctx.close(); return }
   840→   840→        analyser.getByteFrequencyData(data)
   841→   841→        const avg = data.reduce((s, v) => s + v, 0) / data.length / 255
   842→   842→        this.callbacks?.onAudioLevel(peerId, avg)
   843→   843→        peer.audioLevelChecker = requestAnimationFrame(check) as unknown as number
   844→   844→      }
   845→   845→      check()
   846→   846→    } catch {}
   847→   847→  }
   848→   848→}
