   126→function ProfileSection() {
   127→  const { data: session } = useSession()
   128→  const qc = useQueryClient()
   129→
   130→  const { data: me, isLoading } = useQuery({
   131→    queryKey: ['me'],
   132→    queryFn: async () => {
   133→      const res = await fetch('/api/auth/me')
   134→      const data = await res.json()
   135→      return data.user
   136→    },
   137→  })
   138→
   139→  const [displayName, setDisplayName] = useState('')
   140→  const [bio, setBio] = useState('')
   141→  const [avatarUrl, setAvatarUrl] = useState('')
   142→  const [customStatus, setCustomStatus] = useState('')
   143→  const [uploading, setUploading] = useState(false)
   144→
   145→  // Sync from server once
   146→  useState(() => {
   147→    if (me) {
   148→      setDisplayName(me.displayName || '')
   149→      setBio(me.bio || '')
   150→      setAvatarUrl(me.avatarUrl || '')
   151→      setCustomStatus(me.customStatus || '')
   152→    }
   153→  })
   154→
   155→  // Also sync on data change
   156→  if (me && displayName === '' && me.displayName) {
   157→    setDisplayName(me.displayName)
   158→    setBio(me.bio || '')
   159→    setAvatarUrl(me.avatarUrl || '')
   160→    setCustomStatus(me.customStatus || '')
   161→  }
   162→
   163→  const update = useMutation({
   164→    mutationFn: async (data: any) => {
   165→      const res = await fetch(`/api/users/me`, {
   166→        method: 'PATCH',
   167→        headers: { 'Content-Type': 'application/json' },
   168→        body: JSON.stringify(data),
   169→      })
   170→      if (!res.ok) throw new Error('failed')
   171→      return res.json()
   172→    },
   173→    onSuccess: () => {
   174→      qc.invalidateQueries({ queryKey: ['me'] })
   175→      toast.success('Saved')
   176→    },
   177→    onError: () => toast.error('Failed to save'),
   178→  })
   179→
   180→  const handleSave = () => {
   181→    update.mutate({ displayName, bio, avatarUrl, customStatus })
   182→  }
   183→
   184→  const handleAvatar = async (e: React.ChangeEvent<HTMLInputElement>) => {
   185→    const file = e.target.files?.[0]
   186→    if (!file) return
   187→    setUploading(true)
   188→    try {
   189→      const formData = new FormData()
   190→      formData.append('file', file)
   191→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   192→      if (!res.ok) {
   193→        const text = await res.text()
   194→        throw new Error(text.slice(0, 120) || 'Upload failed')
   195→      }
   196→      const data = await res.json()
   197→      setAvatarUrl(data.url)
   198→      toast.success('Avatar uploaded — click Save to apply')
   199→    } catch (e: any) {
   200→      toast.error(e.message || 'Upload failed')
   201→    } finally {
   202→      setUploading(false)
   203→      e.target.value = ''
   204→    }
   205→  }
   206→
   207→  if (isLoading) return <div className="flex justify-center py-12"><Loader2 className="w-6 h-6 animate-spin" /></div>
   208→
   209→  return (
   210→    <Card className="p-6 space-y-5">
   211→      {/* Avatar picker — generated avatars + upload option */}
   212→      <AvatarPicker
   213→        currentAvatarUrl={avatarUrl}
   214→        userId={me?.id || 'seed'}
   215→        onPick={(url) => { setAvatarUrl(url); toast.success('Avatar selected — click Save to apply') }}
   216→        onUpload={handleAvatar}
   217→        uploading={uploading}
   218→      />
   219→
   220→      <div className="space-y-2">
   221→        <Label htmlFor="displayName">Display name</Label>
   222→        <Input id="displayName" value={displayName} onChange={(e) => setDisplayName(e.target.value)} className="h-11" />
   223→      </div>
   224→
   225→      <div className="space-y-2">
   226→        <Label htmlFor="username">Username</Label>
   227→        <Input id="username" value={me?.username || ''} disabled className="h-11" />
   228→        <p className="text-xs text-muted-foreground">Username cannot be changed.</p>
   229→      </div>
   230→
   231→      <div className="space-y-2">
   232→        <Label htmlFor="bio">Bio</Label>
   233→        <Textarea
   234→          id="bio"
   235→          value={bio}
   236→          onChange={(e) => setBio(e.target.value)}
   237→          placeholder="Tell your friends a bit about yourself."
   238→          rows={3}
   239→        />
   240→      </div>
   241→
   242→      <div className="space-y-2">
   243→        <Label htmlFor="customStatus">Custom status</Label>
   244→        <Input
   245→          id="customStatus"
   246→          value={customStatus}
   247→          onChange={(e) => setCustomStatus(e.target.value)}
   248→          placeholder="🎧 Listening to lofi"
   249→          maxLength={80}
   250→          className="h-11"
   251→        />
   252→      </div>
   253→
   254→      <Button onClick={handleSave} disabled={update.isPending} className="h-11">
   255→        {update.isPending ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Save className="w-4 h-4 mr-2" />}
   256→        Save changes
   257→      </Button>
   258→    </Card>
   259→  )
   260→}

   274→
   275→  const [lastSeenVisible, setLastSeenVisible] = useState(true)

   277→  const [typingIndicators, setTypingIndicators] = useState(true)
   278→
   279→  if (me && lastSeenVisible && !me.lastSeenVisible === false) {
   280→    // initial sync — only run once
   281→  }
   282→  useState(() => {
   283→    if (me) {
   284→      setLastSeenVisible(me.lastSeenVisible ?? true)
   285→      setReadReceipts(me.readReceiptsEnabled ?? true)
   286→      setTypingIndicators(me.typingIndicatorsEnabled ?? true)
   287→    }
   288→  })
   289→
   290→  const update = useMutation({
   291→    mutationFn: async (data: any) => {
   292→      const res = await fetch(`/api/users/me`, {
   293→        method: 'PATCH',
   294→        headers: { 'Content-Type': 'application/json' },
   295→        body: JSON.stringify(data),
   296→      })
   297→      if (!res.ok) throw new Error('failed')
   298→      return res.json()
   299→    },
   300→    onSuccess: () => {
   301→      qc.invalidateQueries({ queryKey: ['me'] })
   302→      toast.success('Saved')
   303→    },
   304→  })
   305→
   306→  const handleToggle = (key: 'lastSeenVisible' | 'readReceiptsEnabled' | 'typingIndicatorsEnabled', value: boolean) => {
   307→    if (key === 'lastSeenVisible') setLastSeenVisible(value)
   308→    if (key === 'readReceiptsEnabled') setReadReceipts(value)
   309→    if (key === 'typingIndicatorsEnabled') setTypingIndicators(value)
   310→    update.mutate({ [key]: value })
   311→  }
   312→
   313→  return (
   314→    <Card className="p-2">
   315→      <PrivacyRow
   316→        title="Last seen visible"
   317→        desc="Show when you were last active"

   319→        onChange={(v) => handleToggle('lastSeenVisible', v)}
   320→      />
   321→      <PrivacyRow
   322→        title="Read receipts"
   323→        desc="Let others know when you've read their messages"
   324→        checked={readReceipts}
   325→        onChange={(v) => handleToggle('readReceiptsEnabled', v)}
   326→      />
   327→      <PrivacyRow
   328→        title="Typing indicators"
   329→        desc="Show others when you're typing"
   330→        checked={typingIndicators}
   331→        onChange={(v) => handleToggle('typingIndicatorsEnabled', v)}
   332→      />
   333→    </Card>
   334→  )
