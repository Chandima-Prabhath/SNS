     1→/**
     2→ * Visual Bot Builder — Types & Execution Engine (v2)
     3→ *
     4→ * Architecture
     5→ * ────────────
     6→ * A bot flow is a directed graph stored as JSON in `Bot.flow`:
     7→ *   { nodes: FlowNode[], edges: FlowEdge[] }
     8→ *
     9→ * The engine walks the graph starting from a TRIGGER node. Crucially, it
    10→ * supports **pause/resume** for INPUT and CHOICE nodes — when the engine
    11→ * reaches one, it stops, persists `{ currentNodeId, variables }` to
    12→ * ConversationSession, and resumes on the user's next message.
    13→ *
    14→ * Node categories (used by the UI for palette grouping):
    15→ *   - trigger  : starts the flow (1 type, multiple subtypes)
    16→ *   - output   : bot sends something (message, choice, typing)
    17→ *   - input    : wait for user reply (text input, choice input)
    18→ *   - logic    : control flow (condition, set_var, delay, stop)
    19→ *   - advanced : power features (api_call, random)
    20→ */
    21→
    22→// ─────────────────────────────────────────────────────────────────────────────

    73→export interface FlowNodeData {
    74→  // ── trigger ──
    75→  triggerType?: 'any_message' | 'command' | 'mention'
    76→  command?: string // without leading slash
    77→
    78→  // ── message ──
    79→  text?: string
    80→
    81→  // ── wait_choice ──
    82→  prompt?: string
    83→  options?: string[] // quick-reply choices
    84→  variableName?: string // where to store the reply
    85→
    86→  // ── typing ──
    87→  seconds?: number
    88→
    89→  // ── input ──
    90→  // (uses prompt + variableName from above)
    91→
    92→  // ── condition ──
    93→  variable?: string
    94→  operator?: 'equals' | 'not_equals' | 'contains' | 'starts_with' | 'exists' | 'not_exists'
    95→  value?: string
    96→
    97→  // ── set_var ──
    98→  // (uses variable + value from above)
    99→
   100→  // ── api_call ──
   101→  url?: string
   102→  method?: 'GET' | 'POST'
   103→  headers?: string // JSON string
   104→  body?: string
   105→  // (also uses variableName to store response)
   106→
   107→  // ── random ──
   108→  // (no data — uses N outgoing edges; engine picks one)
   109→
   110→  // ── ai_generate ──
   111→  /** The prompt to send to the LLM. Supports {{var}} interpolation. */
   112→  aiPrompt?: string
   113→  /** Optional system prompt to set the LLM's persona/behavior. */
   114→  aiSystemPrompt?: string
   115→  /** Ollama model name, e.g. 'gemma3:270m'. */
   116→  aiModel?: string
   117→  /** Sampling temperature (0 = deterministic, 1 = creative). */
   118→  aiTemperature?: number
   119→  /** Max tokens to generate. */
   120→  aiMaxTokens?: number
   121→  // (also uses variableName to store the generated response)
   122→
   123→  // ── switch_case ──
   124→  /** Variable to switch on. */
   125→  switchVariable?: string
   126→  /** Case values — each gets its own outgoing handle (case_0, case_1, ...). */
   127→  cases?: string[]
   128→  // (uses 'default' handle for unmatched values)
   129→
   130→  // ── counter ──
   131→  // (uses `variable` to hold the counter name)
   132→  /** Amount to increment by (negative = decrement). */
   133→  increment?: number
   134→  /** Starting value when the variable doesn't exist yet. */
   135→  startValue?: number
   136→
   137→  // ── format_string ──
   138→  // (uses `text` as the template with {{var}} placeholders)
   139→  // (uses `variableName` to store the result)
   140→
   141→  // ── send_media ──
   142→  /** URL of the media to send (image/video/audio). */
   143→  mediaUrl?: string
   144→  /** Caption text. Supports {{var}} interpolation. */
   145→  caption?: string
   146→  /** Media type: image, video, audio. */
   147→  mediaType?: 'image' | 'video' | 'audio'
   148→
   149→  // ── log ──
   150→  /** Message to log. Supports {{var}} interpolation. */
   151→  logMessage?: string
   152→  /** Log level. */
   153→  logLevel?: 'info' | 'warn' | 'error'
   154→
   155→  // ── tts ──
   156→  /** Text to speak. Supports {{var}} interpolation. */
   157→  ttsText?: string
   158→  /** Voice name (alba, charles, etc.) or custom voice ID. */
   159→  ttsVoice?: string
   160→  // (sends the audio as a voice message — no variable needed)
   161→
   162→  // ── asr_transcribe ──
   163→  /** URL of the audio file to transcribe. Supports {{var}} interpolation.
   164→   *  Typically set to {{mediaUrl}} which the bot framework populates with
   165→   *  the incoming message's mediaUrl when the user sends a voice message. */
   166→  asrAudioUrl?: string
   167→  /** Language hint (currently ignored — Moonshine v1 is English-only). */
   168→  asrLanguage?: string
   169→  /** Whether to also send the transcript as a reply message. If false,
   170→   *  the transcript is only stored in the variable. */
   171→  asrReply?: boolean
   172→  // (uses `variableName` to store the transcript text)
   173→
   174→  // ── UI metadata (not used by engine) ──
   175→  /** Node type — stored in data so the editor's CustomNode can read it.
   176→   *  Duplicates FlowNode.type but is required because ReactFlow sets
   177→   *  n.type to 'custom' for all custom nodes. */
   178→  type?: NodeType
   179→  label?: string
   180→}

   193→export interface BotExecutionContext {
   194→  channelId: string
   195→  senderId: string
   196→  senderName: string
   197→  messageId: string
   198→  body: string
   199→  args: string[]
   200→  command?: string
   201→  isMention: boolean
   202→
   203→  /**
   204→   * Variables for the current execution. The caller (visual.ts) is responsible

   207→   */
   208→  variables: Record<string, string>
   209→
   210→  /** Helpers — visual.ts wires these to ConversationSession. */
   211→  getState?: () => Promise<any>
   212→  setState?: (state: any) => Promise<void>
   213→
   214→  /** Reply helper — pushes a message as the bot. Optional keyboard attaches
   215→   *  Telegram-style inline buttons to the message. Returns the message ID. */
   216→  reply: (text: string, keyboard?: BotKeyboardButton[][]) => Promise<string>
   217→
   218→  /** Reply with media (image/video/audio). The URL should be a path like
   219→   *  /api/uploads/xxx.wav. Returns the message ID. */
   220→  replyWithMedia?: (mediaUrl: string, mediaType: string, caption?: string) => Promise<string>
   221→
   222→  /** Edit an existing bot message in-place (body + keyboard). Used for
   223→   *  Telegram-style keyboard updates when a wait_choice loops back. */
   224→  editMessage?: (messageId: string, text: string, keyboard?: BotKeyboardButton[][]) => Promise<void>
   225→
   226→  /** Generate TTS audio from text using Pocket TTS. Returns the media URL
   227→   *  (e.g. /api/uploads/tts-bot-xxx.wav) or null if generation failed.
   228→   *  Server-only — debug mode should return null and let the engine fall
   229→   *  back to sending text. */
   230→  generateTTS?: (text: string, voice: string) => Promise<string | null>
   231→
   232→  /** Transcribe an audio file URL to text using Moonshine ASR. Returns the
   233→   *  transcript text or null if transcription failed (server down, no audio,
   234→   *  etc.). Server-only — debug mode should return null and let the engine
   235→   *  log a clear error. */
   236→  transcribeAudio?: (mediaUrl: string, language?: string) => Promise<string | null>
   237→
   238→  /** Show typing indicator — best-effort, no-op if unsupported. */
   239→  setTyping?: (seconds: number) => Promise<void>
   240→}

   261→// ─────────────────────────────────────────────────────────────────────────────
   262→// Execution trace — emitted by the engine for debugging/test-runs
   263→// ─────────────────────────────────────────────────────────────────────────────
   264→
   265→export type TraceEvent =
   266→  | { type: 'flow_start'; timestamp: number; triggerNodeId?: string; input: { body: string; sender: string; command?: string } }
   267→  | { type: 'node_enter'; timestamp: number; nodeId: string; nodeType: NodeType; nodeLabel: string }
   268→  | { type: 'node_exit'; timestamp: number; nodeId: string; durationMs: number }
   269→  | { type: 'message_sent'; timestamp: number; nodeId: string; text: string }
   270→  | { type: 'variable_set'; timestamp: number; nodeId: string; variable: string; value: string }
   271→  | { type: 'condition_eval'; timestamp: number; nodeId: string; variable: string; value: string; operator: string; compareTo: string; result: boolean }
   272→  | { type: 'branch_taken'; timestamp: number; nodeId: string; handle: string; targetNodeId: string }
   273→  | { type: 'ai_call'; timestamp: number; nodeId: string; model: string; prompt: string; responseLength: number; durationMs: number }
   274→  | { type: 'api_call'; timestamp: number; nodeId: string; url: string; method: string; status: number; durationMs: number }
   275→  | { type: 'asr_call'; timestamp: number; nodeId: string; audioUrl: string; transcriptLength: number; durationMs: number; success: boolean }
   276→  | { type: 'paused'; timestamp: number; nodeId: string; variableName: string }
   277→  | { type: 'resumed'; timestamp: number; nodeId: string; inputText: string }
   278→  | { type: 'error'; timestamp: number; nodeId?: string; message: string }
   279→  | { type: 'flow_end'; timestamp: number; reason: 'stop' | 'no_edge' | 'max_steps' | 'error'; sentCount: number }
   280→  | { type: 'log'; timestamp: number; nodeId: string; level: 'info' | 'warn' | 'error'; message: string }

   282→// ─────────────────────────────────────────────────────────────────────────────
   283→// Engine
   284→// ─────────────────────────────────────────────────────────────────────────────
   285→
   286→const MAX_STEPS = 50
   287→
   288→/**
   289→ * Resume descriptor — when set, the engine resumes from `nodeId` instead of
   290→ * searching for a trigger. The caller passes this when a paused session
   291→ * receives its awaited reply.
   292→ */
   293→export interface ResumeDescriptor {
   294→  nodeId: string
   295→  /** The text to assign to the input variable (usually the user's reply). */
   296→  inputText: string
   297→  /** The variable name to assign inputText to. */
   298→  inputVariable: string
   299→}
   300→
   301→/**
   302→ * Replace {{var}} placeholders in a string.
   303→ */
   304→function interpolate(text: string, ctx: BotExecutionContext): string {
   305→  let out = text
   306→  for (const [k, v] of Object.entries(ctx.variables)) {
   307→    out = out.replace(new RegExp(`\\{\\{${k}\\}\\}`, 'g'), v)
   308→  }
   309→  out = out.replace(/\{\{sender\}\}/g, ctx.senderName)
   310→  out = out.replace(/\{\{args\}\}/g, ctx.args.join(' '))
   311→  out = out.replace(/\{\{body\}\}/g, ctx.body)
   312→  // Convenience: expose the incoming message's media URL/type and any
   313→  // pre-existing ASR transcript as {{mediaUrl}}, {{mediaType}}, {{transcript}}.
   314→  // These are also available as ctx.variables (set by visual.ts at flow start)
   315→  // but the explicit fallbacks here make them work even in debug mode.
   316→  out = out.replace(/\{\{mediaUrl\}\}/g, ctx.variables.__mediaUrl || '')
   317→  out = out.replace(/\{\{mediaType\}\}/g, ctx.variables.__mediaType || '')
   318→  out = out.replace(/\{\{transcript\}\}/g, ctx.variables.__transcript || ctx.variables.transcript || '')
   319→  return out
   320→}

   322→/**
   323→ * Execute a bot flow. Returns when:
   324→ *   - the flow reaches a `stop` node,
   325→ *   - the flow ends naturally (no outgoing edge),
   326→ *   - the flow pauses at an INPUT/WAIT_CHOICE node,
   327→ *   - MAX_STEPS is exceeded.
   328→ *
   329→ * Emits trace events for debugging — each node entry/exit, variable change,
   330→ * message sent, condition evaluation, branch taken, and error is recorded.
   331→ * The caller can pass these to the UI for a test-run visualization.
   332→ */
   333→export async function executeBotFlow(
   334→  flow: BotFlow,
   335→  ctx: BotExecutionContext,
   336→  resume?: ResumeDescriptor
   337→): Promise<BotExecutionResult> {
   338→  let sentCount = 0
   339→  let steps = 0
   340→  let currentNode: FlowNode | undefined
   341→  const trace: TraceEvent[] = []
   342→  const now = () => Date.now()
   343→
   344→  // ── Helpers (declared early so the resume block can use them) ─────────
   345→  const edgesFrom = (nodeId: string, handle?: string | null) =>
   346→    flow.edges.filter((e) => {
   347→      if (e.source !== nodeId) return false
   348→      if (handle === undefined) return true
   349→      if (handle === null) return !e.sourceHandle
   350→      return e.sourceHandle === handle
   351→    })
   352→
   353→  const followEdge = (nodeId: string, handle?: string | null): FlowNode | undefined => {
   354→    const edges = edgesFrom(nodeId, handle)
   355→    if (edges.length === 0) return undefined
   356→    return flow.nodes.find((n) => n.id === edges[0].target)
   357→  }
   358→
   359→  // ── Resolve starting node ─────────────────────────────────────────────
   360→  if (resume) {
   361→    const inputNode = flow.nodes.find((n) => n.id === resume.nodeId)
   362→    if (!inputNode) {
   363→      trace.push({ type: 'error', timestamp: now(), message: 'resume target not found' })
   364→      return { sentCount: 0, paused: false, variables: ctx.variables, trace, error: 'resume target not found' }
   365→    }
   366→    ctx.variables[resume.inputVariable] = resume.inputText
   367→    trace.push({ type: 'resumed', timestamp: now(), nodeId: inputNode.id, inputText: resume.inputText })
   368→    trace.push({ type: 'variable_set', timestamp: now(), nodeId: inputNode.id, variable: resume.inputVariable, value: resume.inputText })
   369→    currentNode = followEdge(inputNode.id, null)
   370→    if (!currentNode) {
   371→      trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   372→      return { sentCount, paused: false, variables: ctx.variables, trace }
   373→    }
   374→  } else {
   375→    const trigger = flow.nodes.find((n) => n.type === 'trigger')
   376→    if (!trigger) {
   377→      trace.push({ type: 'error', timestamp: now(), message: 'No trigger node found' })
   378→      return { sentCount: 0, paused: false, variables: ctx.variables, trace, error: 'No trigger node found' }
   379→    }
   380→
   381→    trace.push({
   382→      type: 'flow_start',
   383→      timestamp: now(),
   384→      triggerNodeId: trigger.id,
   385→      input: { body: ctx.body, sender: ctx.senderName, command: ctx.command },
   386→    })
   387→
   388→    // Trigger matching
   389→    const tt = trigger.data.triggerType || 'any_message'
   390→    if (tt === 'command' && trigger.data.command) {
   391→      if (!ctx.command || ctx.command !== trigger.data.command.replace(/^\//, '')) {
   392→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   393→        return { sentCount: 0, paused: false, variables: ctx.variables, trace }
   394→      }
   395→    } else if (tt === 'mention') {
   396→      if (!ctx.isMention) {
   397→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   398→        return { sentCount: 0, paused: false, variables: ctx.variables, trace }
   399→      }
   400→    }
   401→
   402→    currentNode = trigger
   403→  }
   404→
   405→  // ── Walk the graph ────────────────────────────────────────────────────
   406→  while (currentNode && steps < MAX_STEPS) {
   407→    steps++
   408→    const nodeStartTime = now()
   409→    const nodeLabel = currentNode.data?.label || currentNode.type
   410→
   411→    trace.push({
   412→      type: 'node_enter',
   413→      timestamp: nodeStartTime,
   414→      nodeId: currentNode.id,
   415→      nodeType: currentNode.type,
   416→      nodeLabel,
   417→    })
   418→
   419→    switch (currentNode.type) {
   420→      // ── TRIGGER ────────────────────────────────────────────────────────
   421→      case 'trigger':
   422→        currentNode = followEdge(currentNode.id, null)
   423→        break
   424→
   425→      // ── OUTPUT: message ────────────────────────────────────────────────
   426→      case 'message': {
   427→        const text = interpolate(currentNode.data.text || '', ctx)
   428→        if (text.trim()) {
   429→          await ctx.reply(text)
   430→          sentCount++
   431→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text })
   432→        }
   433→        currentNode = followEdge(currentNode.id, null)
   434→        break
   435→      }
   436→
   437→      // ── OUTPUT: send_media ─────────────────────────────────────────────
   438→      case 'send_media': {
   439→        const url = interpolate(currentNode.data.mediaUrl || '', ctx)
   440→        const caption = interpolate(currentNode.data.caption || '', ctx)
   441→        if (url.trim()) {
   442→          // ctx.reply is text-only; we log it but the actual media send
   443→          // would need an extended reply API. For now, send the URL as text.
   444→          const text = caption ? `${caption}\n${url}` : url
   445→          await ctx.reply(text)
   446→          sentCount++
   447→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text })
   448→        }
   449→        currentNode = followEdge(currentNode.id, null)
   450→        break
   451→      }
   452→
   453→      // ── OUTPUT: typing ─────────────────────────────────────────────────
   454→      case 'typing': {
   455→        const seconds = currentNode.data.seconds || 1
   456→        if (ctx.setTyping) await ctx.setTyping(seconds)
   457→        else await new Promise((r) => setTimeout(r, Math.min(seconds, 2) * 1000))
   458→        currentNode = followEdge(currentNode.id, null)
   459→        break
   460→      }
   461→
   462→      // ── INPUT: pause and wait for next message ────────────────────────
   463→      case 'input': {
   464→        const prompt = interpolate(currentNode.data.prompt || '', ctx)
   465→        if (prompt.trim()) {
   466→          await ctx.reply(prompt)
   467→          sentCount++
   468→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text: prompt })
   469→        }
   470→        trace.push({ type: 'paused', timestamp: now(), nodeId: currentNode.id, variableName: currentNode.data.variableName || 'reply' })
   471→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   472→        return {
   473→          sentCount,
   474→          paused: true,
   475→          pausedAtNodeId: currentNode.id,
   476→          variables: ctx.variables,
   477→          trace,
   478→        }
   479→      }
   480→
   481→      // ── INPUT: wait_choice ────────────────────────────────────────────
   482→      case 'wait_choice': {
   483→        const prompt = interpolate(currentNode.data.prompt || '', ctx)
   484→        const options = currentNode.data.options || []
   485→        // Build a Telegram-style inline keyboard — each option is a button.
   486→        // The button's callbackData is the option text itself, so the resume
   487→        // path in visual.ts can match it directly.
   488→        const keyboard: BotKeyboardButton[][] = options.map((opt) => [{
   489→          text: opt,
   490→          callbackData: opt,
   491→        }])
   492→
   493→        // Telegram-style behavior: if we have an existing keyboard message
   494→        // (from a previous pause at this node, i.e. a loop-back), EDIT it
   495→        // in-place instead of sending a new message. This prevents the bot
   496→        // from spamming new keyboard messages every time the user clicks a
   497→        // button that loops back to this menu.
   498→        //
   499→        // The keyboardMessageId is set in the session state by visual.ts
   500→        // when it persists the pause. We read it from ctx.variables under
   501→        // a reserved key.
   502→        const existingKeyboardMsgId = ctx.variables['__keyboardMsgId']
   503→
   504→        if (prompt.trim()) {
   505→          if (existingKeyboardMsgId && ctx.editMessage) {
   506→            // Edit-in-place: update the existing keyboard message
   507→            await ctx.editMessage(existingKeyboardMsgId, prompt, keyboard)
   508→            trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text: `[edited] ${prompt}` })
   509→          } else {
   510→            // Fresh keyboard: send a new message
   511→            const msgId = await ctx.reply(prompt, keyboard)
   512→            sentCount++
   513→            // Store the message ID so future re-pauses can edit it
   514→            ctx.variables['__keyboardMsgId'] = msgId
   515→            trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text: prompt })
   516→          }
   517→        }
   518→        trace.push({ type: 'paused', timestamp: now(), nodeId: currentNode.id, variableName: currentNode.data.variableName || 'choice' })
   519→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   520→        return {
   521→          sentCount,
   522→          paused: true,
   523→          pausedAtNodeId: currentNode.id,
   524→          variables: ctx.variables,
   525→          trace,
   526→        }
   527→      }
   528→
   529→      // ── LOGIC: condition ──────────────────────────────────────────────
   530→      case 'condition': {
   531→        const varName = currentNode.data.variable || ''
   532→        const varValue = ctx.variables[varName] || ''
   533→        const op = currentNode.data.operator || 'exists'
   534→        const cmp = currentNode.data.value || ''
   535→
   536→        let met = false
   537→        switch (op) {
   538→          case 'equals':       met = varValue === cmp; break
   539→          case 'not_equals':   met = varValue !== cmp; break
   540→          case 'contains':     met = varValue.includes(cmp); break
   541→          case 'starts_with':  met = varValue.startsWith(cmp); break
   542→          case 'exists':       met = !!varValue; break
   543→          case 'not_exists':   met = !varValue; break
   544→        }
   545→
   546→        trace.push({
   547→          type: 'condition_eval',
   548→          timestamp: now(),
   549→          nodeId: currentNode.id,
   550→          variable: varName,
   551→          value: varValue,
   552→          operator: op,
   553→          compareTo: cmp,
   554→          result: met,
   555→        })
   556→
   557→        const handle = met ? 'true' : 'false'
   558→        const next = followEdge(currentNode.id, handle)
   559→        if (next) {
   560→          trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle, targetNodeId: next.id })
   561→          currentNode = next
   562→        } else {
   563→          // No matching branch — fall through to default edge if any
   564→          const fallback = followEdge(currentNode.id, null)
   565→          if (fallback) {
   566→            trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: 'default', targetNodeId: fallback.id })
   567→            currentNode = fallback
   568→          } else {
   569→            trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   570→            currentNode = undefined
   571→          }
   572→        }
   573→        break
   574→      }
   575→
   576→      // ── LOGIC: switch_case (multi-branch) ─────────────────────────────
   577→      case 'switch_case': {
   578→        const varName = currentNode.data.switchVariable || ''
   579→        const varValue = ctx.variables[varName] || ''
   580→        const cases = currentNode.data.cases || []
   581→
   582→        const matchIdx = cases.findIndex((c) => c === varValue)
   583→        const handle = matchIdx >= 0 ? `case_${matchIdx}` : 'default'
   584→
   585→        trace.push({
   586→          type: 'condition_eval',
   587→          timestamp: now(),
   588→          nodeId: currentNode.id,
   589→          variable: varName,
   590→          value: varValue,
   591→          operator: 'switch',
   592→          compareTo: matchIdx >= 0 ? cases[matchIdx] : '(default)',
   593→          result: matchIdx >= 0,
   594→        })
   595→
   596→        const next = followEdge(currentNode.id, handle)
   597→        if (next) {
   598→          trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle, targetNodeId: next.id })
   599→          currentNode = next
   600→        } else {
   601→          const fallback = followEdge(currentNode.id, null)
   602→          if (fallback) {
   603→            trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: 'default', targetNodeId: fallback.id })
   604→            currentNode = fallback
   605→          } else {
   606→            trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   607→            currentNode = undefined
   608→          }
   609→        }
   610→        break
   611→      }
   612→
   613→      // ── LOGIC: set_var ────────────────────────────────────────────────
   614→      case 'set_var': {
   615→        const name = currentNode.data.variable || ''
   616→        const val = interpolate(currentNode.data.value || '', ctx)
   617→        if (name) {
   618→          ctx.variables[name] = val
   619→          trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: name, value: val })
   620→        }
   621→        currentNode = followEdge(currentNode.id, null)
   622→        break
   623→      }
   624→
   625→      // ── LOGIC: counter (increment/decrement a numeric variable) ───────
   626→      case 'counter': {
   627→        const name = currentNode.data.variable || ''
   628→        const inc = currentNode.data.increment ?? 1
   629→        const startVal = currentNode.data.startValue ?? 0
   630→        const current = parseInt(ctx.variables[name] || '', 10)
   631→        const newVal = isNaN(current) ? startVal + inc : current + inc
   632→        ctx.variables[name] = String(newVal)
   633→        trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: name, value: String(newVal) })
   634→        currentNode = followEdge(currentNode.id, null)
   635→        break
   636→      }
   637→
   638→      // ── LOGIC: format_string (template a string with variables) ──────
   639→      case 'format_string': {
   640→        const template = currentNode.data.text || ''
   641→        const result = interpolate(template, ctx)
   642→        const vname = currentNode.data.variableName
   643→        if (vname) {
   644→          ctx.variables[vname] = result
   645→          trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: result })
   646→        }
   647→        currentNode = followEdge(currentNode.id, null)
   648→        break
   649→      }
   650→
   651→      // ── LOGIC: delay ──────────────────────────────────────────────────
   652→      case 'delay': {
   653→        const s = Math.max(0, Math.min(60, currentNode.data.seconds || 1))
   654→        await new Promise((r) => setTimeout(r, s * 1000))
   655→        currentNode = followEdge(currentNode.id, null)
   656→        break
   657→      }
   658→
   659→      // ── LOGIC: log (debug logging — shows in trace, no user output) ───
   660→      case 'log': {
   661→        const msg = interpolate(currentNode.data.logMessage || '', ctx)
   662→        const level = currentNode.data.logLevel || 'info'
   663→        trace.push({ type: 'log', timestamp: now(), nodeId: currentNode.id, level, message: msg })
   664→        currentNode = followEdge(currentNode.id, null)
   665→        break
   666→      }
   667→
   668→      // ── LOGIC: stop ───────────────────────────────────────────────────
   669→      case 'stop':
   670→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'stop', sentCount })
   671→        return { sentCount, paused: false, variables: ctx.variables, trace }
   672→
   673→      // ── OUTPUT: tts (Pocket TTS voice message) ────────────────────────
   674→      case 'tts': {
   675→        const ttsStart = now()
   676→        const ttsNode = currentNode
   677→        const text = interpolate(ttsNode.data.ttsText || '', ctx)
   678→        const voice = ttsNode.data.ttsVoice || 'alba'
   679→
   680→        if (!text.trim()) {
   681→          trace.push({ type: 'error', timestamp: now(), nodeId: ttsNode.id, message: 'TTS node has empty text' })
   682→          currentNode = followEdge(ttsNode.id, null)
   683→          break
   684→        }
   685→
   686→        trace.push({ type: 'log', timestamp: now(), nodeId: ttsNode.id, level: 'info', message: `TTS: generating "${text.slice(0, 60)}…" with voice=${voice}` })
   687→
   688→        let mediaUrl: string | null = null
   689→        if (ctx.generateTTS) {
   690→          try {
   691→            mediaUrl = await ctx.generateTTS(text, voice)
   692→          } catch (e: any) {
   693→            trace.push({ type: 'error', timestamp: now(), nodeId: ttsNode.id, message: `TTS failed: ${e?.message || e}` })
   694→          }
   695→        } else {
   696→          // Debug mode — no actual TTS available
   697→          trace.push({ type: 'error', timestamp: now(), nodeId: ttsNode.id, message: 'TTS not available in debug mode' })
   698→        }
   699→
   700→        if (mediaUrl) {
   701→          // Send as a voice message
   702→          if (ctx.replyWithMedia) {
   703→            await ctx.replyWithMedia(mediaUrl, 'audio')
   704→          } else {
   705→            await ctx.reply(`🔊 ${mediaUrl}`)
   706→          }
   707→          sentCount++
   708→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: ttsNode.id, text: `[voice message: ${text.slice(0, 60)}…]` })
   709→          trace.push({ type: 'ai_call', timestamp: now(), nodeId: ttsNode.id, model: `tts:${voice}`, prompt: text.slice(0, 100), responseLength: 0, durationMs: now() - ttsStart })
   710→        } else {
   711→          // Fallback: send the text as a plain message
   712→          await ctx.reply(`🔊 ${text}`)
   713→          sentCount++
   714→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: ttsNode.id, text: `[TTS fallback] ${text.slice(0, 80)}` })
   715→        }
   716→        currentNode = followEdge(ttsNode.id, null)
   717→        break
   718→      }
   719→
   720→      // ── ADVANCED: asr_transcribe (Moonshine ASR voice-to-text) ───────
   721→      case 'asr_transcribe': {
   722→        const asrStart = now()
   723→        const asrNode = currentNode
   724→        const audioUrl = interpolate(asrNode.data.asrAudioUrl || '{{mediaUrl}}', ctx)
   725→        const language = asrNode.data.asrLanguage || 'en'
   726→        const shouldReply = asrNode.data.asrReply !== false // default true
   727→        const vname = asrNode.data.variableName || 'transcript'
   728→
   729→        if (!audioUrl || !audioUrl.trim()) {
   730→          trace.push({ type: 'error', timestamp: now(), nodeId: asrNode.id, message: 'ASR node has no audio URL (set asrAudioUrl or send a voice message)' })
   731→          currentNode = followEdge(asrNode.id, null)
   732→          break
   733→        }
   734→
   735→        trace.push({ type: 'log', timestamp: now(), nodeId: asrNode.id, level: 'info', message: `ASR: transcribing "${audioUrl.slice(0, 60)}…" (lang=${language})` })
   736→
   737→        let transcript: string | null = null
   738→        if (ctx.transcribeAudio) {
   739→          try {
   740→            transcript = await ctx.transcribeAudio(audioUrl, language)
   741→          } catch (e: any) {
   742→            trace.push({ type: 'error', timestamp: now(), nodeId: asrNode.id, message: `ASR failed: ${e?.message || e}` })
   743→          }
   744→        } else {
   745→          // Debug mode — no actual ASR available
   746→          trace.push({ type: 'error', timestamp: now(), nodeId: asrNode.id, message: 'ASR not available in debug mode' })
   747→        }
   748→
   749→        const transcriptText = transcript?.trim() || ''
   750→        const success = !!transcriptText
   751→
   752→        // Store in variable (even if empty — downstream condition nodes can check `exists`)
   753→        ctx.variables[vname] = transcriptText
   754→        trace.push({ type: 'variable_set', timestamp: now(), nodeId: asrNode.id, variable: vname, value: transcriptText.slice(0, 100) })
   755→        trace.push({
   756→          type: 'asr_call',
   757→          timestamp: now(),
   758→          nodeId: asrNode.id,
   759→          audioUrl: audioUrl.slice(0, 100),
   760→          transcriptLength: transcriptText.length,
   761→          durationMs: now() - asrStart,
   762→          success,
   763→        })
   764→
   765→        // Optionally send the transcript as a reply (default behavior)
   766→        if (shouldReply && transcriptText) {
   767→          await ctx.reply(`📝 ${transcriptText}`)
   768→          sentCount++
   769→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: asrNode.id, text: `[transcript] ${transcriptText.slice(0, 80)}` })
   770→        } else if (shouldReply && !transcriptText) {
   771→          await ctx.reply('📝 (transcription unavailable — no speech detected or ASR server down)')
   772→          sentCount++
   773→        }
   774→
   775→        currentNode = followEdge(asrNode.id, null)
   776→        break
   777→      }
   778→
   779→      // ── ADVANCED: api_call ────────────────────────────────────────────
   780→      case 'api_call': {
   781→        const callStart = now()
   782→        try {
   783→          const url = interpolate(currentNode.data.url || '', ctx)
   784→          const method = currentNode.data.method || 'GET'
   785→          const headers = currentNode.data.headers ? JSON.parse(currentNode.data.headers) : {}
   786→          const body = currentNode.data.body ? interpolate(currentNode.data.body, ctx) : undefined
   787→          const res = await fetch(url, { method, headers, body })
   788→          const text = await res.text()
   789→          const vname = currentNode.data.variableName
   790→          if (vname) {
   791→            ctx.variables[vname] = text.slice(0, 4000)
   792→            trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: text.slice(0, 200) + (text.length > 200 ? '…' : '') })
   793→          }
   794→          trace.push({ type: 'api_call', timestamp: now(), nodeId: currentNode.id, url, method, status: res.status, durationMs: now() - callStart })
   795→        } catch (e: any) {
   796→          trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `API call failed: ${e?.message || e}` })
   797→        }
   798→        currentNode = followEdge(currentNode.id, null)
   799→        break
   800→      }
   801→
   802→      // ── ADVANCED: random ──────────────────────────────────────────────
   803→      case 'random': {
   804→        const allEdges = edgesFrom(currentNode.id, undefined)
   805→        if (allEdges.length === 0) {
   806→          trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   807→          currentNode = undefined
   808→          break
   809→        }
   810→        const pick = allEdges[Math.floor(Math.random() * allEdges.length)]
   811→        const target = flow.nodes.find((n) => n.id === pick.target)
   812→        if (target) {
   813→          trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: pick.sourceHandle || 'random', targetNodeId: target.id })
   814→          currentNode = target
   815→        } else {
   816→          currentNode = undefined
   817→        }
   818→        break
   819→      }
   820→
   821→      // ── ADVANCED: ai_generate ──────────────────────────────────────────
   822→      case 'ai_generate': {
   823→        const vname = currentNode.data.variableName || 'aiResponse'
   824→        const callStart = now()
   825→        try {
   826→          const model = currentNode.data.aiModel || 'gemma3:270m'
   827→          const prompt = interpolate(currentNode.data.aiPrompt || '', ctx)
   828→          const systemPrompt = currentNode.data.aiSystemPrompt
   829→            ? interpolate(currentNode.data.aiSystemPrompt, ctx)
   830→            : undefined
   831→          const temperature = currentNode.data.aiTemperature ?? 0.7
   832→          const maxTokens = currentNode.data.aiMaxTokens ?? 256
   833→
   834→          const ollamaUrl = process.env.OLLAMA_URL || 'http://localhost:11434'
   835→          const res = await fetch(`${ollamaUrl}/api/generate`, {
   836→            method: 'POST',
   837→            headers: { 'Content-Type': 'application/json' },
   838→            body: JSON.stringify({
   839→              model, prompt, system: systemPrompt, stream: false,
   840→              options: { temperature, num_predict: maxTokens },
   841→            }),
   842→          })
   843→
   844→          if (!res.ok) {
   845→            const errText = await res.text().catch(() => 'unknown error')
   846→            const errMsg = `Ollama error ${res.status}: ${errText.slice(0, 100)}`
   847→            ctx.variables[vname] = `[AI error: ${res.status}]`
   848→            trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: errMsg })
   849→            trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: ctx.variables[vname] })
   850→            trace.push({ type: 'ai_call', timestamp: now(), nodeId: currentNode.id, model, prompt: prompt.slice(0, 100), responseLength: 0, durationMs: now() - callStart })
   851→          } else {
   852→            const data = await res.json()
   853→            const generated: string = data.response || ''
   854→            ctx.variables[vname] = generated.slice(0, 4000)
   855→            trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: generated.slice(0, 200) + (generated.length > 200 ? '…' : '') })
   856→            trace.push({ type: 'ai_call', timestamp: now(), nodeId: currentNode.id, model, prompt: prompt.slice(0, 100), responseLength: generated.length, durationMs: now() - callStart })
   857→          }
   858→        } catch (e: any) {
   859→          ctx.variables[vname] = `[AI error: ${e?.message || 'unavailable'}]`
   860→          trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `AI generate failed: ${e?.message || e}` })
   861→          trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: ctx.variables[vname] })
   862→        }
   863→        currentNode = followEdge(currentNode.id, null)
   864→        break
   865→      }
   866→
   867→      default:
   868→        trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `unknown node type: ${currentNode.type}` })
   869→        return { sentCount, paused: false, variables: ctx.variables, trace, error: `unknown node type: ${currentNode.type}` }
   870→    }
   871→
   872→    // Record node exit
   873→    if (currentNode) {
   874→      trace.push({ type: 'node_exit', timestamp: now(), nodeId: currentNode.id, durationMs: 0 })
   875→    }
   876→  }
   877→
   878→  if (steps >= MAX_STEPS) {
   879→    await ctx.reply('⚠️ Bot flow exceeded maximum steps (possible infinite loop).')
   880→    trace.push({ type: 'error', timestamp: now(), message: 'max steps exceeded' })
   881→    trace.push({ type: 'flow_end', timestamp: now(), reason: 'max_steps', sentCount })
   882→    return { sentCount, paused: false, variables: ctx.variables, trace, error: 'max steps' }
   883→  }
   884→
   885→  trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   886→  return { sentCount, paused: false, variables: ctx.variables, trace }
   887→}
