   235→/**
   236→ * useVoiceRecorder — hook that manages MediaRecorder state for recording voice
   237→ * messages. Tap the mic to start, tap stop to upload + send, or cancel to
   238→ * discard. The recording is uploaded via /api/upload and sent as a message
   239→ * with mediaType 'audio/webm' (or the browser's best supported format).
   240→ */
   241→function useVoiceRecorder({
   242→  onSend,
   243→}: {
   244→  onSend: (url: string, mediaType: string) => Promise<void>
   245→}) {
   246→  const [isRecording, setIsRecording] = useState(false)
   247→  const [isUploading, setIsUploading] = useState(false)
   248→  const [seconds, setSeconds] = useState(0)
   249→  const mediaRecorderRef = useRef<MediaRecorder | null>(null)
   250→  const chunksRef = useRef<Blob[]>([])
   251→  const streamRef = useRef<MediaStream | null>(null)
   252→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   253→  const mimeTypeRef = useRef<string>('audio/webm')
   254→
   255→  // Stop the timer + release the mic on unmount
   256→  useEffect(() => {
   257→    return () => {
   258→      if (timerRef.current) clearInterval(timerRef.current)
   259→      if (streamRef.current) streamRef.current.getTracks().forEach((t) => t.stop())
   260→    }
   261→  }, [])
   262→
   263→  const pickMimeType = useCallback(() => {
   264→    const candidates = [
   265→      'audio/webm;codecs=opus',
   266→      'audio/webm',
   267→      'audio/ogg;codecs=opus',
   268→      'audio/mp4',
   269→    ]
   270→    for (const t of candidates) {
   271→      if (typeof MediaRecorder !== 'undefined' && MediaRecorder.isTypeSupported(t)) {
   272→        return t
   273→      }
   274→    }
   275→    return ''
   276→  }, [])
   277→
   278→  const stopTracks = useCallback(() => {
   279→    if (timerRef.current) {
   280→      clearInterval(timerRef.current)
   281→      timerRef.current = null
   282→    }
   283→    if (streamRef.current) {
   284→      streamRef.current.getTracks().forEach((t) => t.stop())
   285→      streamRef.current = null
   286→    }
   287→  }, [])
   288→
   289→  const start = useCallback(async () => {
   290→    try {
   291→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   292→      streamRef.current = stream
   293→      const mimeType = pickMimeType()
   294→      mimeTypeRef.current = mimeType || 'audio/webm'
   295→      const recorder = mimeType
   296→        ? new MediaRecorder(stream, { mimeType })
   297→        : new MediaRecorder(stream)
   298→      chunksRef.current = []
   299→      recorder.ondataavailable = (e) => {
   300→        if (e.data && e.data.size > 0) chunksRef.current.push(e.data)
   301→      }
   302→      recorder.start()
   303→      mediaRecorderRef.current = recorder
   304→      setIsRecording(true)
   305→      setSeconds(0)
   306→      timerRef.current = setInterval(() => {
   307→        setSeconds((s) => s + 1)
   308→      }, 1000)
   309→    } catch {
   310→      toast.error('Microphone access denied')
   311→      if (streamRef.current) {
   312→        streamRef.current.getTracks().forEach((t) => t.stop())
   313→        streamRef.current = null
   314→      }
   315→    }
   316→  }, [pickMimeType])
   317→
   318→  const cancel = useCallback(() => {
   319→    const recorder = mediaRecorderRef.current
   320→    if (recorder && recorder.state !== 'inactive') {
   321→      // Detach onstop so the pending send is discarded
   322→      recorder.onstop = null
   323→      try {
   324→        recorder.stop()
   325→      } catch {
   326→        // ignore
   327→      }
   328→    }
   329→    mediaRecorderRef.current = null
   330→    chunksRef.current = []
   331→    setIsRecording(false)
   332→    setSeconds(0)
   333→    stopTracks()
   334→  }, [stopTracks])
   335→
   336→  const stopAndSend = useCallback(async () => {
   337→    const recorder = mediaRecorderRef.current
   338→    if (!recorder || recorder.state === 'inactive') {
   339→      cancel()
   340→      return
   341→    }
   342→    const mime = mimeTypeRef.current
   343→    const blobPromise = new Promise<Blob>((resolve) => {
   344→      recorder.onstop = () => {
   345→        const blob = new Blob(chunksRef.current, { type: mime || 'audio/webm' })
   346→        resolve(blob)
   347→      }
   348→    })
   349→    try {
   350→      recorder.stop()
   351→    } catch {
   352→      cancel()
   353→      return
   354→    }
   355→    setIsRecording(false)
   356→    stopTracks()
   357→    setIsUploading(true)
   358→    try {
   359→      const blob = await blobPromise
   360→      if (blob.size === 0) {
   361→        toast.error('Recording was empty')
   362→        return
   363→      }
   364→      const ext = mime.includes('mp4') ? 'm4a' : mime.includes('ogg') ? 'ogg' : 'webm'
   365→      const file = new File([blob], `voice-${Date.now()}.${ext}`, { type: blob.type || mime })
   366→      const formData = new FormData()
   367→      formData.append('file', file)
   368→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   369→      if (!res.ok) {
   370→        const t = await res.text().catch(() => 'Upload failed')
   371→        throw new Error(t.slice(0, 120) || 'Upload failed')
   372→      }
   373→      const data = await res.json()
   374→      await onSend(data.url, data.type || mime || 'audio/webm')
   375→      toast.success('Voice message sent')
   376→    } catch (e: any) {
   377→      toast.error(e?.message || 'Failed to send voice message')
   378→    } finally {
   379→      setIsUploading(false)
   380→      setSeconds(0)
   381→      chunksRef.current = []
   382→      mediaRecorderRef.current = null
   383→    }
   384→  }, [cancel, onSend, stopTracks])
   385→
   386→  return { isRecording, isUploading, seconds, start, cancel, stopAndSend }
   387→}

   389→/**
   390→ * TTS Dialog — lets the user type text, pick a voice (built-in or custom),
   391→ * preview, and send. Also supports creating custom voices from audio clips.
   392→ */
   393→function TtsDialog({
   394→  open,
   395→  onOpenChange,
   396→  onSend,
   397→}: {
   398→  open: boolean
   399→  onOpenChange: (o: boolean) => void
   400→  onSend: (url: string) => Promise<void>
   401→}) {
   402→  const [tab, setTab] = useState<'generate' | 'voices'>('generate')
   403→  const [text, setText] = useState('')
   404→  const [voice, setVoice] = useState('alba')
   405→  const [selectedCustomVoiceId, setSelectedCustomVoiceId] = useState<string | null>(null)
   406→  const [generating, setGenerating] = useState(false)
   407→  const [previewUrl, setPreviewUrl] = useState<string | null>(null)
   408→  const [sendUrl, setSendUrl] = useState<string | null>(null)
   409→  const [sending, setSending] = useState(false)
   410→  const qc = useQueryClient()
   411→
   412→  // Fetch built-in voices from the API (single source of truth for labels)
   413→  const { data: builtinVoicesData } = useQuery({
   414→    queryKey: ['tts-builtin-voices'],
   415→    queryFn: async () => {
   416→      const res = await fetch('/api/tts')
   417→      if (!res.ok) throw new Error('failed')
   418→      return res.json()
   419→    },
   420→    enabled: open,
   421→  })
   422→  const builtinVoices: { id: string; name: string; language: string; gender: string }[] =
   423→    builtinVoicesData?.voices || []
   424→
   425→  // Fetch custom voices
   426→  const { data: customVoicesData } = useQuery({
   427→    queryKey: ['tts-voices'],
   428→    queryFn: async () => {
   429→      const res = await fetch('/api/tts/voices')
   430→      if (!res.ok) throw new Error('failed')
   431→      return res.json()
   432→    },
   433→    enabled: open,
   434→  })
   435→  const customVoices = customVoicesData?.voices || []
   436→
   437→  const handleGenerate = async () => {
   438→    if (!text.trim() || generating) return
   439→    setGenerating(true)
   440→    setPreviewUrl(null)
   441→    setSendUrl(null)
   442→    try {
   443→      const body: any = { text }
   444→      if (selectedCustomVoiceId) {
   445→        body.customVoiceId = selectedCustomVoiceId
   446→      } else {
   447→        body.voice = voice
   448→      }
   449→      const res = await fetch('/api/tts', {
   450→        method: 'POST',
   451→        headers: { 'Content-Type': 'application/json' },
   452→        body: JSON.stringify(body),
   453→      })
   454→      if (!res.ok) {
   455→        // Error responses are JSON, streaming success responses are audio
   456→        const err = await res.json().catch(() => ({ error: 'Generation failed' }))
   457→        throw new Error(err.error || 'Generation failed')
   458→      }
   459→      // The route streams the WAV audio back to us and puts the saved-file
   460→      // URL in the X-Tts-Url header (the file is written to disk in the
   461→      // background while we receive the stream). We use the streamed bytes
   462→      // for an instant preview and the header URL when sending the message.
   463→      const serverUrl = res.headers.get('X-Tts-Url')
   464→      const blob = await res.blob()
   465→      const blobUrl = URL.createObjectURL(blob)
   466→      setPreviewUrl(blobUrl)
   467→      setSendUrl(serverUrl)
   468→      toast.success('Voice generated — preview and send')
   469→    } catch (e: any) {
   470→      toast.error(e.message || 'Failed to generate voice')
   471→    } finally {
   472→      setGenerating(false)
   473→    }
   474→  }
   475→
   476→  const handleSend = async () => {
   477→    if (!previewUrl || !sendUrl || sending) return
   478→    setSending(true)
   479→    try {
   480→      await onSend(sendUrl)
   481→      setText('')
   482→      if (previewUrl.startsWith('blob:')) URL.revokeObjectURL(previewUrl)
   483→      setPreviewUrl(null)
   484→      setSendUrl(null)
   485→    } catch {
   486→      toast.error('Failed to send voice message')
   487→    } finally {
   488→      setSending(false)
   489→    }
   490→  }
   491→
   492→  const handleClose = (o: boolean) => {
   493→    if (!o) {
   494→      setText('')
   495→      if (previewUrl?.startsWith('blob:')) URL.revokeObjectURL(previewUrl)
   496→      setPreviewUrl(null)
   497→      setSendUrl(null)
   498→      setSelectedCustomVoiceId(null)
   499→    }
   500→    onOpenChange(o)
   501→  }
   502→
   503→  // Format a built-in voice as "Name (Language, Gender)" for the dropdown
   504→  const formatVoice = (v: { name: string; language: string; gender: string }) => {
   505→    const gender = v.gender.charAt(0).toUpperCase() + v.gender.slice(1)
   506→    return `${v.name} (${v.language}, ${gender})`
   507→  }
   508→
   509→  const builtinSelect = (
   510→    <Select value={voice} onValueChange={setVoice}>
   511→      <SelectTrigger>
   512→        <SelectValue />
   513→      </SelectTrigger>
   514→      <SelectContent>
   515→        {builtinVoices.length === 0 ? (
   516→          <SelectItem value={voice} disabled>
   517→            Loading voices…
   518→          </SelectItem>
   519→        ) : (
   520→          builtinVoices.map((v) => (
   521→            <SelectItem key={v.id} value={v.id}>
   522→              {formatVoice(v)}
   523→            </SelectItem>
   524→          ))
   525→        )}
   526→      </SelectContent>
   527→    </Select>
   528→  )
   529→
   530→  return (
   531→    <Dialog open={open} onOpenChange={handleClose}>
   532→      <DialogContent className="sm:max-w-md">
   533→        <DialogHeader>
   534→          <DialogTitle className="flex items-center gap-2">
   535→            <Sparkles className="w-4 h-4 text-primary" />
   536→            AI Voice Message
   537→          </DialogTitle>
   538→          <DialogDescription>
   539→            Type a message and send it as an AI-generated voice note. Create custom voices from audio clips.
   540→          </DialogDescription>
   541→        </DialogHeader>
   542→
   543→        {/* Tabs */}
   544→        <div className="flex gap-1 p-1 bg-muted rounded-lg">
   545→          <button
   546→            onClick={() => setTab('generate')}
   547→            className={cn(
   548→              'flex-1 py-1.5 text-sm font-medium rounded-md transition-colors',
   549→              tab === 'generate' ? 'bg-background text-foreground' : 'text-muted-foreground'
   550→            )}
   551→          >
   552→            Generate
   553→          </button>
   554→          <button
   555→            onClick={() => setTab('voices')}
   556→            className={cn(
   557→              'flex-1 py-1.5 text-sm font-medium rounded-md transition-colors flex items-center justify-center gap-1.5',
   558→              tab === 'voices' ? 'bg-background text-foreground' : 'text-muted-foreground'
   559→            )}
   560→          >
   561→            My Voices
   562→            {customVoices.length > 0 && (
   563→              <span className="text-[10px] bg-primary/20 text-primary px-1.5 py-0.5 rounded-full">
   564→                {customVoices.length}
   565→              </span>
   566→            )}
   567→          </button>
   568→        </div>
   569→
   570→        {tab === 'generate' ? (
   571→          <div className="space-y-4">
   572→            <div className="space-y-2">
   573→              <Label>Message</Label>
   574→              <Textarea
   575→                value={text}
   576→                onChange={(e) => setText(e.target.value)}
   577→                placeholder="Type what the voice should say..."
   578→                rows={3}
   579→                maxLength={500}
   580→                className="resize-none"
   581→              />
   582→              <div className="text-xs text-muted-foreground text-right">
   583→                {text.length}/500
   584→              </div>
   585→            </div>
   586→
   587→            <div className="space-y-2">
   588→              <Label>Voice</Label>
   589→              {customVoices.length > 0 && (
   590→                <div className="space-y-2">
   591→                  <div className="flex gap-2">
   592→                    <button
   593→                      onClick={() => setSelectedCustomVoiceId(null)}
   594→                      className={cn(
   595→                        'flex-1 py-2 px-3 rounded-lg text-sm font-medium transition-colors border',
   596→                        !selectedCustomVoiceId
   597→                          ? 'bg-primary/15 text-primary border-primary/30'
   598→                          : 'bg-muted/50 text-muted-foreground border-transparent hover:bg-muted'
   599→                      )}
   600→                    >
   601→                      Built-in
   602→                    </button>
   603→                    <button
   604→                      onClick={() => setSelectedCustomVoiceId(customVoices[0]?.id || null)}
   605→                      className={cn(
   606→                        'flex-1 py-2 px-3 rounded-lg text-sm font-medium transition-colors border',
   607→                        selectedCustomVoiceId
   608→                          ? 'bg-primary/15 text-primary border-primary/30'
   609→                          : 'bg-muted/50 text-muted-foreground border-transparent hover:bg-muted'
   610→                      )}
   611→                    >
   612→                      Custom
   613→                    </button>
   614→                  </div>
   615→
   616→                  {selectedCustomVoiceId ? (
   617→                    <Select value={selectedCustomVoiceId} onValueChange={setSelectedCustomVoiceId}>
   618→                      <SelectTrigger>
   619→                        <SelectValue />
   620→                      </SelectTrigger>
   621→                      <SelectContent>
   622→                        {customVoices.map((v: any) => (
   623→                          <SelectItem key={v.id} value={v.id}>
   624→                            {v.name}
   625→                          </SelectItem>
   626→                        ))}
   627→                      </SelectContent>
   628→                    </Select>
   629→                  ) : (
   630→                    builtinSelect
   631→                  )}
   632→                </div>
   633→              )}
   634→
   635→              {!selectedCustomVoiceId && customVoices.length === 0 && builtinSelect}
   636→
   637→              {customVoices.length === 0 && (
   638→                <button
   639→                  onClick={() => setTab('voices')}
   640→                  className="text-xs text-primary hover:underline flex items-center gap-1"
   641→                >
   642→                  <Plus className="w-3 h-3" />
   643→                  Create a custom voice from your voice clip
   644→                </button>
   645→              )}
   646→            </div>
   647→
   648→            {/* Preview player */}
   649→            {previewUrl && (
   650→              <div className="space-y-2">
   651→                <Label>Preview</Label>
   652→                <audio controls src={previewUrl} className="w-full" />
   653→              </div>
   654→            )}
   655→          </div>
   656→        ) : (
   657→          <CustomVoicesTab onUseVoice={(id) => { setSelectedCustomVoiceId(id); setTab('generate') }} />
   658→        )}
   659→
   660→        <DialogFooter>
   661→          <Button variant="outline" onClick={() => handleClose(false)}>Cancel</Button>
   662→          {tab === 'generate' && (
   663→            !previewUrl ? (
   664→              <Button onClick={handleGenerate} disabled={!text.trim() || generating}>
   665→                {generating ? (
   666→                  <>
   667→                    <Loader2 className="w-4 h-4 mr-1.5 animate-spin" />
   668→                    Generating...
   669→                  </>
   670→                ) : (
   671→                  <>
   672→                    <Sparkles className="w-4 h-4 mr-1.5" />
   673→                    Generate voice
   674→                  </>
   675→                )}
   676→              </Button>
   677→            ) : (
   678→              <>
   679→                <Button variant="outline" onClick={() => {
   680→                  if (previewUrl.startsWith('blob:')) URL.revokeObjectURL(previewUrl)
   681→                  setPreviewUrl(null)
   682→                  setSendUrl(null)
   683→                }} disabled={sending}>
   684→                  Regenerate
   685→                </Button>
   686→                <Button onClick={handleSend} disabled={sending}>
   687→                  {sending ? (
   688→                    <Loader2 className="w-4 h-4 mr-1.5 animate-spin" />
   689→                  ) : (
   690→                    <Send className="w-4 h-4 mr-1.5" />
   691→                  )}
   692→                  Send
   693→                </Button>
   694→              </>
   695→            )
   696→          )}
   697→        </DialogFooter>
   698→      </DialogContent>
   699→    </Dialog>
   700→  )
   701→}

   703→/**
   704→ * Custom Voices tab — lets the user record/upload a voice clip and save it
   705→ * as a custom voice model for one-shot voice cloning.
   706→ */
   707→function CustomVoicesTab({ onUseVoice }: { onUseVoice: (id: string) => void }) {
   708→  const qc = useQueryClient()
   709→  const [name, setName] = useState('')
   710→  const [audioUrl, setAudioUrl] = useState<string | null>(null)
   711→  const [uploading, setUploading] = useState(false)
   712→  const [recording, setRecording] = useState(false)
   713→  const [mediaRecorder, setMediaRecorder] = useState<MediaRecorder | null>(null)
   714→  const [audioChunks, setAudioChunks] = useState<Blob[]>([])
   715→
   716→  const { data: customVoicesData } = useQuery({
   717→    queryKey: ['tts-voices'],
   718→    queryFn: async () => {
   719→      const res = await fetch('/api/tts/voices')
   720→      if (!res.ok) throw new Error('failed')
   721→      return res.json()
   722→    },
   723→  })
   724→  const customVoices = customVoicesData?.voices || []
   725→
   726→  const deleteVoice = useMutation({
   727→    mutationFn: async (id: string) => {
   728→      await fetch(`/api/tts/voices/${id}`, { method: 'DELETE' })
   729→    },
   730→    onSuccess: () => {
   731→      qc.invalidateQueries({ queryKey: ['tts-voices'] })
   732→      toast.success('Voice deleted')
   733→    },
   734→  })
   735→
   736→  const saveVoice = useMutation({
   737→    mutationFn: async () => {
   738→      const res = await fetch('/api/tts/voices', {
   739→        method: 'POST',
   740→        headers: { 'Content-Type': 'application/json' },
   741→        body: JSON.stringify({ name, audioUrl }),
   742→      })
   743→      if (!res.ok) throw new Error('failed')
   744→      return res.json()
   745→    },
   746→    onSuccess: () => {
   747→      qc.invalidateQueries({ queryKey: ['tts-voices'] })
   748→      toast.success('Custom voice created')
   749→      setName('')
   750→      setAudioUrl(null)
   751→    },
   752→  })
   753→
   754→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   755→    const file = e.target.files?.[0]
   756→    if (!file) return
   757→    setUploading(true)
   758→    try {
   759→      const formData = new FormData()
   760→      formData.append('file', file)
   761→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   762→      if (!res.ok) throw new Error('Upload failed')
   763→      const data = await res.json()
   764→      setAudioUrl(data.url)
   765→    } catch (e: any) {
   766→      toast.error(e.message || 'Upload failed')
   767→    } finally {
   768→      setUploading(false)
   769→      e.target.value = ''
   770→    }
   771→  }
   772→
   773→  const startRecording = async () => {
   774→    try {
   775→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   776→      const recorder = new MediaRecorder(stream)
   777→      const chunks: Blob[] = []
   778→      recorder.ondataavailable = (e) => chunks.push(e.data)
   779→      recorder.onstop = async () => {
   780→        const blob = new Blob(chunks, { type: 'audio/webm' })
   781→        // Upload the recording
   782→        setUploading(true)
   783→        try {
   784→          const formData = new FormData()
   785→          formData.append('file', new File([blob], 'voice.webm', { type: 'audio/webm' }))
   786→          const res = await fetch('/api/upload', { method: 'POST', body: formData })
   787→          if (!res.ok) throw new Error('Upload failed')
   788→          const data = await res.json()
   789→          setAudioUrl(data.url)
   790→        } catch (e: any) {
   791→          toast.error(e.message || 'Upload failed')
   792→        } finally {
   793→          setUploading(false)
   794→        }
   795→        stream.getTracks().forEach((t) => t.stop())
   796→      }
   797→      recorder.start()
   798→      setMediaRecorder(recorder)
   799→      setRecording(true)
   800→      setAudioChunks(chunks)
   801→    } catch {
   802→      toast.error('Microphone access denied')
   803→    }
   804→  }
   805→
   806→  const stopRecording = () => {
   807→    if (mediaRecorder && mediaRecorder.state === 'recording') {
   808→      mediaRecorder.stop()
   809→      setRecording(false)
   810→    }
   811→  }
   812→
   813→  return (
   814→    <div className="space-y-4">
   815→      {/* Create new voice */}
   816→      <div className="space-y-3 p-3 rounded-xl bg-muted/50 border border-border/30">
   817→        <div className="flex items-center gap-2 text-sm font-medium">
   818→          <Mic className="w-4 h-4 text-primary" />
   819→          Create a Custom Voice
   820→        </div>
   821→        <p className="text-xs text-muted-foreground">
   822→          Upload or record a 10-30 second voice clip. We'll clone the voice so you can generate speech in it.
   823→        </p>
   824→        <div className="space-y-2">
   825→          <Input
   826→            value={name}
   827→            onChange={(e) => setName(e.target.value)}
   828→            placeholder="Voice name (e.g. My Voice)"
   829→          />
   830→        </div>
   831→        <div className="flex gap-2">
   832→          {!audioUrl ? (
   833→            <>
   834→              {!recording ? (
   835→                <Button onClick={startRecording} variant="outline" size="sm" className="flex-1">
   836→                  <Mic className="w-4 h-4 mr-1.5" />
   837→                  Record
   838→                </Button>
   839→              ) : (
   840→                <Button onClick={stopRecording} variant="destructive" size="sm" className="flex-1">
   841→                  <span className="w-2 h-2 rounded-full bg-white animate-pulse mr-1.5" />
   842→                  Stop
   843→                </Button>
   844→              )}
   845→              <label className="cursor-pointer flex-1">
   846→                <Button asChild variant="outline" size="sm" className="w-full">
   847→                  <span>
   848→                    <Plus className="w-4 h-4 mr-1.5" />
   849→                    Upload
   850→                  </span>
   851→                </Button>
   852→                <input type="file" className="hidden" accept="audio/*" onChange={handleUpload} disabled={uploading || recording} />
   853→              </label>
   854→            </>
   855→          ) : (
   856→            <div className="flex-1 space-y-2">
   857→              <audio controls src={audioUrl} className="w-full h-8" />
   858→              <div className="flex gap-2">
   859→                <Button
   860→                  onClick={() => saveVoice.mutate()}
   861→                  disabled={!name.trim() || saveVoice.isPending}
   862→                  size="sm"
   863→                  className="flex-1 gradient-primary"
   864→                >
   865→                  {saveVoice.isPending ? (
   866→                    <Loader2 className="w-4 h-4 mr-1.5 animate-spin" />
   867→                  ) : (
   868→                    <Sparkles className="w-4 h-4 mr-1.5" />
   869→                  )}
   870→                  Save voice
   871→                </Button>
   872→                <Button
   873→                  onClick={() => setAudioUrl(null)}
   874→                  variant="outline"
   875→                  size="sm"
   876→                >
   877→                  Discard
   878→                </Button>
   879→              </div>
   880→            </div>
   881→          )}
   882→        </div>
   883→        {uploading && (
   884→          <div className="text-xs text-muted-foreground flex items-center gap-1.5">
   885→            <Loader2 className="w-3 h-3 animate-spin" />
   886→            Uploading...
   887→          </div>
   888→        )}
   889→      </div>
   890→
   891→      {/* Existing custom voices */}
   892→      {customVoices.length > 0 && (
   893→        <div className="space-y-2">
   894→          <Label className="text-xs uppercase tracking-wider text-muted-foreground">
   895→            Saved Voices
   896→          </Label>
   897→          <div className="space-y-1.5 max-h-48 overflow-y-auto">
   898→            {customVoices.map((v: any) => (
   899→              <div
   900→                key={v.id}
   901→                className="flex items-center gap-3 p-2.5 rounded-xl bg-card border border-border/30 hover:border-primary/30 transition-colors group"
   902→              >
   903→                <div className="w-9 h-9 rounded-full bg-primary/15 flex items-center justify-center shrink-0">
   904→                  <User className="w-4 h-4 text-primary" />
   905→                </div>
   906→                <div className="flex-1 min-w-0">
   907→                  <div className="text-sm font-medium truncate">{v.name}</div>
   908→                  <div className="text-xs text-muted-foreground">
   909→                    Created {new Date(v.createdAt).toLocaleDateString()}
   910→                  </div>
   911→                </div>
   912→                <button
   913→                  onClick={() => onUseVoice(v.id)}
   914→                  className="text-xs px-2.5 py-1.5 rounded-lg bg-primary/10 text-primary hover:bg-primary/20 transition-colors font-medium"
   915→                >
   916→                  Use
   917→                </button>
   918→                <button
   919→                  onClick={() => {
   920→                    if (confirm(`Delete "${v.name}"?`)) deleteVoice.mutate(v.id)
   921→                  }}
   922→                  className="p-1.5 rounded-lg text-red-400 hover:bg-red-500/10 transition-colors"
   923→                >
   924→                  <Trash2 className="w-3.5 h-3.5" />
   925→                </button>
   926→              </div>
   927→            ))}
   928→          </div>
   929→        </div>
   930→      )}
   931→
   932→      {customVoices.length === 0 && !audioUrl && (
   933→        <div className="text-center py-6">
   934→          <div className="w-12 h-12 mx-auto rounded-2xl bg-primary/10 flex items-center justify-center mb-2">
   935→            <Mic className="w-5 h-5 text-primary" />
   936→          </div>
   937→          <p className="text-sm text-muted-foreground">No custom voices yet</p>
   938→        </div>
   939→      )}
   940→    </div>
   941→  )
   942→}
