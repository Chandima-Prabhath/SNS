    23→interface MessageComposerProps {
    24→  channelId: string
    25→}
    26→
    27→export function MessageComposer({ channelId }: MessageComposerProps) {
    28→  const { send, sendTyping, replyTo, setReplyTo } = useChannel(channelId)
    29→  const [text, setText] = useState('')
    30→  const [sending, setSending] = useState(false)
    31→  const [uploading, setUploading] = useState(false)
    32→  const [ttsOpen, setTtsOpen] = useState(false)
    33→  const textareaRef = useRef<HTMLTextAreaElement>(null)
    34→
    35→  useEffect(() => {
    36→    const el = textareaRef.current
    37→    if (!el) return
    38→    el.style.height = 'auto'
    39→    el.style.height = Math.min(el.scrollHeight, 160) + 'px'
    40→  }, [text])
    41→
    42→  useEffect(() => {
    43→    if (replyTo) textareaRef.current?.focus()
    44→  }, [replyTo])
    45→
    46→  const handleSend = async () => {
    47→    const trimmed = text.trim()
    48→    if (!trimmed || sending) return
    49→    setSending(true)
    50→    try {
    51→      await send({ body: trimmed, replyToId: replyTo?.id || null })
    52→      setText('')
    53→      setReplyTo(null)
    54→      sendTyping(false)
    55→    } catch {
    56→      toast.error('Failed to send message')
    57→    } finally {
    58→      setSending(false)
    59→    }
    60→  }
    61→
    62→  const handleKeyDown = (e: React.KeyboardEvent) => {
    63→    if (e.key === 'Enter' && !e.shiftKey) {
    64→      e.preventDefault()
    65→      handleSend()
    66→    }
    67→  }
    68→
    69→  const handleChange = (e: React.ChangeEvent<HTMLTextAreaElement>) => {
    70→    setText(e.target.value)
    71→    if (e.target.value.length > 0) sendTyping(true)
    72→    else sendTyping(false)
    73→  }
    74→
    75→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    76→    const file = e.target.files?.[0]
    77→    if (!file) return
    78→    setUploading(true)
    79→    try {
    80→      const formData = new FormData()
    81→      formData.append('file', file)
    82→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
    83→      if (!res.ok) {
    84→        const text = await res.text()
    85→        throw new Error(text.slice(0, 120) || 'Upload failed')
    86→      }
    87→      const data = await res.json()
    88→      await send({
    89→        body: data.type.startsWith('image') ? '📷' : '📎',
    90→        mediaUrl: data.url,
    91→        mediaType: data.type,
    92→      })
    93→    } catch (e: any) {
    94→      toast.error(e.message || 'Upload failed')
    95→    } finally {
    96→      setUploading(false)
    97→      e.target.value = ''
    98→    }
    99→  }
   100→
   101→  return (
   102→    <div className="border-t border-border/60 bg-muted/50 backdrop-blur-2xl px-3 md:px-6 pt-3 pb-3 pb-safe relative z-20 shadow-[0_-8px_24px_-12px_rgba(0,0,0,0.25)]">
   103→      {/* Reply banner */}
   104→      {replyTo && (
   105→        <div className="flex items-center gap-3 text-xs bg-background/70 border border-border/60 backdrop-blur-md rounded-xl p-3 mb-2.5 shadow-sm">
   106→          <Reply className="w-4 h-4 shrink-0 text-primary" />
   107→          <div className="flex-1 min-w-0">
   108→            <div className="font-semibold text-primary">{replyTo.senderName}</div>
   109→            <div className="text-muted-foreground truncate mt-0.5">{replyTo.body}</div>
   110→          </div>
   111→          <button onClick={() => setReplyTo(null)} className="p-1.5 hover:bg-muted rounded-full transition-colors">
   112→            <X className="w-4 h-4" />
   113→          </button>
   114→        </div>
   115→      )}
   116→
   117→      {/* Cohesive glassmorphic composer bar — wraps the action buttons + input
   118→          in a single translucent pill so they read as one unified control. */}
   119→      <div className="flex items-end gap-1.5 bg-background/60 backdrop-blur-xl rounded-[26px] p-1.5 pl-2 border border-border/60 shadow-lg ring-1 ring-black/5 transition-shadow focus-within:shadow-xl">
   120→        {/* Upload button */}
   121→        <label
   122→          className="cursor-pointer w-10 h-10 flex items-center justify-center rounded-full transition-all shrink-0 text-muted-foreground hover:bg-primary/10 hover:text-primary active:scale-90"
   123→          title="Upload image, video, or audio"
   124→        >
   125→          <ImageIcon className="w-[22px] h-[22px] transition-colors" />
   126→          <input type="file" className="hidden" accept="image/*,video/*,audio/*" onChange={handleUpload} disabled={uploading} />
   127→        </label>
   128→
   129→        {/* TTS voice message button */}
   130→        <button
   131→          onClick={() => setTtsOpen(true)}
   132→          className="w-10 h-10 flex items-center justify-center rounded-full transition-all shrink-0 text-muted-foreground hover:bg-primary/10 hover:text-primary active:scale-90"
   133→          title="Send AI voice message"
   134→        >
   135→          <AudioLines className="w-[22px] h-[22px] transition-colors" />
   136→        </button>
   137→
   138→        {/* Text input — grows up to ~4 lines */}
   139→        <div className="flex-1 min-w-0 bg-transparent rounded-2xl px-3 py-2 flex items-end focus-within:bg-white/5 transition-colors">
   140→          <Textarea
   141→            ref={textareaRef}
   142→            value={text}
   143→            onChange={handleChange}
   144→            onKeyDown={handleKeyDown}
   145→            placeholder="iMessage..."
   146→            className="flex-1 resize-none min-h-[24px] max-h-[160px] bg-transparent border-0 focus-visible:ring-0 focus-visible:ring-offset-0 p-0 text-[16px] leading-relaxed shadow-none placeholder:text-muted-foreground/60"
   147→            disabled={sending || uploading}
   148→            rows={1}
   149→          />
   150→        </div>
   151→
   152→        {/* Send button */}
   153→        <Button
   154→          onClick={handleSend}
   155→          disabled={!text.trim() || sending}
   156→          size="icon"
   157→          className="rounded-full h-11 w-11 shrink-0 transition-transform hover:scale-105 active:scale-95 gradient-primary shadow-glow border border-primary/30 disabled:opacity-40 disabled:shadow-none disabled:scale-100"
   158→        >
   159→          {sending ? <Loader2 className="w-5 h-5 animate-spin" /> : <Send className="w-5 h-5" />}
   160→        </Button>
   161→      </div>
   162→
   163→      {/* TTS Voice Message Dialog */}
   164→      <TtsDialog
   165→        open={ttsOpen}
   166→        onOpenChange={setTtsOpen}
   167→        onSend={async (url) => {
   168→          await send({
   169→            body: '🎙️ AI voice message',
   170→            mediaUrl: url,
   171→            mediaType: 'audio',
   172→          })
   173→          setTtsOpen(false)
   174→          toast.success('Voice message sent')
   175→        }}
   176→      />
   177→    </div>
   178→  )
   179→}

   187→  onOpenChange,
   188→  onSend,
   189→}: {
   190→  open: boolean
   191→  onOpenChange: (o: boolean) => void
   192→  onSend: (url: string) => Promise<void>
   193→}) {
   194→  const [tab, setTab] = useState<'generate' | 'voices'>('generate')
   195→  const [text, setText] = useState('')

   227→        method: 'POST',

   279→        </DialogHeader>
   280→
   281→        {/* Tabs */}
   282→        <div className="flex gap-1 p-1 bg-muted rounded-lg">
   283→          <button
   284→            onClick={() => setTab('generate')}
   285→            className={cn(
   286→              'flex-1 py-1.5 text-sm font-medium rounded-md transition-colors',
   287→              tab === 'generate' ? 'bg-background text-foreground' : 'text-muted-foreground'
   288→            )}
   289→          >
   290→            Generate
   291→          </button>
   292→          <button
   293→            onClick={() => setTab('voices')}
   294→            className={cn(
   295→              'flex-1 py-1.5 text-sm font-medium rounded-md transition-colors flex items-center justify-center gap-1.5',
   296→              tab === 'voices' ? 'bg-background text-foreground' : 'text-muted-foreground'
   297→            )}
   298→          >
   299→            My Voices
   300→            {customVoices.length > 0 && (
   301→              <span className="text-[10px] bg-primary/20 text-primary px-1.5 py-0.5 rounded-full">
   302→                {customVoices.length}
   303→              </span>
   304→            )}
   305→          </button>
   306→        </div>
   307→
   308→        {tab === 'generate' ? (
   309→          <div className="space-y-4">
   310→            <div className="space-y-2">
   311→              <Label>Message</Label>
   312→              <Textarea
   313→                value={text}
   314→                onChange={(e) => setText(e.target.value)}
   315→                placeholder="Type what the voice should say..."
   316→                rows={3}

   333→                        'flex-1 py-2 px-3 rounded-lg text-sm font-medium transition-colors border',
   334→                        !selectedCustomVoiceId

   401→                    <SelectItem value="estelle">Estelle (French, Female)</SelectItem>

   471→/**
   472→ * Custom Voices tab — lets the user record/upload a voice clip and save it
   473→ * as a custom voice model for one-shot voice cloning.
   474→ */
   475→function CustomVoicesTab({ onUseVoice }: { onUseVoice: (id: string) => void }) {
   476→  const qc = useQueryClient()
   477→  const [name, setName] = useState('')
   478→  const [audioUrl, setAudioUrl] = useState<string | null>(null)
   479→  const [uploading, setUploading] = useState(false)
   480→  const [recording, setRecording] = useState(false)
   481→  const [mediaRecorder, setMediaRecorder] = useState<MediaRecorder | null>(null)
   482→  const [audioChunks, setAudioChunks] = useState<Blob[]>([])
   483→
   484→  const { data: customVoicesData } = useQuery({
   485→    queryKey: ['tts-voices'],
   486→    queryFn: async () => {
   487→      const res = await fetch('/api/tts/voices')
   488→      if (!res.ok) throw new Error('failed')
   489→      return res.json()
   490→    },
   491→  })
   492→  const customVoices = customVoicesData?.voices || []
   493→
   494→  const deleteVoice = useMutation({
   495→    mutationFn: async (id: string) => {
   496→      await fetch(`/api/tts/voices/${id}`, { method: 'DELETE' })
   497→    },
   498→    onSuccess: () => {
   499→      qc.invalidateQueries({ queryKey: ['tts-voices'] })
   500→      toast.success('Voice deleted')
   501→    },
   502→  })
   503→
   504→  const saveVoice = useMutation({
   505→    mutationFn: async () => {
   506→      const res = await fetch('/api/tts/voices', {
   507→        method: 'POST',
   508→        headers: { 'Content-Type': 'application/json' },
   509→        body: JSON.stringify({ name, audioUrl }),
   510→      })
   511→      if (!res.ok) throw new Error('failed')
   512→      return res.json()
   513→    },
   514→    onSuccess: () => {
   515→      qc.invalidateQueries({ queryKey: ['tts-voices'] })
   516→      toast.success('Custom voice created')
   517→      setName('')
   518→      setAudioUrl(null)
   519→    },
   520→  })
   521→
   522→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   523→    const file = e.target.files?.[0]
   524→    if (!file) return
   525→    setUploading(true)
   526→    try {
   527→      const formData = new FormData()
   528→      formData.append('file', file)
   529→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   530→      if (!res.ok) throw new Error('Upload failed')
   531→      const data = await res.json()
   532→      setAudioUrl(data.url)
   533→    } catch (e: any) {
   534→      toast.error(e.message || 'Upload failed')
   535→    } finally {
   536→      setUploading(false)
   537→      e.target.value = ''
   538→    }
   539→  }
   540→
   541→  const startRecording = async () => {
   542→    try {
   543→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   544→      const recorder = new MediaRecorder(stream)
   545→      const chunks: Blob[] = []
   546→      recorder.ondataavailable = (e) => chunks.push(e.data)
   547→      recorder.onstop = async () => {
   548→        const blob = new Blob(chunks, { type: 'audio/webm' })
   549→        // Upload the recording
   550→        setUploading(true)
   551→        try {
   552→          const formData = new FormData()
   553→          formData.append('file', new File([blob], 'voice.webm', { type: 'audio/webm' }))
   554→          const res = await fetch('/api/upload', { method: 'POST', body: formData })
   555→          if (!res.ok) throw new Error('Upload failed')
   556→          const data = await res.json()
   557→          setAudioUrl(data.url)
   558→        } catch (e: any) {
   559→          toast.error(e.message || 'Upload failed')
   560→        } finally {
   561→          setUploading(false)
   562→        }
   563→        stream.getTracks().forEach((t) => t.stop())
   564→      }
   565→      recorder.start()
   566→      setMediaRecorder(recorder)
   567→      setRecording(true)
   568→      setAudioChunks(chunks)
   569→    } catch {
   570→      toast.error('Microphone access denied')
   571→    }
   572→  }
   573→
   574→  const stopRecording = () => {
   575→    if (mediaRecorder && mediaRecorder.state === 'recording') {
   576→      mediaRecorder.stop()
   577→      setRecording(false)
   578→    }
   579→  }
   580→
   581→  return (
   582→    <div className="space-y-4">
   583→      {/* Create new voice */}
   584→      <div className="space-y-3 p-3 rounded-xl bg-muted/50 border border-border/30">
   585→        <div className="flex items-center gap-2 text-sm font-medium">
   586→          <Mic className="w-4 h-4 text-primary" />
   587→          Create a Custom Voice
   588→        </div>
   589→        <p className="text-xs text-muted-foreground">
   590→          Upload or record a 10-30 second voice clip. We'll clone the voice so you can generate speech in it.
   591→        </p>
   592→        <div className="space-y-2">
   593→          <Input
   594→            value={name}
   595→            onChange={(e) => setName(e.target.value)}
   596→            placeholder="Voice name (e.g. My Voice)"
   597→          />
   598→        </div>
   599→        <div className="flex gap-2">
   600→          {!audioUrl ? (
   601→            <>
   602→              {!recording ? (
   603→                <Button onClick={startRecording} variant="outline" size="sm" className="flex-1">
   604→                  <Mic className="w-4 h-4 mr-1.5" />
   605→                  Record
   606→                </Button>
   607→              ) : (
   608→                <Button onClick={stopRecording} variant="destructive" size="sm" className="flex-1">
   609→                  <span className="w-2 h-2 rounded-full bg-white animate-pulse mr-1.5" />
   610→                  Stop
   611→                </Button>
   612→              )}
   613→              <label className="cursor-pointer flex-1">
   614→                <Button asChild variant="outline" size="sm" className="w-full">
   615→                  <span>
   616→                    <Plus className="w-4 h-4 mr-1.5" />
   617→                    Upload
   618→                  </span>
   619→                </Button>
   620→                <input type="file" className="hidden" accept="audio/*" onChange={handleUpload} disabled={uploading || recording} />
   621→              </label>
   622→            </>
   623→          ) : (
   624→            <div className="flex-1 space-y-2">
   625→              <audio controls src={audioUrl} className="w-full h-8" />
   626→              <div className="flex gap-2">
   627→                <Button
   628→                  onClick={() => saveVoice.mutate()}
   629→                  disabled={!name.trim() || saveVoice.isPending}
   630→                  size="sm"
   631→                  className="flex-1 gradient-primary"
   632→                >
   633→                  {saveVoice.isPending ? (
   634→                    <Loader2 className="w-4 h-4 mr-1.5 animate-spin" />
   635→                  ) : (
   636→                    <Sparkles className="w-4 h-4 mr-1.5" />
   637→                  )}
   638→                  Save voice
   639→                </Button>
   640→                <Button
   641→                  onClick={() => setAudioUrl(null)}
   642→                  variant="outline"
   643→                  size="sm"
   644→                >
   645→                  Discard
   646→                </Button>
   647→              </div>
   648→            </div>
   649→          )}
   650→        </div>
   651→        {uploading && (
   652→          <div className="text-xs text-muted-foreground flex items-center gap-1.5">
   653→            <Loader2 className="w-3 h-3 animate-spin" />
   654→            Uploading...
   655→          </div>
   656→        )}
   657→      </div>
   658→
   659→      {/* Existing custom voices */}
   660→      {customVoices.length > 0 && (
   661→        <div className="space-y-2">
   662→          <Label className="text-xs uppercase tracking-wider text-muted-foreground">
   663→            Saved Voices
   664→          </Label>
   665→          <div className="space-y-1.5 max-h-48 overflow-y-auto">
   666→            {customVoices.map((v: any) => (
   667→              <div
   668→                key={v.id}
   669→                className="flex items-center gap-3 p-2.5 rounded-xl bg-card border border-border/30 hover:border-primary/30 transition-colors group"
   670→              >
   671→                <div className="w-9 h-9 rounded-full bg-primary/15 flex items-center justify-center shrink-0">
   672→                  <User className="w-4 h-4 text-primary" />
   673→                </div>
   674→                <div className="flex-1 min-w-0">
   675→                  <div className="text-sm font-medium truncate">{v.name}</div>
   676→                  <div className="text-xs text-muted-foreground">
   677→                    Created {new Date(v.createdAt).toLocaleDateString()}
   678→                  </div>
   679→                </div>
   680→                <button
   681→                  onClick={() => onUseVoice(v.id)}
   682→                  className="text-xs px-2.5 py-1.5 rounded-lg bg-primary/10 text-primary hover:bg-primary/20 transition-colors font-medium"
   683→                >
   684→                  Use
   685→                </button>
   686→                <button
   687→                  onClick={() => {
   688→                    if (confirm(`Delete "${v.name}"?`)) deleteVoice.mutate(v.id)
   689→                  }}
   690→                  className="p-1.5 rounded-lg text-red-400 hover:bg-red-500/10 transition-colors"
   691→                >
   692→                  <Trash2 className="w-3.5 h-3.5" />
   693→                </button>
   694→              </div>
   695→            ))}
   696→          </div>
   697→        </div>
   698→      )}
   699→
   700→      {customVoices.length === 0 && !audioUrl && (
   701→        <div className="text-center py-6">
   702→          <div className="w-12 h-12 mx-auto rounded-2xl bg-primary/10 flex items-center justify-center mb-2">
   703→            <Mic className="w-5 h-5 text-primary" />
   704→          </div>
   705→          <p className="text-sm text-muted-foreground">No custom voices yet</p>
   706→        </div>
   707→      )}
   708→    </div>
   709→  )
   710→}
