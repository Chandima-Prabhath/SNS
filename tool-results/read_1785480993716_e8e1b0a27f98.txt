     1→/**
     2→ * Visual Bot Builder — Types & Execution Engine (v2)
     3→ *
     4→ * Architecture
     5→ * ────────────
     6→ * A bot flow is a directed graph stored as JSON in `Bot.flow`:
     7→ *   { nodes: FlowNode[], edges: FlowEdge[] }
     8→ *
     9→ * The engine walks the graph starting from a TRIGGER node. Crucially, it
    10→ * supports **pause/resume** for INPUT and CHOICE nodes — when the engine
    11→ * reaches one, it stops, persists `{ currentNodeId, variables }` to
    12→ * ConversationSession, and resumes on the user's next message.
    13→ *
    14→ * Node categories (used by the UI for palette grouping):
    15→ *   - trigger  : starts the flow (1 type, multiple subtypes)
    16→ *   - output   : bot sends something (message, choice, typing)
    17→ *   - input    : wait for user reply (text input, choice input)
    18→ *   - logic    : control flow (condition, set_var, delay, stop)
    19→ *   - advanced : power features (api_call, random)
    20→ */
    21→
    22→// ─────────────────────────────────────────────────────────────────────────────
    23→// Node type catalog

    72→export interface FlowNodeData {
    73→  // ── trigger ──
    74→  triggerType?: 'any_message' | 'command' | 'mention'
    75→  command?: string // without leading slash
    76→
    77→  // ── message ──
    78→  text?: string
    79→
    80→  // ── wait_choice ──
    81→  prompt?: string
    82→  options?: string[] // quick-reply choices
    83→  variableName?: string // where to store the reply
    84→
    85→  // ── typing ──
    86→  seconds?: number
    87→
    88→  // ── input ──
    89→  // (uses prompt + variableName from above)
    90→
    91→  // ── condition ──
    92→  variable?: string
    93→  operator?: 'equals' | 'not_equals' | 'contains' | 'starts_with' | 'exists' | 'not_exists'
    94→  value?: string
    95→
    96→  // ── set_var ──
    97→  // (uses variable + value from above)
    98→
    99→  // ── api_call ──
   100→  url?: string
   101→  method?: 'GET' | 'POST'
   102→  headers?: string // JSON string
   103→  body?: string
   104→  // (also uses variableName to store response)
   105→
   106→  // ── random ──
   107→  // (no data — uses N outgoing edges; engine picks one)
   108→
   109→  // ── ai_generate ──
   110→  /** The prompt to send to the LLM. Supports {{var}} interpolation. */
   111→  aiPrompt?: string
   112→  /** Optional system prompt to set the LLM's persona/behavior. */
   113→  aiSystemPrompt?: string
   114→  /** Ollama model name, e.g. 'gemma3:270m'. */
   115→  aiModel?: string
   116→  /** Sampling temperature (0 = deterministic, 1 = creative). */
   117→  aiTemperature?: number
   118→  /** Max tokens to generate. */
   119→  aiMaxTokens?: number
   120→  // (also uses variableName to store the generated response)
   121→
   122→  // ── switch_case ──
   123→  /** Variable to switch on. */
   124→  switchVariable?: string
   125→  /** Case values — each gets its own outgoing handle (case_0, case_1, ...). */
   126→  cases?: string[]
   127→  // (uses 'default' handle for unmatched values)
   128→
   129→  // ── counter ──
   130→  // (uses `variable` to hold the counter name)
   131→  /** Amount to increment by (negative = decrement). */
   132→  increment?: number
   133→  /** Starting value when the variable doesn't exist yet. */
   134→  startValue?: number
   135→
   136→  // ── format_string ──
   137→  // (uses `text` as the template with {{var}} placeholders)
   138→  // (uses `variableName` to store the result)
   139→
   140→  // ── send_media ──
   141→  /** URL of the media to send (image/video/audio). */
   142→  mediaUrl?: string
   143→  /** Caption text. Supports {{var}} interpolation. */
   144→  caption?: string
   145→  /** Media type: image, video, audio. */
   146→  mediaType?: 'image' | 'video' | 'audio'
   147→
   148→  // ── log ──
   149→  /** Message to log. Supports {{var}} interpolation. */
   150→  logMessage?: string
   151→  /** Log level. */
   152→  logLevel?: 'info' | 'warn' | 'error'
   153→
   154→  // ── tts ──
   155→  /** Text to speak. Supports {{var}} interpolation. */
   156→  ttsText?: string
   157→  /** Voice name (alba, charles, etc.) or custom voice ID. */
   158→  ttsVoice?: string
   159→  // (sends the audio as a voice message — no variable needed)
   160→
   161→  // ── UI metadata (not used by engine) ──
   162→  /** Node type — stored in data so the editor's CustomNode can read it.
   163→   *  Duplicates FlowNode.type but is required because ReactFlow sets
   164→   *  n.type to 'custom' for all custom nodes. */
   165→  type?: NodeType
   166→  label?: string
   167→}

   180→export interface BotExecutionContext {
   181→  channelId: string
   182→  senderId: string
   183→  senderName: string
   184→  messageId: string
   185→  body: string
   186→  args: string[]
   187→  command?: string
   188→  isMention: boolean
   189→
   190→  /**
   191→   * Variables for the current execution. The caller (visual.ts) is responsible
   192→   * for loading these from ConversationSession at the start and persisting
   193→   * them when execution pauses.
   194→   */
   195→  variables: Record<string, string>
   196→
   197→  /** Helpers — visual.ts wires these to ConversationSession. */
   198→  getState?: () => Promise<any>
   199→  setState?: (state: any) => Promise<void>
   200→
   201→  /** Reply helper — pushes a message as the bot. Optional keyboard attaches
   202→   *  Telegram-style inline buttons to the message. Returns the message ID. */
   203→  reply: (text: string, keyboard?: BotKeyboardButton[][]) => Promise<string>
   204→
   205→  /** Reply with media (image/video/audio). The URL should be a path like
   206→   *  /api/uploads/xxx.wav. Returns the message ID. */
   207→  replyWithMedia?: (mediaUrl: string, mediaType: string, caption?: string) => Promise<string>
   208→
   209→  /** Edit an existing bot message in-place (body + keyboard). Used for
   210→   *  Telegram-style keyboard updates when a wait_choice loops back. */
   211→  editMessage?: (messageId: string, text: string, keyboard?: BotKeyboardButton[][]) => Promise<void>
   212→
   213→  /** Generate TTS audio from text using Pocket TTS. Returns the media URL
   214→   *  (e.g. /api/uploads/tts-bot-xxx.wav) or null if generation failed.
   215→   *  Server-only — debug mode should return null and let the engine fall
   216→   *  back to sending text. */
   217→  generateTTS?: (text: string, voice: string) => Promise<string | null>
   218→
   219→  /** Show typing indicator — best-effort, no-op if unsupported. */
   220→  setTyping?: (seconds: number) => Promise<void>
   221→}

   242→// ─────────────────────────────────────────────────────────────────────────────
   243→// Execution trace — emitted by the engine for debugging/test-runs
   244→// ─────────────────────────────────────────────────────────────────────────────
   245→
   246→export type TraceEvent =
   247→  | { type: 'flow_start'; timestamp: number; triggerNodeId?: string; input: { body: string; sender: string; command?: string } }
   248→  | { type: 'node_enter'; timestamp: number; nodeId: string; nodeType: NodeType; nodeLabel: string }
   249→  | { type: 'node_exit'; timestamp: number; nodeId: string; durationMs: number }
   250→  | { type: 'message_sent'; timestamp: number; nodeId: string; text: string }
   251→  | { type: 'variable_set'; timestamp: number; nodeId: string; variable: string; value: string }
   252→  | { type: 'condition_eval'; timestamp: number; nodeId: string; variable: string; value: string; operator: string; compareTo: string; result: boolean }
   253→  | { type: 'branch_taken'; timestamp: number; nodeId: string; handle: string; targetNodeId: string }
   254→  | { type: 'ai_call'; timestamp: number; nodeId: string; model: string; prompt: string; responseLength: number; durationMs: number }
   255→  | { type: 'api_call'; timestamp: number; nodeId: string; url: string; method: string; status: number; durationMs: number }
   256→  | { type: 'paused'; timestamp: number; nodeId: string; variableName: string }
   257→  | { type: 'resumed'; timestamp: number; nodeId: string; inputText: string }
   258→  | { type: 'error'; timestamp: number; nodeId?: string; message: string }
   259→  | { type: 'flow_end'; timestamp: number; reason: 'stop' | 'no_edge' | 'max_steps' | 'error'; sentCount: number }
   260→  | { type: 'log'; timestamp: number; nodeId: string; level: 'info' | 'warn' | 'error'; message: string }

   262→// ─────────────────────────────────────────────────────────────────────────────
   263→// Engine
   264→// ─────────────────────────────────────────────────────────────────────────────
   265→
   266→const MAX_STEPS = 50
   267→
   268→/**
   269→ * Resume descriptor — when set, the engine resumes from `nodeId` instead of
   270→ * searching for a trigger. The caller passes this when a paused session
   271→ * receives its awaited reply.
   272→ */
   273→export interface ResumeDescriptor {
   274→  nodeId: string
   275→  /** The text to assign to the input variable (usually the user's reply). */
   276→  inputText: string
   277→  /** The variable name to assign inputText to. */
   278→  inputVariable: string
   279→}
   280→
   281→/**
   282→ * Replace {{var}} placeholders in a string.
   283→ */
   284→function interpolate(text: string, ctx: BotExecutionContext): string {
   285→  let out = text
   286→  for (const [k, v] of Object.entries(ctx.variables)) {
   287→    out = out.replace(new RegExp(`\\{\\{${k}\\}\\}`, 'g'), v)
   288→  }
   289→  out = out.replace(/\{\{sender\}\}/g, ctx.senderName)
   290→  out = out.replace(/\{\{args\}\}/g, ctx.args.join(' '))
   291→  out = out.replace(/\{\{body\}\}/g, ctx.body)
   292→  return out
   293→}

   295→/**
   296→ * Execute a bot flow. Returns when:
   297→ *   - the flow reaches a `stop` node,
   298→ *   - the flow ends naturally (no outgoing edge),
   299→ *   - the flow pauses at an INPUT/WAIT_CHOICE node,
   300→ *   - MAX_STEPS is exceeded.
   301→ *
   302→ * Emits trace events for debugging — each node entry/exit, variable change,
   303→ * message sent, condition evaluation, branch taken, and error is recorded.
   304→ * The caller can pass these to the UI for a test-run visualization.
   305→ */
   306→export async function executeBotFlow(
   307→  flow: BotFlow,
   308→  ctx: BotExecutionContext,
   309→  resume?: ResumeDescriptor
   310→): Promise<BotExecutionResult> {
   311→  let sentCount = 0
   312→  let steps = 0
   313→  let currentNode: FlowNode | undefined
   314→  const trace: TraceEvent[] = []
   315→  const now = () => Date.now()
   316→
   317→  // ── Helpers (declared early so the resume block can use them) ─────────
   318→  const edgesFrom = (nodeId: string, handle?: string | null) =>
   319→    flow.edges.filter((e) => {
   320→      if (e.source !== nodeId) return false
   321→      if (handle === undefined) return true
   322→      if (handle === null) return !e.sourceHandle
   323→      return e.sourceHandle === handle
   324→    })
   325→
   326→  const followEdge = (nodeId: string, handle?: string | null): FlowNode | undefined => {
   327→    const edges = edgesFrom(nodeId, handle)
   328→    if (edges.length === 0) return undefined
   329→    return flow.nodes.find((n) => n.id === edges[0].target)
   330→  }
   331→
   332→  // ── Resolve starting node ─────────────────────────────────────────────
   333→  if (resume) {
   334→    const inputNode = flow.nodes.find((n) => n.id === resume.nodeId)
   335→    if (!inputNode) {
   336→      trace.push({ type: 'error', timestamp: now(), message: 'resume target not found' })
   337→      return { sentCount: 0, paused: false, variables: ctx.variables, trace, error: 'resume target not found' }
   338→    }
   339→    ctx.variables[resume.inputVariable] = resume.inputText
   340→    trace.push({ type: 'resumed', timestamp: now(), nodeId: inputNode.id, inputText: resume.inputText })
   341→    trace.push({ type: 'variable_set', timestamp: now(), nodeId: inputNode.id, variable: resume.inputVariable, value: resume.inputText })
   342→    currentNode = followEdge(inputNode.id, null)
   343→    if (!currentNode) {
   344→      trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   345→      return { sentCount, paused: false, variables: ctx.variables, trace }
   346→    }
   347→  } else {
   348→    const trigger = flow.nodes.find((n) => n.type === 'trigger')
   349→    if (!trigger) {
   350→      trace.push({ type: 'error', timestamp: now(), message: 'No trigger node found' })
   351→      return { sentCount: 0, paused: false, variables: ctx.variables, trace, error: 'No trigger node found' }
   352→    }
   353→
   354→    trace.push({
   355→      type: 'flow_start',
   356→      timestamp: now(),
   357→      triggerNodeId: trigger.id,
   358→      input: { body: ctx.body, sender: ctx.senderName, command: ctx.command },
   359→    })
   360→
   361→    // Trigger matching
   362→    const tt = trigger.data.triggerType || 'any_message'
   363→    if (tt === 'command' && trigger.data.command) {
   364→      if (!ctx.command || ctx.command !== trigger.data.command.replace(/^\//, '')) {
   365→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   366→        return { sentCount: 0, paused: false, variables: ctx.variables, trace }
   367→      }
   368→    } else if (tt === 'mention') {
   369→      if (!ctx.isMention) {
   370→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   371→        return { sentCount: 0, paused: false, variables: ctx.variables, trace }
   372→      }
   373→    }
   374→
   375→    currentNode = trigger
   376→  }
   377→
   378→  // ── Walk the graph ────────────────────────────────────────────────────
   379→  while (currentNode && steps < MAX_STEPS) {
   380→    steps++
   381→    const nodeStartTime = now()
   382→    const nodeLabel = currentNode.data?.label || currentNode.type
   383→
   384→    trace.push({
   385→      type: 'node_enter',
   386→      timestamp: nodeStartTime,
   387→      nodeId: currentNode.id,
   388→      nodeType: currentNode.type,
   389→      nodeLabel,
   390→    })
   391→
   392→    switch (currentNode.type) {
   393→      // ── TRIGGER ────────────────────────────────────────────────────────
   394→      case 'trigger':
   395→        currentNode = followEdge(currentNode.id, null)
   396→        break
   397→
   398→      // ── OUTPUT: message ────────────────────────────────────────────────
   399→      case 'message': {
   400→        const text = interpolate(currentNode.data.text || '', ctx)
   401→        if (text.trim()) {
   402→          await ctx.reply(text)
   403→          sentCount++
   404→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text })
   405→        }
   406→        currentNode = followEdge(currentNode.id, null)
   407→        break
   408→      }
   409→
   410→      // ── OUTPUT: send_media ─────────────────────────────────────────────
   411→      case 'send_media': {
   412→        const url = interpolate(currentNode.data.mediaUrl || '', ctx)
   413→        const caption = interpolate(currentNode.data.caption || '', ctx)
   414→        if (url.trim()) {
   415→          // ctx.reply is text-only; we log it but the actual media send
   416→          // would need an extended reply API. For now, send the URL as text.
   417→          const text = caption ? `${caption}\n${url}` : url
   418→          await ctx.reply(text)
   419→          sentCount++
   420→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text })
   421→        }
   422→        currentNode = followEdge(currentNode.id, null)
   423→        break
   424→      }
   425→
   426→      // ── OUTPUT: typing ─────────────────────────────────────────────────
   427→      case 'typing': {
   428→        const seconds = currentNode.data.seconds || 1
   429→        if (ctx.setTyping) await ctx.setTyping(seconds)
   430→        else await new Promise((r) => setTimeout(r, Math.min(seconds, 2) * 1000))
   431→        currentNode = followEdge(currentNode.id, null)
   432→        break
   433→      }
   434→
   435→      // ── INPUT: pause and wait for next message ────────────────────────
   436→      case 'input': {
   437→        const prompt = interpolate(currentNode.data.prompt || '', ctx)
   438→        if (prompt.trim()) {
   439→          await ctx.reply(prompt)
   440→          sentCount++
   441→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text: prompt })
   442→        }
   443→        trace.push({ type: 'paused', timestamp: now(), nodeId: currentNode.id, variableName: currentNode.data.variableName || 'reply' })
   444→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   445→        return {
   446→          sentCount,
   447→          paused: true,
   448→          pausedAtNodeId: currentNode.id,
   449→          variables: ctx.variables,
   450→          trace,
   451→        }
   452→      }
   453→
   454→      // ── INPUT: wait_choice ────────────────────────────────────────────
   455→      case 'wait_choice': {
   456→        const prompt = interpolate(currentNode.data.prompt || '', ctx)
   457→        const options = currentNode.data.options || []
   458→        // Build a Telegram-style inline keyboard — each option is a button.
   459→        // The button's callbackData is the option text itself, so the resume
   460→        // path in visual.ts can match it directly.
   461→        const keyboard: BotKeyboardButton[][] = options.map((opt) => [{
   462→          text: opt,
   463→          callbackData: opt,
   464→        }])
   465→
   466→        // Telegram-style behavior: if we have an existing keyboard message
   467→        // (from a previous pause at this node, i.e. a loop-back), EDIT it
   468→        // in-place instead of sending a new message. This prevents the bot
   469→        // from spamming new keyboard messages every time the user clicks a
   470→        // button that loops back to this menu.
   471→        //
   472→        // The keyboardMessageId is set in the session state by visual.ts
   473→        // when it persists the pause. We read it from ctx.variables under
   474→        // a reserved key.
   475→        const existingKeyboardMsgId = ctx.variables['__keyboardMsgId']
   476→
   477→        if (prompt.trim()) {
   478→          if (existingKeyboardMsgId && ctx.editMessage) {
   479→            // Edit-in-place: update the existing keyboard message
   480→            await ctx.editMessage(existingKeyboardMsgId, prompt, keyboard)
   481→            trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text: `[edited] ${prompt}` })
   482→          } else {
   483→            // Fresh keyboard: send a new message
   484→            const msgId = await ctx.reply(prompt, keyboard)
   485→            sentCount++
   486→            // Store the message ID so future re-pauses can edit it
   487→            ctx.variables['__keyboardMsgId'] = msgId
   488→            trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text: prompt })
   489→          }
   490→        }
   491→        trace.push({ type: 'paused', timestamp: now(), nodeId: currentNode.id, variableName: currentNode.data.variableName || 'choice' })
   492→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   493→        return {
   494→          sentCount,
   495→          paused: true,
   496→          pausedAtNodeId: currentNode.id,
   497→          variables: ctx.variables,
   498→          trace,
   499→        }
   500→      }
   501→
   502→      // ── LOGIC: condition ──────────────────────────────────────────────
   503→      case 'condition': {
   504→        const varName = currentNode.data.variable || ''
   505→        const varValue = ctx.variables[varName] || ''
   506→        const op = currentNode.data.operator || 'exists'
   507→        const cmp = currentNode.data.value || ''
   508→
   509→        let met = false
   510→        switch (op) {
   511→          case 'equals':       met = varValue === cmp; break
   512→          case 'not_equals':   met = varValue !== cmp; break
   513→          case 'contains':     met = varValue.includes(cmp); break
   514→          case 'starts_with':  met = varValue.startsWith(cmp); break
   515→          case 'exists':       met = !!varValue; break
   516→          case 'not_exists':   met = !varValue; break
   517→        }
   518→
   519→        trace.push({
   520→          type: 'condition_eval',
   521→          timestamp: now(),
   522→          nodeId: currentNode.id,
   523→          variable: varName,
   524→          value: varValue,
   525→          operator: op,
   526→          compareTo: cmp,
   527→          result: met,
   528→        })
   529→
   530→        const handle = met ? 'true' : 'false'
   531→        const next = followEdge(currentNode.id, handle)
   532→        if (next) {
   533→          trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle, targetNodeId: next.id })
   534→          currentNode = next
   535→        } else {
   536→          // No matching branch — fall through to default edge if any
   537→          const fallback = followEdge(currentNode.id, null)
   538→          if (fallback) {
   539→            trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: 'default', targetNodeId: fallback.id })
   540→            currentNode = fallback
   541→          } else {
   542→            trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   543→            currentNode = undefined
   544→          }
   545→        }
   546→        break
   547→      }
   548→
   549→      // ── LOGIC: switch_case (multi-branch) ─────────────────────────────
   550→      case 'switch_case': {
   551→        const varName = currentNode.data.switchVariable || ''
   552→        const varValue = ctx.variables[varName] || ''
   553→        const cases = currentNode.data.cases || []
   554→
   555→        const matchIdx = cases.findIndex((c) => c === varValue)
   556→        const handle = matchIdx >= 0 ? `case_${matchIdx}` : 'default'
   557→
   558→        trace.push({
   559→          type: 'condition_eval',
   560→          timestamp: now(),
   561→          nodeId: currentNode.id,
   562→          variable: varName,
   563→          value: varValue,
   564→          operator: 'switch',
   565→          compareTo: matchIdx >= 0 ? cases[matchIdx] : '(default)',
   566→          result: matchIdx >= 0,
   567→        })
   568→
   569→        const next = followEdge(currentNode.id, handle)
   570→        if (next) {
   571→          trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle, targetNodeId: next.id })
   572→          currentNode = next
   573→        } else {
   574→          const fallback = followEdge(currentNode.id, null)
   575→          if (fallback) {
   576→            trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: 'default', targetNodeId: fallback.id })
   577→            currentNode = fallback
   578→          } else {
   579→            trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   580→            currentNode = undefined
   581→          }
   582→        }
   583→        break
   584→      }
   585→
   586→      // ── LOGIC: set_var ────────────────────────────────────────────────
   587→      case 'set_var': {
   588→        const name = currentNode.data.variable || ''
   589→        const val = interpolate(currentNode.data.value || '', ctx)
   590→        if (name) {
   591→          ctx.variables[name] = val
   592→          trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: name, value: val })
   593→        }
   594→        currentNode = followEdge(currentNode.id, null)
   595→        break
   596→      }
   597→
   598→      // ── LOGIC: counter (increment/decrement a numeric variable) ───────
   599→      case 'counter': {
   600→        const name = currentNode.data.variable || ''
   601→        const inc = currentNode.data.increment ?? 1
   602→        const startVal = currentNode.data.startValue ?? 0
   603→        const current = parseInt(ctx.variables[name] || '', 10)
   604→        const newVal = isNaN(current) ? startVal + inc : current + inc
   605→        ctx.variables[name] = String(newVal)
   606→        trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: name, value: String(newVal) })
   607→        currentNode = followEdge(currentNode.id, null)
   608→        break
   609→      }
   610→
   611→      // ── LOGIC: format_string (template a string with variables) ──────
   612→      case 'format_string': {
   613→        const template = currentNode.data.text || ''
   614→        const result = interpolate(template, ctx)
   615→        const vname = currentNode.data.variableName
   616→        if (vname) {
   617→          ctx.variables[vname] = result
   618→          trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: result })
   619→        }
   620→        currentNode = followEdge(currentNode.id, null)
   621→        break
   622→      }
   623→
   624→      // ── LOGIC: delay ──────────────────────────────────────────────────
   625→      case 'delay': {
   626→        const s = Math.max(0, Math.min(60, currentNode.data.seconds || 1))
   627→        await new Promise((r) => setTimeout(r, s * 1000))
   628→        currentNode = followEdge(currentNode.id, null)
   629→        break
   630→      }
   631→
   632→      // ── LOGIC: log (debug logging — shows in trace, no user output) ───
   633→      case 'log': {
   634→        const msg = interpolate(currentNode.data.logMessage || '', ctx)
   635→        const level = currentNode.data.logLevel || 'info'
   636→        trace.push({ type: 'log', timestamp: now(), nodeId: currentNode.id, level, message: msg })
   637→        currentNode = followEdge(currentNode.id, null)
   638→        break
   639→      }
   640→
   641→      // ── LOGIC: stop ───────────────────────────────────────────────────
   642→      case 'stop':
   643→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'stop', sentCount })
   644→        return { sentCount, paused: false, variables: ctx.variables, trace }
   645→
   646→      // ── OUTPUT: tts (Pocket TTS voice message) ────────────────────────
   647→      case 'tts': {
   648→        const ttsStart = now()
   649→        const ttsNode = currentNode
   650→        const text = interpolate(ttsNode.data.ttsText || '', ctx)
   651→        const voice = ttsNode.data.ttsVoice || 'alba'
   652→
   653→        if (!text.trim()) {
   654→          trace.push({ type: 'error', timestamp: now(), nodeId: ttsNode.id, message: 'TTS node has empty text' })
   655→          currentNode = followEdge(ttsNode.id, null)
   656→          break
   657→        }
   658→
   659→        trace.push({ type: 'log', timestamp: now(), nodeId: ttsNode.id, level: 'info', message: `TTS: generating "${text.slice(0, 60)}…" with voice=${voice}` })
   660→
   661→        let mediaUrl: string | null = null
   662→        if (ctx.generateTTS) {
   663→          try {
   664→            mediaUrl = await ctx.generateTTS(text, voice)
   665→          } catch (e: any) {
   666→            trace.push({ type: 'error', timestamp: now(), nodeId: ttsNode.id, message: `TTS failed: ${e?.message || e}` })
   667→          }
   668→        } else {
   669→          // Debug mode — no actual TTS available
   670→          trace.push({ type: 'error', timestamp: now(), nodeId: ttsNode.id, message: 'TTS not available in debug mode' })
   671→        }
   672→
   673→        if (mediaUrl) {
   674→          // Send as a voice message
   675→          if (ctx.replyWithMedia) {
   676→            await ctx.replyWithMedia(mediaUrl, 'audio')
   677→          } else {
   678→            await ctx.reply(`🔊 ${mediaUrl}`)
   679→          }
   680→          sentCount++
   681→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: ttsNode.id, text: `[voice message: ${text.slice(0, 60)}…]` })
   682→          trace.push({ type: 'ai_call', timestamp: now(), nodeId: ttsNode.id, model: `tts:${voice}`, prompt: text.slice(0, 100), responseLength: 0, durationMs: now() - ttsStart })
   683→        } else {
   684→          // Fallback: send the text as a plain message
   685→          await ctx.reply(`🔊 ${text}`)
   686→          sentCount++
   687→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: ttsNode.id, text: `[TTS fallback] ${text.slice(0, 80)}` })
   688→        }
   689→        currentNode = followEdge(ttsNode.id, null)
   690→        break
   691→      }
   692→
   693→      // ── ADVANCED: api_call ────────────────────────────────────────────
   694→      case 'api_call': {
   695→        const callStart = now()
   696→        try {
   697→          const url = interpolate(currentNode.data.url || '', ctx)
   698→          const method = currentNode.data.method || 'GET'
   699→          const headers = currentNode.data.headers ? JSON.parse(currentNode.data.headers) : {}
   700→          const body = currentNode.data.body ? interpolate(currentNode.data.body, ctx) : undefined
   701→          const res = await fetch(url, { method, headers, body })
   702→          const text = await res.text()
   703→          const vname = currentNode.data.variableName
   704→          if (vname) {
   705→            ctx.variables[vname] = text.slice(0, 4000)
   706→            trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: text.slice(0, 200) + (text.length > 200 ? '…' : '') })
   707→          }
   708→          trace.push({ type: 'api_call', timestamp: now(), nodeId: currentNode.id, url, method, status: res.status, durationMs: now() - callStart })
   709→        } catch (e: any) {
   710→          trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `API call failed: ${e?.message || e}` })
   711→        }
   712→        currentNode = followEdge(currentNode.id, null)
   713→        break
   714→      }
   715→
   716→      // ── ADVANCED: random ──────────────────────────────────────────────
   717→      case 'random': {
   718→        const allEdges = edgesFrom(currentNode.id, undefined)
   719→        if (allEdges.length === 0) {
   720→          trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   721→          currentNode = undefined
   722→          break
   723→        }
   724→        const pick = allEdges[Math.floor(Math.random() * allEdges.length)]
   725→        const target = flow.nodes.find((n) => n.id === pick.target)
   726→        if (target) {
   727→          trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: pick.sourceHandle || 'random', targetNodeId: target.id })
   728→          currentNode = target
   729→        } else {
   730→          currentNode = undefined
   731→        }
   732→        break
   733→      }
   734→
   735→      // ── ADVANCED: ai_generate ──────────────────────────────────────────
   736→      case 'ai_generate': {
   737→        const vname = currentNode.data.variableName || 'aiResponse'
   738→        const callStart = now()
   739→        try {
   740→          const model = currentNode.data.aiModel || 'gemma3:270m'
   741→          const prompt = interpolate(currentNode.data.aiPrompt || '', ctx)
   742→          const systemPrompt = currentNode.data.aiSystemPrompt
   743→            ? interpolate(currentNode.data.aiSystemPrompt, ctx)
   744→            : undefined
   745→          const temperature = currentNode.data.aiTemperature ?? 0.7
   746→          const maxTokens = currentNode.data.aiMaxTokens ?? 256
   747→
   748→          const ollamaUrl = process.env.OLLAMA_URL || 'http://localhost:11434'
   749→          const res = await fetch(`${ollamaUrl}/api/generate`, {
   750→            method: 'POST',
   751→            headers: { 'Content-Type': 'application/json' },
   752→            body: JSON.stringify({
   753→              model, prompt, system: systemPrompt, stream: false,
   754→              options: { temperature, num_predict: maxTokens },
   755→            }),
   756→          })
   757→
   758→          if (!res.ok) {
   759→            const errText = await res.text().catch(() => 'unknown error')
   760→            const errMsg = `Ollama error ${res.status}: ${errText.slice(0, 100)}`
   761→            ctx.variables[vname] = `[AI error: ${res.status}]`
   762→            trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: errMsg })
   763→            trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: ctx.variables[vname] })
   764→            trace.push({ type: 'ai_call', timestamp: now(), nodeId: currentNode.id, model, prompt: prompt.slice(0, 100), responseLength: 0, durationMs: now() - callStart })
   765→          } else {
   766→            const data = await res.json()
   767→            const generated: string = data.response || ''
   768→            ctx.variables[vname] = generated.slice(0, 4000)
   769→            trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: generated.slice(0, 200) + (generated.length > 200 ? '…' : '') })
   770→            trace.push({ type: 'ai_call', timestamp: now(), nodeId: currentNode.id, model, prompt: prompt.slice(0, 100), responseLength: generated.length, durationMs: now() - callStart })
   771→          }
   772→        } catch (e: any) {
   773→          ctx.variables[vname] = `[AI error: ${e?.message || 'unavailable'}]`
   774→          trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `AI generate failed: ${e?.message || e}` })
   775→          trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: ctx.variables[vname] })
   776→        }
   777→        currentNode = followEdge(currentNode.id, null)
   778→        break
   779→      }
   780→
   781→      default:
   782→        trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `unknown node type: ${currentNode.type}` })
   783→        return { sentCount, paused: false, variables: ctx.variables, trace, error: `unknown node type: ${currentNode.type}` }
   784→    }
   785→
   786→    // Record node exit
   787→    if (currentNode) {
   788→      trace.push({ type: 'node_exit', timestamp: now(), nodeId: currentNode.id, durationMs: 0 })
   789→    }
   790→  }
   791→
   792→  if (steps >= MAX_STEPS) {
   793→    await ctx.reply('⚠️ Bot flow exceeded maximum steps (possible infinite loop).')
   794→    trace.push({ type: 'error', timestamp: now(), message: 'max steps exceeded' })
   795→    trace.push({ type: 'flow_end', timestamp: now(), reason: 'max_steps', sentCount })
   796→    return { sentCount, paused: false, variables: ctx.variables, trace, error: 'max steps' }
   797→  }
   798→
   799→  trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   800→  return { sentCount, paused: false, variables: ctx.variables, trace }
   801→}

   960→    case 'input':
   961→      return { prompt: 'What is your name?', variableName: 'userName', label: 'Wait for Reply' }

   963→      return { prompt: 'Pick one:', options: ['Yes', 'No'], variableName: 'choice', label: 'Wait for Choice' }

   965→      return { variable: '', operator: 'exists', value: '', label: 'Condition' }

   967→      return { switchVariable: '', cases: ['yes', 'no'], label: 'Switch' }

   969→      return { variable: '', value: '', label: 'Set Variable' }
