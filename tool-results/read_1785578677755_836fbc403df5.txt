    19→/**
    20→ * Detect whether a media URL + mediaType combination represents an audio file.
    21→ *
    22→ * We check BOTH the mediaType AND the file extension, because some browsers
    23→ * tag WebM audio recordings as 'video/webm' (WebM is primarily a video
    24→ * container) or 'application/octet-stream' (if the type is lost in FormData
    25→ * upload). Without this check, voice messages would render as a black
    26→ * <video> square instead of the custom VoiceMessagePlayer.
    27→ */
    28→function isAudioFile(mediaUrl: string | null | undefined, mediaType: string | null | undefined): boolean {
    29→  if (mediaType?.startsWith('audio')) return true
    30→  if (!mediaUrl) return false
    31→  const AUDIO_EXTENSIONS = ['.webm', '.mp3', '.wav', '.ogg', '.m4a', '.flac', '.aac']
    32→  const lower = mediaUrl.toLowerCase()
    33→  return AUDIO_EXTENSIONS.some((ext) => lower.endsWith(ext))
    34→}

    40→export function MessageList({ channelId }: MessageListProps) {
    41→  const { messages, isLoading, send, edit, remove, sendCallback, typing, markRead, replyTo, setReplyTo } =
    42→    useChannel(channelId)
    43→  const { data: session } = useSession()
    44→  const myId = (session?.user as any)?.id
    45→  const scrollRef = useRef<HTMLDivElement>(null)
    46→  const lastMessageIdRef = useRef<string | null>(null)
    47→  const isAtBottomRef = useRef(true)
    48→
    49→  // Scroll-to-bottom button state.
    50→  //   - showScrollButton: visible when the user has scrolled >200px from the
    51→  //     bottom (matches the threshold in the task spec).
    52→  //   - unreadBelow: count of new messages that have arrived BELOW the user's
    53→  //     current scroll position. Resets to 0 when the user scrolls back to the
    54→  //     bottom (or clicks the button).
    55→  const [showScrollButton, setShowScrollButton] = useState(false)
    56→  const [unreadBelow, setUnreadBelow] = useState(0)
    57→
    58→  // Reset scroll state when switching channels — otherwise the auto-scroll
    59→  // logic thinks we already saw the last message and skips the initial scroll.
    60→  // Refs are reset here (no rerender needed); the button state is reset in
    61→  // the channel-switch derived-state check below to keep it in sync with
    62→  // React's render cycle.
    63→  useEffect(() => {
    64→    lastMessageIdRef.current = null
    65→    isAtBottomRef.current = true
    66→  }, [channelId])
    67→
    68→  // Reset the scroll-button React state when the channel changes. We use the
    69→  // "derived state" pattern (calling setState during render in a conditional)
    70→  // instead of useEffect+setState — React specifically endorses this for
    71→  // "reset state when a prop changes" because it avoids the extra render
    72→  // cycle that useEffect+setState triggers.
    73→  // https://react.dev/reference/react/useState#storing-information-from-previous-renders
    74→  const [prevChannelId, setPrevChannelId] = useState(channelId)
    75→  if (prevChannelId !== channelId) {
    76→    setPrevChannelId(channelId)
    77→    setShowScrollButton(false)
    78→    setUnreadBelow(0)
    79→  }
    80→
    81→  // Track whether the user is at the bottom of the scroll container.
    82→  // We use a ref so the scroll handler doesn't trigger re-renders, but we
    83→  // also mirror `showScrollButton` into React state so the button can
    84→  // appear/disappear. The 80px threshold for "at bottom" matches the
    85→  // existing auto-scroll logic; the 200px threshold for showing the button
    86→  // is per the task spec.
    87→  useEffect(() => {
    88→    const el = scrollRef.current
    89→    if (!el) return
    90→    const onScroll = () => {
    91→      const distanceFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight
    92→      const atBottom = distanceFromBottom < 80
    93→      isAtBottomRef.current = atBottom
    94→      // Show the floating button only when scrolled meaningfully up
    95→      setShowScrollButton(distanceFromBottom > 200)
    96→      // Reset the "new messages below" counter when the user reaches the
    97→      // bottom — they've now seen everything.
    98→      if (atBottom) {
    99→        setUnreadBelow(0)
   100→      }
   101→    }
   102→    el.addEventListener('scroll', onScroll, { passive: true })
   103→    return () => el.removeEventListener('scroll', onScroll)
   104→  }, [])
   105→
   106→  // Auto-scroll — handles BOTH the initial load (channel switch / first
   107→  // messages arriving) and subsequent new messages.
   108→  //
   109→  // • Initial load: wait 200ms so avatars/images have time to render and
   110→  //   lay out before we measure scrollHeight. Double rAF was too short for
   111→  //   channels with media-heavy history (scroll would land on "yesterday"
   112→  //   instead of the latest message).
   113→  // • New message: use requestAnimationFrame so the DOM updates before we
   114→  //   measure. Scroll smoothly so the user sees the message slide in.
   115→  // • Own message: ALWAYS scroll, even if the user scrolled up (so sending
   116→  //   a message snaps back to the bottom).
   117→  // • New message while scrolled up: DON'T scroll, but increment the
   118→  //   "unread below" counter so the scroll-to-bottom button shows a badge.
   119→  //   Multiple messages can land in a single render (e.g., a burst of bot
   120→  //   replies) — count all of them by comparing indices.
   121→  useEffect(() => {
   122→    const el = scrollRef.current
   123→    if (!el) return
   124→    const lastMsg = messages[messages.length - 1]
   125→    if (!lastMsg) return
   126→    // Already processed this exact message — nothing to do.
   127→    if (lastMessageIdRef.current === lastMsg.id) return
   128→
   129→    const isMine = lastMsg.senderId === myId
   130→    const isInitialLoad = !lastMessageIdRef.current
   131→    const shouldScroll = isInitialLoad || isMine || isAtBottomRef.current
   132→
   133→    // Count how many new messages arrived since the last one we processed.
   134→    // If the previous last-seen id is still in the array, the delta is the
   135→    // number of messages after it. Fall back to 1 if we can't find it (e.g.,
   136→    // the array was refetched and the old id is gone).
   137→    let newCount = 1
   138→    if (lastMessageIdRef.current) {
   139→      const lastSeenIdx = messages.findIndex((m) => m.id === lastMessageIdRef.current)
   140→      if (lastSeenIdx >= 0) {
   141→        newCount = Math.max(1, messages.length - 1 - lastSeenIdx)
   142→      }
   143→    }
   144→
   145→    lastMessageIdRef.current = lastMsg.id
   146→    markRead(lastMsg.id)
   147→
   148→    if (!shouldScroll) {
   149→      // User is scrolled up — count this burst as new messages below the fold.
   150→      setUnreadBelow((n) => n + newCount)
   151→      return
   152→    }
   153→
   154→    if (isInitialLoad) {
   155→      // Give the browser time to lay out all the historical messages (and
   156→      // their media) before we jump. 200ms is enough for most avatars and
   157→      // thumbnails to load without making the channel switch feel slow.
   158→      const t = window.setTimeout(() => {
   159→        const e = scrollRef.current
   160→        if (e) e.scrollTop = e.scrollHeight
   161→      }, 200)
   162→      return () => window.clearTimeout(t)
   163→    }
   164→
   165→    // New message — wait one frame so the new bubble has rendered before
   166→    // we measure scrollHeight. Otherwise we'd scroll to the OLD bottom and
   167→    // the latest message would sit just below the fold.
   168→    const raf = requestAnimationFrame(() => {
   169→      const e = scrollRef.current
   170→      if (e) {
   171→        e.scrollTo({ top: e.scrollHeight, behavior: 'smooth' })
   172→      }
   173→    })
   174→    // We scrolled to the bottom — clear the unread-below counter.
   175→    setUnreadBelow(0)
   176→    return () => cancelAnimationFrame(raf)
   177→  }, [messages, myId, markRead])
   178→
   179→  // Group consecutive messages by same sender within 5 min
   180→  const grouped = useMemo(() => {
   181→    const groups: ChannelMessage[][] = []
   182→    let current: ChannelMessage[] = []
   183→    for (const m of messages) {
   184→      if (current.length === 0) {
   185→        current.push(m)
   186→      } else {
   187→        const prev = current[current.length - 1]
   188→        const sameSender = prev.senderId === m.senderId && prev.senderType === m.senderType
   189→        const timeDiff = new Date(m.createdAt).getTime() - new Date(prev.createdAt).getTime()
   190→        if (sameSender && timeDiff < 5 * 60 * 1000) {
   191→          current.push(m)
   192→        } else {
   193→          groups.push(current)
   194→          current = [m]
   195→        }
   196→      }
   197→    }
   198→    if (current.length > 0) groups.push(current)
   199→    return groups
   200→  }, [messages])
   201→
   202→  return (
   203→    <div className="flex-1 flex flex-col min-h-0 bg-background relative overflow-hidden">
   204→      {/* Subtle cinematic glow in the background */}
   205→      <div className="absolute inset-0 bg-[radial-gradient(ellipse_80%_80%_at_50%_-20%,rgba(var(--primary),0.05),transparent)] pointer-events-none" />
   206→      <div ref={scrollRef as any} className="flex-1 min-h-0 overflow-y-auto relative z-10 scroll-smooth">
   207→        <div className="px-3 md:px-6 py-4 space-y-3 max-w-4xl mx-auto">
   208→          {isLoading && <LoadingState />}
   209→
   210→          {!isLoading && messages.length === 0 && (
   211→            <div className="text-center py-20 px-4">
   212→              <div className="w-16 h-16 mx-auto rounded-3xl bg-primary/10 flex items-center justify-center mb-3 ring-1 ring-primary/15">
   213→                <Reply className="w-7 h-7 text-primary" strokeWidth={1.5} />
   214→              </div>
   215→              <p className="text-base font-semibold">No messages yet</p>
   216→              <p className="text-sm text-muted-foreground mt-1">Be the first to say hello</p>
   217→            </div>
   218→          )}
   219→
   220→          {grouped.map((group, gi) => {
   221→            const first = group[0]
   222→            const isMine = first.senderId === myId && first.senderType === 'user'
   223→            const isBot = first.senderType === 'bot'
   224→            const senderDeleted = first.senderType === 'user' && !first.sender
   225→            const senderName =
   226→              first.senderType === 'bot'
   227→                ? first.sender?.displayName || 'Bot'
   228→                : first.sender?.displayName || (senderDeleted ? 'Deleted User' : 'Unknown')
   229→            const dayLabel = formatDay(first.createdAt)
   230→            const prevGroup = gi > 0 ? grouped[gi - 1] : null
   231→            const showDayDivider = !prevGroup || formatDay(prevGroup[0].createdAt) !== dayLabel
   232→
   233→            return (
   234→              <div key={gi}>
   235→                {showDayDivider && (
   236→                  <div className="flex items-center gap-3 my-5">
   237→                    <div className="flex-1 h-px bg-border" />
   238→                    <span className="text-xs text-muted-foreground font-medium">{dayLabel}</span>
   239→                    <div className="flex-1 h-px bg-border" />
   240→                  </div>
   241→                )}
   242→                <MessageGroup
   243→                  group={group}
   244→                  isMine={isMine}
   245→                  isBot={isBot}
   246→                  senderDeleted={senderDeleted}
   247→                  senderName={senderName}
   248→                  avatarUrl={first.sender?.avatarUrl || null}
   249→                  myId={myId}
   250→                  onReply={(m) =>
   251→                    setReplyTo({ id: m.id, body: m.body, senderName })
   252→                  }
   253→                  onDelete={(m) => remove(m.id)}
   254→                  onEditSubmit={async (messageId, text) => edit(messageId, text)}
   255→                  onCallback={async (messageId, callbackData) => sendCallback(messageId, callbackData)}
   256→                />
   257→              </div>
   258→            )
   259→          })}
   260→
   261→          {/* Typing indicator */}
   262→          <AnimatePresence>
   263→            {typing.length > 0 && (
   264→              <motion.div
   265→                initial={{ opacity: 0, y: 8 }}
   266→                animate={{ opacity: 1, y: 0 }}
   267→                exit={{ opacity: 0, y: 8 }}
   268→                className="flex items-center gap-2 text-xs text-muted-foreground italic px-2"
   269→              >
   270→                <span className="flex gap-1">
   271→                  <span className="w-1.5 h-1.5 rounded-full bg-current animate-bounce [animation-delay:0ms]" />
   272→                  <span className="w-1.5 h-1.5 rounded-full bg-current animate-bounce [animation-delay:120ms]" />
   273→                  <span className="w-1.5 h-1.5 rounded-full bg-current animate-bounce [animation-delay:240ms]" />
   274→                </span>
   275→                {typing.map((t) => t.username).join(', ')} {typing.length === 1 ? 'is' : 'are'} typing...
   276→              </motion.div>
   277→            )}
   278→          </AnimatePresence>
   279→        </div>
   280→      </div>
   281→
   282→      {/* Scroll-to-bottom button — appears when the user has scrolled >200px
   283→          up from the bottom. Shows an unread count badge for messages that
   284→          arrived below the current scroll position. Smooth-scrolls to the
   285→          bottom on click and hides itself when already at the bottom. */}
   286→      <AnimatePresence>
   287→        {showScrollButton && (
   288→          <motion.button
   289→            initial={{ opacity: 0, scale: 0.8, y: 8 }}
   290→            animate={{ opacity: 1, scale: 1, y: 0 }}
   291→            exit={{ opacity: 0, scale: 0.8, y: 8 }}
   292→            transition={{ duration: 0.15 }}
   293→            onClick={() => {
   294→              const e = scrollRef.current
   295→              if (e) {
   296→                e.scrollTo({ top: e.scrollHeight, behavior: 'smooth' })
   297→                // Optimistically clear — the scroll handler will confirm
   298→                // once the smooth scroll settles at the bottom.
   299→                setUnreadBelow(0)
   300→                setShowScrollButton(false)
   301→              }
   302→            }}
   303→            aria-label="Scroll to latest messages"
   304→            title="Scroll to latest"
   305→            className="absolute bottom-4 right-4 z-20 w-11 h-11 rounded-full bg-card border border-border/60 shadow-lg flex items-center justify-center hover:bg-accent active:scale-95 transition-all"
   306→          >
   307→            <ChevronDown className="w-5 h-5" />
   308→            {unreadBelow > 0 && (
   309→              <span className="absolute -top-1.5 -right-1.5 min-w-[20px] h-5 px-1.5 rounded-full bg-primary text-primary-foreground text-[10px] font-bold flex items-center justify-center ring-2 ring-card">
   310→                {unreadBelow > 99 ? '99+' : unreadBelow}
   311→              </span>
   312→            )}
   313→          </motion.button>
   314→        )}
   315→      </AnimatePresence>
   316→    </div>
   317→  )
   318→}

   336→function formatDay(dateStr: string | Date) {
   337→  const d = new Date(dateStr)
   338→  if (isToday(d)) return 'Today'
   339→  if (isYesterday(d)) return 'Yesterday'
   340→  return format(d, 'MMM d, yyyy')
   341→}

   440→function MessageItem(props: MessageItemProps) {
   441→  const { message: m, isMine, isLastOfGroup, isEditing, setEditing, onEditSubmit } = props
   442→  const [editText, setEditText] = useState(m.body)
   443→  const isDeleted = !!m.deletedAt
   444→  const ctxMenu = useContextMenu()
   445→  const { startCall } = useCall()
   446→  const setActiveRoomId = useMusicStore((s) => s.setActiveRoomId)
   447→
   448→  // Handle invitation Join button — joins a call or music room
   449→  const handleInviteJoin = async (invite: any) => {
   450→    try {
   451→      if (invite.type === 'call') {
   452→        // Join the call via the API (idempotent — joins existing or creates)
   453→        const payload = invite.dmGroupId
   454→          ? { dmGroupId: invite.dmGroupId }
   455→          : { channelId: invite.channelId }
   456→        const res = await fetch('/api/calls', {
   457→          method: 'POST',
   458→          headers: { 'Content-Type': 'application/json' },
   459→          body: JSON.stringify(payload),
   460→        })
   461→        if (!res.ok) throw new Error('Failed to join call')
   462→        const data = await res.json()
   463→        // Start our side (mic + WebRTC)
   464→        await startCall({
   465→          callId: data.call.id,
   466→          channelId: invite.channelId,
   467→          dmGroupId: invite.dmGroupId,
   468→          enableVideo: invite.isVideo ?? false,
   469→        })
   470→        toast.success('Joined the call')
   471→      } else if (invite.type === 'music') {
   472→        // Join the music room — set active room + auto-join as member
   473→        setActiveRoomId(invite.targetId)
   474→        await fetch(`/api/music/rooms/${invite.targetId}`).catch(() => {})
   475→        toast.success(`Joined music room "${invite.roomName || 'Unnamed'}"`)
   476→        // Navigate to music view
   477→        window.location.hash = 'music'
   478→      }
   479→    } catch (e: any) {
   480→      toast.error(e.message || 'Failed to join')
   481→    }
   482→  }
   483→
   484→  const showMessageContextMenu = (e: React.MouseEvent | React.TouchEvent) => {
   485→    let x: number, y: number
   486→    if ('touches' in e) {
   487→      const t = e.touches[0] || (e.changedTouches?.[0])
   488→      if (!t) return
   489→      x = t.clientX; y = t.clientY
   490→    } else {
   491→      x = e.clientX; y = e.clientY
   492→    }
   493→    if (!ctxMenu) return
   494→    const items: Array<{ label: string; icon: React.ReactNode; onClick: () => void; variant?: 'default' | 'danger' }> = [
   495→      {
   496→        label: 'Reply',
   497→        icon: <Reply className="w-4 h-4" />,
   498→        onClick: () => props.onReply(),
   499→      },
   500→      {
   501→        label: 'Copy text',
   502→        icon: <Copy className="w-4 h-4" />,

   777→// ─── Transcript button for voice messages ───────────────────────────────────
   778→//
   779→// Shows a "Show transcript" toggle below voice messages. The transcript can
   780→// come from three sources (checked in order):
   781→//   1. m.transcript — populated by the server's auto-transcription background
   782→//      job (which fires when the voice message is uploaded)
   783→//   2. A manual /api/asr call — if auto-transcription hasn't completed yet
   784→//      (or the server was down when the message was sent), the user can
   785→//      click the button to force transcription on demand
   786→//   3. If both fail, we show "(transcription unavailable)"
   787→
   788→function TranscriptButton({ message, isMine }: { message: ChannelMessage; isMine: boolean }) {
   789→  const [expanded, setExpanded] = useState(false)
   790→  const [loading, setLoading] = useState(false)
   791→  const [manualTranscript, setManualTranscript] = useState<string | null>(null)
   792→  const [error, setError] = useState<string | null>(null)
   793→  // Track whether the user manually re-transcribed. If so, the manual result
   794→  // takes priority over the (possibly stale/garbage) cached message.transcript.
   795→  const [manualOverride, setManualOverride] = useState(false)
   796→
   797→  // If auto-transcription already produced a transcript, use it directly —
   798→  // UNLESS the user has manually re-transcribed (manualOverride), in which
   799→  // case the manual result wins. This fixes the cache bug where a failed
   800→  // auto-transcription (e.g. "..." or null) would persist even after a
   801→  // successful manual retry.
   802→  const transcript = manualOverride ? manualTranscript : (manualTranscript || message.transcript)
   803→
   804→  const handleShow = async () => {
   805→    // If we already have a valid transcript and no error, just toggle
   806→    if (transcript && !error) {
   807→      setExpanded((v) => !v)
   808→      return
   809→    }
   810→
   811→    // Otherwise, fetch it on demand via /api/asr
   812→    setLoading(true)
   813→    setError(null) // Clear any previous error before retrying
   814→    try {
   815→      const res = await fetch('/api/asr', {
   816→        method: 'POST',
   817→        headers: { 'Content-Type': 'application/json' },
   818→        body: JSON.stringify({
   819→          mediaUrl: message.mediaUrl,
   820→          messageId: message.id,
   821→        }),
   822→      })
   823→      if (!res.ok) {
   824→        const err = await res.json().catch(() => ({}))
   825→        throw new Error(err.error || `HTTP ${res.status}`)
   826→      }
   827→      const data = await res.json()
   828→      if (data.text && /[a-zA-Z0-9]/.test(data.text)) {
   829→        // Valid transcription — update state and override the cached value
   830→        setManualTranscript(data.text)
   831→        setManualOverride(true)
   832→        setExpanded(true)
   833→      } else {
   834→        setError('No speech detected (transcription returned empty or garbage)')
   835→      }
   836→    } catch (e: any) {
   837→      setError(e?.message || 'Transcription failed — is the ASR server running?')
   838→    } finally {
   839→      setLoading(false)
   840→    }
   841→  }
   842→
   843→  // Don't render the button at all if there's no transcript and no mediaUrl
   844→  // to transcribe
   845→  if (!message.mediaUrl) return null
   846→
   847→  return (
   848→    <div className="mt-1">
   849→      <button
   850→        onClick={handleShow}
   851→        disabled={loading}
   852→        className={cn(
   853→          'flex items-center gap-1.5 px-2 py-1 rounded-lg text-[11px] font-medium transition-colors',
   854→          isMine
   855→            ? 'bg-black/10 hover:bg-black/20 text-foreground/70'
   856→            : 'bg-black/10 hover:bg-black/20 text-foreground/70',
   857→          loading && 'opacity-60 cursor-wait'
   858→        )}
   859→      >
   860→        {loading ? (
   861→          <Loader2 className="w-3 h-3 animate-spin" />
   862→        ) : (
   863→          <FileText className="w-3 h-3" />
   864→        )}
   865→        {transcript
   866→          ? (expanded ? 'Hide transcript' : 'Show transcript')
   867→          : (loading ? 'Transcribing…' : 'Transcribe')}
   868→      </button>
   869→
   870→      {error && (
   871→        <p className="mt-1 text-[11px] text-red-500/80 px-2">{error}</p>
   872→      )}
   873→
   874→      {expanded && transcript && (
   875→        <div className={cn(
   876→          'mt-1.5 px-2.5 py-2 rounded-lg text-xs leading-relaxed border-l-2',
   877→          isMine
   878→            ? 'bg-black/10 border-primary/50 text-foreground/80'
   879→            : 'bg-black/15 border-primary/50 text-foreground/80'
   880→        )}>
   881→          <p className="font-medium text-[10px] uppercase tracking-wider text-muted-foreground mb-1">
   882→            Transcript
   883→          </p>
   884→          <p>{transcript}</p>
   885→        </div>
   886→      )}
   887→    </div>
   888→  )
   889→}
