   273→/**
   274→ * useVoiceRecorder — hook that manages MediaRecorder state for recording voice
   275→ * messages. Tap the mic to start, tap stop to upload + send, or cancel to
   276→ * discard. The recording is uploaded via /api/upload and sent as a message
   277→ * with mediaType 'audio/webm' (or the browser's best supported format).
   278→ */
   279→function useVoiceRecorder({
   280→  onSend,
   281→}: {
   282→  onSend: (url: string, mediaType: string) => Promise<void>
   283→}) {
   284→  const [isRecording, setIsRecording] = useState(false)
   285→  const [isUploading, setIsUploading] = useState(false)
   286→  const [seconds, setSeconds] = useState(0)
   287→  const mediaRecorderRef = useRef<MediaRecorder | null>(null)
   288→  const chunksRef = useRef<Blob[]>([])
   289→  const streamRef = useRef<MediaStream | null>(null)
   290→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   291→  const mimeTypeRef = useRef<string>('audio/webm')
   292→
   293→  // Stop the timer + release the mic on unmount
   294→  useEffect(() => {
   295→    return () => {
   296→      if (timerRef.current) clearInterval(timerRef.current)
   297→      if (streamRef.current) streamRef.current.getTracks().forEach((t) => t.stop())
   298→    }
   299→  }, [])
   300→
   301→  const pickMimeType = useCallback(() => {
   302→    const candidates = [
   303→      'audio/webm;codecs=opus',
   304→      'audio/webm',
   305→      'audio/ogg;codecs=opus',
   306→      'audio/mp4',
   307→    ]
   308→    for (const t of candidates) {
   309→      if (typeof MediaRecorder !== 'undefined' && MediaRecorder.isTypeSupported(t)) {
   310→        return t
   311→      }
   312→    }
   313→    return ''
   314→  }, [])
   315→
   316→  const stopTracks = useCallback(() => {
   317→    if (timerRef.current) {
   318→      clearInterval(timerRef.current)
   319→      timerRef.current = null
   320→    }
   321→    if (streamRef.current) {
   322→      streamRef.current.getTracks().forEach((t) => t.stop())
   323→      streamRef.current = null
   324→    }
   325→  }, [])
   326→
   327→  const start = useCallback(async () => {
   328→    try {
   329→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   330→      streamRef.current = stream
   331→      const mimeType = pickMimeType()
   332→      mimeTypeRef.current = mimeType || 'audio/webm'
   333→      const recorder = mimeType
   334→        ? new MediaRecorder(stream, { mimeType })
   335→        : new MediaRecorder(stream)
   336→      chunksRef.current = []
   337→      recorder.ondataavailable = (e) => {
   338→        if (e.data && e.data.size > 0) chunksRef.current.push(e.data)
   339→      }
   340→      recorder.start()
   341→      mediaRecorderRef.current = recorder
   342→      setIsRecording(true)
   343→      setSeconds(0)
   344→      timerRef.current = setInterval(() => {
   345→        setSeconds((s) => s + 1)
   346→      }, 1000)
   347→    } catch {
   348→      toast.error('Microphone access denied')
   349→      if (streamRef.current) {
   350→        streamRef.current.getTracks().forEach((t) => t.stop())
   351→        streamRef.current = null
   352→      }
   353→    }
   354→  }, [pickMimeType])
   355→
   356→  const cancel = useCallback(() => {
   357→    const recorder = mediaRecorderRef.current
   358→    if (recorder && recorder.state !== 'inactive') {
   359→      // Detach onstop so the pending send is discarded
   360→      recorder.onstop = null
   361→      try {
   362→        recorder.stop()
   363→      } catch {
   364→        // ignore
   365→      }
   366→    }
   367→    mediaRecorderRef.current = null
   368→    chunksRef.current = []
   369→    setIsRecording(false)
   370→    setSeconds(0)
   371→    stopTracks()
   372→  }, [stopTracks])
   373→
   374→  const stopAndSend = useCallback(async () => {
   375→    const recorder = mediaRecorderRef.current
   376→    if (!recorder || recorder.state === 'inactive') {
   377→      cancel()
   378→      return
   379→    }
   380→    const mime = mimeTypeRef.current
   381→    const blobPromise = new Promise<Blob>((resolve) => {
   382→      recorder.onstop = () => {
   383→        const blob = new Blob(chunksRef.current, { type: mime || 'audio/webm' })
   384→        resolve(blob)
   385→      }
   386→    })
   387→    try {
   388→      recorder.stop()
   389→    } catch {
   390→      cancel()
   391→      return
   392→    }
   393→    setIsRecording(false)
   394→    stopTracks()
   395→    setIsUploading(true)
   396→    try {
   397→      const blob = await blobPromise
   398→      if (blob.size === 0) {
   399→        toast.error('Recording was empty')
   400→        return
   401→      }
   402→      const ext = mime.includes('mp4') ? 'm4a' : mime.includes('ogg') ? 'ogg' : 'webm'
   403→      const file = new File([blob], `voice-${Date.now()}.${ext}`, { type: blob.type || mime })
   404→      const formData = new FormData()
   405→      formData.append('file', file)
   406→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   407→      if (!res.ok) {
   408→        const t = await res.text().catch(() => 'Upload failed')
   409→        throw new Error(t.slice(0, 120) || 'Upload failed')
   410→      }
   411→      const data = await res.json()
   412→      await onSend(data.url, data.type || mime || 'audio/webm')
   413→      toast.success('Voice message sent')
   414→    } catch (e: any) {
   415→      toast.error(e?.message || 'Failed to send voice message')
   416→    } finally {
   417→      setIsUploading(false)
   418→      setSeconds(0)
   419→      chunksRef.current = []
   420→      mediaRecorderRef.current = null
   421→    }
   422→  }, [cancel, onSend, stopTracks])
   423→
   424→  return { isRecording, isUploading, seconds, start, cancel, stopAndSend }
   425→}

   427→/**
   428→ * TTS Dialog — lets the user type text, pick a voice (built-in or custom),
   429→ * preview, and send. Also supports creating custom voices from audio clips.
   430→ */
   431→function TtsDialog({
   432→  open,
   433→  onOpenChange,
   434→  onSend,
   435→}: {
   436→  open: boolean
   437→  onOpenChange: (o: boolean) => void
   438→  onSend: (url: string) => Promise<void>
   439→}) {
   440→  const [tab, setTab] = useState<'generate' | 'voices'>('generate')
   441→  const [text, setText] = useState('')
   442→  const [voice, setVoice] = useState('alba')
   443→  const [selectedCustomVoiceId, setSelectedCustomVoiceId] = useState<string | null>(null)
   444→  const [generating, setGenerating] = useState(false)
   445→  const [previewUrl, setPreviewUrl] = useState<string | null>(null)
   446→  const [isStreaming, setIsStreaming] = useState(false)
   447→  const [sending, setSending] = useState(false)
   448→  const qc = useQueryClient()
   449→
   450→  // Fetch built-in voices from the API (single source of truth for labels)
   451→  const { data: builtinVoicesData } = useQuery({
   452→    queryKey: ['tts-builtin-voices'],
   453→    queryFn: async () => {
   454→      const res = await fetch('/api/tts')
   455→      if (!res.ok) throw new Error('failed')
   456→      return res.json()
   457→    },
   458→    enabled: open,
   459→  })
   460→  const builtinVoices: { id: string; name: string; language: string; gender: string }[] =
   461→    builtinVoicesData?.voices || []
   462→
   463→  // Fetch custom voices
   464→  const { data: customVoicesData } = useQuery({
   465→    queryKey: ['tts-voices'],
   466→    queryFn: async () => {
   467→      const res = await fetch('/api/tts/voices')
   468→      if (!res.ok) throw new Error('failed')
   469→      return res.json()
   470→    },
   471→    enabled: open,
   472→  })
   473→  const customVoices = customVoicesData?.voices || []
   474→
   475→  const handleGenerate = async () => {
   476→    if (!text.trim() || generating) return
   477→    setGenerating(true)
   478→    setPreviewUrl(null)
   479→    setIsStreaming(true)
   480→    try {
   481→      const body: any = { text }
   482→      if (selectedCustomVoiceId) {
   483→        body.customVoiceId = selectedCustomVoiceId
   484→      } else {
   485→        body.voice = voice
   486→      }
   487→      const res = await fetch('/api/tts', {
   488→        method: 'POST',
   489→        headers: { 'Content-Type': 'application/json' },
   490→        body: JSON.stringify(body),
   491→      })
   492→      if (!res.ok) {
   493→        const err = await res.json().catch(() => ({ error: 'Generation failed' }))
   494→        throw new Error(err.error || 'Generation failed')
   495→      }
   496→
   497→      // The API route streams the TTS audio AND saves to disk in the
   498→      // background. It returns the saved URL in the X-Tts-Url header.
   499→      // We read the stream for immediate playback via Web Audio API,
   500→      // then use the header URL for the preview player and sending.
   501→      const ttsUrl = res.headers.get('X-Tts-Url')
   502→
   503→      // Stream playback via Web Audio API (like the official Pocket TTS UI)
   504→      const reader = res.body!.getReader()
   505→      let audioCtx: AudioContext | null = null
   506→      let headerParsed = false
   507→      let headerBuf = new Uint8Array(44)
   508→      let headerBytes = 0
   509→      let pcmBuffer = new Uint8Array(0)
   510→      let sampleRate = 24000
   511→      let nextStartTime = 0
   512→
   513→      const playChunk = (data: Uint8Array) => {
   514→        if (!audioCtx) return
   515→        const samples = Math.floor(data.length / 2)
   516→        if (samples === 0) return
   517→        const audioBuffer = audioCtx.createBuffer(1, samples, sampleRate)
   518→        const int16 = new Int16Array(data.buffer, data.byteOffset, samples)
   519→        const channelData = audioBuffer.getChannelData(0)
   520→        for (let i = 0; i < samples; i++) {
   521→          channelData[i] = int16[i] / 32768
   522→        }
   523→        const source = audioCtx.createBufferSource()
   524→        source.buffer = audioBuffer
   525→        source.connect(audioCtx.destination)
   526→        const startTime = Math.max(audioCtx.currentTime, nextStartTime)
   527→        source.start(startTime)
   528→        nextStartTime = startTime + audioBuffer.duration
   529→      }
   530→
   531→      while (true) {
   532→        const { done, value } = await reader.read()
   533→        if (done) break
   534→        if (value) {
   535→          if (!headerParsed) {
   536→            const needed = 44 - headerBytes
   537→            const copy = Math.min(needed, value.length)
   538→            headerBuf.set(value.slice(0, copy), headerBytes)
   539→            headerBytes += copy
   540→            if (headerBytes >= 44) {
   541→              const view = new DataView(headerBuf.buffer)
   542→              sampleRate = view.getUint32(24, true)
   543→              headerParsed = true
   544→              if (!audioCtx) {
   545→                audioCtx = new AudioContext({ latencyHint: 'playback' })
   546→              }
   547→              if (value.length > copy) {
   548→                pcmBuffer = new Uint8Array(value.slice(copy))
   549→              }
   550→            }
   551→          } else {
   552→            const merged = new Uint8Array(pcmBuffer.length + value.length)
   553→            merged.set(pcmBuffer)
   554→            merged.set(value, pcmBuffer.length)
   555→            if (merged.length >= 16384) {
   556→              playChunk(merged)
   557→              pcmBuffer = new Uint8Array(0)
   558→            } else {
   559→              pcmBuffer = merged
   560→            }
   561→          }
   562→        }
   563→      }
   564→      if (pcmBuffer.length > 0) playChunk(pcmBuffer)
   565→
   566→      // Use the server-saved URL for the preview player and sending.
   567→      // NO blob upload — the file is already on disk via the background tee.
   568→      // This fixes the "audio length 0" issue in production.
   569→      if (ttsUrl) {
   570→        setPreviewUrl(ttsUrl)
   571→      }
   572→      setIsStreaming(false)
   573→      toast.success('Voice generated — preview and send')
   574→    } catch (e: any) {
   575→      setIsStreaming(false)
   576→      toast.error(e.message || 'Failed to generate voice')
   577→    } finally {
   578→      setGenerating(false)
   579→    }
   580→  }
   581→
   582→  const handleSend = async () => {
   583→    if (!previewUrl || sending) return
   584→    setSending(true)
   585→    try {
   586→      // The audio is already saved on the server (by the background tee in
   587→      // the API route). Just send the URL directly — no blob upload needed.
   588→      await onSend(previewUrl)
   589→      setText('')
   590→      setPreviewUrl(null)
   591→    } catch {
   592→      toast.error('Failed to send voice message')
   593→    } finally {
   594→      setSending(false)
   595→    }
   596→  }

   602→      setPreviewUrl(null)
   603→      setPreviewBlob(null)
   604→      setSelectedCustomVoiceId(null)
   605→    }
   606→    onOpenChange(o)
   607→  }

   642→            AI Voice Message

   646→          </DialogDescription>

   992→                  Save voice
   993→                </Button>

  1008→            Uploading...
