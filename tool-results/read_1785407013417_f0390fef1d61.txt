    23→interface MessageComposerProps {
    24→  channelId: string

    26→
    27→export function MessageComposer({ channelId }: MessageComposerProps) {
    28→  const { send, sendTyping, replyTo, setReplyTo } = useChannel(channelId)
    29→  const [text, setText] = useState('')
    30→  const [sending, setSending] = useState(false)
    31→  const [uploading, setUploading] = useState(false)
    32→  const [ttsOpen, setTtsOpen] = useState(false)
    33→  const textareaRef = useRef<HTMLTextAreaElement>(null)
    34→
    35→  useEffect(() => {
    36→    const el = textareaRef.current
    37→    if (!el) return
    38→    el.style.height = 'auto'
    39→    el.style.height = Math.min(el.scrollHeight, 160) + 'px'
    40→  }, [text])
    41→
    42→  useEffect(() => {
    43→    if (replyTo) textareaRef.current?.focus()
    44→  }, [replyTo])
    45→
    46→  const handleSend = async () => {
    47→    const trimmed = text.trim()
    48→    if (!trimmed || sending) return
    49→    setSending(true)
    50→    try {
    51→      await send({ body: trimmed, replyToId: replyTo?.id || null })
    52→      setText('')
    53→      setReplyTo(null)
    54→      sendTyping(false)
    55→    } catch {
    56→      toast.error('Failed to send message')
    57→    } finally {
    58→      setSending(false)
    59→    }
    60→  }
    61→
    62→  const handleKeyDown = (e: React.KeyboardEvent) => {
    63→    if (e.key === 'Enter' && !e.shiftKey) {
    64→      e.preventDefault()
    65→      handleSend()
    66→    }
    67→  }
    68→
    69→  const handleChange = (e: React.ChangeEvent<HTMLTextAreaElement>) => {
    70→    setText(e.target.value)
    71→    if (e.target.value.length > 0) sendTyping(true)
    72→    else sendTyping(false)
    73→  }
    74→
    75→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
    76→    const file = e.target.files?.[0]
    77→    if (!file) return
    78→    setUploading(true)
    79→    try {
    80→      const formData = new FormData()
    81→      formData.append('file', file)
    82→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
    83→      if (!res.ok) {
    84→        const text = await res.text()
    85→        throw new Error(text.slice(0, 120) || 'Upload failed')
    86→      }
    87→      const data = await res.json()
    88→      await send({
    89→        body: data.type.startsWith('image') ? '📷' : '📎',
    90→        mediaUrl: data.url,
    91→        mediaType: data.type,
    92→      })
    93→    } catch (e: any) {
    94→      toast.error(e.message || 'Upload failed')
    95→    } finally {
    96→      setUploading(false)
    97→      e.target.value = ''
    98→    }
    99→  }
   100→
   101→  return (
   102→    <div className="border-t border-white/5 bg-background/60 backdrop-blur-3xl px-3 md:px-6 py-4 pb-safe relative z-20">
   103→      {/* Reply banner */}
   104→      {replyTo && (
   105→        <div className="flex items-center gap-3 text-xs bg-black/40 border border-white/10 backdrop-blur-md rounded-xl p-3 mb-3 shadow-lg">
   106→          <Reply className="w-4 h-4 shrink-0 text-primary" />
   107→          <div className="flex-1 min-w-0">
   108→            <div className="font-semibold text-primary">{replyTo.senderName}</div>
   109→            <div className="text-muted-foreground truncate mt-0.5">{replyTo.body}</div>
   110→          </div>
   111→          <button onClick={() => setReplyTo(null)} className="p-1.5 hover:bg-white/10 rounded-full transition-colors">
   112→            <X className="w-4 h-4" />
   113→          </button>
   114→        </div>
   115→      )}
   116→
   117→      <div className="flex items-end gap-2.5">
   118→        {/* Upload button */}
   119→        <label className="cursor-pointer w-10 h-10 flex items-center justify-center hover:bg-white/5 rounded-full transition-colors shrink-0 border border-transparent hover:border-white/10">
   120→          <ImageIcon className="w-[22px] h-[22px] text-muted-foreground hover:text-foreground transition-colors" />
   121→          <input type="file" className="hidden" accept="image/*,video/*,audio/*" onChange={handleUpload} disabled={uploading} />
   122→        </label>
   123→
   124→        {/* TTS voice message button */}
   125→        <button
   126→          onClick={() => setTtsOpen(true)}
   127→          className="w-10 h-10 flex items-center justify-center hover:bg-white/5 rounded-full transition-colors shrink-0 group border border-transparent hover:border-white/10"
   128→          title="Send AI voice message"
   129→        >
   130→          <AudioLines className="w-[22px] h-[22px] text-muted-foreground group-hover:text-primary transition-colors" />
   131→        </button>
   132→
   133→        {/* Text input — grows up to ~4 lines */}
   134→        <div className="flex-1 bg-black/20 backdrop-blur-xl rounded-[24px] px-5 py-3 flex items-end border border-white/10 shadow-inner ring-1 ring-transparent focus-within:ring-primary/30 transition-all">
   135→          <Textarea
   136→            ref={textareaRef}
   137→            value={text}
   138→            onChange={handleChange}
   139→            onKeyDown={handleKeyDown}
   140→            placeholder="iMessage..."
   141→            className="flex-1 resize-none min-h-[24px] max-h-[160px] bg-transparent border-0 focus-visible:ring-0 focus-visible:ring-offset-0 p-0 text-[16px] leading-relaxed shadow-none placeholder:text-muted-foreground/50"
   142→            disabled={sending || uploading}
   143→            rows={1}
   144→          />
   145→        </div>
   146→
   147→        {/* Send button */}
   148→        <Button
   149→          onClick={handleSend}
   150→          disabled={!text.trim() || sending}
   151→          size="icon"
   152→          className="rounded-full h-11 w-11 shrink-0 transition-transform hover:scale-105 active:scale-95 gradient-primary shadow-glow disabled:opacity-50 disabled:shadow-none"
   153→        >
   154→          {sending ? <Loader2 className="w-5 h-5 animate-spin" /> : <Send className="w-5 h-5" />}
   155→        </Button>
   156→      </div>
   157→
   158→      {/* TTS Voice Message Dialog */}
   159→      <TtsDialog
   160→        open={ttsOpen}
   161→        onOpenChange={setTtsOpen}
   162→        onSend={async (url) => {
   163→          await send({
   164→            body: '🎙️ AI voice message',
   165→            mediaUrl: url,
   166→            mediaType: 'audio',
   167→          })
   168→          setTtsOpen(false)
   169→          toast.success('Voice message sent')
   170→        }}
   171→      />
   172→    </div>
   173→  )
   174→}

   176→/**
   177→ * TTS Dialog — lets the user type text, pick a voice (built-in or custom),
   178→ * preview, and send. Also supports creating custom voices from audio clips.
   179→ */
   180→function TtsDialog({
   181→  open,
   182→  onOpenChange,
   183→  onSend,
   184→}: {
   185→  open: boolean
   186→  onOpenChange: (o: boolean) => void
   187→  onSend: (url: string) => Promise<void>
   188→}) {
   189→  const [tab, setTab] = useState<'generate' | 'voices'>('generate')
   190→  const [text, setText] = useState('')
   191→  const [voice, setVoice] = useState('alba')
   192→  const [selectedCustomVoiceId, setSelectedCustomVoiceId] = useState<string | null>(null)
   193→  const [generating, setGenerating] = useState(false)
   194→  const [previewUrl, setPreviewUrl] = useState<string | null>(null)
   195→  const [sending, setSending] = useState(false)
   196→  const qc = useQueryClient()
   197→
   198→  // Fetch custom voices
   199→  const { data: customVoicesData } = useQuery({
   200→    queryKey: ['tts-voices'],
   201→    queryFn: async () => {
   202→      const res = await fetch('/api/tts/voices')
   203→      if (!res.ok) throw new Error('failed')
   204→      return res.json()
   205→    },
   206→    enabled: open,
   207→  })
   208→  const customVoices = customVoicesData?.voices || []
   209→
   210→  const handleGenerate = async () => {
   211→    if (!text.trim() || generating) return
   212→    setGenerating(true)
   213→    setPreviewUrl(null)
   214→    try {
   215→      const body: any = { text }
   216→      if (selectedCustomVoiceId) {
   217→        body.customVoiceId = selectedCustomVoiceId
   218→      } else {
   219→        body.voice = voice
   220→      }
   221→      const res = await fetch('/api/tts', {
   222→        method: 'POST',
   223→        headers: { 'Content-Type': 'application/json' },
   224→        body: JSON.stringify(body),
   225→      })
   226→      if (!res.ok) {
   227→        const err = await res.json()
   228→        throw new Error(err.error || 'Generation failed')
   229→      }
   230→      const data = await res.json()
   231→      setPreviewUrl(data.url)
   232→      toast.success('Voice generated — preview and send')
   233→    } catch (e: any) {
   234→      toast.error(e.message || 'Failed to generate voice')
   235→    } finally {
   236→      setGenerating(false)
   237→    }
   238→  }
   239→
   240→  const handleSend = async () => {
   241→    if (!previewUrl || sending) return
   242→    setSending(true)
   243→    try {
   244→      await onSend(previewUrl)
   245→      setText('')
   246→      setPreviewUrl(null)
   247→    } catch {
   248→      toast.error('Failed to send voice message')
   249→    } finally {
   250→      setSending(false)
   251→    }
   252→  }
   253→
   254→  const handleClose = (o: boolean) => {
   255→    if (!o) {
   256→      setText('')
   257→      setPreviewUrl(null)
   258→      setSelectedCustomVoiceId(null)
   259→    }
   260→    onOpenChange(o)
   261→  }
   262→
   263→  return (
   264→    <Dialog open={open} onOpenChange={handleClose}>
   265→      <DialogContent className="sm:max-w-md">
   266→        <DialogHeader>
   267→          <DialogTitle className="flex items-center gap-2">
   268→            <Sparkles className="w-4 h-4 text-primary" />
   269→            AI Voice Message
   270→          </DialogTitle>
   271→          <DialogDescription>
   272→            Type a message and send it as an AI-generated voice note. Create custom voices from audio clips.
   273→          </DialogDescription>
   274→        </DialogHeader>
   275→
   276→        {/* Tabs */}
   277→        <div className="flex gap-1 p-1 bg-muted rounded-lg">
   278→          <button
   279→            onClick={() => setTab('generate')}
   280→            className={cn(
   281→              'flex-1 py-1.5 text-sm font-medium rounded-md transition-colors',
   282→              tab === 'generate' ? 'bg-background text-foreground' : 'text-muted-foreground'
   283→            )}
   284→          >
   285→            Generate
   286→          </button>
   287→          <button
   288→            onClick={() => setTab('voices')}
   289→            className={cn(
   290→              'flex-1 py-1.5 text-sm font-medium rounded-md transition-colors flex items-center justify-center gap-1.5',
   291→              tab === 'voices' ? 'bg-background text-foreground' : 'text-muted-foreground'
   292→            )}
   293→          >
   294→            My Voices
   295→            {customVoices.length > 0 && (
   296→              <span className="text-[10px] bg-primary/20 text-primary px-1.5 py-0.5 rounded-full">
   297→                {customVoices.length}
   298→              </span>
   299→            )}
   300→          </button>
   301→        </div>
   302→
   303→        {tab === 'generate' ? (
   304→          <div className="space-y-4">
   305→            <div className="space-y-2">
   306→              <Label>Message</Label>
   307→              <Textarea
   308→                value={text}
   309→                onChange={(e) => setText(e.target.value)}
   310→                placeholder="Type what the voice should say..."
   311→                rows={3}
   312→                maxLength={500}
   313→                className="resize-none"
   314→              />
   315→              <div className="text-xs text-muted-foreground text-right">
   316→                {text.length}/500
   317→              </div>
   318→            </div>
   319→
   320→            <div className="space-y-2">
   321→              <Label>Voice</Label>
   322→              {customVoices.length > 0 && (
   323→                <div className="space-y-2">
   324→                  <div className="flex gap-2">
   325→                    <button
   326→                      onClick={() => setSelectedCustomVoiceId(null)}
   327→                      className={cn(
   328→                        'flex-1 py-2 px-3 rounded-lg text-sm font-medium transition-colors border',
   329→                        !selectedCustomVoiceId
   330→                          ? 'bg-primary/15 text-primary border-primary/30'
   331→                          : 'bg-muted/50 text-muted-foreground border-transparent hover:bg-muted'
   332→                      )}
   333→                    >
   334→                      Built-in
   335→                    </button>
   336→                    <button
   337→                      onClick={() => setSelectedCustomVoiceId(customVoices[0]?.id || null)}
   338→                      className={cn(
   339→                        'flex-1 py-2 px-3 rounded-lg text-sm font-medium transition-colors border',
   340→                        selectedCustomVoiceId
   341→                          ? 'bg-primary/15 text-primary border-primary/30'

   344→                    >

   347→                  </div>

   349→                  {selectedCustomVoiceId ? (
   350→                    <Select value={selectedCustomVoiceId} onValueChange={setSelectedCustomVoiceId}>
   351→                      <SelectTrigger>
   352→                        <SelectValue />
   353→                      </SelectTrigger>
   354→                      <SelectContent>
   355→                        {customVoices.map((v: any) => (
   356→                          <SelectItem key={v.id} value={v.id}>
   357→                            {v.name}
   358→                          </SelectItem>
   359→                        ))}
   360→                      </SelectContent>
   361→                    </Select>
   362→                  ) : (
   363→                    <Select value={voice} onValueChange={setVoice}>
   364→                      <SelectTrigger>
   365→                        <SelectValue />
   366→                      </SelectTrigger>
   367→                      <SelectContent>
   368→                        <SelectItem value="alba">Alba (English, Female)</SelectItem>
   369→                        <SelectItem value="charles">Charles (English, Male)</SelectItem>
   370→                        <SelectItem value="jane">Jane (English, Female)</SelectItem>
   371→                        <SelectItem value="michael">Michael (English, Male)</SelectItem>
   372→                        <SelectItem value="vera">Vera (English, Female)</SelectItem>
   373→                        <SelectItem value="paul">Paul (English, Male)</SelectItem>
   374→                        <SelectItem value="estelle">Estelle (French, Female)</SelectItem>
   375→                        <SelectItem value="giovanni">Giovanni (Italian, Male)</SelectItem>
   376→                        <SelectItem value="juergen">Juergen (German, Male)</SelectItem>
   377→                        <SelectItem value="lola">Lola (Spanish, Female)</SelectItem>
   378→                      </SelectContent>
   379→                    </Select>
   380→                  )}
   381→                </div>
   382→              )}
   383→
   384→              {!selectedCustomVoiceId && customVoices.length === 0 && (
   385→                <Select value={voice} onValueChange={setVoice}>
   386→                  <SelectTrigger>
   387→                    <SelectValue />
   388→                  </SelectTrigger>
   389→                  <SelectContent>
   390→                    <SelectItem value="alba">Alba (English, Female)</SelectItem>
   391→                    <SelectItem value="charles">Charles (English, Male)</SelectItem>
   392→                    <SelectItem value="jane">Jane (English, Female)</SelectItem>
   393→                    <SelectItem value="michael">Michael (English, Male)</SelectItem>
   394→                    <SelectItem value="vera">Vera (English, Female)</SelectItem>
   395→                    <SelectItem value="paul">Paul (English, Male)</SelectItem>
   396→                    <SelectItem value="estelle">Estelle (French, Female)</SelectItem>
   397→                    <SelectItem value="giovanni">Giovanni (Italian, Male)</SelectItem>
   398→                    <SelectItem value="juergen">Juergen (German, Male)</SelectItem>
   399→                    <SelectItem value="lola">Lola (Spanish, Female)</SelectItem>
   400→                  </SelectContent>
   401→                </Select>
   402→              )}
   403→
   404→              {customVoices.length === 0 && (
   405→                <button
   406→                  onClick={() => setTab('voices')}

   408→                >

   427→        <DialogFooter>

   430→            !previewUrl ? (
   431→              <Button onClick={handleGenerate} disabled={!text.trim() || generating}>

   466→/**
   467→ * Custom Voices tab — lets the user record/upload a voice clip and save it
   468→ * as a custom voice model for one-shot voice cloning.
   469→ */
   470→function CustomVoicesTab({ onUseVoice }: { onUseVoice: (id: string) => void }) {
   471→  const qc = useQueryClient()
   472→  const [name, setName] = useState('')
   473→  const [audioUrl, setAudioUrl] = useState<string | null>(null)
   474→  const [uploading, setUploading] = useState(false)
   475→  const [recording, setRecording] = useState(false)
   476→  const [mediaRecorder, setMediaRecorder] = useState<MediaRecorder | null>(null)
   477→  const [audioChunks, setAudioChunks] = useState<Blob[]>([])
   478→
   479→  const { data: customVoicesData } = useQuery({
   480→    queryKey: ['tts-voices'],
   481→    queryFn: async () => {
   482→      const res = await fetch('/api/tts/voices')
   483→      if (!res.ok) throw new Error('failed')
   484→      return res.json()
   485→    },
   486→  })
   487→  const customVoices = customVoicesData?.voices || []
   488→
   489→  const deleteVoice = useMutation({
   490→    mutationFn: async (id: string) => {
   491→      await fetch(`/api/tts/voices/${id}`, { method: 'DELETE' })
   492→    },
   493→    onSuccess: () => {
   494→      qc.invalidateQueries({ queryKey: ['tts-voices'] })
   495→      toast.success('Voice deleted')
   496→    },
   497→  })
   498→
   499→  const saveVoice = useMutation({
   500→    mutationFn: async () => {
   501→      const res = await fetch('/api/tts/voices', {
   502→        method: 'POST',
   503→        headers: { 'Content-Type': 'application/json' },
   504→        body: JSON.stringify({ name, audioUrl }),
   505→      })
   506→      if (!res.ok) throw new Error('failed')
   507→      return res.json()
   508→    },
   509→    onSuccess: () => {
   510→      qc.invalidateQueries({ queryKey: ['tts-voices'] })
   511→      toast.success('Custom voice created')
   512→      setName('')
   513→      setAudioUrl(null)
   514→    },
   515→  })
   516→
   517→  const handleUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   518→    const file = e.target.files?.[0]
   519→    if (!file) return
   520→    setUploading(true)
   521→    try {
   522→      const formData = new FormData()
   523→      formData.append('file', file)
   524→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   525→      if (!res.ok) throw new Error('Upload failed')
   526→      const data = await res.json()
   527→      setAudioUrl(data.url)
   528→    } catch (e: any) {
   529→      toast.error(e.message || 'Upload failed')
   530→    } finally {
   531→      setUploading(false)
   532→      e.target.value = ''
   533→    }
   534→  }
   535→
   536→  const startRecording = async () => {
   537→    try {
   538→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   539→      const recorder = new MediaRecorder(stream)
   540→      const chunks: Blob[] = []
   541→      recorder.ondataavailable = (e) => chunks.push(e.data)
   542→      recorder.onstop = async () => {
   543→        const blob = new Blob(chunks, { type: 'audio/webm' })
   544→        // Upload the recording
   545→        setUploading(true)
   546→        try {
   547→          const formData = new FormData()
   548→          formData.append('file', new File([blob], 'voice.webm', { type: 'audio/webm' }))
   549→          const res = await fetch('/api/upload', { method: 'POST', body: formData })
   550→          if (!res.ok) throw new Error('Upload failed')
   551→          const data = await res.json()
   552→          setAudioUrl(data.url)
   553→        } catch (e: any) {
   554→          toast.error(e.message || 'Upload failed')
   555→        } finally {
   556→          setUploading(false)
   557→        }
   558→        stream.getTracks().forEach((t) => t.stop())
   559→      }
   560→      recorder.start()
   561→      setMediaRecorder(recorder)
   562→      setRecording(true)
   563→      setAudioChunks(chunks)
   564→    } catch {
   565→      toast.error('Microphone access denied')
   566→    }
   567→  }
   568→
   569→  const stopRecording = () => {
   570→    if (mediaRecorder && mediaRecorder.state === 'recording') {
   571→      mediaRecorder.stop()
   572→      setRecording(false)
   573→    }
   574→  }
   575→
   576→  return (
   577→    <div className="space-y-4">
   578→      {/* Create new voice */}
   579→      <div className="space-y-3 p-3 rounded-xl bg-muted/50 border border-border/30">
   580→        <div className="flex items-center gap-2 text-sm font-medium">
   581→          <Mic className="w-4 h-4 text-primary" />
   582→          Create a Custom Voice
   583→        </div>
   584→        <p className="text-xs text-muted-foreground">
   585→          Upload or record a 10-30 second voice clip. We'll clone the voice so you can generate speech in it.
   586→        </p>
   587→        <div className="space-y-2">
   588→          <Input
   589→            value={name}
   590→            onChange={(e) => setName(e.target.value)}
   591→            placeholder="Voice name (e.g. My Voice)"
   592→          />
   593→        </div>
   594→        <div className="flex gap-2">
   595→          {!audioUrl ? (
   596→            <>
   597→              {!recording ? (
   598→                <Button onClick={startRecording} variant="outline" size="sm" className="flex-1">
   599→                  <Mic className="w-4 h-4 mr-1.5" />
   600→                  Record
   601→                </Button>
   602→              ) : (
   603→                <Button onClick={stopRecording} variant="destructive" size="sm" className="flex-1">
   604→                  <span className="w-2 h-2 rounded-full bg-white animate-pulse mr-1.5" />
   605→                  Stop
   606→                </Button>
   607→              )}
   608→              <label className="cursor-pointer flex-1">
   609→                <Button asChild variant="outline" size="sm" className="w-full">
   610→                  <span>
   611→                    <Plus className="w-4 h-4 mr-1.5" />
   612→                    Upload
   613→                  </span>
   614→                </Button>
   615→                <input type="file" className="hidden" accept="audio/*" onChange={handleUpload} disabled={uploading || recording} />
   616→              </label>
   617→            </>
   618→          ) : (
   619→            <div className="flex-1 space-y-2">
   620→              <audio controls src={audioUrl} className="w-full h-8" />
   621→              <div className="flex gap-2">
   622→                <Button
   623→                  onClick={() => saveVoice.mutate()}
   624→                  disabled={!name.trim() || saveVoice.isPending}
   625→                  size="sm"
   626→                  className="flex-1 gradient-primary"
   627→                >
   628→                  {saveVoice.isPending ? (
   629→                    <Loader2 className="w-4 h-4 mr-1.5 animate-spin" />
   630→                  ) : (
   631→                    <Sparkles className="w-4 h-4 mr-1.5" />
   632→                  )}
   633→                  Save voice
   634→                </Button>
   635→                <Button
   636→                  onClick={() => setAudioUrl(null)}
   637→                  variant="outline"
   638→                  size="sm"
   639→                >
   640→                  Discard
   641→                </Button>
   642→              </div>
   643→            </div>
   644→          )}
   645→        </div>
   646→        {uploading && (
   647→          <div className="text-xs text-muted-foreground flex items-center gap-1.5">
   648→            <Loader2 className="w-3 h-3 animate-spin" />
   649→            Uploading...
   650→          </div>
   651→        )}
   652→      </div>
   653→
   654→      {/* Existing custom voices */}
   655→      {customVoices.length > 0 && (
   656→        <div className="space-y-2">
   657→          <Label className="text-xs uppercase tracking-wider text-muted-foreground">
   658→            Saved Voices
   659→          </Label>
   660→          <div className="space-y-1.5 max-h-48 overflow-y-auto">
   661→            {customVoices.map((v: any) => (
   662→              <div
   663→                key={v.id}
   664→                className="flex items-center gap-3 p-2.5 rounded-xl bg-card border border-border/30 hover:border-primary/30 transition-colors group"
   665→              >
   666→                <div className="w-9 h-9 rounded-full bg-primary/15 flex items-center justify-center shrink-0">
   667→                  <User className="w-4 h-4 text-primary" />
   668→                </div>
   669→                <div className="flex-1 min-w-0">
   670→                  <div className="text-sm font-medium truncate">{v.name}</div>
   671→                  <div className="text-xs text-muted-foreground">
   672→                    Created {new Date(v.createdAt).toLocaleDateString()}
   673→                  </div>
   674→                </div>
   675→                <button
   676→                  onClick={() => onUseVoice(v.id)}
   677→                  className="text-xs px-2.5 py-1.5 rounded-lg bg-primary/10 text-primary hover:bg-primary/20 transition-colors font-medium"
   678→                >
   679→                  Use
   680→                </button>
   681→                <button
   682→                  onClick={() => {
   683→                    if (confirm(`Delete "${v.name}"?`)) deleteVoice.mutate(v.id)
   684→                  }}
   685→                  className="p-1.5 rounded-lg text-red-400 hover:bg-red-500/10 transition-colors"
   686→                >
   687→                  <Trash2 className="w-3.5 h-3.5" />
   688→                </button>
   689→              </div>
   690→            ))}
   691→          </div>
   692→        </div>
   693→      )}
   694→
   695→      {customVoices.length === 0 && !audioUrl && (
   696→        <div className="text-center py-6">
   697→          <div className="w-12 h-12 mx-auto rounded-2xl bg-primary/10 flex items-center justify-center mb-2">
   698→            <Mic className="w-5 h-5 text-primary" />
   699→          </div>
   700→          <p className="text-sm text-muted-foreground">No custom voices yet</p>
   701→        </div>
   702→      )}
   703→    </div>
   704→  )
   705→}
