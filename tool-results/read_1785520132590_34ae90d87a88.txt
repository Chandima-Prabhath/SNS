    64→// ─── Custom Node Component ───────────────────────────────────────────────────
    65→/**
    66→ * Reads the FlowNode type from `data.type` (NOT the ReactFlow `type` prop,
    67→ * which is always 'custom' in this app). This was the root cause of every
    68→ * node rendering as a green "Send Message" node.
    69→ */
    70→function CustomNode({ data, selected }: { data: any; selected?: boolean }) {
    71→  const nodeType = (data.type as NodeType) || 'message'
    72→  const def = NODE_DEFS[nodeType] || NODE_DEFS.message
    73→  const Icon = ICONS[def.icon] || Zap
    74→
    75→  const isTrigger = nodeType === 'trigger'
    76→  const isStop = nodeType === 'stop'
    77→  const isCondition = nodeType === 'condition'
    78→  const isRandom = nodeType === 'random'
    79→  const isSwitch = nodeType === 'switch_case'
    80→
    81→  // ── Build preview content per node type ──────────────────────────────
    82→  let preview: { label: string; value: string } | null = null
    83→  switch (nodeType) {
    84→    case 'trigger': {
    85→      const tt = data.triggerType || 'any_message'
    86→      const label =
    87→        tt === 'command' ? `/${data.command || 'command'}` :
    88→        tt === 'mention' ? '@mentioned' :
    89→        'any message'
    90→      preview = { label: 'When', value: label }
    91→      break
    92→    }
    93→    case 'message':
    94→      preview = { label: 'Says', value: data.text || '(empty)' }
    95→      break
    96→    case 'typing':
    97→      preview = { label: 'For', value: `${data.seconds || 1}s` }
    98→      break
    99→    case 'input':
   100→      preview = { label: 'Asks', value: data.prompt || '(empty)' }
   101→      break
   102→    case 'wait_choice':
   103→      preview = { label: 'Buttons', value: (data.options || []).join(' · ') || '(none)' }
   104→      break
   105→    case 'condition':
   106→      preview = { label: 'If', value: `${data.variable || '∅'} ${data.operator || 'exists'} ${data.value || ''}` }
   107→      break
   108→    case 'set_var':
   109→      preview = { label: 'Sets', value: `${data.variable || '∅'} = ${data.value || '∅'}` }
   110→      break
   111→    case 'delay':
   112→      preview = { label: 'Wait', value: `${data.seconds || 1}s` }
   113→      break
   114→    case 'stop':
   115→      preview = null
   116→      break
   117→    case 'api_call':
   118→      preview = { label: 'Calls', value: `${data.method || 'GET'} ${data.url || '(no url)'}` }
   119→      break
   120→    case 'random':
   121→      preview = { label: 'Picks', value: 'a random branch' }
   122→      break
   123→    case 'ai_generate':
   124→      preview = { label: 'Asks AI', value: data.aiPrompt ? (data.aiPrompt.length > 40 ? data.aiPrompt.slice(0, 40) + '…' : data.aiPrompt) : '(no prompt)' }
   125→      break
   126→    case 'send_media':
   127→      preview = { label: 'Sends', value: data.mediaUrl || '(no url)' }
   128→      break
   129→    case 'switch_case':
   130→      preview = { label: 'Switch on', value: data.switchVariable || '(no var)' }
   131→      break
   132→    case 'counter':
   133→      preview = { label: 'Incr', value: `${data.variable || 'count'} by ${data.increment ?? 1}` }
   134→      break
   135→    case 'format_string':
   136→      preview = { label: 'Formats', value: data.text || '(no template)' }
   137→      break
   138→    case 'log':
   139→      preview = { label: 'Logs', value: data.logMessage ? (data.logMessage.length > 40 ? data.logMessage.slice(0, 40) + '…' : data.logMessage) : '(no message)' }
   140→      break
   141→    case 'tts':
   142→      preview = { label: 'Speaks', value: data.ttsText ? (data.ttsText.length > 40 ? data.ttsText.slice(0, 40) + '…' : data.ttsText) : '(no text)' }
   143→      break
   144→  }
   145→
   146→  return (
   147→    <div
   148→      className="bg-[#2b2d31] rounded-xl min-w-[200px] max-w-[260px] shadow-lg border overflow-hidden transition-shadow"
   149→      style={{
   150→        borderColor: selected ? def.color : 'rgba(255,255,255,0.08)',
   151→        boxShadow: selected ? `0 0 0 2px ${def.color}40` : undefined,
   152→      }}
   153→    >
   154→      {/* Target handle (not for trigger — it has no incoming edges) */}
   155→      {!isTrigger && (
   156→        <Handle
   157→          type="target"
   158→          position={Position.Top}
   159→          className="adoo-handle"
   160→          style={{ background: def.color, width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   161→        />
   162→      )}
   163→
   164→      {/* Header */}
   165→      <div
   166→        className="flex items-center gap-2 px-3 py-2 border-b"
   167→        style={{ background: def.bg, borderColor: `${def.color}30` }}
   168→      >
   169→        <div
   170→          className="w-6 h-6 rounded-md flex items-center justify-center shrink-0"
   171→          style={{ background: def.color + '30' }}
   172→        >
   173→          <Icon className="w-3.5 h-3.5" style={{ color: def.color }} />
   174→        </div>
   175→        <span
   176→          className="text-[11px] font-bold uppercase tracking-wider flex-1"
   177→          style={{ color: def.color }}
   178→        >
   179→          {def.label}
   180→        </span>
   181→        {def.pauses && (
   182→          <span className="text-[9px] uppercase tracking-wider px-1.5 py-0.5 rounded bg-white/10 text-white/60">
   183→            pause
   184→          </span>
   185→        )}
   186→        {def.terminates && (
   187→          <span className="text-[9px] uppercase tracking-wider px-1.5 py-0.5 rounded bg-white/10 text-white/60">
   188→            end
   189→          </span>
   190→        )}
   191→      </div>
   192→
   193→      {/* Body */}
   194→      <div className="px-3 py-2 text-xs">
   195→        {preview ? (
   196→          <div className="space-y-1">
   197→            <div className="text-white/40 uppercase tracking-wider text-[10px]">{preview.label}</div>
   198→            <div className="text-white/80 break-words line-clamp-2">{preview.value}</div>
   199→            {(nodeType === 'input' || nodeType === 'wait_choice' || nodeType === 'api_call') && data.variableName && (
   200→              <div className="text-[10px] text-white/40">→ saves as <code className="text-white/60">{`{{${data.variableName}}}`}</code></div>
   201→            )}
   202→          </div>
   203→        ) : (
   204→          <div className="text-white/50 italic text-center py-1">{def.description}</div>
   205→        )}
   206→      </div>
   207→
   208→      {/* Source handles */}
   209→      {isCondition ? (
   210→        <div className="flex justify-between px-3 pb-2 pt-1">
   211→          <div className="flex flex-col items-center">
   212→            <span className="text-[10px] font-semibold text-emerald-400 mb-1">TRUE</span>
   213→            <Handle
   214→              type="source"
   215→              position={Position.Bottom}
   216→              id="true"
   217→              className="adoo-handle"
   218→              style={{ background: '#34D399', width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   219→            />
   220→          </div>
   221→          <div className="flex flex-col items-center">
   222→            <span className="text-[10px] font-semibold text-red-400 mb-1">FALSE</span>
   223→            <Handle
   224→              type="source"
   225→              position={Position.Bottom}
   226→              id="false"
   227→              className="adoo-handle"
   228→              style={{ background: '#F87171', width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   229→            />
   230→          </div>
   231→        </div>
   232→      ) : isSwitch ? (
   233→        <div className="px-3 pb-2 pt-1 space-y-1.5">
   234→          {((data.cases as string[]) || []).map((c, i) => (
   235→            <div key={i} className="flex items-center gap-2">
   236→              <span className="text-[10px] font-mono text-white/40 shrink-0 w-10">case_{i}</span>
   237→              <span className="text-[10px] text-white/70 truncate flex-1 px-1.5 py-0.5 rounded bg-white/5">{c || `(empty)`}</span>
   238→              <Handle
   239→                type="source"
   240→                position={Position.Right}
   241→                id={`case_${i}`}
   242→                className="adoo-handle"
   243→                style={{ background: '#FB7185', width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%', position: 'relative' }}
   244→              />
   245→            </div>
   246→          ))}
   247→          <div className="flex items-center gap-2 pt-1.5 border-t border-white/10">
   248→            <span className="text-[10px] font-mono text-white/40 shrink-0 w-10">default</span>
   249→            <span className="text-[10px] text-white/40 italic flex-1 px-1.5 py-0.5 rounded bg-white/5">fallback</span>
   250→            <Handle
   251→              type="source"
   252→              position={Position.Right}
   253→              id="default"
   254→              className="adoo-handle"
   255→              style={{ background: '#64748B', width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%', position: 'relative' }}
   256→            />
   257→          </div>
   258→        </div>
   259→      ) : isRandom ? (
   260→        <div className="px-3 pb-2 pt-1 text-center">
   261→          <span className="text-[10px] text-white/40 mb-1 block">All outgoing edges are random picks</span>
   262→          <Handle
   263→            type="source"
   264→            position={Position.Bottom}
   265→            className="adoo-handle"
   266→            style={{ background: def.color, width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   267→          />
   268→        </div>
   269→      ) : !isStop ? (
   270→        <Handle
   271→          type="source"
   272→          position={Position.Bottom}
   273→          className="adoo-handle"
   274→          style={{ background: def.color, width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   275→        />
   276→      ) : null}
   277→    </div>
   278→  )
   279→}

   291→export function BotBuilderEditor({ initialFlow, onSave, bot }: BotBuilderEditorProps) {
   292→  const [selectedNodeId, setSelectedNodeId] = useState<string | null>(null)
   293→  const [selectedEdgeId, setSelectedEdgeId] = useState<string | null>(null)
   294→  const [rightTab, setRightTab] = useState<'inspector' | 'issues' | 'debug'>('inspector')
   295→  const [showShortcuts, setShowShortcuts] = useState(false)
   296→  const confirm = useConfirm()
   297→
   298→  // ── Initial nodes/edges ──────────────────────────────────────────────
   299→  // FIX: previously the .map() overwrote data.type with ReactFlow's type
   300→  // ('custom'), so the inspector crashed on click. Now we use data.type
   301→  // everywhere and never let n.type leak into data.
   302→  const { initialNodes, initialEdges } = useMemo(() => {
   303→    if (initialFlow?.nodes?.length) {
   304→      return {
   305→        initialNodes: initialFlow.nodes.map((n) => ({
   306→          id: n.id,
   307→          type: 'custom',
   308→          position: n.position,
   309→          data: { ...n.data },
   310→        })) as Node[],
   311→        initialEdges: initialFlow.edges.map((e) => ({
   312→          id: e.id,
   313→          source: e.source,
   314→          target: e.target,
   315→          sourceHandle: e.sourceHandle || undefined,
   316→          label: e.label || undefined,
   317→          animated: true,
   318→          style: { stroke: '#5865F2', strokeWidth: 2 },
   319→          markerEnd: { type: MarkerType.ArrowClosed, color: '#5865F2' },
   320→        })) as Edge[],
   321→      }
   322→    }
   323→    // Brand-new bot: seed with a sensible default flow
   324→    const triggerData = defaultNodeData('trigger')
   325→    const msgData = defaultNodeData('message')
   326→    return {
   327→      initialNodes: [
   328→        { id: 'trigger-1', type: 'custom', position: { x: 320, y: 80 }, data: { ...triggerData, type: 'trigger' } },
   329→        { id: 'message-1', type: 'custom', position: { x: 320, y: 240 }, data: { ...msgData, type: 'message' } },
   330→      ] as Node[],
   331→      initialEdges: [
   332→        {
   333→          id: 'e-trigger-1-message-1',
   334→          source: 'trigger-1',
   335→          target: 'message-1',
   336→          animated: true,
   337→          style: { stroke: '#5865F2', strokeWidth: 2 },
   338→          markerEnd: { type: MarkerType.ArrowClosed, color: '#5865F2' },
   339→        },
   340→      ] as Edge[],
   341→    }
   342→  }, [initialFlow])
   343→
   344→  const [nodes, setNodes, onNodesChange] = useNodesState(initialNodes)
   345→  const [edges, setEdges, onEdgesChange] = useEdgesState(initialEdges)
   346→
   347→  const onConnect = useCallback(
   348→    (params: Connection) =>
   349→      setEdges((eds) =>
   350→        addEdge(
   351→          {
   352→            ...params,
   353→            animated: true,
   354→            style: { stroke: '#5865F2', strokeWidth: 2 },
   355→            markerEnd: { type: MarkerType.ArrowClosed, color: '#5865F2' },
   356→          },
   357→          eds
   358→        )
   359→      ),
   360→    [setEdges]
   361→  )
   362→
   363→  // ── Add node — position near the canvas center, offset by count ──────
   364→  const addNode = (type: NodeType) => {
   365→    const id = `${type}-${Date.now()}`
   366→    const data = { ...defaultNodeData(type), type }
   367→    // Place near the top-center with a slight cascade so it's always visible
   368→    const idx = nodes.length
   369→    const position = {
   370→      x: 240 + (idx % 4) * 90,
   371→      y: 80 + idx * 30,
   372→    }
   373→    setNodes((nds) => [...nds, { id, type: 'custom', position, data }])
   374→    setSelectedNodeId(id)
   375→  }
   376→
   377→  const updateNodeData = (nodeId: string, patch: Record<string, any>) =>
   378→    setNodes((nds) =>
   379→      nds.map((n) => (n.id === nodeId ? { ...n, data: { ...n.data, ...patch } } : n))
   380→    )
   381→
   382→  const deleteNode = (nodeId: string) => {
   383→    setNodes((nds) => nds.filter((n) => n.id !== nodeId))
   384→    setEdges((eds) => eds.filter((e) => e.source !== nodeId && e.target !== nodeId))
   385→    setSelectedNodeId(null)
   386→  }
   387→
   388→  // Disconnect an edge — used by the edge context menu and the Delete key.
   389→  const deleteEdge = (edgeId: string) => {
   390→    setEdges((eds) => eds.filter((e) => e.id !== edgeId))
   391→    setSelectedEdgeId(null)
   392→  }
   393→
   394→  // ── Clipboard for copy/paste ─────────────────────────────────────────
   395→  const clipboardRef = useRef<{ nodes: Node[]; edges: Edge[] } | null>(null)
   396→
   397→  // Get all currently selected nodes (supports multi-select)
   398→  const selectedNodes = useMemo(() => nodes.filter((n) => n.selected), [nodes])
   399→  const selectedEdges = useMemo(
   400→    () => edges.filter((e) => e.selected || (selectedNodeId && (e.source === selectedNodeId || e.target === selectedNodeId))),
   401→    [edges, selectedNodeId]
   402→  )
   403→
   404→  // Duplicate selected nodes (Ctrl+D)
   405→  const duplicateSelected = useCallback(() => {
   406→    if (selectedNodes.length === 0) return
   407→    const newNodes: Node[] = []
   408→    const newEdges: Edge[] = []
   409→    const idMap = new Map<string, string>()
   410→
   411→    // Create copies of each selected node, offset by 40px
   412→    for (const n of selectedNodes) {
   413→      const newId = `${(n.data as any)?.type || 'node'}-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`
   414→      idMap.set(n.id, newId)
   415→      newNodes.push({
   416→        ...n,
   417→        id: newId,
   418→        position: { x: n.position.x + 40, y: n.position.y + 40 },
   419→        selected: true,
   420→        data: { ...n.data },
   421→      })
   422→    }
   423→
   424→    // Copy edges between duplicated nodes
   425→    for (const e of edges) {
   426→      const newSource = idMap.get(e.source)
   427→      const newTarget = idMap.get(e.target)
   428→      if (newSource && newTarget) {
   429→        newEdges.push({
   430→          ...e,
   431→          id: `e-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
   432→          source: newSource,
   433→          target: newTarget,
   434→          selected: false,
   435→        })
   436→      }
   437→    }
   438→
   439→    // Deselect original nodes, add the copies
   440→    setNodes((nds) => [...nds.map((n) => ({ ...n, selected: false })), ...newNodes])
   441→    setEdges((eds) => [...eds, ...newEdges])
   442→  }, [selectedNodes, edges, setNodes, setEdges])
   443→
   444→  // Copy selected nodes to clipboard (Ctrl+C)
   445→  const copySelected = useCallback(() => {
   446→    if (selectedNodes.length === 0) return
   447→    const connectedEdges = edges.filter((e) =>
   448→      selectedNodes.some((n) => n.id === e.source) &&
   449→      selectedNodes.some((n) => n.id === e.target)
   450→    )
   451→    clipboardRef.current = { nodes: selectedNodes, edges: connectedEdges }
   452→    toast.success(`Copied ${selectedNodes.length} node${selectedNodes.length > 1 ? 's' : ''}`)
   453→  }, [selectedNodes, edges])
   454→
   455→  // Paste from clipboard (Ctrl+V)
   456→  const paste = useCallback(() => {
   457→    if (!clipboardRef.current || clipboardRef.current.nodes.length === 0) return
   458→    const { nodes: clipNodes, edges: clipEdges } = clipboardRef.current
   459→    const idMap = new Map<string, string>()
   460→    const newNodes: Node[] = []
   461→    const newEdges: Edge[] = []
   462→
   463→    for (const n of clipNodes) {
   464→      const newId = `${(n.data as any)?.type || 'node'}-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`
   465→      idMap.set(n.id, newId)
   466→      newNodes.push({
   467→        ...n,
   468→        id: newId,
   469→        position: { x: n.position.x + 60, y: n.position.y + 60 },
   470→        selected: true,
   471→        data: { ...n.data },
   472→      })
   473→    }
   474→
   475→    for (const e of clipEdges) {
   476→      const newSource = idMap.get(e.source)
   477→      const newTarget = idMap.get(e.target)
   478→      if (newSource && newTarget) {
   479→        newEdges.push({
   480→          ...e,
   481→          id: `e-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
   482→          source: newSource,
   483→          target: newTarget,
   484→          selected: false,
   485→        })
   486→      }
   487→    }
   488→
   489→    setNodes((nds) => [...nds.map((n) => ({ ...n, selected: false })), ...newNodes])
   490→    setEdges((eds) => [...eds, ...newEdges])
   491→    toast.success(`Pasted ${newNodes.length} node${newNodes.length > 1 ? 's' : ''}`)
   492→  }, [clipboardRef, setNodes, setEdges])
   493→
   494→  // Select all nodes (Ctrl+A)
   495→  const selectAll = useCallback(() => {
   496→    setNodes((nds) => nds.map((n) => ({ ...n, selected: true })))
   497→  }, [setNodes])
   498→
   499→  // Delete all selected nodes (handles multi-select Delete)
   500→  const deleteSelected = useCallback(() => {
   501→    const toDelete = nodes.filter((n) => n.selected).map((n) => n.id)
   502→    if (toDelete.length === 0) {
   503→      if (selectedNodeId) deleteNode(selectedNodeId)
   504→      else if (selectedEdgeId) deleteEdge(selectedEdgeId)
   505→      return
   506→    }
   507→    setNodes((nds) => nds.filter((n) => !n.selected))
   508→    setEdges((eds) => eds.filter((e) => !toDelete.includes(e.source) && !toDelete.includes(e.target)))
   509→    setSelectedNodeId(null)
   510→  }, [nodes, selectedNodeId, selectedEdgeId, setNodes, setEdges]) // eslint-disable-line react-hooks/exhaustive-deps
   511→

   529→        return
   530→      }

   532→      // Ctrl/Cmd+V — paste

   534→        e.preventDefault()
   535→        paste()
   536→        return
   537→      }
   538→
   539→      // Ctrl/Cmd+D — duplicate
   540→      if (mod && e.key === 'd') {
   541→        e.preventDefault()
   542→        duplicateSelected()
   543→        return
   544→      }
   545→
   546→      // Ctrl/Cmd+A — select all
   547→      if (mod && e.key === 'a') {
   548→        e.preventDefault()
   549→        selectAll()
   550→        return
   551→      }
   552→
   553→      // Delete/Backspace — delete selected
   554→      if (e.key === 'Delete' || e.key === 'Backspace') {
   555→        e.preventDefault()
   556→        deleteSelected()
   557→        return
   558→      }

   560→      // Escape — deselect all

   564→        setSelectedEdgeId(null)

  1904→// ─── Switch Case inspector ───────────────────────────────────────────────────
  1905→function SwitchCaseInspector({
  1906→  data,
  1907→  onUpdate,
  1908→}: {
  1909→  data: any
  1910→  onUpdate: (patch: Record<string, any>) => void
  1911→}) {
  1912→  const inputCls = 'bg-[#1e1f22] border-white/10 text-white placeholder:text-white/30'
  1913→  const cases: string[] = data.cases || []
  1914→
  1915→  const addCase = () => {
  1916→    onUpdate({ cases: [...cases, ''] })
  1917→  }
  1918→  const updateCase = (i: number, value: string) => {
  1919→    const next = [...cases]
  1920→    next[i] = value
  1921→    onUpdate({ cases: next })
  1922→  }
  1923→  const removeCase = (i: number) => {
  1924→    onUpdate({ cases: cases.filter((_, idx) => idx !== i) })
  1925→  }
  1926→
  1927→  return (
  1928→    <div className="space-y-4">
  1929→      <VariableNameField
  1930→        value={data.switchVariable || ''}
  1931→        onChange={(v) => onUpdate({ switchVariable: v })}
  1932→        label="Variable to switch on"
  1933→        placeholder="choice"
  1934→      />
  1935→      <div className="space-y-2">
  1936→        <div className="flex items-center justify-between">
  1937→          <Label className="text-white/60 text-xs">Cases ({cases.length})</Label>
  1938→          <Button onClick={addCase} variant="ghost" size="sm" className="h-6 text-xs">
  1939→            <Plus className="w-3 h-3 mr-1" /> Add
  1940→          </Button>
  1941→        </div>
  1942→        <div className="space-y-1.5">
  1943→          {cases.map((c, i) => (
  1944→            <div key={i} className="flex items-center gap-1.5">
  1945→              <span className="text-[10px] text-white/40 font-mono shrink-0 w-12">case_{i}</span>
  1946→              <Input
  1947→                value={c}
  1948→                onChange={(e) => updateCase(i, e.target.value)}
  1949→                placeholder="value to match"
  1950→                className={inputCls + ' text-xs flex-1'}
