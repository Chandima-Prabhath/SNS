     1→     1→/**
     2→     2→ * Visual Bot Builder — Types & Execution Engine (v2)
     3→     3→ *
     4→     4→ * Architecture
     5→     5→ * ────────────
     6→     6→ * A bot flow is a directed graph stored as JSON in `Bot.flow`:
     7→     7→ *   { nodes: FlowNode[], edges: FlowEdge[] }
     8→     8→ *
     9→     9→ * The engine walks the graph starting from a TRIGGER node. Crucially, it
    10→    10→ * supports **pause/resume** for INPUT and CHOICE nodes — when the engine
    11→    11→ * reaches one, it stops, persists `{ currentNodeId, variables }` to
    12→    12→ * ConversationSession, and resumes on the user's next message.
    13→    13→ *
    14→    14→ * Node categories (used by the UI for palette grouping):
    15→    15→ *   - trigger  : starts the flow (1 type, multiple subtypes)
    16→    16→ *   - output   : bot sends something (message, choice, typing)
    17→    17→ *   - input    : wait for user reply (text input, choice input)
    18→    18→ *   - logic    : control flow (condition, set_var, delay, stop)
    19→    19→ *   - advanced : power features (api_call, random)
    20→    20→ */

   258→   258→  | { type: 'error'; timestamp: number; nodeId?: string; message: string }
   259→   259→  | { type: 'flow_end'; timestamp: number; reason: 'stop' | 'no_edge' | 'max_steps' | 'error'; sentCount: number }
   260→   260→  | { type: 'log'; timestamp: number; nodeId: string; level: 'info' | 'warn' | 'error'; message: string }

   263→   263→// Engine
   264→   264→// ─────────────────────────────────────────────────────────────────────────────
   265→   265→
   266→   266→const MAX_STEPS = 50
   267→   267→
   268→   268→/**
   269→   269→ * Resume descriptor — when set, the engine resumes from `nodeId` instead of
   270→   270→ * searching for a trigger. The caller passes this when a paused session
   271→   271→ * receives its awaited reply.
   272→   272→ */
   273→   273→export interface ResumeDescriptor {
   274→   274→  nodeId: string
   275→   275→  /** The text to assign to the input variable (usually the user's reply). */
   276→   276→  inputText: string
   277→   277→  /** The variable name to assign inputText to. */
   278→   278→  inputVariable: string
   279→   279→}
   280→   280→
   281→   281→/**
   282→   282→ * Replace {{var}} placeholders in a string.
   283→   283→ */
   284→   284→function interpolate(text: string, ctx: BotExecutionContext): string {
   285→   285→  let out = text
   286→   286→  for (const [k, v] of Object.entries(ctx.variables)) {
   287→   287→    out = out.replace(new RegExp(`\\{\\{${k}\\}\\}`, 'g'), v)
   288→   288→  }
   289→   289→  out = out.replace(/\{\{sender\}\}/g, ctx.senderName)
   290→   290→  out = out.replace(/\{\{args\}\}/g, ctx.args.join(' '))
   291→   291→  out = out.replace(/\{\{body\}\}/g, ctx.body)
   292→   292→  return out
   293→   293→}
   294→   294→
   295→   295→/**
   296→   296→ * Execute a bot flow. Returns when:
   297→   297→ *   - the flow reaches a `stop` node,
   298→   298→ *   - the flow ends naturally (no outgoing edge),
   299→   299→ *   - the flow pauses at an INPUT/WAIT_CHOICE node,
   300→   300→ *   - MAX_STEPS is exceeded.
   301→   301→ *
   302→   302→ * Emits trace events for debugging — each node entry/exit, variable change,
   303→   303→ * message sent, condition evaluation, branch taken, and error is recorded.
   304→   304→ * The caller can pass these to the UI for a test-run visualization.
   305→   305→ */
   306→   306→export async function executeBotFlow(
   307→   307→  flow: BotFlow,
   308→   308→  ctx: BotExecutionContext,
   309→   309→  resume?: ResumeDescriptor
   310→   310→): Promise<BotExecutionResult> {
   311→   311→  let sentCount = 0
   312→   312→  let steps = 0
   313→   313→  let currentNode: FlowNode | undefined
   314→   314→  const trace: TraceEvent[] = []
   315→   315→  const now = () => Date.now()
   316→   316→
   317→   317→  // ── Helpers (declared early so the resume block can use them) ─────────
   318→   318→  const edgesFrom = (nodeId: string, handle?: string | null) =>
   319→   319→    flow.edges.filter((e) => {
   320→   320→      if (e.source !== nodeId) return false
   321→   321→      if (handle === undefined) return true
   322→   322→      if (handle === null) return !e.sourceHandle
   323→   323→      return e.sourceHandle === handle
   324→   324→    })
   325→   325→
   326→   326→  const followEdge = (nodeId: string, handle?: string | null): FlowNode | undefined => {
   327→   327→    const edges = edgesFrom(nodeId, handle)
   328→   328→    if (edges.length === 0) return undefined
   329→   329→    return flow.nodes.find((n) => n.id === edges[0].target)
   330→   330→  }
   331→   331→
   332→   332→  // ── Resolve starting node ─────────────────────────────────────────────
   333→   333→  if (resume) {
   334→   334→    const inputNode = flow.nodes.find((n) => n.id === resume.nodeId)
   335→   335→    if (!inputNode) {
   336→   336→      trace.push({ type: 'error', timestamp: now(), message: 'resume target not found' })
   337→   337→      return { sentCount: 0, paused: false, variables: ctx.variables, trace, error: 'resume target not found' }
   338→   338→    }
   339→   339→    ctx.variables[resume.inputVariable] = resume.inputText
   340→   340→    trace.push({ type: 'resumed', timestamp: now(), nodeId: inputNode.id, inputText: resume.inputText })
   341→   341→    trace.push({ type: 'variable_set', timestamp: now(), nodeId: inputNode.id, variable: resume.inputVariable, value: resume.inputText })
   342→   342→    currentNode = followEdge(inputNode.id, null)
   343→   343→    if (!currentNode) {
   344→   344→      trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   345→   345→      return { sentCount, paused: false, variables: ctx.variables, trace }
   346→   346→    }
   347→   347→  } else {
   348→   348→    const trigger = flow.nodes.find((n) => n.type === 'trigger')
   349→   349→    if (!trigger) {
   350→   350→      trace.push({ type: 'error', timestamp: now(), message: 'No trigger node found' })
   351→   351→      return { sentCount: 0, paused: false, variables: ctx.variables, trace, error: 'No trigger node found' }
   352→   352→    }
   353→   353→
   354→   354→    trace.push({
   355→   355→      type: 'flow_start',
   356→   356→      timestamp: now(),
   357→   357→      triggerNodeId: trigger.id,
   358→   358→      input: { body: ctx.body, sender: ctx.senderName, command: ctx.command },
   359→   359→    })
   360→   360→
   361→   361→    // Trigger matching
   362→   362→    const tt = trigger.data.triggerType || 'any_message'
   363→   363→    if (tt === 'command' && trigger.data.command) {
   364→   364→      if (!ctx.command || ctx.command !== trigger.data.command.replace(/^\//, '')) {
   365→   365→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   366→   366→        return { sentCount: 0, paused: false, variables: ctx.variables, trace }
   367→   367→      }
   368→   368→    } else if (tt === 'mention') {
   369→   369→      if (!ctx.isMention) {
   370→   370→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   371→   371→        return { sentCount: 0, paused: false, variables: ctx.variables, trace }
   372→   372→      }
   373→   373→    }
   374→   374→
   375→   375→    currentNode = trigger
   376→   376→  }
   377→   377→
   378→   378→  // ── Walk the graph ────────────────────────────────────────────────────
   379→   379→  while (currentNode && steps < MAX_STEPS) {
   380→   380→    steps++
   381→   381→    const nodeStartTime = now()
   382→   382→    const nodeLabel = currentNode.data?.label || currentNode.type
   383→   383→
   384→   384→    trace.push({
   385→   385→      type: 'node_enter',
   386→   386→      timestamp: nodeStartTime,
   387→   387→      nodeId: currentNode.id,
   388→   388→      nodeType: currentNode.type,
   389→   389→      nodeLabel,
   390→   390→    })
   391→   391→
   392→   392→    switch (currentNode.type) {
   393→   393→      // ── TRIGGER ────────────────────────────────────────────────────────
   394→   394→      case 'trigger':
   395→   395→        currentNode = followEdge(currentNode.id, null)
   396→   396→        break
   397→   397→
   398→   398→      // ── OUTPUT: message ────────────────────────────────────────────────
   399→   399→      case 'message': {
   400→   400→        const text = interpolate(currentNode.data.text || '', ctx)
   401→   401→        if (text.trim()) {
   402→   402→          await ctx.reply(text)
   403→   403→          sentCount++
   404→   404→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text })
   405→   405→        }
   406→   406→        currentNode = followEdge(currentNode.id, null)
   407→   407→        break
   408→   408→      }
   409→   409→
   410→   410→      // ── OUTPUT: send_media ─────────────────────────────────────────────
   411→   411→      case 'send_media': {
   412→   412→        const url = interpolate(currentNode.data.mediaUrl || '', ctx)
   413→   413→        const caption = interpolate(currentNode.data.caption || '', ctx)
   414→   414→        if (url.trim()) {
   415→   415→          // ctx.reply is text-only; we log it but the actual media send
   416→   416→          // would need an extended reply API. For now, send the URL as text.
   417→   417→          const text = caption ? `${caption}\n${url}` : url
   418→   418→          await ctx.reply(text)
   419→   419→          sentCount++
   420→   420→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text })
   421→   421→        }
   422→   422→        currentNode = followEdge(currentNode.id, null)
   423→   423→        break
   424→   424→      }
   425→   425→
   426→   426→      // ── OUTPUT: typing ─────────────────────────────────────────────────
   427→   427→      case 'typing': {
   428→   428→        const seconds = currentNode.data.seconds || 1
   429→   429→        if (ctx.setTyping) await ctx.setTyping(seconds)
   430→   430→        else await new Promise((r) => setTimeout(r, Math.min(seconds, 2) * 1000))
   431→   431→        currentNode = followEdge(currentNode.id, null)
   432→   432→        break
   433→   433→      }
   434→   434→
   435→   435→      // ── INPUT: pause and wait for next message ────────────────────────
   436→   436→      case 'input': {
   437→   437→        const prompt = interpolate(currentNode.data.prompt || '', ctx)
   438→   438→        if (prompt.trim()) {
   439→   439→          await ctx.reply(prompt)
   440→   440→          sentCount++
   441→   441→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text: prompt })
   442→   442→        }
   443→   443→        trace.push({ type: 'paused', timestamp: now(), nodeId: currentNode.id, variableName: currentNode.data.variableName || 'reply' })
   444→   444→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   445→   445→        return {
   446→   446→          sentCount,
   447→   447→          paused: true,
   448→   448→          pausedAtNodeId: currentNode.id,
   449→   449→          variables: ctx.variables,
   450→   450→          trace,
   451→   451→        }
   452→   452→      }
   453→   453→
   454→   454→      // ── INPUT: wait_choice ────────────────────────────────────────────
   455→   455→      case 'wait_choice': {
   456→   456→        const prompt = interpolate(currentNode.data.prompt || '', ctx)
   457→   457→        const options = currentNode.data.options || []
   458→   458→        // Build a Telegram-style inline keyboard — each option is a button.
   459→   459→        // The button's callbackData is the option text itself, so the resume
   460→   460→        // path in visual.ts can match it directly.
   461→   461→        const keyboard: BotKeyboardButton[][] = options.map((opt) => [{
   462→   462→          text: opt,
   463→   463→          callbackData: opt,
   464→   464→        }])
   465→   465→
   466→   466→        // Telegram-style behavior: if we have an existing keyboard message
   467→   467→        // (from a previous pause at this node, i.e. a loop-back), EDIT it
   468→   468→        // in-place instead of sending a new message. This prevents the bot
   469→   469→        // from spamming new keyboard messages every time the user clicks a
   470→   470→        // button that loops back to this menu.
   471→   471→        //
   472→   472→        // The keyboardMessageId is set in the session state by visual.ts
   473→   473→        // when it persists the pause. We read it from ctx.variables under
   474→   474→        // a reserved key.
   475→   475→        const existingKeyboardMsgId = ctx.variables['__keyboardMsgId']
   476→   476→
   477→   477→        if (prompt.trim()) {
   478→   478→          if (existingKeyboardMsgId && ctx.editMessage) {
   479→   479→            // Edit-in-place: update the existing keyboard message
   480→   480→            await ctx.editMessage(existingKeyboardMsgId, prompt, keyboard)
   481→   481→            trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text: `[edited] ${prompt}` })
   482→   482→          } else {
   483→   483→            // Fresh keyboard: send a new message
   484→   484→            const msgId = await ctx.reply(prompt, keyboard)
   485→   485→            sentCount++
   486→   486→            // Store the message ID so future re-pauses can edit it
   487→   487→            ctx.variables['__keyboardMsgId'] = msgId
   488→   488→            trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text: prompt })
   489→   489→          }
   490→   490→        }
   491→   491→        trace.push({ type: 'paused', timestamp: now(), nodeId: currentNode.id, variableName: currentNode.data.variableName || 'choice' })
   492→   492→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   493→   493→        return {
   494→   494→          sentCount,
   495→   495→          paused: true,
   496→   496→          pausedAtNodeId: currentNode.id,
   497→   497→          variables: ctx.variables,
   498→   498→          trace,
   499→   499→        }
   500→   500→      }
   501→   501→
   502→   502→      // ── LOGIC: condition ──────────────────────────────────────────────
   503→   503→      case 'condition': {
   504→   504→        const varName = currentNode.data.variable || ''
   505→   505→        const varValue = ctx.variables[varName] || ''
   506→   506→        const op = currentNode.data.operator || 'exists'
   507→   507→        const cmp = currentNode.data.value || ''
   508→   508→
   509→   509→        let met = false
   510→   510→        switch (op) {
   511→   511→          case 'equals':       met = varValue === cmp; break
   512→   512→          case 'not_equals':   met = varValue !== cmp; break
   513→   513→          case 'contains':     met = varValue.includes(cmp); break
   514→   514→          case 'starts_with':  met = varValue.startsWith(cmp); break
   515→   515→          case 'exists':       met = !!varValue; break
   516→   516→          case 'not_exists':   met = !varValue; break
   517→   517→        }
   518→   518→
   519→   519→        trace.push({
   520→   520→          type: 'condition_eval',
   521→   521→          timestamp: now(),
   522→   522→          nodeId: currentNode.id,
   523→   523→          variable: varName,
   524→   524→          value: varValue,
   525→   525→          operator: op,
   526→   526→          compareTo: cmp,
   527→   527→          result: met,
   528→   528→        })
   529→   529→
   530→   530→        const handle = met ? 'true' : 'false'
   531→   531→        const next = followEdge(currentNode.id, handle)
   532→   532→        if (next) {
   533→   533→          trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle, targetNodeId: next.id })
   534→   534→          currentNode = next
   535→   535→        } else {
   536→   536→          // No matching branch — fall through to default edge if any
   537→   537→          const fallback = followEdge(currentNode.id, null)
   538→   538→          if (fallback) {
   539→   539→            trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: 'default', targetNodeId: fallback.id })
   540→   540→            currentNode = fallback
   541→   541→          } else {
   542→   542→            trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   543→   543→            currentNode = undefined
   544→   544→          }
   545→   545→        }
   546→   546→        break
   547→   547→      }
   548→   548→
   549→   549→      // ── LOGIC: switch_case (multi-branch) ─────────────────────────────
   550→   550→      case 'switch_case': {
   551→   551→        const varName = currentNode.data.switchVariable || ''
   552→   552→        const varValue = ctx.variables[varName] || ''
   553→   553→        const cases = currentNode.data.cases || []
   554→   554→
   555→   555→        const matchIdx = cases.findIndex((c) => c === varValue)
   556→   556→        const handle = matchIdx >= 0 ? `case_${matchIdx}` : 'default'
   557→   557→
   558→   558→        trace.push({
   559→   559→          type: 'condition_eval',
   560→   560→          timestamp: now(),
   561→   561→          nodeId: currentNode.id,
   562→   562→          variable: varName,
   563→   563→          value: varValue,
   564→   564→          operator: 'switch',
   565→   565→          compareTo: matchIdx >= 0 ? cases[matchIdx] : '(default)',
   566→   566→          result: matchIdx >= 0,
   567→   567→        })
   568→   568→
   569→   569→        const next = followEdge(currentNode.id, handle)
   570→   570→        if (next) {
   571→   571→          trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle, targetNodeId: next.id })
   572→   572→          currentNode = next
   573→   573→        } else {
   574→   574→          const fallback = followEdge(currentNode.id, null)
   575→   575→          if (fallback) {
   576→   576→            trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: 'default', targetNodeId: fallback.id })
   577→   577→            currentNode = fallback
   578→   578→          } else {
   579→   579→            trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   580→   580→            currentNode = undefined
   581→   581→          }
   582→   582→        }
   583→   583→        break
   584→   584→      }
   585→   585→
   586→   586→      // ── LOGIC: set_var ────────────────────────────────────────────────
   587→   587→      case 'set_var': {
   588→   588→        const name = currentNode.data.variable || ''
   589→   589→        const val = interpolate(currentNode.data.value || '', ctx)
   590→   590→        if (name) {
   591→   591→          ctx.variables[name] = val
   592→   592→          trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: name, value: val })
   593→   593→        }
   594→   594→        currentNode = followEdge(currentNode.id, null)
   595→   595→        break
   596→   596→      }
   597→   597→
   598→   598→      // ── LOGIC: counter (increment/decrement a numeric variable) ───────
   599→   599→      case 'counter': {
   600→   600→        const name = currentNode.data.variable || ''
   601→   601→        const inc = currentNode.data.increment ?? 1
   602→   602→        const startVal = currentNode.data.startValue ?? 0
   603→   603→        const current = parseInt(ctx.variables[name] || '', 10)
   604→   604→        const newVal = isNaN(current) ? startVal + inc : current + inc
   605→   605→        ctx.variables[name] = String(newVal)
   606→   606→        trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: name, value: String(newVal) })
   607→   607→        currentNode = followEdge(currentNode.id, null)
   608→   608→        break
   609→   609→      }
   610→   610→
   611→   611→      // ── LOGIC: format_string (template a string with variables) ──────
   612→   612→      case 'format_string': {
   613→   613→        const template = currentNode.data.text || ''
   614→   614→        const result = interpolate(template, ctx)
   615→   615→        const vname = currentNode.data.variableName
   616→   616→        if (vname) {
   617→   617→          ctx.variables[vname] = result
   618→   618→          trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: result })
   619→   619→        }
   620→   620→        currentNode = followEdge(currentNode.id, null)
   621→   621→        break
   622→   622→      }
   623→   623→
   624→   624→      // ── LOGIC: delay ──────────────────────────────────────────────────
   625→   625→      case 'delay': {
   626→   626→        const s = Math.max(0, Math.min(60, currentNode.data.seconds || 1))
   627→   627→        await new Promise((r) => setTimeout(r, s * 1000))
   628→   628→        currentNode = followEdge(currentNode.id, null)
   629→   629→        break
   630→   630→      }
   631→   631→
   632→   632→      // ── LOGIC: log (debug logging — shows in trace, no user output) ───
   633→   633→      case 'log': {
   634→   634→        const msg = interpolate(currentNode.data.logMessage || '', ctx)
   635→   635→        const level = currentNode.data.logLevel || 'info'
   636→   636→        trace.push({ type: 'log', timestamp: now(), nodeId: currentNode.id, level, message: msg })
   637→   637→        currentNode = followEdge(currentNode.id, null)
   638→   638→        break
   639→   639→      }
   640→   640→
   641→   641→      // ── LOGIC: stop ───────────────────────────────────────────────────
   642→   642→      case 'stop':
   643→   643→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'stop', sentCount })
   644→   644→        return { sentCount, paused: false, variables: ctx.variables, trace }
   645→   645→
   646→   646→      // ── OUTPUT: tts (Pocket TTS voice message) ────────────────────────
   647→   647→      case 'tts': {
   648→   648→        const ttsStart = now()
   649→   649→        const ttsNode = currentNode
   650→   650→        const text = interpolate(ttsNode.data.ttsText || '', ctx)
   651→   651→        const voice = ttsNode.data.ttsVoice || 'alba'
   652→   652→
   653→   653→        if (!text.trim()) {
   654→   654→          trace.push({ type: 'error', timestamp: now(), nodeId: ttsNode.id, message: 'TTS node has empty text' })
   655→   655→          currentNode = followEdge(ttsNode.id, null)
   656→   656→          break
   657→   657→        }
   658→   658→
   659→   659→        trace.push({ type: 'log', timestamp: now(), nodeId: ttsNode.id, level: 'info', message: `TTS: generating "${text.slice(0, 60)}…" with voice=${voice}` })
   660→   660→
   661→   661→        let mediaUrl: string | null = null
   662→   662→        if (ctx.generateTTS) {
   663→   663→          try {
   664→   664→            mediaUrl = await ctx.generateTTS(text, voice)
   665→   665→          } catch (e: any) {
   666→   666→            trace.push({ type: 'error', timestamp: now(), nodeId: ttsNode.id, message: `TTS failed: ${e?.message || e}` })
   667→   667→          }
   668→   668→        } else {
   669→   669→          // Debug mode — no actual TTS available
   670→   670→          trace.push({ type: 'error', timestamp: now(), nodeId: ttsNode.id, message: 'TTS not available in debug mode' })
   671→   671→        }
   672→   672→
   673→   673→        if (mediaUrl) {
   674→   674→          // Send as a voice message
   675→   675→          if (ctx.replyWithMedia) {
   676→   676→            await ctx.replyWithMedia(mediaUrl, 'audio')
   677→   677→          } else {
   678→   678→            await ctx.reply(`🔊 ${mediaUrl}`)
   679→   679→          }
   680→   680→          sentCount++
   681→   681→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: ttsNode.id, text: `[voice message: ${text.slice(0, 60)}…]` })
   682→   682→          trace.push({ type: 'ai_call', timestamp: now(), nodeId: ttsNode.id, model: `tts:${voice}`, prompt: text.slice(0, 100), responseLength: 0, durationMs: now() - ttsStart })
   683→   683→        } else {
   684→   684→          // Fallback: send the text as a plain message
   685→   685→          await ctx.reply(`🔊 ${text}`)
   686→   686→          sentCount++
   687→   687→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: ttsNode.id, text: `[TTS fallback] ${text.slice(0, 80)}` })
   688→   688→        }
   689→   689→        currentNode = followEdge(ttsNode.id, null)
   690→   690→        break
   691→   691→      }
   692→   692→
   693→   693→      // ── ADVANCED: api_call ────────────────────────────────────────────
   694→   694→      case 'api_call': {
   695→   695→        const callStart = now()
   696→   696→        try {
   697→   697→          const url = interpolate(currentNode.data.url || '', ctx)
   698→   698→          const method = currentNode.data.method || 'GET'
   699→   699→          const headers = currentNode.data.headers ? JSON.parse(currentNode.data.headers) : {}
   700→   700→          const body = currentNode.data.body ? interpolate(currentNode.data.body, ctx) : undefined
   701→   701→          const res = await fetch(url, { method, headers, body })
   702→   702→          const text = await res.text()
   703→   703→          const vname = currentNode.data.variableName
   704→   704→          if (vname) {
   705→   705→            ctx.variables[vname] = text.slice(0, 4000)
   706→   706→            trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: text.slice(0, 200) + (text.length > 200 ? '…' : '') })
   707→   707→          }
   708→   708→          trace.push({ type: 'api_call', timestamp: now(), nodeId: currentNode.id, url, method, status: res.status, durationMs: now() - callStart })
   709→   709→        } catch (e: any) {
   710→   710→          trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `API call failed: ${e?.message || e}` })
   711→   711→        }
   712→   712→        currentNode = followEdge(currentNode.id, null)
   713→   713→        break
   714→   714→      }
   715→   715→
   716→   716→      // ── ADVANCED: random ──────────────────────────────────────────────
   717→   717→      case 'random': {
   718→   718→        const allEdges = edgesFrom(currentNode.id, undefined)
   719→   719→        if (allEdges.length === 0) {
   720→   720→          trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   721→   721→          currentNode = undefined
   722→   722→          break
   723→   723→        }
   724→   724→        const pick = allEdges[Math.floor(Math.random() * allEdges.length)]
   725→   725→        const target = flow.nodes.find((n) => n.id === pick.target)
   726→   726→        if (target) {
   727→   727→          trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: pick.sourceHandle || 'random', targetNodeId: target.id })
   728→   728→          currentNode = target
   729→   729→        } else {
   730→   730→          currentNode = undefined
   731→   731→        }
   732→   732→        break
   733→   733→      }
   734→   734→
   735→   735→      // ── ADVANCED: ai_generate ──────────────────────────────────────────
   736→   736→      case 'ai_generate': {
   737→   737→        const vname = currentNode.data.variableName || 'aiResponse'
   738→   738→        const callStart = now()
   739→   739→        try {
   740→   740→          const model = currentNode.data.aiModel || 'gemma3:270m'
   741→   741→          const prompt = interpolate(currentNode.data.aiPrompt || '', ctx)
   742→   742→          const systemPrompt = currentNode.data.aiSystemPrompt
   743→   743→            ? interpolate(currentNode.data.aiSystemPrompt, ctx)
   744→   744→            : undefined
   745→   745→          const temperature = currentNode.data.aiTemperature ?? 0.7
   746→   746→          const maxTokens = currentNode.data.aiMaxTokens ?? 256
   747→   747→
   748→   748→          const ollamaUrl = process.env.OLLAMA_URL || 'http://localhost:11434'
   749→   749→          const res = await fetch(`${ollamaUrl}/api/generate`, {
   750→   750→            method: 'POST',
   751→   751→            headers: { 'Content-Type': 'application/json' },
   752→   752→            body: JSON.stringify({
   753→   753→              model, prompt, system: systemPrompt, stream: false,
   754→   754→              options: { temperature, num_predict: maxTokens },
   755→   755→            }),
   756→   756→          })
   757→   757→
   758→   758→          if (!res.ok) {
   759→   759→            const errText = await res.text().catch(() => 'unknown error')
   760→   760→            const errMsg = `Ollama error ${res.status}: ${errText.slice(0, 100)}`
   761→   761→            ctx.variables[vname] = `[AI error: ${res.status}]`
   762→   762→            trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: errMsg })
   763→   763→            trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: ctx.variables[vname] })
   764→   764→            trace.push({ type: 'ai_call', timestamp: now(), nodeId: currentNode.id, model, prompt: prompt.slice(0, 100), responseLength: 0, durationMs: now() - callStart })
   765→   765→          } else {
   766→   766→            const data = await res.json()
   767→   767→            const generated: string = data.response || ''
   768→   768→            ctx.variables[vname] = generated.slice(0, 4000)
   769→   769→            trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: generated.slice(0, 200) + (generated.length > 200 ? '…' : '') })
   770→   770→            trace.push({ type: 'ai_call', timestamp: now(), nodeId: currentNode.id, model, prompt: prompt.slice(0, 100), responseLength: generated.length, durationMs: now() - callStart })
   771→   771→          }
   772→   772→        } catch (e: any) {
   773→   773→          ctx.variables[vname] = `[AI error: ${e?.message || 'unavailable'}]`
   774→   774→          trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `AI generate failed: ${e?.message || e}` })
   775→   775→          trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: ctx.variables[vname] })
   776→   776→        }
   777→   777→        currentNode = followEdge(currentNode.id, null)
   778→   778→        break
   779→   779→      }
   780→   780→
   781→   781→      default:
   782→   782→        trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `unknown node type: ${currentNode.type}` })
   783→   783→        return { sentCount, paused: false, variables: ctx.variables, trace, error: `unknown node type: ${currentNode.type}` }
   784→   784→    }
   785→   785→
   786→   786→    // Record node exit
   787→   787→    if (currentNode) {
   788→   788→      trace.push({ type: 'node_exit', timestamp: now(), nodeId: currentNode.id, durationMs: 0 })
   789→   789→    }
   790→   790→  }
   791→   791→
   792→   792→  if (steps >= MAX_STEPS) {
   793→   793→    await ctx.reply('⚠️ Bot flow exceeded maximum steps (possible infinite loop).')
   794→   794→    trace.push({ type: 'error', timestamp: now(), message: 'max steps exceeded' })
   795→   795→    trace.push({ type: 'flow_end', timestamp: now(), reason: 'max_steps', sentCount })
   796→   796→    return { sentCount, paused: false, variables: ctx.variables, trace, error: 'max steps' }
   797→   797→  }
   798→   798→
   799→   799→  trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   800→   800→  return { sentCount, paused: false, variables: ctx.variables, trace }
   801→   801→}
