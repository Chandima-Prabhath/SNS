   274→/**
   275→ * useVoiceRecorder — hook that manages MediaRecorder state for recording voice
   276→ * messages. Tap the mic to start, tap stop to upload + send, or cancel to
   277→ * discard. The recording is uploaded via /api/upload and sent as a message
   278→ * with mediaType 'audio/webm' (or the browser's best supported format).
   279→ */
   280→function useVoiceRecorder({
   281→  onSend,
   282→}: {
   283→  onSend: (url: string, mediaType: string) => Promise<void>
   284→}) {
   285→  const [isRecording, setIsRecording] = useState(false)
   286→  const [isUploading, setIsUploading] = useState(false)
   287→  const [seconds, setSeconds] = useState(0)
   288→  const mediaRecorderRef = useRef<MediaRecorder | null>(null)
   289→  const chunksRef = useRef<Blob[]>([])
   290→  const streamRef = useRef<MediaStream | null>(null)
   291→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   292→  const mimeTypeRef = useRef<string>('audio/webm')
   293→
   294→  // Stop the timer + release the mic on unmount
   295→  useEffect(() => {
   296→    return () => {
   297→      if (timerRef.current) clearInterval(timerRef.current)
   298→      if (streamRef.current) streamRef.current.getTracks().forEach((t) => t.stop())
   299→    }
   300→  }, [])
   301→
   302→  const pickMimeType = useCallback(() => {
   303→    const candidates = [
   304→      'audio/webm;codecs=opus',
   305→      'audio/webm',
   306→      'audio/ogg;codecs=opus',
   307→      'audio/mp4',
   308→    ]
   309→    for (const t of candidates) {
   310→      if (typeof MediaRecorder !== 'undefined' && MediaRecorder.isTypeSupported(t)) {
   311→        return t
   312→      }
   313→    }
   314→    return ''
   315→  }, [])
   316→
   317→  const stopTracks = useCallback(() => {
   318→    if (timerRef.current) {
   319→      clearInterval(timerRef.current)
   320→      timerRef.current = null
   321→    }
   322→    if (streamRef.current) {
   323→      streamRef.current.getTracks().forEach((t) => t.stop())
   324→      streamRef.current = null
   325→    }
   326→  }, [])
   327→
   328→  const start = useCallback(async () => {
   329→    try {
   330→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   331→      streamRef.current = stream
   332→      const mimeType = pickMimeType()
   333→      mimeTypeRef.current = mimeType || 'audio/webm'
   334→      const recorder = mimeType
   335→        ? new MediaRecorder(stream, { mimeType })
   336→        : new MediaRecorder(stream)
   337→      chunksRef.current = []
   338→      recorder.ondataavailable = (e) => {
   339→        if (e.data && e.data.size > 0) chunksRef.current.push(e.data)
   340→      }
   341→      recorder.start()
   342→      mediaRecorderRef.current = recorder
   343→      setIsRecording(true)
   344→      setSeconds(0)
   345→      timerRef.current = setInterval(() => {
   346→        setSeconds((s) => s + 1)
   347→      }, 1000)
   348→    } catch {
   349→      toast.error('Microphone access denied')
   350→      if (streamRef.current) {
   351→        streamRef.current.getTracks().forEach((t) => t.stop())
   352→        streamRef.current = null
   353→      }
   354→    }
   355→  }, [pickMimeType])
   356→
   357→  const cancel = useCallback(() => {
   358→    const recorder = mediaRecorderRef.current
   359→    if (recorder && recorder.state !== 'inactive') {
   360→      // Detach onstop so the pending send is discarded
   361→      recorder.onstop = null
   362→      try {
   363→        recorder.stop()
   364→      } catch {
   365→        // ignore
   366→      }
   367→    }
   368→    mediaRecorderRef.current = null
   369→    chunksRef.current = []
   370→    setIsRecording(false)
   371→    setSeconds(0)
   372→    stopTracks()
   373→  }, [stopTracks])
   374→
   375→  const stopAndSend = useCallback(async () => {
   376→    const recorder = mediaRecorderRef.current
   377→    if (!recorder || recorder.state === 'inactive') {
   378→      cancel()
   379→      return
   380→    }
   381→    const mime = mimeTypeRef.current
   382→    const blobPromise = new Promise<Blob>((resolve) => {
   383→      recorder.onstop = () => {
   384→        const blob = new Blob(chunksRef.current, { type: mime || 'audio/webm' })
   385→        resolve(blob)
   386→      }
   387→    })
   388→    try {
   389→      recorder.stop()
   390→    } catch {
   391→      cancel()
   392→      return
   393→    }
   394→    setIsRecording(false)
   395→    stopTracks()
   396→    setIsUploading(true)
   397→    try {
   398→      const blob = await blobPromise
   399→      if (blob.size === 0) {
   400→        toast.error('Recording was empty')
   401→        return
   402→      }
   403→      const ext = mime.includes('mp4') ? 'm4a' : mime.includes('ogg') ? 'ogg' : 'webm'
   404→      const file = new File([blob], `voice-${Date.now()}.${ext}`, { type: blob.type || mime })
   405→      const formData = new FormData()
   406→      formData.append('file', file)
   407→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   408→      if (!res.ok) {
   409→        const t = await res.text().catch(() => 'Upload failed')
   410→        throw new Error(t.slice(0, 120) || 'Upload failed')
   411→      }
   412→      const data = await res.json()
   413→      await onSend(data.url, data.type || mime || 'audio/webm')
   414→      toast.success('Voice message sent')
   415→    } catch (e: any) {
   416→      toast.error(e?.message || 'Failed to send voice message')
   417→    } finally {
   418→      setIsUploading(false)
   419→      setSeconds(0)
   420→      chunksRef.current = []
   421→      mediaRecorderRef.current = null
   422→    }
   423→  }, [cancel, onSend, stopTracks])
   424→
   425→  return { isRecording, isUploading, seconds, start, cancel, stopAndSend }
   426→}

   428→/**
   429→ * TTS Dialog — lets the user type text, pick a voice (built-in or custom),
   430→ * preview, and send. Also supports creating custom voices from audio clips.
   431→ */
   432→function TtsDialog({
   433→  open,
   434→  onOpenChange,
   435→  onSend,
   436→}: {
   437→  open: boolean
   438→  onOpenChange: (o: boolean) => void
   439→  onSend: (url: string) => Promise<void>
   440→}) {
   441→  const [tab, setTab] = useState<'generate' | 'voices'>('generate')
   442→  const [text, setText] = useState('')
   443→  const [voice, setVoice] = useState('alba')
   444→  const [selectedCustomVoiceId, setSelectedCustomVoiceId] = useState<string | null>(null)
   445→  const [generating, setGenerating] = useState(false)
   446→  const [previewUrl, setPreviewUrl] = useState<string | null>(null)
   447→  const [previewBlob, setPreviewBlob] = useState<Blob | null>(null)
   448→  const [isStreaming, setIsStreaming] = useState(false)
   449→  const [sending, setSending] = useState(false)
   450→  const qc = useQueryClient()
   451→
   452→  // Fetch built-in voices from the API (single source of truth for labels)
   453→  const { data: builtinVoicesData } = useQuery({
   454→    queryKey: ['tts-builtin-voices'],
   455→    queryFn: async () => {
   456→      const res = await fetch('/api/tts')
   457→      if (!res.ok) throw new Error('failed')
   458→      return res.json()
   459→    },
   460→    enabled: open,
   461→  })
   462→  const builtinVoices: { id: string; name: string; language: string; gender: string }[] =
   463→    builtinVoicesData?.voices || []
   464→
   465→  // Fetch custom voices
   466→  const { data: customVoicesData } = useQuery({
   467→    queryKey: ['tts-voices'],
   468→    queryFn: async () => {
   469→      const res = await fetch('/api/tts/voices')
   470→      if (!res.ok) throw new Error('failed')
   471→      return res.json()
   472→    },
   473→    enabled: open,
   474→  })
   475→  const customVoices = customVoicesData?.voices || []
   476→
   477→  const handleGenerate = async () => {
   478→    if (!text.trim() || generating) return
   479→    setGenerating(true)
   480→    // Clear any previous preview (revoke blob URL to avoid leaks)
   481→    if (previewUrl?.startsWith('blob:')) URL.revokeObjectURL(previewUrl)
   482→    setPreviewUrl(null)
   483→    setPreviewBlob(null)
   484→    setIsStreaming(true)
   485→    try {
   486→      const body: any = { text }
   487→      if (selectedCustomVoiceId) {
   488→        body.customVoiceId = selectedCustomVoiceId
   489→      } else {
   490→        body.voice = voice
   491→      }
   492→      const res = await fetch('/api/tts', {
   493→        method: 'POST',
   494→        headers: { 'Content-Type': 'application/json' },
   495→        body: JSON.stringify(body),
   496→      })
   497→      if (!res.ok) {
   498→        const err = await res.json().catch(() => ({ error: 'Generation failed' }))
   499→        throw new Error(err.error || 'Generation failed')
   500→      }
   501→
   502→      // The API route streams the TTS audio directly (no server-side save).
   503→      // We read the stream for TWO purposes simultaneously:
   504→      //   1. Instant playback via Web Audio API (chunks played as they arrive)
   505→      //   2. Collect ALL chunks into a Blob for upload when the user sends
   506→      //
   507→      // When the user clicks "Send", we upload the Blob to /api/upload which
   508→      // saves synchronously and returns a URL. This avoids the race condition
   509→      // where a background server-side save hadn't finished before the
   510→      // message URL was accessed (which caused 0-length audio in production).
   511→      const reader = res.body!.getReader()
   512→      let audioCtx: AudioContext | null = null
   513→      let headerParsed = false
   514→      let headerBuf = new Uint8Array(44)
   515→      let headerBytes = 0
   516→      let pcmBuffer = new Uint8Array(0)
   517→      let sampleRate = 24000
   518→      let nextStartTime = 0
   519→      const collectedChunks: Uint8Array[] = []
   520→
   521→      const playChunk = (data: Uint8Array) => {
   522→        if (!audioCtx) return
   523→        const samples = Math.floor(data.length / 2)
   524→        if (samples === 0) return
   525→        const audioBuffer = audioCtx.createBuffer(1, samples, sampleRate)
   526→        const int16 = new Int16Array(data.buffer, data.byteOffset, samples)
   527→        const channelData = audioBuffer.getChannelData(0)
   528→        for (let i = 0; i < samples; i++) {
   529→          channelData[i] = int16[i] / 32768
   530→        }
   531→        const source = audioCtx.createBufferSource()
   532→        source.buffer = audioBuffer
   533→        source.connect(audioCtx.destination)
   534→        const startTime = Math.max(audioCtx.currentTime, nextStartTime)
   535→        source.start(startTime)
   536→        nextStartTime = startTime + audioBuffer.duration
   537→      }
   538→
   539→      while (true) {
   540→        const { done, value } = await reader.read()
   541→        if (done) break
   542→        if (value) {
   543→          // Collect every chunk for the final Blob
   544→          collectedChunks.push(value)
   545→
   546→          if (!headerParsed) {
   547→            const needed = 44 - headerBytes
   548→            const copy = Math.min(needed, value.length)
   549→            headerBuf.set(value.slice(0, copy), headerBytes)
   550→            headerBytes += copy
   551→            if (headerBytes >= 44) {
   552→              const view = new DataView(headerBuf.buffer)
   553→              sampleRate = view.getUint32(24, true)
   554→              headerParsed = true
   555→              if (!audioCtx) {
   556→                audioCtx = new AudioContext({ latencyHint: 'playback' })
   557→              }
   558→              if (value.length > copy) {
   559→                pcmBuffer = new Uint8Array(value.slice(copy))
   560→              }
   561→            }
   562→          } else {
   563→            const merged = new Uint8Array(pcmBuffer.length + value.length)
   564→            merged.set(pcmBuffer)
   565→            merged.set(value, pcmBuffer.length)
   566→            if (merged.length >= 16384) {
   567→              playChunk(merged)
   568→              pcmBuffer = new Uint8Array(0)
   569→            } else {
   570→              pcmBuffer = merged
   571→            }
   572→          }
   573→        }
   574→      }
   575→      if (pcmBuffer.length > 0) playChunk(pcmBuffer)
   576→
   577→      // Build the final Blob from all collected chunks and create a blob URL
   578→      // for the preview player. The Blob is kept in state so handleSend can
   579→      // upload it to /api/upload.
   580→      const wavBlob = new Blob(collectedChunks as BlobPart[], { type: 'audio/wav' })
   581→      console.log(`[tts] collected ${collectedChunks.length} chunks, blob size = ${wavBlob.size} bytes`)
   582→      if (wavBlob.size === 0) {
   583→        throw new Error('TTS produced empty audio (0 bytes collected from stream)')
   584→      }
   585→      const blobUrl = URL.createObjectURL(wavBlob)
   586→      setPreviewBlob(wavBlob)
   587→      setPreviewUrl(blobUrl)
   588→      setIsStreaming(false)
   589→      toast.success('Voice generated — preview and send')
   590→    } catch (e: any) {
   591→      setIsStreaming(false)
   592→      toast.error(e.message || 'Failed to generate voice')
   593→    } finally {
   594→      setGenerating(false)
   595→    }
   596→  }
   597→
   598→  const handleSend = async () => {
   599→    if (!previewBlob || sending) return
   600→    setSending(true)
   601→    try {
   602→      // Upload the collected Blob to /api/upload.
   603→      const formData = new FormData()
   604→      formData.append('file', previewBlob, `tts-${Date.now()}.wav`)
   605→      console.log('[tts] uploading blob, size=', previewBlob.size, 'type=', previewBlob.type)
   606→      const uploadRes = await fetch('/api/upload', {
   607→        method: 'POST',
   608→        body: formData,
   609→      })
   610→      if (!uploadRes.ok) {
   611→        const errText = await uploadRes.text().catch(() => 'unknown')
   612→        console.error('[tts] upload failed:', uploadRes.status, errText)
   613→        throw new Error(`Upload failed (${uploadRes.status}): ${errText.slice(0, 100)}`)
   614→      }
   615→      const data = await uploadRes.json()
   616→      console.log('[tts] upload success, url=', data.url)
   617→
   618→      // Revoke the blob URL — the uploaded file URL is now the source of truth
   619→      if (previewUrl?.startsWith('blob:')) URL.revokeObjectURL(previewUrl)
   620→
   621→      await onSend(data.url)
   622→      setText('')
   623→      setPreviewUrl(null)
   624→      setPreviewBlob(null)
   625→    } catch (e: any) {
   626→      console.error('[tts] send failed:', e)
   627→      toast.error(e?.message || 'Failed to send voice message')
   628→    } finally {
   629→      setSending(false)
   630→    }
   631→  }
   632→
   633→  const handleClose = (o: boolean) => {

   640→    }
   641→    onOpenChange(o)
   642→  }

   673→      <DialogContent className="sm:max-w-md">
   674→        <DialogHeader>
   675→          <DialogTitle className="flex items-center gap-2">
   676→            <Sparkles className="w-4 h-4 text-primary" />
   677→            AI Voice Message

   808→                <Label>Preview</Label>
   809→                <audio controls src={previewUrl} className="w-full" />

   816→
   817→        <DialogFooter>
   818→          <Button variant="outline" onClick={() => handleClose(false)}>Cancel</Button>
   819→          {tab === 'generate' && (
   820→            !previewUrl ? (
