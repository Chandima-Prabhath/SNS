   273→   273→/**
   274→   274→ * useVoiceRecorder — hook that manages MediaRecorder state for recording voice
   275→   275→ * messages. Tap the mic to start, tap stop to upload + send, or cancel to
   276→   276→ * discard. The recording is uploaded via /api/upload and sent as a message
   277→   277→ * with mediaType 'audio/webm' (or the browser's best supported format).
   278→   278→ */
   279→   279→function useVoiceRecorder({
   280→   280→  onSend,
   281→   281→}: {
   282→   282→  onSend: (url: string, mediaType: string) => Promise<void>
   283→   283→}) {
   284→   284→  const [isRecording, setIsRecording] = useState(false)
   285→   285→  const [isUploading, setIsUploading] = useState(false)
   286→   286→  const [seconds, setSeconds] = useState(0)
   287→   287→  const mediaRecorderRef = useRef<MediaRecorder | null>(null)
   288→   288→  const chunksRef = useRef<Blob[]>([])
   289→   289→  const streamRef = useRef<MediaStream | null>(null)
   290→   290→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   291→   291→  const mimeTypeRef = useRef<string>('audio/webm')
   292→   292→
   293→   293→  // Stop the timer + release the mic on unmount
   294→   294→  useEffect(() => {
   295→   295→    return () => {
   296→   296→      if (timerRef.current) clearInterval(timerRef.current)
   297→   297→      if (streamRef.current) streamRef.current.getTracks().forEach((t) => t.stop())
   298→   298→    }
   299→   299→  }, [])
   300→   300→
   301→   301→  const pickMimeType = useCallback(() => {
   302→   302→    const candidates = [
   303→   303→      'audio/webm;codecs=opus',
   304→   304→      'audio/webm',
   305→   305→      'audio/ogg;codecs=opus',
   306→   306→      'audio/mp4',
   307→   307→    ]
   308→   308→    for (const t of candidates) {
   309→   309→      if (typeof MediaRecorder !== 'undefined' && MediaRecorder.isTypeSupported(t)) {
   310→   310→        return t
   311→   311→      }
   312→   312→    }
   313→   313→    return ''
   314→   314→  }, [])
   315→   315→
   316→   316→  const stopTracks = useCallback(() => {
   317→   317→    if (timerRef.current) {
   318→   318→      clearInterval(timerRef.current)
   319→   319→      timerRef.current = null
   320→   320→    }
   321→   321→    if (streamRef.current) {
   322→   322→      streamRef.current.getTracks().forEach((t) => t.stop())
   323→   323→      streamRef.current = null
   324→   324→    }
   325→   325→  }, [])
   326→   326→
   327→   327→  const start = useCallback(async () => {
   328→   328→    try {
   329→   329→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   330→   330→      streamRef.current = stream
   331→   331→      const mimeType = pickMimeType()
   332→   332→      mimeTypeRef.current = mimeType || 'audio/webm'
   333→   333→      const recorder = mimeType
   334→   334→        ? new MediaRecorder(stream, { mimeType })
   335→   335→        : new MediaRecorder(stream)
   336→   336→      chunksRef.current = []
   337→   337→      recorder.ondataavailable = (e) => {
   338→   338→        if (e.data && e.data.size > 0) chunksRef.current.push(e.data)
   339→   339→      }
   340→   340→      recorder.start()
   341→   341→      mediaRecorderRef.current = recorder
   342→   342→      setIsRecording(true)
   343→   343→      setSeconds(0)
   344→   344→      timerRef.current = setInterval(() => {
   345→   345→        setSeconds((s) => s + 1)
   346→   346→      }, 1000)
   347→   347→    } catch {
   348→   348→      toast.error('Microphone access denied')
   349→   349→      if (streamRef.current) {
   350→   350→        streamRef.current.getTracks().forEach((t) => t.stop())
   351→   351→        streamRef.current = null
   352→   352→      }
   353→   353→    }
   354→   354→  }, [pickMimeType])
   355→   355→
   356→   356→  const cancel = useCallback(() => {
   357→   357→    const recorder = mediaRecorderRef.current
   358→   358→    if (recorder && recorder.state !== 'inactive') {
   359→   359→      // Detach onstop so the pending send is discarded
   360→   360→      recorder.onstop = null
   361→   361→      try {
   362→   362→        recorder.stop()
   363→   363→      } catch {
   364→   364→        // ignore
   365→   365→      }
   366→   366→    }
   367→   367→    mediaRecorderRef.current = null
   368→   368→    chunksRef.current = []
   369→   369→    setIsRecording(false)
   370→   370→    setSeconds(0)
   371→   371→    stopTracks()
   372→   372→  }, [stopTracks])
   373→   373→
   374→   374→  const stopAndSend = useCallback(async () => {
   375→   375→    const recorder = mediaRecorderRef.current
   376→   376→    if (!recorder || recorder.state === 'inactive') {
   377→   377→      cancel()
   378→   378→      return
   379→   379→    }
   380→   380→    const mime = mimeTypeRef.current
   381→   381→    const blobPromise = new Promise<Blob>((resolve) => {
   382→   382→      recorder.onstop = () => {
   383→   383→        const blob = new Blob(chunksRef.current, { type: mime || 'audio/webm' })
   384→   384→        resolve(blob)
   385→   385→      }
   386→   386→    })
   387→   387→    try {
   388→   388→      recorder.stop()
   389→   389→    } catch {
   390→   390→      cancel()
   391→   391→      return
   392→   392→    }
   393→   393→    setIsRecording(false)
   394→   394→    stopTracks()
   395→   395→    setIsUploading(true)
   396→   396→    try {
   397→   397→      const blob = await blobPromise
   398→   398→      if (blob.size === 0) {
   399→   399→        toast.error('Recording was empty')
   400→   400→        return
   401→   401→      }
   402→   402→      const ext = mime.includes('mp4') ? 'm4a' : mime.includes('ogg') ? 'ogg' : 'webm'
   403→   403→      const file = new File([blob], `voice-${Date.now()}.${ext}`, { type: blob.type || mime })
   404→   404→      const formData = new FormData()
   405→   405→      formData.append('file', file)
   406→   406→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   407→   407→      if (!res.ok) {
   408→   408→        const t = await res.text().catch(() => 'Upload failed')
   409→   409→        throw new Error(t.slice(0, 120) || 'Upload failed')
   410→   410→      }
   411→   411→      const data = await res.json()
   412→   412→      await onSend(data.url, data.type || mime || 'audio/webm')
   413→   413→      toast.success('Voice message sent')
   414→   414→    } catch (e: any) {
   415→   415→      toast.error(e?.message || 'Failed to send voice message')
   416→   416→    } finally {
   417→   417→      setIsUploading(false)
   418→   418→      setSeconds(0)
   419→   419→      chunksRef.current = []
   420→   420→      mediaRecorderRef.current = null
   421→   421→    }
   422→   422→  }, [cancel, onSend, stopTracks])
   423→   423→
   424→   424→  return { isRecording, isUploading, seconds, start, cancel, stopAndSend }
   425→   425→}

   427→   427→/**
   428→   428→ * TTS Dialog — lets the user type text, pick a voice (built-in or custom),
   429→   429→ * preview, and send. Also supports creating custom voices from audio clips.
   430→   430→ */
   431→   431→function TtsDialog({
   432→   432→  open,
   433→   433→  onOpenChange,
   434→   434→  onSend,
   435→   435→}: {
   436→   436→  open: boolean
   437→   437→  onOpenChange: (o: boolean) => void
   438→   438→  onSend: (url: string) => Promise<void>
   439→   439→}) {
