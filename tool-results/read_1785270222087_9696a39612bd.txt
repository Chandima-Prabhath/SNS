     1→'use client'
     2→
     3→import { useState, useRef, useEffect, useCallback } from 'react'
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
     5→import { motion, AnimatePresence } from 'framer-motion'
     6→import { Avatar, AvatarFallback, AvatarImage } from '@/components/ui/avatar'
     7→import { Button } from '@/components/ui/button'
     8→import { Input } from '@/components/ui/input'
     9→import { Card } from '@/components/ui/card'
    10→import {
    11→  Search, Play, Pause, SkipForward, Plus, Users, Music as MusicIcon,
    12→  Volume2, Loader2, Radio, Headphones, X, ListMusic, Compass,
    13→  Trash2, Repeat, Shuffle, Heart,
    14→} from 'lucide-react'
    15→import { toast } from 'sonner'
    16→import { cn } from '@/lib/utils'
    17→import { useSocket } from '@/hooks/useSocket'
    18→
    19→interface Track {
    20→  videoId: string
    21→  title: string
    22→  artist: string
    23→  thumbnail: string | null
    24→  durationSeconds: number | null
    25→  album?: string | null
    26→}
    27→
    28→interface Room {
    29→  id: string
    30→  name: string
    31→  host: any
    32→  currentVideoId: string | null
    33→  currentState: string
    34→  currentPosition: number
    35→  queue: string
    36→  members: any[]
    37→}
    38→
    39→type Tab = 'browse' | 'rooms' | 'queue'
    40→
    41→export function MusicView() {
    42→  const [tab, setTab] = useState<Tab>('browse')
    43→  const [activeRoomId, setActiveRoomId] = useState<string | null>(null)
    44→  const [searchQuery, setSearchQuery] = useState('')
    45→  const [searchResults, setSearchResults] = useState<Track[]>([])
    46→  const [searching, setSearching] = useState(false)
    47→  const [currentTrack, setCurrentTrack] = useState<Track | null>(null)
    48→  const [isPlaying, setIsPlaying] = useState(false)
    49→  const [position, setPosition] = useState(0)
    50→  const [volume, setVolume] = useState(0.8)
    51→  const [queue, setQueue] = useState<Track[]>([])
    52→  const [autoplay, setAutoplay] = useState(true)
    53→  const [shuffle, setShuffle] = useState(false)
    54→  const [repeat, setRepeat] = useState(false)
    55→  const audioRef = useRef<HTMLAudioElement>(null)
    56→  const searchTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null)
    57→  const qc = useQueryClient()
    58→  const { socket } = useSocket()
    59→
    60→  // Fetch trending
    61→  const { data: trendingData, isLoading: trendingLoading } = useQuery({
    62→    queryKey: ['music-trending'],
    63→    queryFn: async () => {
    64→      const res = await fetch('/api/music/trending')
    65→      if (!res.ok) throw new Error('failed')
    66→      return res.json()
    67→    },
    68→  })
    69→
    70→  // Fetch rooms
    71→  const { data: roomsData, refetch: refetchRooms } = useQuery({
    72→    queryKey: ['music-rooms'],
    73→    queryFn: async () => {
    74→      const res = await fetch('/api/music/rooms')
    75→      if (!res.ok) throw new Error('failed')
    76→      return res.json()
    77→    },
    78→    refetchInterval: 10000,
    79→  })
    80→
    81→  // Create room
    82→  const createRoom = useMutation({
    83→    mutationFn: async (name: string) => {
    84→      const res = await fetch('/api/music/rooms', {
    85→        method: 'POST',
    86→        headers: { 'Content-Type': 'application/json' },
    87→        body: JSON.stringify({ name }),
    88→      })
    89→      if (!res.ok) throw new Error('failed')
    90→      return res.json()
    91→    },
    92→    onSuccess: (data) => {
    93→      qc.invalidateQueries({ queryKey: ['music-rooms'] })
    94→      setActiveRoomId(data.room.id)
    95→      setTab('rooms')
    96→      toast.success('Room created')
    97→    },
    98→  })
    99→
   100→  // Debounced search
   101→  const handleSearchChange = (value: string) => {
   102→    setSearchQuery(value)
   103→    if (searchTimeoutRef.current) clearTimeout(searchTimeoutRef.current)
   104→    if (!value.trim()) {
   105→      setSearchResults([])
   106→      return
   107→    }
   108→    searchTimeoutRef.current = setTimeout(async () => {
   109→      setSearching(true)
   110→      try {
   111→        const res = await fetch(`/api/music/search?q=${encodeURIComponent(value)}`)
   112→        if (!res.ok) throw new Error('Search failed')
   113→        const data = await res.json()
   114→        setSearchResults(data.tracks || [])
   115→      } catch (e: any) {
   116→        toast.error(e.message || 'Search failed')
   117→      } finally {
   118→        setSearching(false)
   119→      }
   120→    }, 500)
   121→  }
   122→
   123→  // Play a track
   124→  const playTrack = useCallback(async (track: Track, addToQueue = false) => {
   125→    if (addToQueue && currentTrack) {
   126→      setQueue((q) => [...q, track])
   127→      toast.success(`Added to queue: ${track.title}`)
   128→      return
   129→    }
   130→
   131→    setCurrentTrack(track)
   132→    setIsPlaying(true)
   133→    setPosition(0)
   134→
   135→    // If in a room, broadcast the track change to all members via Socket.io
   136→    if (activeRoomId && socket) {
   137→      socket.emit('music:sync', {
   138→        roomId: activeRoomId,
   139→        state: 'playing',
   140→        position: 0,
   141→        videoId: track.videoId,
   142→        trackInfo: {
   143→          title: track.title,
   144→          artist: track.artist,
   145→          thumbnail: track.thumbnail,
   146→          durationSeconds: track.durationSeconds,
   147→        },
   148→      })
   149→    }
   150→
   151→    if (audioRef.current) {
   152→      audioRef.current.src = `/api/music/stream/${track.videoId}`
   153→      audioRef.current.volume = volume
   154→      try {
   155→        await audioRef.current.play()
   156→      } catch (e: any) {
   157→        if (e?.name === 'NotSupportedError' || e?.name === 'MediaError') {
   158→          try {
   159→            const res = await fetch(`/api/music/stream/${track.videoId}`, { method: 'HEAD' })
   160→            if (!res.ok) {
   161→              const data = await res.json().catch(() => ({}))
   162→              toast.error(data.error || 'Could not download this track.')
   163→              setCurrentTrack(null)
   164→              setIsPlaying(false)
   165→            }
   166→          } catch {
   167→            toast.error('The server is downloading this track — try again in a moment.')
   168→          }
   169→        }
   170→      }
   171→    }
   172→  }, [activeRoomId, volume, currentTrack, socket])
   173→
   174→  // Play next track from queue or autoplay
   175→  const playNext = useCallback(async () => {
   176→    let nextTrack: Track | null = null
   177→
   178→    if (queue.length > 0) {
   179→      if (shuffle) {
   180→        const idx = Math.floor(Math.random() * queue.length)
   181→        nextTrack = queue[idx]
   182→        setQueue((q) => q.filter((_, i) => i !== idx))
   183→      } else {
   184→        nextTrack = queue[0]
   185→        setQueue((q) => q.slice(1))
   186→      }
   187→    } else if (autoplay && currentTrack) {
   188→      // Fetch related tracks and play the first one
   189→      try {
   190→        const res = await fetch(`/api/music/related/${currentTrack.videoId}`)
   191→        if (res.ok) {
   192→          const data = await res.json()
   193→          if (data.tracks?.length > 0) {
   194→            nextTrack = data.tracks[0]
   195→          }
   196→        }
   197→      } catch {
   198→        // Autoplay failed — just stop
   199→      }
   200→    }
   201→
   202→    if (nextTrack) {
   203→      await playTrack(nextTrack)
   204→    } else if (repeat && currentTrack) {
   205→      await playTrack(currentTrack)
   206→    } else {
   207→      setIsPlaying(false)
   208→      setPosition(0)
   209→    }
   210→  }, [queue, shuffle, autoplay, currentTrack, repeat, playTrack])
   211→
   212→  // Toggle play/pause
   213→  const togglePlay = () => {
   214→    if (!audioRef.current || !currentTrack) return
   215→    if (isPlaying) {
   216→      audioRef.current.pause()
   217→      setIsPlaying(false)
   218→      // Broadcast pause to room
   219→      if (activeRoomId && socket) {
   220→        socket.emit('music:sync', {
   221→          roomId: activeRoomId,
   222→          state: 'paused',
   223→          position: audioRef.current.currentTime,
   224→        })
   225→      }
   226→    } else {
   227→      audioRef.current.play()
   228→      setIsPlaying(true)
   229→      // Broadcast play to room
   230→      if (activeRoomId && socket) {
   231→        socket.emit('music:sync', {
   232→          roomId: activeRoomId,
   233→          state: 'playing',
   234→          position: audioRef.current.currentTime,
   235→        })
   236→      }
   237→    }
   238→  }
   239→
   240→  // Seek
   241→  const handleSeek = (e: React.ChangeEvent<HTMLInputElement>) => {
   242→    const newPos = parseFloat(e.target.value)
   243→    if (audioRef.current) {
   244→      audioRef.current.currentTime = newPos
   245→      setPosition(newPos)
   246→      // Broadcast seek to room
   247→      if (activeRoomId && socket) {
   248→        socket.emit('music:sync', {
   249→          roomId: activeRoomId,
   250→          state: isPlaying ? 'playing' : 'paused',
   251→          position: newPos,
   252→        })
   253→      }
   254→    }
   255→  }
   256→
   257→  // Audio event handlers
   258→  useEffect(() => {
   259→    const audio = audioRef.current
   260→    if (!audio) return
   261→
   262→    const onTimeUpdate = () => setPosition(audio.currentTime)
   263→    const onEnded = () => {
   264→      // Auto-play next track when current ends
   265→      playNext()
   266→    }
   267→    const onError = () => {
   268→      if (currentTrack) {
   269→        // Skip to next on error
   270→        toast.error('Could not play this track — skipping...')
   271→        setQueue((q) => q.slice(1))
   272→        playNext()
   273→      }
   274→    }
   275→
   276→    audio.addEventListener('timeupdate', onTimeUpdate)
   277→    audio.addEventListener('ended', onEnded)
   278→    audio.addEventListener('error', onError)
   279→    return () => {
   280→      audio.removeEventListener('timeupdate', onTimeUpdate)
   281→      audio.removeEventListener('ended', onEnded)
   282→      audio.removeEventListener('error', onError)
   283→    }
   284→  }, [currentTrack, playNext])
   285→
   286→  // Real-time sync: listen for room sync events from the host.
   287→  // This is the CORE of sync play — when the host plays/pauses/seeks/changes
   288→  // track, the event arrives instantly via Socket.io (not DB polling).
   289→  useEffect(() => {
   290→    if (!socket || !activeRoomId) return
   291→
   292→    // Join the music room's socket channel
   293→    socket.emit('music:join', activeRoomId)
   294→
   295→    // Request the host's current state immediately after joining
   296→    setTimeout(() => {
   297→      socket.emit('music:request-sync', activeRoomId)
   298→    }, 500)
   299→
   300→    const onSync = (data: {
   301→      roomId: string
   302→      state: string
   303→      position: number
   304→      videoId?: string
   305→      trackInfo?: Track
   306→      serverTimestamp: number
   307→    }) => {
   308→      if (data.roomId !== activeRoomId) return
   309→
   310→      // Calculate network delay and adjust position
   311→      const networkDelay = (Date.now() - data.serverTimestamp) / 1000
   312→      const adjustedPosition = data.position + (data.state === 'playing' ? networkDelay : 0)
   313→
   314→      // Track changed — load the new track and sync to host's position
   315→      if (data.videoId && (!currentTrack || currentTrack.videoId !== data.videoId)) {
   316→        const track: Track = data.trackInfo || {
   317→          videoId: data.videoId,
   318→          title: 'Now Playing',
   319→          artist: '',
   320→          thumbnail: null,
   321→          durationSeconds: null,
   322→        }
   323→        setCurrentTrack(track)
   324→        setTimeout(() => {
   325→          if (audioRef.current) {
   326→            audioRef.current.src = `/api/music/stream/${data.videoId}`
   327→            // Set the position to match the host's current position
   328→            audioRef.current.currentTime = adjustedPosition
   329→            if (data.state === 'playing') {
   330→              audioRef.current.play().catch(() => {})
   331→              setIsPlaying(true)
   332→            } else {
   333→              setIsPlaying(false)
   334→            }
   335→          }
   336→        }, 200)
   337→      } else if (audioRef.current && currentTrack) {
   338→        // Same track — just sync position and state
   339→        const drift = Math.abs(audioRef.current.currentTime - adjustedPosition)
   340→
   341→        // Only force-seek if drift > 1.5 seconds (avoid jitter from minor differences)
   342→        if (drift > 1.5) {
   343→          audioRef.current.currentTime = adjustedPosition
   344→        }
   345→
   346→        // Sync play/pause state
   347→        if (data.state === 'playing' && !isPlaying) {
   348→          audioRef.current.play().catch(() => {})
   349→          setIsPlaying(true)
   350→        } else if (data.state === 'paused' && isPlaying) {
   351→          audioRef.current.pause()
   352→          setIsPlaying(false)
   353→        }
   354→      }
   355→    }
   356→
   357→    // When the host receives a sync request, broadcast current state
   358→    const onRequestSync = (data: { roomId: string; fromUserId: string }) => {
   359→      if (data.roomId !== activeRoomId) return
   360→      // Only the host responds to sync requests
   361→      if (currentTrack && socket) {
   362→        socket.emit('music:sync', {
   363→          roomId: activeRoomId,
   364→          state: isPlaying ? 'playing' : 'paused',
   365→          position: audioRef.current?.currentTime || 0,
   366→          videoId: currentTrack.videoId,
   367→          trackInfo: {
   368→            title: currentTrack.title,
   369→            artist: currentTrack.artist,
   370→            thumbnail: currentTrack.thumbnail,
   371→            durationSeconds: currentTrack.durationSeconds,
   372→          },
   373→        })
   374→      }
   375→    }
   376→
   377→    socket.on('music:sync', onSync)
   378→    socket.on('music:request-sync', onRequestSync)
   379→
   380→    return () => {
   381→      socket.off('music:sync', onSync)
   382→      socket.off('music:request-sync', onRequestSync)
   383→      socket.emit('music:leave', activeRoomId)
   384→    }
   385→  }, [socket, activeRoomId, currentTrack, isPlaying])
   386→
   387→  // Delete room
   388→  const deleteRoom = useMutation({
   389→    mutationFn: async (roomId: string) => {
   390→      const res = await fetch(`/api/music/rooms/${roomId}`, { method: 'DELETE' })
   391→      if (!res.ok) throw new Error('failed')
   392→    },
   393→    onSuccess: () => {
   394→      qc.invalidateQueries({ queryKey: ['music-rooms'] })
   395→      if (activeRoomId) setActiveRoomId(null)
   396→      toast.success('Room deleted')
   397→    },
   398→    onError: () => toast.error('Failed to delete room'),
   399→  })
   400→
   401→  const trendingTracks = trendingData?.tracks || []
   402→  const rooms = roomsData?.rooms || []
   403→  const activeRoom = rooms.find((r: Room) => r.id === activeRoomId)
   404→
   405→  return (
   406→    <div className="h-full flex flex-col mesh-gradient overflow-hidden">
   407→      <audio ref={audioRef} />
   408→
   409→      {/* Header with tabs */}
   410→      <div className="shrink-0 px-4 pt-4 pb-2 border-b border-border/30">
   411→        <div className="max-w-5xl mx-auto">
   412→          <h1 className="text-2xl md:text-3xl font-bold tracking-tight flex items-center gap-2 mb-3">
   413→            <MusicIcon className="w-7 h-7 text-primary" />
   414→            Musical
   415→          </h1>
   416→
   417→          {/* Tab navigation */}
   418→          <div className="flex gap-1 p-1 bg-muted/50 rounded-xl max-w-xs">
   419→            {([
   420→              ['browse', 'Browse', Compass],
   421→              ['rooms', 'Rooms', Radio],
   422→              ['queue', 'Queue', ListMusic],
   423→            ] as const).map(([key, label, Icon]) => (
   424→              <button
   425→                key={key}
   426→                onClick={() => setTab(key)}
   427→                className={cn(
   428→                  'flex-1 flex items-center justify-center gap-1.5 py-2 px-3 rounded-lg text-sm font-medium transition-all',
   429→                  tab === key
   430→                    ? 'bg-background text-primary shadow-sm'
   431→                    : 'text-muted-foreground hover:text-foreground'
   432→                )}
   433→              >
   434→                <Icon className="w-4 h-4" />
   435→                {label}
   436→                {key === 'queue' && queue.length > 0 && (
   437→                  <span className="text-[10px] bg-primary/20 text-primary px-1.5 py-0.5 rounded-full">
   438→                    {queue.length}
   439→                  </span>
   440→                )}
   441→              </button>
   442→            ))}
   443→          </div>
   444→        </div>
   445→      </div>
   446→
   447→      {/* Main content area */}
   448→      <div className="flex-1 overflow-y-auto pb-40 lg:pb-28">
   449→        <div className="max-w-5xl mx-auto p-4 md:p-6">
   450→          <AnimatePresence mode="wait">
   451→            {tab === 'browse' && (
   452→              <motion.div
   453→                key="browse"
   454→                initial={{ opacity: 0, y: 10 }}
   455→                animate={{ opacity: 1, y: 0 }}
   456→                exit={{ opacity: 0, y: -10 }}
   457→                transition={{ duration: 0.2 }}
   458→                className="space-y-6"
   459→              >
   460→                {/* Search */}
   461→                <div className="relative">
   462→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   463→                  <Input
   464→                    value={searchQuery}
   465→                    onChange={(e) => handleSearchChange(e.target.value)}
   466→                    placeholder="Search songs, artists..."
   467→                    className="pl-9 h-11 bg-card/50 backdrop-blur-sm border-border/30"
   468→                  />
   469→                  {searching && (
   470→                    <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 animate-spin text-muted-foreground" />
   471→                  )}
   472→                </div>
   473→
   474→                {/* Search results */}
   475→                {searchResults.length > 0 ? (
   476→                  <div className="space-y-1.5">
   477→                    <h2 className="text-xs font-semibold uppercase tracking-wider text-muted-foreground mb-2">
   478→                      Results
   479→                    </h2>
   480→                    {searchResults.map((track) => (
   481→                      <TrackRow
   482→                        key={track.videoId}
   483→                        track={track}
   484→                        onPlay={() => playTrack(track)}
   485→                        onAddToQueue={() => playTrack(track, true)}
   486→                        isCurrent={currentTrack?.videoId === track.videoId}
   487→                        isPlaying={isPlaying && currentTrack?.videoId === track.videoId}
   488→                      />
   489→                    ))}
   490→                  </div>
   491→                ) : (
   492→                  <>
   493→                    {/* Trending */}
   494→                    <div>
   495→                      <h2 className="text-xs font-semibold uppercase tracking-wider text-muted-foreground mb-3">
   496→                        Trending Now
   497→                      </h2>
   498→                      {trendingLoading ? (
   499→                        <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-3">
   500→                          {[1, 2, 3, 4, 5, 6].map((i) => (
   501→                            <div key={i} className="space-y-2 animate-pulse">
   502→                              <div className="aspect-square rounded-xl bg-muted" />
   503→                              <div className="h-3 bg-muted rounded w-3/4" />
   504→                              <div className="h-2 bg-muted rounded w-1/2" />
   505→                            </div>
   506→                          ))}
   507→                        </div>
   508→                      ) : (
   509→                        <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-3">
   510→                          {trendingTracks.map((track: Track) => (
   511→                            <TrackCard
   512→                              key={track.videoId}
   513→                              track={track}
   514→                              onPlay={() => playTrack(track)}
   515→                              onAddToQueue={() => playTrack(track, true)}
   516→                              isCurrent={currentTrack?.videoId === track.videoId}
   517→                              isPlaying={isPlaying && currentTrack?.videoId === track.videoId}
   518→                            />
   519→                          ))}
   520→                        </div>
   521→                      )}
   522→                    </div>
   523→                  </>
   524→                )}
   525→              </motion.div>
   526→            )}
   527→
   528→            {tab === 'rooms' && (
   529→              <motion.div
   530→                key="rooms"
   531→                initial={{ opacity: 0, y: 10 }}
   532→                animate={{ opacity: 1, y: 0 }}
   533→                exit={{ opacity: 0, y: -10 }}
   534→                transition={{ duration: 0.2 }}
   535→                className="space-y-4"
   536→              >
   537→                <div className="flex items-center justify-between">
   538→                  <h2 className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   539→                    Listening Rooms
   540→                  </h2>
   541→                  <Button
   542→                    variant="outline"
   543→                    size="sm"
   544→                    onClick={() => {
   545→                      const name = prompt('Room name?')
   546→                      if (name?.trim()) createRoom.mutate(name)
   547→                    }}
   548→                  >
   549→                    <Plus className="w-4 h-4 mr-1" />
   550→                    Create Room
   551→                  </Button>
   552→                </div>
   553→
   554→                {rooms.length === 0 ? (
   555→                  <Card className="p-12 text-center border-dashed">
   556→                    <Headphones className="w-12 h-12 mx-auto text-muted-foreground mb-3" strokeWidth={1.5} />
   557→                    <p className="font-medium">No rooms yet</p>
   558→                    <p className="text-sm text-muted-foreground mt-1">
   559→                      Create a room to listen together with friends.
   560→                    </p>
   561→                  </Card>
   562→                ) : (
   563→                  <div className="grid gap-2">
   564→                    {rooms.map((room: Room) => (
   565→                      <RoomCard
   566→                        key={room.id}
   567→                        room={room}
   568→                        isActive={activeRoomId === room.id}
   569→                        onJoin={() => {
   570→                          // Just set the active room — the Socket.io sync
   571→                          // effect will join the socket channel and request
   572→                          // the host's current state automatically.
   573→                          setActiveRoomId(room.id)
   574→                          // Also call the API to auto-join as a member
   575→                          fetch(`/api/music/rooms/${room.id}`).catch(() => {})
   576→                        }}
   577→                        onDelete={() => deleteRoom.mutate(room.id)}
   578→                      />
   579→                    ))}
   580→                  </div>
   581→                )}
   582→              </motion.div>
   583→            )}
   584→
   585→            {tab === 'queue' && (
   586→              <motion.div
   587→                key="queue"
   588→                initial={{ opacity: 0, y: 10 }}
   589→                animate={{ opacity: 1, y: 0 }}
   590→                exit={{ opacity: 0, y: -10 }}
