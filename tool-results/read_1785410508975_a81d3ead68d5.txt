     1→'use client'
     2→
     3→import { useState, useRef } from 'react'
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
     5→import { motion, AnimatePresence } from 'framer-motion'
     6→import { Button } from '@/components/ui/button'
     7→import { Input } from '@/components/ui/input'
     8→import { Card } from '@/components/ui/card'
     9→import {
    10→  Search, Play, Pause, SkipForward, Plus,
    11→  Loader2, Radio, Headphones, X, ListMusic, Compass,
    12→  Trash2, Repeat, Shuffle, Music as MusicIcon, Users,
    13→} from 'lucide-react'
    14→import { toast } from 'sonner'
    15→import { cn } from '@/lib/utils'
    16→import { useMusicStore, type Track } from '@/stores/useMusicStore'
    17→import { useMusicPlayer } from '@/components/music/global-music-player'
    18→import { SpotlightCard, GlassSurface, GradientText, BorderGlow, ShinyText, StarBorder } from '@/components/reactbits'
    19→
    20→interface Room {
    21→  id: string
    22→  name: string
    23→  host: any
    24→  currentVideoId: string | null
    25→  currentState: string
    26→  currentPosition: number
    27→  queue: string
    28→  members: any[]
    29→}
    30→
    31→type Tab = 'browse' | 'rooms' | 'queue'
    32→
    33→/**
    34→ * MusicView — Browse / Rooms / Queue UI for the music feature.
    35→ *
    36→ * NOTE: This component does NOT own the audio element or the player bar.
    37→ * Those live in <GlobalMusicPlayer /> (rendered at the app root) so that
    38→ * playback persists across all tabs (Chats, Status, Calls, Settings, Music).
    39→ *
    40→ * This view reads playback state from `useMusicStore` and triggers actions
    41→ * via `useMusicPlayer()`.
    42→ */
    43→export function MusicView() {
    44→  const [tab, setTab] = useState<Tab>('browse')
    45→  const [searchQuery, setSearchQuery] = useState('')
    46→  const [searchResults, setSearchResults] = useState<Track[]>([])
    47→  const [searching, setSearching] = useState(false)
    48→  const searchTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null)
    49→  const qc = useQueryClient()
    50→
    51→  // ─── Playback state from the global store ─────────────────────────────
    52→  const currentTrack = useMusicStore((s) => s.currentTrack)
    53→  const isPlaying = useMusicStore((s) => s.isPlaying)
    54→  const queue = useMusicStore((s) => s.queue)
    55→  const autoplay = useMusicStore((s) => s.autoplay)
    56→  const shuffle = useMusicStore((s) => s.shuffle)
    57→  const repeat = useMusicStore((s) => s.repeat)
    58→  const activeRoomId = useMusicStore((s) => s.activeRoomId)
    59→  const setActiveRoomId = useMusicStore((s) => s.setActiveRoomId)
    60→  const setShuffle = useMusicStore((s) => s.setShuffle)
    61→  const setRepeat = useMusicStore((s) => s.setRepeat)
    62→  const setAutoplay = useMusicStore((s) => s.setAutoplay)
    63→
    64→  // ─── Player actions (broadcast + audio handled by global player) ──────
    65→  const { playTrack, playNext, removeFromQueue, clearQueue } = useMusicPlayer()
    66→
    67→  // Fetch trending
    68→  const { data: trendingData, isLoading: trendingLoading } = useQuery({
    69→    queryKey: ['music-trending'],
    70→    queryFn: async () => {
    71→      const res = await fetch('/api/music/trending')
    72→      if (!res.ok) throw new Error('failed')
    73→      return res.json()
    74→    },
    75→  })
    76→
    77→  // Fetch rooms
    78→  const { data: roomsData } = useQuery({
    79→    queryKey: ['music-rooms'],
    80→    queryFn: async () => {
    81→      const res = await fetch('/api/music/rooms')
    82→      if (!res.ok) throw new Error('failed')
    83→      return res.json()
    84→    },
    85→    refetchInterval: 10000,
    86→  })
    87→
    88→  // Create room
    89→  const createRoom = useMutation({
    90→    mutationFn: async (name: string) => {
    91→      const res = await fetch('/api/music/rooms', {
    92→        method: 'POST',
    93→        headers: { 'Content-Type': 'application/json' },
    94→        body: JSON.stringify({ name }),
    95→      })
    96→      if (!res.ok) throw new Error('failed')
    97→      return res.json()
    98→    },
    99→    onSuccess: (data) => {
   100→      qc.invalidateQueries({ queryKey: ['music-rooms'] })
   101→      setActiveRoomId(data.room.id)
   102→      setTab('rooms')
   103→      toast.success('Room created')
   104→    },
   105→  })
   106→
   107→  // Delete room
   108→  const deleteRoom = useMutation({
   109→    mutationFn: async (roomId: string) => {
   110→      const res = await fetch(`/api/music/rooms/${roomId}`, {
   111→        method: 'DELETE',
   112→      })
   113→      if (!res.ok) throw new Error('failed')
   114→    },
   115→    onSuccess: () => {
   116→      qc.invalidateQueries({ queryKey: ['music-rooms'] })
   117→      if (activeRoomId) setActiveRoomId(null)
   118→      toast.success('Room deleted')
   119→    },
   120→    onError: () => toast.error('Failed to delete room'),
   121→  })
   122→
   123→  // Debounced search
   124→  const handleSearchChange = (value: string) => {
   125→    setSearchQuery(value)
   126→    if (searchTimeoutRef.current) clearTimeout(searchTimeoutRef.current)
   127→    if (!value.trim()) {
   128→      setSearchResults([])
   129→      return
   130→    }
   131→    searchTimeoutRef.current = setTimeout(async () => {
   132→      setSearching(true)
   133→      try {
   134→        const res = await fetch(`/api/music/search?q=${encodeURIComponent(value)}`)
   135→        if (!res.ok) throw new Error('Search failed')
   136→        const data = await res.json()
   137→        setSearchResults(data.tracks || [])
   138→      } catch (e: any) {
   139→        toast.error(e.message || 'Search failed')
   140→      } finally {
   141→        setSearching(false)
   142→      }
   143→    }, 500)
   144→  }
   145→
   146→  const trendingTracks = trendingData?.tracks || []
   147→  const rooms = roomsData?.rooms || []
   148→
   149→  return (
   150→    <div className="h-full flex flex-col mesh-gradient overflow-hidden">
   151→      {/* Header with gradient title and glassmorphic tabs */}
   152→      <div className="shrink-0 px-4 pt-4 pb-2 border-b border-border/30">
   153→        <div className="max-w-5xl mx-auto">
   154→          <h1 className="text-2xl md:text-3xl font-bold tracking-tight flex items-center gap-2 mb-3">
   155→            <MusicIcon className="w-7 h-7 text-primary" />
   156→            <GradientText>Musical</GradientText>
   157→          </h1>
   158→
   159→          {/* Tab navigation — glassmorphic pill bar */}
   160→          <GlassSurface className="max-w-xs" blur={12} opacity={0.05}>
   161→            <div className="flex gap-1 p-1">
   162→              {([
   163→                ['browse', 'Browse', Compass],
   164→                ['rooms', 'Rooms', Radio],
   165→                ['queue', 'Queue', ListMusic],
   166→              ] as const).map(([key, label, Icon]) => (
   167→                <button
   168→                  key={key}
   169→                  onClick={() => setTab(key)}
   170→                  className={cn(
   171→                    'flex-1 flex items-center justify-center gap-1.5 py-2 px-3 rounded-xl text-sm font-medium transition-all',
   172→                    tab === key
   173→                      ? 'gradient-primary text-primary-foreground shadow-glow'
   174→                      : 'text-muted-foreground hover:text-foreground hover:bg-white/5',
   175→                  )}
   176→                >
   177→                  <Icon className="w-4 h-4" />
   178→                  {label}
   179→                  {key === 'queue' && queue.length > 0 && (
   180→                    <span className="text-[10px] bg-primary-foreground/20 text-primary-foreground px-1.5 py-0.5 rounded-full">
   181→                      {queue.length}
   182→                    </span>
   183→                  )}
   184→                </button>
   185→              ))}
   186→            </div>
   187→          </GlassSurface>
   188→        </div>
   189→      </div>
   190→
   191→      {/* Main content area — extra bottom padding so content isn't hidden
   192→          behind the persistent global player bar. */}
   193→      <div className="flex-1 overflow-y-auto pb-40 lg:pb-28">
   194→        <div className="max-w-5xl mx-auto p-4 md:p-6">
   195→          <AnimatePresence mode="wait">
   196→            {tab === 'browse' && (
   197→              <motion.div
   198→                key="browse"
   199→                initial={{ opacity: 0, y: 10 }}
   200→                animate={{ opacity: 1, y: 0 }}
   201→                exit={{ opacity: 0, y: -10 }}
   202→                transition={{ duration: 0.2 }}
   203→                className="space-y-6"
   204→              >
   205→                {/* Search */}
   206→                <div className="relative group">
   207→                  <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-muted-foreground group-focus-within:text-primary transition-colors" />
   208→                  <Input
   209→                    value={searchQuery}
   210→                    onChange={(e) => handleSearchChange(e.target.value)}
   211→                    placeholder="Search songs, artists..."
   212→                    className="pl-12 h-14 bg-black/20 backdrop-blur-xl border-white/10 rounded-2xl text-lg focus-visible:ring-primary/50 focus-visible:border-primary/50 transition-all shadow-inner"
   213→                  />
   214→                  {searching && (
   215→                    <Loader2 className="absolute right-4 top-1/2 -translate-y-1/2 w-5 h-5 animate-spin text-muted-foreground" />
   216→                  )}
   217→                </div>
   218→
   219→                {/* Search results */}
   220→                {searchResults.length > 0 ? (
   221→                  <div className="space-y-1.5">
   222→                    <h2 className="text-sm font-bold uppercase tracking-wider mb-3">
   223→                      <ShinyText shimmerDuration={4} className="text-sm">
   224→                        Results
   225→                      </ShinyText>
   226→                    </h2>
   227→                    {searchResults.map((track) => (
   228→                      <TrackRow
   229→                        key={track.videoId}
   230→                        track={track}
   231→                        onPlay={() => playTrack(track)}
   232→                        onAddToQueue={() => playTrack(track, true)}
   233→                        isCurrent={currentTrack?.videoId === track.videoId}
   234→                        isPlaying={
   235→                          isPlaying && currentTrack?.videoId === track.videoId
   236→                        }
   237→                      />
   238→                    ))}
   239→                  </div>
   240→                ) : (
   241→                  <>
   242→                    {/* Trending */}
   243→                    <div>
   244→                      <h2 className="text-sm font-bold uppercase tracking-wider mb-4">
   245→                        <ShinyText shimmerDuration={5} className="text-sm">
   246→                          🔥 Trending Now
   247→                        </ShinyText>
   248→                      </h2>
   249→                      {trendingLoading ? (
   250→                        <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-3">
   251→                          {[1, 2, 3, 4, 5, 6].map((i) => (
   252→                            <div key={i} className="space-y-2 animate-pulse">
   253→                              <div className="aspect-square rounded-xl bg-muted" />
   254→                              <div className="h-3 bg-muted rounded w-3/4" />
   255→                              <div className="h-2 bg-muted rounded w-1/2" />
   256→                            </div>
   257→                          ))}
   258→                        </div>
   259→                      ) : (
   260→                        <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-3">
   261→                          {trendingTracks.map((track: Track) => (
   262→                            <TrackCard
   263→                              key={track.videoId}
   264→                              track={track}
   265→                              onPlay={() => playTrack(track)}
   266→                              onAddToQueue={() => playTrack(track, true)}
   267→                              isCurrent={currentTrack?.videoId === track.videoId}
   268→                              isPlaying={
   269→                                isPlaying &&
   270→                                currentTrack?.videoId === track.videoId
   271→                              }
   272→                            />
   273→                          ))}
   274→                        </div>
   275→                      )}
   276→                    </div>
   277→                  </>
   278→                )}
   279→              </motion.div>
   280→            )}
   281→
   282→            {tab === 'rooms' && (
   283→              <motion.div
   284→                key="rooms"
   285→                initial={{ opacity: 0, y: 10 }}
   286→                animate={{ opacity: 1, y: 0 }}
   287→                exit={{ opacity: 0, y: -10 }}
   288→                transition={{ duration: 0.2 }}
   289→                className="space-y-4"
   290→              >
   291→                <div className="flex items-center justify-between">
   292→                  <h2 className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   293→                    Listening Rooms
   294→                  </h2>
   295→                  <Button
   296→                    variant="outline"
   297→                    size="sm"
   298→                    onClick={() => {
   299→                      const name = prompt('Room name?')
   300→                      if (name?.trim()) createRoom.mutate(name)
   301→                    }}
   302→                  >
   303→                    <Plus className="w-4 h-4 mr-1" />
   304→                    Create Room
   305→                  </Button>
   306→                </div>
   307→
   308→                {rooms.length === 0 ? (
   309→                  <Card className="p-12 text-center border-dashed">
   310→                    <Headphones
   311→                      className="w-12 h-12 mx-auto text-muted-foreground mb-3"
   312→                      strokeWidth={1.5}
   313→                    />
   314→                    <p className="font-medium">No rooms yet</p>
   315→                    <p className="text-sm text-muted-foreground mt-1">
   316→                      Create a room to listen together with friends.
   317→                    </p>
   318→                  </Card>
   319→                ) : (
   320→                  <div className="grid gap-2">
   321→                    {rooms.map((room: Room) => (
   322→                      <RoomCard
   323→                        key={room.id}
   324→                        room={room}
   325→                        isActive={activeRoomId === room.id}
   326→                        onJoin={() => {
   327→                          // Just set the active room — the global player's
   328→                          // socket effect joins the channel and requests
   329→                          // the host's current state automatically.
   330→                          setActiveRoomId(room.id)
   331→                          // Also call the API to auto-join as a member
   332→                          fetch(`/api/music/rooms/${room.id}`).catch(() => {})
   333→                        }}
   334→                        onDelete={() => deleteRoom.mutate(room.id)}
   335→                      />
   336→                    ))}
   337→                  </div>
   338→                )}
   339→              </motion.div>
   340→            )}
   341→
   342→            {tab === 'queue' && (
   343→              <motion.div
   344→                key="queue"
   345→                initial={{ opacity: 0, y: 10 }}
   346→                animate={{ opacity: 1, y: 0 }}
   347→                exit={{ opacity: 0, y: -10 }}
   348→                transition={{ duration: 0.2 }}
   349→                className="space-y-4"
   350→              >
   351→                <div className="flex items-center justify-between">
   352→                  <h2 className="text-sm font-bold uppercase tracking-wider">
   353→                    <ShinyText shimmerDuration={4} className="text-sm">
   354→                      🎧 Play Queue
   355→                    </ShinyText>
   356→                  </h2>
   357→                  {queue.length > 0 && (
   358→                    <Button
   359→                      variant="ghost"
   360→                      size="sm"
   361→                      onClick={clearQueue}
   362→                      className="text-red-400 hover:text-red-300"
   363→                    >
   364→                      <Trash2 className="w-4 h-4 mr-1" />
   365→                      Clear
   366→                    </Button>
   367→                  )}
   368→                </div>
   369→
   370→                {/* Currently playing */}
   371→                {currentTrack && (
   372→                  <StarBorder color="oklch(0.68 0.24 264)" className="w-full">
   373→                    <SpotlightCard spotlightColor="rgba(var(--primary), 0.15)" className="p-5 rounded-2xl border-0 bg-gradient-to-br from-primary/8 to-transparent">
   374→                      <div className="text-[10px] uppercase tracking-widest text-primary font-bold mb-3 flex items-center gap-2">
   375→                        <span className="w-2 h-2 rounded-full bg-primary animate-pulse shadow-[0_0_8px_oklch(0.68_0.24_264/0.6)]" />
   376→                        <ShinyText shimmerDuration={3} className="text-[10px]">Now Playing</ShinyText>
   377→                      </div>
   378→                    <div className="flex items-center gap-4">
   379→                      {currentTrack.thumbnail ? (
   380→                        <div className="relative w-16 h-16 rounded-xl overflow-hidden shadow-lg ring-1 ring-white/10">
   381→                          <img
   382→                            src={currentTrack.thumbnail}
   383→                            alt=""
   384→                            className="w-full h-full object-cover"
   385→                          />
   386→                        </div>
   387→                      ) : (
   388→                        <div className="w-16 h-16 rounded-xl gradient-primary flex items-center justify-center shadow-lg">
   389→                          <Radio className="w-6 h-6 text-primary-foreground" />
   390→                        </div>
   391→                      )}
   392→                      <div className="flex-1 min-w-0">
   393→                        <div className="text-base font-semibold truncate text-foreground">
   394→                          {currentTrack.title}
   395→                        </div>
   396→                        <div className="text-sm text-muted-foreground truncate">
   397→                          {currentTrack.artist}
   398→                        </div>
   399→                      </div>
   400→                      <div className="flex items-center gap-1.5">
   401→                        <button
   402→                          onClick={() => setShuffle(!shuffle)}
   403→                          className={cn(
   404→                            'p-2.5 rounded-xl transition-all',
   405→                            shuffle
   406→                              ? 'text-primary bg-primary/10 shadow-glow'
   407→                              : 'text-muted-foreground hover:text-foreground hover:bg-white/5',
   408→                          )}
   409→                          aria-label="Toggle shuffle"
   410→                        >
   411→                          <Shuffle className="w-4 h-4" />
   412→                        </button>
   413→                        <button
   414→                          onClick={() => setRepeat(!repeat)}
   415→                          className={cn(
   416→                            'p-2.5 rounded-xl transition-all',
   417→                            repeat
   418→                              ? 'text-primary bg-primary/10 shadow-glow'
   419→                              : 'text-muted-foreground hover:text-foreground hover:bg-white/5',
   420→                          )}
   421→                          aria-label="Toggle repeat"
   422→                        >
   423→                          <Repeat className="w-4 h-4" />
   424→                        </button>
   425→                        <button
   426→                          onClick={() => setAutoplay(!autoplay)}
   427→                          className={cn(
   428→                            'p-2.5 rounded-xl transition-all',
   429→                            autoplay
   430→                              ? 'text-primary bg-primary/10 shadow-glow'
   431→                              : 'text-muted-foreground hover:text-foreground hover:bg-white/5',
   432→                          )}
   433→                          title="Autoplay recommendations"
   434→                          aria-label="Toggle autoplay"
   435→                        >
   436→                          <Radio className="w-4 h-4" />
   437→                        </button>
   438→                      </div>
   439→                    </div>
   440→                    </SpotlightCard>
   441→                  </StarBorder>
   442→                )}
   443→
   444→                {/* Queue list */}
   445→                {queue.length === 0 ? (
   446→                  <Card className="p-8 text-center border-dashed">
   447→                    <ListMusic
   448→                      className="w-10 h-10 mx-auto text-muted-foreground mb-2"
   449→                      strokeWidth={1.5}
   450→                    />
   451→                    <p className="text-sm text-muted-foreground">
   452→                      Queue is empty. Add songs from Browse or search.
   453→                    </p>
   454→                  </Card>
   455→                ) : (
   456→                    <div className="space-y-1.5">
   457→                      {queue.map((track, i) => (
   458→                        <div
   459→                          key={`${track.videoId}-${i}`}
   460→                          className="flex items-center gap-3 p-2.5 rounded-xl hover:bg-white/[0.06] transition-all group border border-transparent hover:border-white/5">
   461→                          <span className="text-xs text-muted-foreground w-6 text-center font-medium">
   462→                            {i + 1}
   463→                          </span>
   464→                        {track.thumbnail ? (
   465→                          <img
   466→                            src={track.thumbnail}
   467→                            alt=""
   468→                            className="w-10 h-10 rounded object-cover"
   469→                          />
   470→                        ) : (
   471→                          <div className="w-10 h-10 rounded bg-muted flex items-center justify-center">
   472→                            <ListMusic className="w-4 h-4 text-muted-foreground" />
   473→                          </div>
   474→                        )}
   475→                        <div className="flex-1 min-w-0">
   476→                          <div className="text-sm font-medium truncate">
   477→                            {track.title}
   478→                          </div>
   479→                          <div className="text-xs text-muted-foreground truncate">
   480→                            {track.artist}
   481→                          </div>
   482→                        </div>
   483→                        <button
   484→                          onClick={() => playTrack(track)}
   485→                          className="p-1.5 rounded-lg text-muted-foreground opacity-0 group-hover:opacity-100 hover:text-primary transition-opacity"
   486→                          title="Play this track now"
   487→                          aria-label="Play this track now"
   488→                        >
   489→                          <Play className="w-3.5 h-3.5" />
   490→                        </button>
   491→                          <button
   492→                            onClick={() => removeFromQueue(i)}
   493→                            className="p-1.5 rounded-lg text-red-400 opacity-0 group-hover:opacity-100 transition-opacity"
   494→                            title="Remove from queue"
   495→                            aria-label="Remove from queue"
   496→                          >
   497→                            <X className="w-3.5 h-3.5" />
   498→                          </button>
   499→                        </div>
   500→                      ))}
   501→                    </div>
   502→                )}
   503→
   504→                {/* Quick action: skip to next */}
   505→                {currentTrack && (
   506→                  <Button
   507→                    variant="outline"
   508→                    size="sm"
   509→                    onClick={() => playNext()}
   510→                    className="w-full"
   511→                  >
   512→                    <SkipForward className="w-4 h-4 mr-2" />
   513→                    Skip to next
   514→                  </Button>
   515→                )}
   516→              </motion.div>
   517→            )}
   518→          </AnimatePresence>
   519→        </div>
   520→      </div>
   521→    </div>
   522→  )
   523→}

   525→// ─── Track Row (list view) ─────────────────────────────────────────────────
   526→function TrackRow({
   527→  track,
   528→  onPlay,
   529→  onAddToQueue,
   530→  isCurrent,
   531→  isPlaying,
   532→}: {
   533→  track: Track
   534→  onPlay: () => void
   535→  onAddToQueue: () => void
   536→  isCurrent: boolean
   537→  isPlaying: boolean
   538→}) {
   539→  return (
   540→    <div
   541→      className={cn(
   542→        'flex items-center gap-4 p-3 rounded-2xl transition-all group hover:bg-white/[0.04]',
   543→        isCurrent && 'bg-primary/5 hover:bg-primary/10',
   544→      )}
   545→    >
   546→      <button
   547→        onClick={onPlay}
   548→        className="flex items-center gap-4 flex-1 min-w-0 text-left"
   549→      >
   550→        {track.thumbnail ? (
   551→          <div className="relative w-14 h-14 rounded-xl overflow-hidden shrink-0 shadow-sm ring-1 ring-white/5">
   552→            <img
   553→              src={track.thumbnail}
   554→              alt=""
   555→              className="w-full h-full object-cover"
   556→            />
   557→            <div className="absolute inset-0 bg-black/40 opacity-0 group-hover:opacity-100 transition-opacity flex items-center justify-center backdrop-blur-[2px]">

   669→// ─── Room Card ─────────────────────────────────────────────────────────────
   670→// Note: we no longer show the raw member count — it includes stale members
   671→// who joined and left (we don't track leftAt for music rooms). Instead we
   672→// show a "Listening" indicator that's only lit when the host is actively
   673→// playing, which is a more accurate signal of "is anything happening here
   674→// right now".
   675→function RoomCard({
   676→  room,
   677→  isActive,
   678→  onJoin,
   679→  onDelete,
   680→}: {
   681→  room: Room
   682→  isActive: boolean
   683→  onJoin: () => void
   684→  onDelete?: () => void
   685→}) {
   686→  const isPlaying = room.currentState === 'playing'
   687→  return (
   688→    <div className="relative group">
   689→      {isActive && (
   690→        <div className="absolute -inset-0.5 rounded-2xl bg-gradient-to-r from-primary/50 to-purple-500/50 opacity-50 blur-md transition-opacity group-hover:opacity-75" />
   691→      )}
   692→      <GlassSurface
   693→        blur={12}
   694→        opacity={isActive ? 0.08 : 0.02}
   695→        className={cn(
   696→          'w-full flex items-center gap-4 p-4 transition-all',
   697→          isActive
   698→            ? 'border-primary/50'
   699→            : 'border-white/5 hover:bg-white/[0.03]',
   700→        )}
   701→      >
   702→        <button
   703→          onClick={onJoin}
   704→          className="flex items-center gap-4 flex-1 min-w-0 text-left"
   705→        >
   706→          <div
   707→            className={cn(
   708→              'w-14 h-14 rounded-xl flex items-center justify-center shrink-0 transition-all shadow-sm',
   709→              isPlaying
   710→                ? 'bg-primary/20 text-primary pulse-glow ring-1 ring-primary/30'
   711→                : 'bg-black/30 text-muted-foreground ring-1 ring-white/10',
   712→            )}
   713→          >
   714→            <Radio className={cn('w-6 h-6', isPlaying && 'animate-pulse')} />
   715→          </div>
   716→          <div className="flex-1 min-w-0">
   717→            <div className="font-semibold text-base truncate text-foreground">{room.name}</div>
   718→            <div className="text-sm text-muted-foreground flex items-center gap-2 mt-0.5">
   719→              <span>Hosted by {room.host?.displayName || 'Unknown'}</span>
   720→              {isPlaying ? (
   721→                <>
   722→                  <span className="text-white/30">•</span>
   723→                  <span className="text-status-online flex items-center gap-1.5 font-medium">
   724→                    <span className="w-2 h-2 rounded-full bg-status-online shadow-[0_0_8px_rgba(34,197,94,0.6)] animate-pulse" />
   725→                    Listening now
   726→                  </span>
   727→                </>
   728→              ) : (
   729→                <>
   730→                  <span className="text-white/30">•</span>
   731→                  <span className="text-muted-foreground">Idle</span>
   732→                </>
   733→              )}
   734→            </div>
   735→          </div>
   736→          <div className={cn(
   737→            "shrink-0 text-sm font-semibold px-4 py-2 rounded-xl transition-all shadow-sm",
   738→            isActive ? "bg-primary text-primary-foreground shadow-glow" : "bg-white/5 text-foreground hover:bg-white/10 border border-white/10"
   739→          )}>
   740→            {isActive ? 'In Room' : 'Join'}
   741→          </div>
   742→        </button>
   743→        {onDelete && (
   744→          <button
   745→            onClick={(e) => {
   746→              e.stopPropagation()
   747→              if (confirm(`Delete room "${room.name}"?`)) onDelete()
   748→            }}
   749→            className="shrink-0 p-2.5 rounded-full text-red-400 opacity-0 group-hover:opacity-100 hover:bg-red-500/10 hover:text-red-300 transition-all"
   750→            title="Delete room"
   751→            aria-label="Delete room"
   752→          >
   753→            <Trash2 className="w-5 h-5" />
   754→          </button>
   755→        )}
   756→      </GlassSurface>
   757→    </div>
   758→  )
   759→}
