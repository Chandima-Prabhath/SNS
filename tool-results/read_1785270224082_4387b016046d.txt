    20→    20→  videoId: string
    21→    21→  title: string
    22→    22→  artist: string
    23→    23→  thumbnail: string | null
    24→    24→  durationSeconds: number | null
    25→    25→  album?: string | null

    27→    27→
    28→    28→interface Room {
    29→    29→  id: string
    30→    30→  name: string
    31→    31→  host: any
    32→    32→  currentVideoId: string | null

    40→    40→
    41→    41→export function MusicView() {
    42→    42→  const [tab, setTab] = useState<Tab>('browse')
    43→    43→  const [activeRoomId, setActiveRoomId] = useState<string | null>(null)
    44→    44→  const [searchQuery, setSearchQuery] = useState('')
    45→    45→  const [searchResults, setSearchResults] = useState<Track[]>([])
    46→    46→  const [searching, setSearching] = useState(false)
    47→    47→  const [currentTrack, setCurrentTrack] = useState<Track | null>(null)
    48→    48→  const [isPlaying, setIsPlaying] = useState(false)
    49→    49→  const [position, setPosition] = useState(0)
    50→    50→  const [volume, setVolume] = useState(0.8)
    51→    51→  const [queue, setQueue] = useState<Track[]>([])
    52→    52→  const [autoplay, setAutoplay] = useState(true)
    53→    53→  const [shuffle, setShuffle] = useState(false)
    54→    54→  const [repeat, setRepeat] = useState(false)
    55→    55→  const audioRef = useRef<HTMLAudioElement>(null)
    56→    56→  const searchTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null)
    57→    57→  const qc = useQueryClient()
    58→    58→  const { socket } = useSocket()
    59→    59→
    60→    60→  // Fetch trending
    61→    61→  const { data: trendingData, isLoading: trendingLoading } = useQuery({
    62→    62→    queryKey: ['music-trending'],
    63→    63→    queryFn: async () => {
    64→    64→      const res = await fetch('/api/music/trending')
    65→    65→      if (!res.ok) throw new Error('failed')
    66→    66→      return res.json()
    67→    67→    },
    68→    68→  })
    69→    69→
    70→    70→  // Fetch rooms
    71→    71→  const { data: roomsData, refetch: refetchRooms } = useQuery({
    72→    72→    queryKey: ['music-rooms'],
    73→    73→    queryFn: async () => {
    74→    74→      const res = await fetch('/api/music/rooms')
    75→    75→      if (!res.ok) throw new Error('failed')
    76→    76→      return res.json()
    77→    77→    },
    78→    78→    refetchInterval: 10000,
    79→    79→  })
    80→    80→
    81→    81→  // Create room
    82→    82→  const createRoom = useMutation({
    83→    83→    mutationFn: async (name: string) => {
    84→    84→      const res = await fetch('/api/music/rooms', {
    85→    85→        method: 'POST',
    86→    86→        headers: { 'Content-Type': 'application/json' },
    87→    87→        body: JSON.stringify({ name }),
    88→    88→      })
    89→    89→      if (!res.ok) throw new Error('failed')
    90→    90→      return res.json()
    91→    91→    },
    92→    92→    onSuccess: (data) => {
    93→    93→      qc.invalidateQueries({ queryKey: ['music-rooms'] })
    94→    94→      setActiveRoomId(data.room.id)
    95→    95→      setTab('rooms')
    96→    96→      toast.success('Room created')
    97→    97→    },
    98→    98→  })
    99→    99→
   100→   100→  // Debounced search
   101→   101→  const handleSearchChange = (value: string) => {
   102→   102→    setSearchQuery(value)
   103→   103→    if (searchTimeoutRef.current) clearTimeout(searchTimeoutRef.current)
   104→   104→    if (!value.trim()) {
   105→   105→      setSearchResults([])
   106→   106→      return
   107→   107→    }
   108→   108→    searchTimeoutRef.current = setTimeout(async () => {
   109→   109→      setSearching(true)
   110→   110→      try {
   111→   111→        const res = await fetch(`/api/music/search?q=${encodeURIComponent(value)}`)
   112→   112→        if (!res.ok) throw new Error('Search failed')
   113→   113→        const data = await res.json()
   114→   114→        setSearchResults(data.tracks || [])
   115→   115→      } catch (e: any) {
   116→   116→        toast.error(e.message || 'Search failed')
   117→   117→      } finally {
   118→   118→        setSearching(false)
   119→   119→      }
   120→   120→    }, 500)
   121→   121→  }
   122→   122→
   123→   123→  // Play a track
   124→   124→  const playTrack = useCallback(async (track: Track, addToQueue = false) => {
   125→   125→    if (addToQueue && currentTrack) {
   126→   126→      setQueue((q) => [...q, track])
   127→   127→      toast.success(`Added to queue: ${track.title}`)
   128→   128→      return
   129→   129→    }
   130→   130→
   131→   131→    setCurrentTrack(track)
   132→   132→    setIsPlaying(true)
   133→   133→    setPosition(0)
   134→   134→
   135→   135→    // If in a room, broadcast the track change to all members via Socket.io
   136→   136→    if (activeRoomId && socket) {
   137→   137→      socket.emit('music:sync', {
   138→   138→        roomId: activeRoomId,
   139→   139→        state: 'playing',
   140→   140→        position: 0,
   141→   141→        videoId: track.videoId,
   142→   142→        trackInfo: {
   143→   143→          title: track.title,
   144→   144→          artist: track.artist,
   145→   145→          thumbnail: track.thumbnail,
   146→   146→          durationSeconds: track.durationSeconds,
   147→   147→        },
   148→   148→      })
   149→   149→    }
   150→   150→
   151→   151→    if (audioRef.current) {
   152→   152→      audioRef.current.src = `/api/music/stream/${track.videoId}`
   153→   153→      audioRef.current.volume = volume
   154→   154→      try {
   155→   155→        await audioRef.current.play()
   156→   156→      } catch (e: any) {
   157→   157→        if (e?.name === 'NotSupportedError' || e?.name === 'MediaError') {
   158→   158→          try {
   159→   159→            const res = await fetch(`/api/music/stream/${track.videoId}`, { method: 'HEAD' })
   160→   160→            if (!res.ok) {
   161→   161→              const data = await res.json().catch(() => ({}))
   162→   162→              toast.error(data.error || 'Could not download this track.')
   163→   163→              setCurrentTrack(null)
   164→   164→              setIsPlaying(false)
   165→   165→            }
   166→   166→          } catch {
   167→   167→            toast.error('The server is downloading this track — try again in a moment.')
   168→   168→          }
   169→   169→        }
   170→   170→      }
   171→   171→    }
   172→   172→  }, [activeRoomId, volume, currentTrack, socket])
   173→   173→
   174→   174→  // Play next track from queue or autoplay
   175→   175→  const playNext = useCallback(async () => {
   176→   176→    let nextTrack: Track | null = null
   177→   177→
   178→   178→    if (queue.length > 0) {
   179→   179→      if (shuffle) {
   180→   180→        const idx = Math.floor(Math.random() * queue.length)
   181→   181→        nextTrack = queue[idx]
   182→   182→        setQueue((q) => q.filter((_, i) => i !== idx))
   183→   183→      } else {
   184→   184→        nextTrack = queue[0]
   185→   185→        setQueue((q) => q.slice(1))
   186→   186→      }
   187→   187→    } else if (autoplay && currentTrack) {
   188→   188→      // Fetch related tracks and play the first one
   189→   189→      try {
   190→   190→        const res = await fetch(`/api/music/related/${currentTrack.videoId}`)
   191→   191→        if (res.ok) {
   192→   192→          const data = await res.json()
   193→   193→          if (data.tracks?.length > 0) {
   194→   194→            nextTrack = data.tracks[0]
   195→   195→          }
   196→   196→        }
   197→   197→      } catch {
   198→   198→        // Autoplay failed — just stop
   199→   199→      }
   200→   200→    }
   201→   201→
   202→   202→    if (nextTrack) {
   203→   203→      await playTrack(nextTrack)
   204→   204→    } else if (repeat && currentTrack) {
   205→   205→      await playTrack(currentTrack)
   206→   206→    } else {
   207→   207→      setIsPlaying(false)
   208→   208→      setPosition(0)
   209→   209→    }
   210→   210→  }, [queue, shuffle, autoplay, currentTrack, repeat, playTrack])
   211→   211→
   212→   212→  // Toggle play/pause
   213→   213→  const togglePlay = () => {
   214→   214→    if (!audioRef.current || !currentTrack) return
   215→   215→    if (isPlaying) {
   216→   216→      audioRef.current.pause()
   217→   217→      setIsPlaying(false)
   218→   218→      // Broadcast pause to room
   219→   219→      if (activeRoomId && socket) {
   220→   220→        socket.emit('music:sync', {
   221→   221→          roomId: activeRoomId,
   222→   222→          state: 'paused',
   223→   223→          position: audioRef.current.currentTime,
   224→   224→        })
   225→   225→      }
   226→   226→    } else {
   227→   227→      audioRef.current.play()
   228→   228→      setIsPlaying(true)
   229→   229→      // Broadcast play to room
   230→   230→      if (activeRoomId && socket) {
   231→   231→        socket.emit('music:sync', {
   232→   232→          roomId: activeRoomId,
   233→   233→          state: 'playing',
   234→   234→          position: audioRef.current.currentTime,
   235→   235→        })
   236→   236→      }
   237→   237→    }
   238→   238→  }
   239→   239→
   240→   240→  // Seek
   241→   241→  const handleSeek = (e: React.ChangeEvent<HTMLInputElement>) => {
   242→   242→    const newPos = parseFloat(e.target.value)
   243→   243→    if (audioRef.current) {
   244→   244→      audioRef.current.currentTime = newPos
   245→   245→      setPosition(newPos)
   246→   246→      // Broadcast seek to room
   247→   247→      if (activeRoomId && socket) {
   248→   248→        socket.emit('music:sync', {
   249→   249→          roomId: activeRoomId,
   250→   250→          state: isPlaying ? 'playing' : 'paused',
   251→   251→          position: newPos,
   252→   252→        })
   253→   253→      }
   254→   254→    }
   255→   255→  }
   256→   256→
   257→   257→  // Audio event handlers
   258→   258→  useEffect(() => {
   259→   259→    const audio = audioRef.current
   260→   260→    if (!audio) return
   261→   261→
   262→   262→    const onTimeUpdate = () => setPosition(audio.currentTime)
   263→   263→    const onEnded = () => {
   264→   264→      // Auto-play next track when current ends
   265→   265→      playNext()
   266→   266→    }
   267→   267→    const onError = () => {
   268→   268→      if (currentTrack) {
   269→   269→        // Skip to next on error
   270→   270→        toast.error('Could not play this track — skipping...')
   271→   271→        setQueue((q) => q.slice(1))
   272→   272→        playNext()
   273→   273→      }
   274→   274→    }
   275→   275→
   276→   276→    audio.addEventListener('timeupdate', onTimeUpdate)
   277→   277→    audio.addEventListener('ended', onEnded)
   278→   278→    audio.addEventListener('error', onError)
   279→   279→    return () => {
   280→   280→      audio.removeEventListener('timeupdate', onTimeUpdate)
   281→   281→      audio.removeEventListener('ended', onEnded)
   282→   282→      audio.removeEventListener('error', onError)
   283→   283→    }
   284→   284→  }, [currentTrack, playNext])
   285→   285→
   286→   286→  // Real-time sync: listen for room sync events from the host.
   287→   287→  // This is the CORE of sync play — when the host plays/pauses/seeks/changes
   288→   288→  // track, the event arrives instantly via Socket.io (not DB polling).
   289→   289→  useEffect(() => {
   290→   290→    if (!socket || !activeRoomId) return
   291→   291→
   292→   292→    // Join the music room's socket channel
   293→   293→    socket.emit('music:join', activeRoomId)
   294→   294→
   295→   295→    // Request the host's current state immediately after joining
   296→   296→    setTimeout(() => {
   297→   297→      socket.emit('music:request-sync', activeRoomId)
   298→   298→    }, 500)
   299→   299→
   300→   300→    const onSync = (data: {
   301→   301→      roomId: string
   302→   302→      state: string
   303→   303→      position: number
   304→   304→      videoId?: string
   305→   305→      trackInfo?: Track
   306→   306→      serverTimestamp: number
   307→   307→    }) => {
   308→   308→      if (data.roomId !== activeRoomId) return
   309→   309→
   310→   310→      // Calculate network delay and adjust position
   311→   311→      const networkDelay = (Date.now() - data.serverTimestamp) / 1000
   312→   312→      const adjustedPosition = data.position + (data.state === 'playing' ? networkDelay : 0)
   313→   313→
   314→   314→      // Track changed — load the new track and sync to host's position
   315→   315→      if (data.videoId && (!currentTrack || currentTrack.videoId !== data.videoId)) {
   316→   316→        const track: Track = data.trackInfo || {
   317→   317→          videoId: data.videoId,
   318→   318→          title: 'Now Playing',
   319→   319→          artist: '',
   320→   320→          thumbnail: null,
   321→   321→          durationSeconds: null,
   322→   322→        }
   323→   323→        setCurrentTrack(track)
   324→   324→        setTimeout(() => {
   325→   325→          if (audioRef.current) {
   326→   326→            audioRef.current.src = `/api/music/stream/${data.videoId}`
   327→   327→            // Set the position to match the host's current position
   328→   328→            audioRef.current.currentTime = adjustedPosition
   329→   329→            if (data.state === 'playing') {
   330→   330→              audioRef.current.play().catch(() => {})
   331→   331→              setIsPlaying(true)
   332→   332→            } else {
   333→   333→              setIsPlaying(false)
   334→   334→            }
   335→   335→          }
   336→   336→        }, 200)
   337→   337→      } else if (audioRef.current && currentTrack) {
   338→   338→        // Same track — just sync position and state
   339→   339→        const drift = Math.abs(audioRef.current.currentTime - adjustedPosition)
   340→   340→
   341→   341→        // Only force-seek if drift > 1.5 seconds (avoid jitter from minor differences)
   342→   342→        if (drift > 1.5) {
   343→   343→          audioRef.current.currentTime = adjustedPosition
   344→   344→        }
   345→   345→
   346→   346→        // Sync play/pause state
   347→   347→        if (data.state === 'playing' && !isPlaying) {
   348→   348→          audioRef.current.play().catch(() => {})
   349→   349→          setIsPlaying(true)
   350→   350→        } else if (data.state === 'paused' && isPlaying) {
   351→   351→          audioRef.current.pause()
   352→   352→          setIsPlaying(false)
   353→   353→        }
   354→   354→      }
   355→   355→    }
   356→   356→
   357→   357→    // When the host receives a sync request, broadcast current state
   358→   358→    const onRequestSync = (data: { roomId: string; fromUserId: string }) => {
   359→   359→      if (data.roomId !== activeRoomId) return
   360→   360→      // Only the host responds to sync requests
   361→   361→      if (currentTrack && socket) {
   362→   362→        socket.emit('music:sync', {
   363→   363→          roomId: activeRoomId,
   364→   364→          state: isPlaying ? 'playing' : 'paused',
   365→   365→          position: audioRef.current?.currentTime || 0,
   366→   366→          videoId: currentTrack.videoId,
   367→   367→          trackInfo: {
   368→   368→            title: currentTrack.title,
   369→   369→            artist: currentTrack.artist,
   370→   370→            thumbnail: currentTrack.thumbnail,
   371→   371→            durationSeconds: currentTrack.durationSeconds,
   372→   372→          },
   373→   373→        })
   374→   374→      }
   375→   375→    }
   376→   376→
   377→   377→    socket.on('music:sync', onSync)
   378→   378→    socket.on('music:request-sync', onRequestSync)
   379→   379→
   380→   380→    return () => {
   381→   381→      socket.off('music:sync', onSync)
   382→   382→      socket.off('music:request-sync', onRequestSync)
   383→   383→      socket.emit('music:leave', activeRoomId)
   384→   384→    }
   385→   385→  }, [socket, activeRoomId, currentTrack, isPlaying])
   386→   386→
   387→   387→  // Delete room
   388→   388→  const deleteRoom = useMutation({
   389→   389→    mutationFn: async (roomId: string) => {
   390→   390→      const res = await fetch(`/api/music/rooms/${roomId}`, { method: 'DELETE' })
   391→   391→      if (!res.ok) throw new Error('failed')
   392→   392→    },
   393→   393→    onSuccess: () => {
   394→   394→      qc.invalidateQueries({ queryKey: ['music-rooms'] })
   395→   395→      if (activeRoomId) setActiveRoomId(null)
   396→   396→      toast.success('Room deleted')
   397→   397→    },
   398→   398→    onError: () => toast.error('Failed to delete room'),
   399→   399→  })
   400→   400→
   401→   401→  const trendingTracks = trendingData?.tracks || []
   402→   402→  const rooms = roomsData?.rooms || []
   403→   403→  const activeRoom = rooms.find((r: Room) => r.id === activeRoomId)
   404→   404→
   405→   405→  return (
   406→   406→    <div className="h-full flex flex-col mesh-gradient overflow-hidden">
   407→   407→      <audio ref={audioRef} />
   408→   408→
   409→   409→      {/* Header with tabs */}
   410→   410→      <div className="shrink-0 px-4 pt-4 pb-2 border-b border-border/30">
   411→   411→        <div className="max-w-5xl mx-auto">
   412→   412→          <h1 className="text-2xl md:text-3xl font-bold tracking-tight flex items-center gap-2 mb-3">
   413→   413→            <MusicIcon className="w-7 h-7 text-primary" />
   414→   414→            Musical
   415→   415→          </h1>
   416→   416→
   417→   417→          {/* Tab navigation */}
   418→   418→          <div className="flex gap-1 p-1 bg-muted/50 rounded-xl max-w-xs">
   419→   419→            {([
   420→   420→              ['browse', 'Browse', Compass],
   421→   421→              ['rooms', 'Rooms', Radio],
   422→   422→              ['queue', 'Queue', ListMusic],
   423→   423→            ] as const).map(([key, label, Icon]) => (
   424→   424→              <button
   425→   425→                key={key}
   426→   426→                onClick={() => setTab(key)}
   427→   427→                className={cn(
   428→   428→                  'flex-1 flex items-center justify-center gap-1.5 py-2 px-3 rounded-lg text-sm font-medium transition-all',
   429→   429→                  tab === key
   430→   430→                    ? 'bg-background text-primary shadow-sm'
   431→   431→                    : 'text-muted-foreground hover:text-foreground'
   432→   432→                )}
   433→   433→              >
   434→   434→                <Icon className="w-4 h-4" />
   435→   435→                {label}
   436→   436→                {key === 'queue' && queue.length > 0 && (
   437→   437→                  <span className="text-[10px] bg-primary/20 text-primary px-1.5 py-0.5 rounded-full">
   438→   438→                    {queue.length}
   439→   439→                  </span>
   440→   440→                )}
   441→   441→              </button>
   442→   442→            ))}
   443→   443→          </div>
   444→   444→        </div>
   445→   445→      </div>
   446→   446→
   447→   447→      {/* Main content area */}
   448→   448→      <div className="flex-1 overflow-y-auto pb-40 lg:pb-28">
   449→   449→        <div className="max-w-5xl mx-auto p-4 md:p-6">
   450→   450→          <AnimatePresence mode="wait">
   451→   451→            {tab === 'browse' && (
   452→   452→              <motion.div
   453→   453→                key="browse"
   454→   454→                initial={{ opacity: 0, y: 10 }}
   455→   455→                animate={{ opacity: 1, y: 0 }}
   456→   456→                exit={{ opacity: 0, y: -10 }}
   457→   457→                transition={{ duration: 0.2 }}
   458→   458→                className="space-y-6"
   459→   459→              >
   460→   460→                {/* Search */}
   461→   461→                <div className="relative">
   462→   462→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   463→   463→                  <Input
   464→   464→                    value={searchQuery}
   465→   465→                    onChange={(e) => handleSearchChange(e.target.value)}
   466→   466→                    placeholder="Search songs, artists..."
   467→   467→                    className="pl-9 h-11 bg-card/50 backdrop-blur-sm border-border/30"
   468→   468→                  />
   469→   469→                  {searching && (
   470→   470→                    <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 animate-spin text-muted-foreground" />
   471→   471→                  )}
   472→   472→                </div>
   473→   473→
   474→   474→                {/* Search results */}
   475→   475→                {searchResults.length > 0 ? (
   476→   476→                  <div className="space-y-1.5">
   477→   477→                    <h2 className="text-xs font-semibold uppercase tracking-wider text-muted-foreground mb-2">
   478→   478→                      Results
   479→   479→                    </h2>
   480→   480→                    {searchResults.map((track) => (
   481→   481→                      <TrackRow
   482→   482→                        key={track.videoId}
   483→   483→                        track={track}
   484→   484→                        onPlay={() => playTrack(track)}
   485→   485→                        onAddToQueue={() => playTrack(track, true)}
