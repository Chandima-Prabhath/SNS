   141→function ProfileSection() {
   142→  const { data: session } = useSession()
   143→  const qc = useQueryClient()
   144→
   145→  const { data: me, isLoading } = useQuery({
   146→    queryKey: ['me'],
   147→    queryFn: async () => {
   148→      const res = await fetch('/api/auth/me')
   149→      const data = await res.json()
   150→      return data.user
   151→    },
   152→  })
   153→
   154→  const [displayName, setDisplayName] = useState('')
   155→  const [bio, setBio] = useState('')
   156→  const [avatarUrl, setAvatarUrl] = useState('')
   157→  const [customStatus, setCustomStatus] = useState('')
   158→  const [uploading, setUploading] = useState(false)
   159→
   160→  // Sync from server once
   161→  useState(() => {
   162→    if (me) {
   163→      setDisplayName(me.displayName || '')
   164→      setBio(me.bio || '')
   165→      setAvatarUrl(me.avatarUrl || '')
   166→      setCustomStatus(me.customStatus || '')
   167→    }
   168→  })
   169→
   170→  // Also sync on data change
   171→  if (me && displayName === '' && me.displayName) {
   172→    setDisplayName(me.displayName)
   173→    setBio(me.bio || '')
   174→    setAvatarUrl(me.avatarUrl || '')
   175→    setCustomStatus(me.customStatus || '')
   176→  }
   177→
   178→  const update = useMutation({
   179→    mutationFn: async (data: any) => {
   180→      const res = await fetch(`/api/users/me`, {
   181→        method: 'PATCH',
   182→        headers: { 'Content-Type': 'application/json' },
   183→        body: JSON.stringify(data),
   184→      })
   185→      if (!res.ok) throw new Error('failed')
   186→      return res.json()
   187→    },
   188→    onSuccess: () => {
   189→      qc.invalidateQueries({ queryKey: ['me'] })
   190→      toast.success('Saved')
   191→    },
   192→    onError: () => toast.error('Failed to save'),
   193→  })
   194→
   195→  const handleSave = () => {
   196→    update.mutate({ displayName, bio, avatarUrl, customStatus })
   197→  }
   198→
   199→  const handleAvatar = async (e: React.ChangeEvent<HTMLInputElement>) => {
   200→    const file = e.target.files?.[0]
   201→    if (!file) return
   202→    setUploading(true)
   203→    try {
   204→      const formData = new FormData()
   205→      formData.append('file', file)
   206→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   207→      if (!res.ok) {
   208→        const text = await res.text()
   209→        throw new Error(text.slice(0, 120) || 'Upload failed')
   210→      }
   211→      const data = await res.json()
   212→      setAvatarUrl(data.url)
   213→      toast.success('Avatar uploaded — click Save to apply')
   214→    } catch (e: any) {
   215→      toast.error(e.message || 'Upload failed')
   216→    } finally {
   217→      setUploading(false)
   218→      e.target.value = ''
   219→    }
   220→  }
   221→
   222→  if (isLoading) return <div className="flex justify-center py-12"><Loader2 className="w-6 h-6 animate-spin" /></div>
   223→
   224→  return (
   225→    <div className="space-y-6">
   226→      <GlassSurface blur={16} opacity={0.03} className="p-6">
   227→        <div className="flex flex-col md:flex-row gap-8 items-start md:items-center">
   228→          <AvatarPicker
   229→            currentAvatarUrl={avatarUrl}
   230→            userId={me?.id || 'seed'}
   231→            onPick={(url) => { setAvatarUrl(url); toast.success('Avatar selected — click Save to apply') }}
   232→            onUpload={handleAvatar}
   233→            uploading={uploading}
   234→          />
   235→          <div className="flex-1 space-y-1">
   236→            <h2 className="text-xl font-semibold"><ShinyText shimmerDuration={3}>{displayName || 'Your Profile'}</ShinyText></h2>
   237→            <p className="text-sm text-muted-foreground">Manage your public persona</p>
   238→          </div>
   239→        </div>
   240→      </GlassSurface>
   241→
   242→      <GlassSurface blur={16} opacity={0.03} className="p-6 space-y-6">
   243→        <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
   244→          <div className="space-y-2">
   245→            <Label htmlFor="displayName" className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Display name</Label>
   246→            <Input id="displayName" value={displayName} onChange={(e) => setDisplayName(e.target.value)} className="h-12 bg-black/20 border-white/10 rounded-xl px-4 focus-visible:ring-primary/50 transition-all" />
   247→          </div>
   248→
   249→          <div className="space-y-2">
   250→            <Label htmlFor="username" className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Username</Label>
   251→            <Input id="username" value={me?.username || ''} disabled className="h-12 bg-black/20 border-white/5 rounded-xl px-4 opacity-70 cursor-not-allowed" />
   252→            <p className="text-[10px] text-muted-foreground pl-1">Username cannot be changed.</p>
   253→          </div>
   254→        </div>
   255→
   256→        <div className="space-y-2">
   257→          <Label htmlFor="customStatus" className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Custom status</Label>
   258→          <Input
   259→            id="customStatus"
   260→            value={customStatus}
   261→            onChange={(e) => setCustomStatus(e.target.value)}
   262→            placeholder="Listening to lofi beats"
   263→            maxLength={80}
   264→            className="h-12 bg-black/20 border-white/10 rounded-xl px-4 focus-visible:ring-primary/50 transition-all"
   265→          />
   266→        </div>
   267→
   268→        <div className="space-y-2">
   269→          <Label htmlFor="bio" className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Bio</Label>
   270→          <Textarea
   271→            id="bio"
   272→            value={bio}
   273→            onChange={(e) => setBio(e.target.value)}
   274→            placeholder="Tell your friends a bit about yourself."
   275→            rows={3}
   276→            className="bg-black/20 border-white/10 rounded-xl p-4 focus-visible:ring-primary/50 transition-all resize-none"
   277→          />
   278→        </div>
   279→
   280→        <div className="pt-2 flex justify-end">
   281→          <Button onClick={handleSave} disabled={update.isPending} className="h-11 px-6 rounded-xl gradient-primary shadow-glow hover:scale-105 active:scale-95 transition-all font-semibold">
   282→            {update.isPending ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Save className="w-4 h-4 mr-2" />}
   283→            Save changes
   284→          </Button>
   285→        </div>
   286→      </GlassSurface>
   287→    </div>
   288→  )
   289→}

   293→function PrivacySection() {
   294→  const qc = useQueryClient()
   295→  const { data: me } = useQuery({
   296→    queryKey: ['me'],
   297→    queryFn: async () => {
   298→      const res = await fetch('/api/auth/me')
   299→      const data = await res.json()
   300→      return data.user
   301→    },
   302→  })
   303→
   304→  const [lastSeenVisible, setLastSeenVisible] = useState(true)
   305→  const [readReceipts, setReadReceipts] = useState(true)
   306→  const [typingIndicators, setTypingIndicators] = useState(true)
   307→
   308→  if (me && lastSeenVisible && !me.lastSeenVisible === false) {
   309→    // initial sync — only run once
   310→  }
   311→  useState(() => {
   312→    if (me) {
   313→      setLastSeenVisible(me.lastSeenVisible ?? true)
   314→      setReadReceipts(me.readReceiptsEnabled ?? true)
   315→      setTypingIndicators(me.typingIndicatorsEnabled ?? true)
   316→    }
   317→  })
   318→
   319→  const update = useMutation({
   320→    mutationFn: async (data: any) => {
   321→      const res = await fetch(`/api/users/me`, {
   322→        method: 'PATCH',
   323→        headers: { 'Content-Type': 'application/json' },
   324→        body: JSON.stringify(data),
   325→      })
   326→      if (!res.ok) throw new Error('failed')
   327→      return res.json()
   328→    },
   329→    onSuccess: () => {
   330→      qc.invalidateQueries({ queryKey: ['me'] })
   331→      toast.success('Saved')
   332→    },
   333→  })
   334→
   335→  const handleToggle = (key: 'lastSeenVisible' | 'readReceiptsEnabled' | 'typingIndicatorsEnabled', value: boolean) => {
   336→    if (key === 'lastSeenVisible') setLastSeenVisible(value)
   337→    if (key === 'readReceiptsEnabled') setReadReceipts(value)
   338→    if (key === 'typingIndicatorsEnabled') setTypingIndicators(value)
   339→    update.mutate({ [key]: value })
   340→  }
   341→
   342→  return (
   343→    <GlassSurface blur={16} opacity={0.03} className="overflow-hidden">
   344→      <div className="divide-y divide-white/5">
   345→        <PrivacyRow
   346→          title="Last seen visible"
   347→          desc="Show when you were last active"
   348→          checked={lastSeenVisible}
   349→          onChange={(v) => handleToggle('lastSeenVisible', v)}
   350→        />
   351→        <PrivacyRow
   352→          title="Read receipts"
   353→          desc="Let others know when you've read their messages"
   354→          checked={readReceipts}
   355→          onChange={(v) => handleToggle('readReceiptsEnabled', v)}
   356→        />
   357→        <PrivacyRow
   358→          title="Typing indicators"
   359→          desc="Show others when you're typing"
   360→          checked={typingIndicators}
   361→          onChange={(v) => handleToggle('typingIndicatorsEnabled', v)}
   362→        />
   363→      </div>
   364→    </GlassSurface>
   365→  )
   366→}

   395→function BotsList({ onEditBot }: { onEditBot: (bot: any) => void }) {
   396→  const { bots, isLoading, create, update, remove } = useBots()
   397→  const confirm = useConfirm()
   398→  const [createOpen, setCreateOpen] = useState(false)
   399→  const [name, setName] = useState('')
   400→  const [username, setUsername] = useState('')
   401→  const [description, setDescription] = useState('')
   402→
   403→  const handleCreate = async () => {
   404→    try {
   405→      await create({ name, username, description, module: 'visual' })
   406→      toast.success('Bot created! Edit it to design your flow.')
   407→      setCreateOpen(false)
   408→      setName('')
   409→      setUsername('')
   410→      setDescription('')
   411→    } catch (e: any) {
   412→      toast.error(e.message)
   413→    }
   414→  }
   415→
   416→  return (
   417→    <Card className="p-4">
   418→      <div className="flex items-center justify-between mb-3">
   419→        <h2 className="font-semibold text-sm">My Bots ({bots.length})</h2>
   420→        <Button size="sm" onClick={() => setCreateOpen(!createOpen)}>
   421→          <Plus className="w-4 h-4 mr-1" /> New Bot

   436→            <Label>Description</Label>
   437→            <Textarea value={description} onChange={(e) => setDescription(e.target.value)} rows={2} placeholder="Runs polls in any channel" />
   438→          </div>
   439→          <p className="text-xs text-muted-foreground">
   440→            Bots are created with the visual flow editor. After creating, click "Edit Flow" to design your bot's behavior.
   441→          </p>
   442→          <div className="flex gap-2">
   443→            <Button size="sm" onClick={handleCreate} disabled={!name.trim() || !username.trim()}>Create</Button>

   446→        </div>

   449→      {isLoading ? (
   450→        <div className="text-center py-6 text-sm text-muted-foreground">Loading...</div>
   451→      ) : bots.length === 0 ? (
   452→        <div className="text-center py-10">
   453→          <div className="w-16 h-16 mx-auto rounded-3xl bg-primary/10 flex items-center justify-center mb-3 ring-1 ring-primary/15">
   454→            <Bot className="w-8 h-8 text-primary" strokeWidth={1.5} />
   455→          </div>
   456→          <p className="font-medium text-base">No bots yet</p>
   457→          <p className="text-sm text-muted-foreground mt-1 max-w-xs mx-auto">
   458→            Create a bot and design its behavior visually with the flow editor.
   459→          </p>
   460→        </div>
   461→      ) : (
   462→        <div className="space-y-2">
   463→          {bots.map((bot: any) => (
   464→            <div key={bot.id} className="p-3 rounded-lg border space-y-2">
   465→              <div className="flex items-center gap-3">
   466→                <div className="w-9 h-9 rounded-full bg-primary/15 flex items-center justify-center shrink-0">
   467→                  <Bot className="w-4 h-4 text-primary" />
   468→                </div>
   469→                <div className="flex-1 min-w-0">
   470→                  <div className="font-medium text-sm truncate">
   471→                    {bot.name} <span className="text-xs text-muted-foreground">@{bot.username}</span>

   473→                  <div className="text-xs text-muted-foreground truncate">
   474→                    {bot.module} · {bot.enabled ? 'enabled' : 'disabled'}
   475→                  </div>
   476→                </div>
   477→                <Button variant="outline" size="sm" onClick={() => onEditBot(bot)}>
   478→                  Edit Flow
   479→                </Button>
   480→                <Button variant="ghost" size="sm" onClick={() => update({ id: bot.id, data: { enabled: !bot.enabled } })}>
   481→                  {bot.enabled ? 'Disable' : 'Enable'}
   482→                </Button>
   483→                <Button variant="ghost" size="icon" className="h-8 w-8 text-muted-foreground hover:text-red-500"
   484→                  onClick={async () => {
   485→                    const ok = await confirm({ title: `Delete bot @${bot.username}?`, message: 'This will permanently delete the bot and all its conversation sessions.', confirmLabel: 'Delete', variant: 'danger' })
   486→                    if (ok) { remove(bot.id).then(() => toast.success('Deleted')) }
   487→                  }}>
   488→                  <Trash2 className="w-4 h-4" />
   489→                </Button>
   490→              </div>
   491→              {/* Add to channel */}
   492→              <AddBotToChannel botId={bot.id} />
   493→            </div>
   494→          ))}
   495→        </div>
   496→      )}
   497→    </Card>
   498→  )
   499→}
