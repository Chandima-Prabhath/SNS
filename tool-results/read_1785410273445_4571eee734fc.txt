   235→   235→/**
   236→   236→ * useVoiceRecorder — hook that manages MediaRecorder state for recording voice
   237→   237→ * messages. Tap the mic to start, tap stop to upload + send, or cancel to
   238→   238→ * discard. The recording is uploaded via /api/upload and sent as a message
   239→   239→ * with mediaType 'audio/webm' (or the browser's best supported format).
   240→   240→ */
   241→   241→function useVoiceRecorder({
   242→   242→  onSend,
   243→   243→}: {
   244→   244→  onSend: (url: string, mediaType: string) => Promise<void>
   245→   245→}) {
   246→   246→  const [isRecording, setIsRecording] = useState(false)
   247→   247→  const [isUploading, setIsUploading] = useState(false)
   248→   248→  const [seconds, setSeconds] = useState(0)
   249→   249→  const mediaRecorderRef = useRef<MediaRecorder | null>(null)
   250→   250→  const chunksRef = useRef<Blob[]>([])
   251→   251→  const streamRef = useRef<MediaStream | null>(null)
   252→   252→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   253→   253→  const mimeTypeRef = useRef<string>('audio/webm')
   254→   254→
   255→   255→  // Stop the timer + release the mic on unmount
   256→   256→  useEffect(() => {
   257→   257→    return () => {
   258→   258→      if (timerRef.current) clearInterval(timerRef.current)
   259→   259→      if (streamRef.current) streamRef.current.getTracks().forEach((t) => t.stop())
   260→   260→    }
   261→   261→  }, [])
   262→   262→
   263→   263→  const pickMimeType = useCallback(() => {
   264→   264→    const candidates = [
   265→   265→      'audio/webm;codecs=opus',
   266→   266→      'audio/webm',
   267→   267→      'audio/ogg;codecs=opus',
   268→   268→      'audio/mp4',
   269→   269→    ]
   270→   270→    for (const t of candidates) {
   271→   271→      if (typeof MediaRecorder !== 'undefined' && MediaRecorder.isTypeSupported(t)) {
   272→   272→        return t
   273→   273→      }
   274→   274→    }
   275→   275→    return ''
   276→   276→  }, [])
   277→   277→
   278→   278→  const stopTracks = useCallback(() => {
   279→   279→    if (timerRef.current) {
   280→   280→      clearInterval(timerRef.current)
   281→   281→      timerRef.current = null
   282→   282→    }
   283→   283→    if (streamRef.current) {
   284→   284→      streamRef.current.getTracks().forEach((t) => t.stop())
   285→   285→      streamRef.current = null
   286→   286→    }
   287→   287→  }, [])
   288→   288→
   289→   289→  const start = useCallback(async () => {
   290→   290→    try {
   291→   291→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   292→   292→      streamRef.current = stream
   293→   293→      const mimeType = pickMimeType()
   294→   294→      mimeTypeRef.current = mimeType || 'audio/webm'
   295→   295→      const recorder = mimeType
   296→   296→        ? new MediaRecorder(stream, { mimeType })
   297→   297→        : new MediaRecorder(stream)
   298→   298→      chunksRef.current = []
   299→   299→      recorder.ondataavailable = (e) => {
   300→   300→        if (e.data && e.data.size > 0) chunksRef.current.push(e.data)
   301→   301→      }
   302→   302→      recorder.start()
   303→   303→      mediaRecorderRef.current = recorder
   304→   304→      setIsRecording(true)
   305→   305→      setSeconds(0)
   306→   306→      timerRef.current = setInterval(() => {
   307→   307→        setSeconds((s) => s + 1)
   308→   308→      }, 1000)
   309→   309→    } catch {
   310→   310→      toast.error('Microphone access denied')
   311→   311→      if (streamRef.current) {
   312→   312→        streamRef.current.getTracks().forEach((t) => t.stop())
   313→   313→        streamRef.current = null
   314→   314→      }
   315→   315→    }
   316→   316→  }, [pickMimeType])
   317→   317→
   318→   318→  const cancel = useCallback(() => {
   319→   319→    const recorder = mediaRecorderRef.current
   320→   320→    if (recorder && recorder.state !== 'inactive') {
   321→   321→      // Detach onstop so the pending send is discarded
   322→   322→      recorder.onstop = null
   323→   323→      try {
   324→   324→        recorder.stop()
   325→   325→      } catch {
   326→   326→        // ignore
   327→   327→      }
   328→   328→    }
   329→   329→    mediaRecorderRef.current = null
   330→   330→    chunksRef.current = []
   331→   331→    setIsRecording(false)
   332→   332→    setSeconds(0)
   333→   333→    stopTracks()
   334→   334→  }, [stopTracks])
   335→   335→
   336→   336→  const stopAndSend = useCallback(async () => {
   337→   337→    const recorder = mediaRecorderRef.current
   338→   338→    if (!recorder || recorder.state === 'inactive') {
   339→   339→      cancel()
   340→   340→      return
   341→   341→    }
   342→   342→    const mime = mimeTypeRef.current
   343→   343→    const blobPromise = new Promise<Blob>((resolve) => {
   344→   344→      recorder.onstop = () => {
   345→   345→        const blob = new Blob(chunksRef.current, { type: mime || 'audio/webm' })
   346→   346→        resolve(blob)
   347→   347→      }
   348→   348→    })
   349→   349→    try {
   350→   350→      recorder.stop()
   351→   351→    } catch {
   352→   352→      cancel()
   353→   353→      return
   354→   354→    }
   355→   355→    setIsRecording(false)
   356→   356→    stopTracks()
   357→   357→    setIsUploading(true)
   358→   358→    try {
   359→   359→      const blob = await blobPromise
   360→   360→      if (blob.size === 0) {
   361→   361→        toast.error('Recording was empty')
   362→   362→        return
   363→   363→      }
   364→   364→      const ext = mime.includes('mp4') ? 'm4a' : mime.includes('ogg') ? 'ogg' : 'webm'
   365→   365→      const file = new File([blob], `voice-${Date.now()}.${ext}`, { type: blob.type || mime })
   366→   366→      const formData = new FormData()
   367→   367→      formData.append('file', file)
   368→   368→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   369→   369→      if (!res.ok) {
   370→   370→        const t = await res.text().catch(() => 'Upload failed')
   371→   371→        throw new Error(t.slice(0, 120) || 'Upload failed')
   372→   372→      }
   373→   373→      const data = await res.json()
   374→   374→      await onSend(data.url, data.type || mime || 'audio/webm')
   375→   375→      toast.success('Voice message sent')
   376→   376→    } catch (e: any) {
   377→   377→      toast.error(e?.message || 'Failed to send voice message')
   378→   378→    } finally {
   379→   379→      setIsUploading(false)
   380→   380→      setSeconds(0)
   381→   381→      chunksRef.current = []
   382→   382→      mediaRecorderRef.current = null
   383→   383→    }
   384→   384→  }, [cancel, onSend, stopTracks])
   385→   385→
   386→   386→  return { isRecording, isUploading, seconds, start, cancel, stopAndSend }
   387→   387→}
   388→   388→
   389→   389→/**
   390→   390→ * TTS Dialog — lets the user type text, pick a voice (built-in or custom),
   391→   391→ * preview, and send. Also supports creating custom voices from audio clips.
   392→   392→ */
   393→   393→function TtsDialog({
   394→   394→  open,
   395→   395→  onOpenChange,
   396→   396→  onSend,
   397→   397→}: {
   398→   398→  open: boolean
   399→   399→  onOpenChange: (o: boolean) => void
   400→   400→  onSend: (url: string) => Promise<void>
   401→   401→}) {
   402→   402→  const [tab, setTab] = useState<'generate' | 'voices'>('generate')
   403→   403→  const [text, setText] = useState('')
   404→   404→  const [voice, setVoice] = useState('alba')
   405→   405→  const [selectedCustomVoiceId, setSelectedCustomVoiceId] = useState<string | null>(null)
   406→   406→  const [generating, setGenerating] = useState(false)
   407→   407→  const [previewUrl, setPreviewUrl] = useState<string | null>(null)
   408→   408→  const [sendUrl, setSendUrl] = useState<string | null>(null)
   409→   409→  const [sending, setSending] = useState(false)
   410→   410→  const qc = useQueryClient()
   411→   411→
   412→   412→  // Fetch built-in voices from the API (single source of truth for labels)
   413→   413→  const { data: builtinVoicesData } = useQuery({
   414→   414→    queryKey: ['tts-builtin-voices'],
   415→   415→    queryFn: async () => {
   416→   416→      const res = await fetch('/api/tts')
   417→   417→      if (!res.ok) throw new Error('failed')
   418→   418→      return res.json()
   419→   419→    },
   420→   420→    enabled: open,
   421→   421→  })
   422→   422→  const builtinVoices: { id: string; name: string; language: string; gender: string }[] =
   423→   423→    builtinVoicesData?.voices || []
   424→   424→
   425→   425→  // Fetch custom voices
   426→   426→  const { data: customVoicesData } = useQuery({
   427→   427→    queryKey: ['tts-voices'],
   428→   428→    queryFn: async () => {
   429→   429→      const res = await fetch('/api/tts/voices')
   430→   430→      if (!res.ok) throw new Error('failed')
   431→   431→      return res.json()
   432→   432→    },
   433→   433→    enabled: open,
   434→   434→  })
   435→   435→  const customVoices = customVoicesData?.voices || []
   436→   436→
   437→   437→  const handleGenerate = async () => {
   438→   438→    if (!text.trim() || generating) return
   439→   439→    setGenerating(true)
   440→   440→    setPreviewUrl(null)
   441→   441→    setSendUrl(null)
   442→   442→    try {
   443→   443→      const body: any = { text }
   444→   444→      if (selectedCustomVoiceId) {
   445→   445→        body.customVoiceId = selectedCustomVoiceId
   446→   446→      } else {
