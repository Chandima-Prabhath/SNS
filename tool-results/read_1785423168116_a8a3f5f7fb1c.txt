    31→
    32→interface ChannelInfo {
    33→  id: string
    34→  name: string
    35→  topic: string | null
    36→  type: string
    37→  order: number
    38→  partner?: {
    39→    id: string
    40→    displayName: string
    41→    username: string
    42→    avatarUrl: string | null
    43→    status: string
    44→  }
    45→  group: {
    46→    id: string
    47→    name: string
    48→    iconUrl: string | null
    49→    isDm: boolean
    50→    inviteCode: string
    51→    ownerId: string
    52→  }
    53→}
    54→
    55→interface ChatRow {
    56→  channel: ChannelInfo
    57→  isDm: boolean
    58→  isGroup: boolean
    59→  isBot: boolean
    60→  partner?: ChannelInfo['partner']
    61→  groupName: string
    62→  groupIcon: string | null

    64→
    65→export function ChatList() {
    66→  const activeChannelId = useAppStore((s) => s.activeChannelId)
    67→  const setActiveChannel = useAppStore((s) => s.setActiveChannel)
    68→  const selectedGroupId = useAppStore((s) => s.selectedGroupId)
    69→  const filter = useAppStore((s) => s.chatFilter)
    70→  const setFilter = useAppStore((s) => s.setChatFilter)
    71→  const presence = usePresence()
    72→  const { data: unreadData } = useUnreadCounts()
    73→  const { data: session } = useSession()
    74→  const myId = (session?.user as any)?.id
    75→
    76→  const [search, setSearch] = useState('')
    77→
    78→  const { data: groups } = useQuery({
    79→    queryKey: ['channels'],
    80→    queryFn: async () => {
    81→      const res = await fetch('/api/channels')
    82→      const data = await res.json()
    83→      return data.groups as any[]
    84→    },
    85→  })
    86→
    87→  // The currently selected group object (or null for DMs)
    88→  const selectedGroup = groups?.find((g) => g.id === selectedGroupId) || null
    89→  const isViewingDms = selectedGroupId === 'dm'
    90→  const setServerRailOpen = useAppStore((s) => s.setServerRailOpen)
    91→  const ctxMenu = useContextMenu()
    92→  const qc = useQueryClient()
    93→
    94→  // Show a context menu for a chat row — supports right-click (desktop) and
    95→  // long-press (mobile, via the custom touch handler).
    96→  const showChatContextMenu = (e: React.MouseEvent | React.TouchEvent, row: ChatRow) => {
    97→    let x: number, y: number
    98→    if ('touches' in e) {
    99→      const t = e.touches[0] || (e.changedTouches?.[0])
   100→      if (!t) return
   101→      x = t.clientX; y = t.clientY
   102→    } else {
   103→      x = e.clientX; y = e.clientY
   104→    }
   105→    if (!ctxMenu) return
   106→    ctxMenu.show(x, y, [
   107→      {
   108→        label: 'Open',
   109→        icon: <MessageCircle className="w-4 h-4" />,
   110→        onClick: () => setActiveChannel(row.channel.id),
   111→      },
   112→      {
   113→        label: 'Mark as read',
   114→        icon: <Check className="w-4 h-4" />,
   115→        onClick: () => {
   116→          fetch(`/api/channels/${row.channel.id}/read`, { method: 'POST' }).then(() => {
   117→            qc.invalidateQueries({ queryKey: ['unread-counts'] })
   118→            qc.invalidateQueries({ queryKey: ['channels'] })
   119→            toast.success('Marked as read')
   120→          })
   121→        },
   122→      },
   123→      {
   124→        label: row.isDm ? 'Delete conversation' : 'Leave channel',
   125→        icon: <Trash2 className="w-4 h-4" />,
   126→        variant: 'danger',
   127→        onClick: () => {
   128→          if (row.isDm) {
   129→            if (confirm('Delete this conversation? Messages will be removed for you.')) {
   130→              // For DMs, we leave the channel membership
   131→              fetch(`/api/channels/${row.channel.id}/members`, {
   132→                method: 'DELETE',
   133→              }).then(() => {
   134→                qc.invalidateQueries({ queryKey: ['channels'] })
   135→                if (activeChannelId === row.channel.id) setActiveChannel(null)
   136→                toast.success('Conversation deleted')
   137→              }).catch(() => toast.error('Failed to delete'))
   138→            }
   139→          } else {
   140→            if (confirm(`Leave #${row.channel.name}?`)) {
   141→              fetch(`/api/channels/${row.channel.id}/members`, {
   142→                method: 'DELETE',
   143→              }).then(() => {
   144→                qc.invalidateQueries({ queryKey: ['channels'] })
   145→                if (activeChannelId === row.channel.id) setActiveChannel(null)
   146→                toast.success('Left channel')
   147→              }).catch(() => toast.error('Failed to leave'))
   148→            }
   149→          }
   150→        },
   151→      },
   152→    ])
   153→  }
   154→
   155→  // Join a voice/video channel — creates a call and switches to the voice view
   156→  const setView = useAppStore((s) => s.setView)
   157→  const { startCall } = useCall()
   158→  const joinCallMutation = useMutation({
   159→    mutationFn: async (channelId: string) => {
   160→      const res = await fetch('/api/calls', {
   161→        method: 'POST',
   162→        headers: { 'Content-Type': 'application/json' },
   163→        body: JSON.stringify({ channelId }),
   164→      })
   165→      if (!res.ok) throw new Error('failed')
   166→      return res.json()
   167→    },
   168→    onSuccess: async (data) => {
   169→      await startCall({ callId: data.call.id, channelId: data.call.channelId })
   170→      unlockAudio()
   171→      setView('voice')
   172→      toast.success('Joined channel')
   173→    },
   174→    onError: () => toast.error('Failed to join channel'),
   175→  })
   176→
   177→  // Auto-select first channel on desktop only (mobile shows the list first)
   178→  useEffect(() => {
   179→    if (!activeChannelId && groups && groups.length > 0 && typeof window !== 'undefined' && window.innerWidth >= 1024) {
   180→      const first = groups[0].channels.find((c: any) => c.type === 'text')
   181→      if (first) setActiveChannel(first.id)
   182→    }
   183→  }, [activeChannelId, groups, setActiveChannel])
   184→
   185→  // Flatten channels into a unified list of "chat rows"
   186→  // When a server (non-DM group) is selected in the rail, show ALL channel
   187→  // types (text, voice, video) for that group. When 'dm' is selected, only
   188→  // show DM text channels.
   189→  const allChats: ChatRow[] = useMemo(() => {
   190→    if (!groups) return []
   191→    const rows: ChatRow[] = []
   192→    for (const g of groups) {
   193→      // Filter by selected group from the server rail
   194→      if (selectedGroupId === 'dm' && !g.isDm) continue
   195→      if (selectedGroupId && selectedGroupId !== 'dm' && g.id !== selectedGroupId) continue
   196→      for (const ch of g.channels) {
   197→        // DMs only have text channels. Groups show all types.
   198→        if (g.isDm && ch.type !== 'text') continue
   199→        rows.push({
   200→          channel: ch,
   201→          isDm: g.isDm,
   202→          isGroup: !g.isDm,
   203→          isBot: false,
   204→          partner: g.partner, // populated by the API for DM groups
   205→          groupName: g.name,
   206→          groupIcon: g.iconUrl,
   207→        })
   208→      }
   209→    }
   210→    return rows
   211→  }, [groups, selectedGroupId])
   212→
   213→  // Apply filter
   214→  const filteredChats = useMemo(() => {
   215→    let list = allChats
   216→    if (filter === 'dms') list = list.filter((c) => c.isDm)
   217→    else if (filter === 'groups') list = list.filter((c) => c.isGroup)
   218→    else if (filter === 'bots') list = list.filter((c) => c.isBot)
   219→
   220→    if (search.trim()) {
   221→      const q = search.toLowerCase()
   222→      list = list.filter(
   223→        (c) =>
   224→          c.channel.name.toLowerCase().includes(q) ||
   225→          c.groupName.toLowerCase().includes(q) ||
   226→          (c.partner?.displayName.toLowerCase().includes(q) ?? false) ||
   227→          (c.partner?.username.toLowerCase().includes(q) ?? false)
   228→      )
   229→    }
   230→    return list
   231→  }, [allChats, filter, search])
   232→
   233→  return (
   234→    <div className="flex flex-col h-full w-full bg-sidebar/50 backdrop-blur-2xl">
   235→      {/* Header — shows the selected group name or "Direct Messages" */}
   236→      <div className="px-4 pt-4 pb-3 space-y-4 border-b border-white/5 bg-background/30 backdrop-blur-3xl shadow-sm">
   237→        <div className="flex items-center justify-between">
   238→          <div className="flex items-center gap-2.5 min-w-0">
   239→            {/* Mobile: hamburger to open server rail drawer */}
   240→            <button
   241→              onClick={() => setServerRailOpen(true)}
   242→              className="md:hidden w-9 h-9 rounded-xl bg-sidebar-accent flex items-center justify-center shrink-0 hover:bg-accent transition-colors active:scale-95"
   243→              title="Groups & servers"
   244→            >
   245→              <Menu className="w-4 h-4" />
   246→            </button>
   247→            {isViewingDms ? (
   248→              <>
   249→                <div className="w-9 h-9 rounded-xl bg-primary/15 flex items-center justify-center shrink-0">
   250→                  <MessageCircle className="w-4 h-4 text-primary" />
   251→                </div>
   252→                <div className="min-w-0">
   253→                  <h1 className="text-lg font-semibold tracking-tight truncate">Direct Messages</h1>
   254→                  <p className="text-[11px] text-muted-foreground">Private conversations</p>
   255→                </div>
   256→              </>
   257→            ) : selectedGroup ? (
   258→              <>
   259→                <div className="w-9 h-9 rounded-xl bg-primary/15 flex items-center justify-center shrink-0 overflow-hidden">
   260→                  {selectedGroup.iconUrl ? (
   261→                    <img src={selectedGroup.iconUrl} alt="" className="w-full h-full object-cover" />
   262→                  ) : (
   263→                    <Users className="w-4 h-4 text-primary" />
   264→                  )}
   265→                </div>
   266→                <div className="min-w-0">
   267→                  <h1 className="text-lg font-semibold tracking-tight truncate">{selectedGroup.name}</h1>
   268→                  {selectedGroup.description && (
   269→                    <p className="text-[11px] text-muted-foreground truncate">{selectedGroup.description}</p>
   270→                  )}
   271→                </div>
   272→              </>
   273→            ) : (
   274→              <h1 className="text-2xl font-semibold tracking-tight">Chats</h1>
   275→            )}
   276→          </div>
   277→          <div className="flex gap-1 shrink-0">
   278→            {isViewingDms && <NewDmButton />}
   279→            {!isViewingDms && selectedGroup && <GroupSettingsButton group={selectedGroup} />}
   280→          </div>
   281→        </div>
   282→
   283→        {/* Search */}
   284→        <div className="relative group">
   285→          <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground group-focus-within:text-primary transition-colors" />
   286→          <Input
   287→            value={search}
   288→            onChange={(e) => setSearch(e.target.value)}
   289→            placeholder={isViewingDms ? "Search DMs..." : "Search channels..."}
   290→            className="pl-10 h-10 bg-black/20 border-white/10 focus-visible:ring-primary/30 rounded-xl shadow-inner transition-all"
   291→          />
   292→        </div>
   293→
   294→        {/* Filter chips — only show for DMs */}
   295→        {isViewingDms && (
   296→          <div className="flex gap-2 overflow-x-auto no-scrollbar -mx-4 px-4">
   297→            {([
   298→              ['all', 'All'],
   299→              ['unread', 'Unread'],
   300→            ] as const).map(([key, label]) => (
   301→              <button
   302→                key={key}
   303→                onClick={() => setFilter(key)}
   304→                className={cn(
   305→                  'px-3.5 py-1.5 rounded-full text-xs font-medium whitespace-nowrap transition-colors',
   306→                  filter === key
   307→                    ? 'bg-primary text-primary-foreground'
   308→                    : 'bg-muted text-muted-foreground hover:bg-accent hover:text-foreground'
   309→                )}
   310→              >
   311→                {label}
   312→              </button>
   313→            ))}
   314→          </div>
   315→        )}
   316→      </div>
   317→
   318→      {/* Chat list */}
   319→      <ScrollArea className="flex-1">
   320→        <div className="px-2 pb-4">
   321→          {filteredChats.length === 0 ? (
   322→            <EmptyChatList />
   323→          ) : (
   324→            <>
   325→              <div className="space-y-0.5">
   326→                {filteredChats.map((row) => {
   327→                  const active = row.channel.id === activeChannelId
   328→                  // A DM partner may have been deleted — detect and show a
   329→                  // graceful "Deleted User" label instead of @user fallback.
   330→                  const partnerDeleted = !row.isGroup && !row.partner
   331→                  const presenceInfo = row.partner ? presence[row.partner.id] : null
   332→                  const presenceStatus = presenceInfo?.status || 'offline'
   333→                  const unreadCount = unreadData?.unread?.[row.channel.id] || 0
   334→                  const displayName = row.isGroup
   335→                    ? row.channel.name
   336→                    : row.partner?.displayName || 'Deleted User'
   337→                  const handleText = row.isGroup
   338→                    ? row.groupName
   339→                    : row.partner ? `@${row.partner.username}` : 'account no longer exists'
   340→
   341→                  // Voice/video channels are rendered as "join" rows, not chat rows
   342→                  const isCallChannel = row.channel.type === 'voice' || row.channel.type === 'video'
   343→                  const CallIcon = row.channel.type === 'video' ? Video : Volume2
   344→
   345→                  if (isCallChannel) {
   346→                    return (
   347→                      <button
   348→                        key={row.channel.id}
   349→                        onClick={() => joinCallMutation.mutate(row.channel.id)}
   350→                        disabled={joinCallMutation.isPending}
   351→                        className={cn(
   352→                          'w-full flex items-center gap-3 p-2.5 rounded-xl transition-colors text-left group/call',
   353→                          'hover:bg-accent/50 disabled:opacity-50'
   354→                        )}
   355→                      >
   356→                        {/* Icon */}
   357→                        <div className={cn(
   358→                          'relative w-12 h-12 rounded-full flex items-center justify-center shrink-0',
   359→                          row.channel.type === 'video'
   360→                            ? 'bg-emerald-500/15 text-emerald-400'
   361→                            : 'bg-primary/15 text-primary'
   362→                        )}>
   363→                          <CallIcon className="w-5 h-5" />
   364→                        </div>
   365→
   366→                        {/* Name + status */}
   367→                        <div className="flex-1 min-w-0">
   368→                          <div className="flex items-baseline justify-between gap-2">
   369→                            <span className="truncate text-[15px] font-medium">
   370→                              {row.channel.name}
   371→                            </span>
   372→                          </div>
   373→                          <div className="text-xs text-muted-foreground truncate flex items-center gap-1.5">
   374→                            <span className="capitalize">{row.channel.type} channel</span>
   375→                            <span>·</span>
   376→                            <span className="text-primary flex items-center gap-0.5">
   377→                              <Phone className="w-3 h-3" />
   378→                              Tap to join
   379→                            </span>
   380→                          </div>
   381→                        </div>
   382→
   383→                        {/* Join button */}
   384→                        <div className="w-9 h-9 rounded-full gradient-primary flex items-center justify-center shrink-0 group-hover/call:scale-110 transition-transform shadow-glow">
   385→                          <Phone className="w-4 h-4 text-primary-foreground" />
   386→                        </div>
   387→                      </button>
   388→                    )
   389→                  }
   390→
   391→                  return (
   392→                    <button
   393→                      key={row.channel.id}
   394→                      onClick={() => setActiveChannel(row.channel.id)}
   395→                      onContextMenu={(e) => { e.preventDefault(); showChatContextMenu(e, row) }}
   396→                      onTouchStart={(e) => {
   397→                        // Long-press detection for mobile
   398→                        const touch = e.touches[0]
   399→                        const timer = setTimeout(() => {
   400→                          showChatContextMenu({
   401→                            touches: [{ clientX: touch.clientX, clientY: touch.clientY }],
   402→                          } as any, row)
   403→                          // Trigger haptic feedback if available
   404→                          if (navigator.vibrate) navigator.vibrate(50)
   405→                        }, 500)
   406→                        const cancel = () => clearTimeout(timer)
   407→                        ;(e.currentTarget as HTMLElement).dataset.longPressTimer = String(timer)
   408→                        // One-time listeners for cancellation
   409→                        const el = e.currentTarget
   410→                        const clear = () => {
   411→                          cancel()
   412→                          el.removeEventListener('touchend', clear)
   413→                          el.removeEventListener('touchmove', clear)
   414→                          el.removeEventListener('touchcancel', clear)
   415→                        }
   416→                        el.addEventListener('touchend', clear, { once: true })
   417→                        el.addEventListener('touchmove', clear, { once: true })
   418→                        el.addEventListener('touchcancel', clear, { once: true })
   419→                      }}
   420→                      className={cn(
   421→                        'w-full flex items-center gap-3.5 p-3 rounded-2xl transition-all text-left select-none relative overflow-hidden',
   422→                        active 
   423→                          ? 'bg-primary/10 ring-1 ring-primary/20 shadow-[inset_0_1px_1px_rgba(255,255,255,0.05)]' 
   424→                          : 'hover:bg-white/[0.04]'
   425→                      )}
   426→                    >
   427→                      {/* Avatar */}
   428→                      {row.isGroup ? (
   429→                        <div className="relative w-12 h-12 rounded-full bg-primary/15 flex items-center justify-center shrink-0">
   430→                          <Hash className="w-5 h-5 text-primary" />
   431→                        </div>
   432→                      ) : partnerDeleted ? (
   433→                        <div className="relative shrink-0">
   434→                          <div className="w-12 h-12 rounded-full bg-muted flex items-center justify-center text-muted-foreground">
   435→                            <UserX className="w-5 h-5" />
   436→                          </div>
   437→                        </div>
   438→                      ) : (
   439→                        <div className="relative shrink-0">
   440→                          <Avatar className="w-12 h-12">
   441→                            <AvatarImage src={row.partner?.avatarUrl || undefined} />
   442→                            <AvatarFallback>
   443→                              {row.partner?.displayName?.charAt(0) || row.channel.name.charAt(0).toUpperCase()}
   444→                            </AvatarFallback>
   445→                          </Avatar>
   446→                          <span
   447→                            className={cn(
   448→                              'absolute bottom-0 right-0 w-3 h-3 rounded-full border-2 border-sidebar',
   449→                              presenceStatus === 'online' && 'bg-status-online',
   450→                              presenceStatus === 'idle' && 'bg-status-idle',
   451→                              presenceStatus === 'dnd' && 'bg-status-dnd',
   452→                              presenceStatus === 'offline' && 'bg-status-offline'
   453→                            )}
   454→                          />
   455→                        </div>
   456→                      )}
   457→
   458→                      {/* Name + preview */}
   459→                      <div className="flex-1 min-w-0">
   460→                        <div className="flex items-baseline justify-between gap-2">
   461→                          <span
   462→                            className={cn(
   463→                              'truncate text-[15px]',
   464→                              unreadCount > 0 ? 'font-semibold text-foreground' : 'font-medium',
   465→                              partnerDeleted && 'text-muted-foreground italic'
   466→                            )}
   467→                          >
   468→                            {displayName}
   469→                          </span>
   470→                          {unreadCount > 0 && !active && (
   471→                            <span className="shrink-0 min-w-[20px] h-5 px-1.5 rounded-full bg-primary text-primary-foreground text-[10px] font-bold flex items-center justify-center">
   472→                              {unreadCount > 99 ? '99+' : unreadCount}
   473→                            </span>
   474→                          )}
   475→                        </div>
   476→                        <div
   477→                          className={cn(
   478→                            'text-xs truncate',
   479→                            unreadCount > 0 ? 'text-foreground/80 font-medium' : 'text-muted-foreground',
   480→                            partnerDeleted && 'italic'
   481→                          )}
   482→                        >
   483→                          {handleText}
   484→                        </div>
   485→                      </div>
   486→                    </button>
   487→                  )
   488→                })}
   489→              </div>
   490→            </>
   491→          )}
   492→        </div>
   493→      </ScrollArea>
   494→    </div>
   495→  )
   496→}

   517→function JoinGroupButton() {
   518→  const [open, setOpen] = useState(false)
   519→  const [code, setCode] = useState('')
   520→  const qc = useQueryClient()
   521→
   522→  const join = useMutation({
   523→    mutationFn: async () => {
   524→      const res = await fetch('/api/groups', {
   525→        method: 'PATCH',
   526→        headers: { 'Content-Type': 'application/json' },
   527→        body: JSON.stringify({ inviteCode: code.trim() }),
   528→      })

   717→/**
   718→ * Group settings button — opens the GroupSettingsDialog for the currently
   719→ * selected group. Shows a Crown for owners and Shield for admins.
   720→ */
   721→function GroupSettingsButton({ group }: { group: any }) {
   722→  const [open, setOpen] = useState(false)
   723→  const { data: session } = useSession()
   724→  const myId = (session?.user as any)?.id
   725→  const isOwner = group.ownerId === myId
   726→
   727→  // Check if I'm an admin via the members API
   728→  const { data: myMembership } = useQuery({
   729→    queryKey: ['my-membership', group.id],
   730→    queryFn: async () => {
   731→      const res = await fetch(`/api/groups/${group.id}/members`)
   732→      if (!res.ok) return null
   733→      const data = await res.json()
   734→      return data.members?.find((m: any) => m.userId === myId) || null
   735→    },
   736→    enabled: !!myId && open,
   737→  })
   738→  const isAdmin = myMembership?.role === 'admin'
   739→  const canManage = isOwner || isAdmin
   740→  const myRole = isOwner ? 'owner' : isAdmin ? 'admin' : 'member'
   741→
   742→  return (
   743→    <>
   744→      <Button
   745→        variant="ghost"
   746→        size="icon"
   747→        className="h-9 w-9"
   748→        onClick={() => setOpen(true)}
   749→        title="Group settings"
   750→      >
   751→        <Settings className="w-4 h-4" />
   752→      </Button>
   753→      {open && (
   754→        <GroupSettingsDialog
   755→          group={group}
   756→          myRole={myRole}
   757→          onClose={() => setOpen(false)}
   758→        />
   759→      )}
   760→    </>
   761→  )
   762→}
