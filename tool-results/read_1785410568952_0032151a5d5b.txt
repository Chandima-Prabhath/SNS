   138→function ProfileSection() {
   139→  const { data: session } = useSession()
   140→  const qc = useQueryClient()
   141→
   142→  const { data: me, isLoading } = useQuery({
   143→    queryKey: ['me'],
   144→    queryFn: async () => {
   145→      const res = await fetch('/api/auth/me')
   146→      const data = await res.json()
   147→      return data.user
   148→    },
   149→  })
   150→
   151→  const [displayName, setDisplayName] = useState('')
   152→  const [bio, setBio] = useState('')
   153→  const [avatarUrl, setAvatarUrl] = useState('')
   154→  const [customStatus, setCustomStatus] = useState('')
   155→  const [uploading, setUploading] = useState(false)
   156→
   157→  // Sync from server once
   158→  useState(() => {
   159→    if (me) {
   160→      setDisplayName(me.displayName || '')
   161→      setBio(me.bio || '')
   162→      setAvatarUrl(me.avatarUrl || '')
   163→      setCustomStatus(me.customStatus || '')
   164→    }
   165→  })
   166→
   167→  // Also sync on data change
   168→  if (me && displayName === '' && me.displayName) {
   169→    setDisplayName(me.displayName)
   170→    setBio(me.bio || '')
   171→    setAvatarUrl(me.avatarUrl || '')
   172→    setCustomStatus(me.customStatus || '')
   173→  }
   174→
   175→  const update = useMutation({
   176→    mutationFn: async (data: any) => {
   177→      const res = await fetch(`/api/users/me`, {
   178→        method: 'PATCH',
   179→        headers: { 'Content-Type': 'application/json' },
   180→        body: JSON.stringify(data),
   181→      })
   182→      if (!res.ok) throw new Error('failed')
   183→      return res.json()
   184→    },
   185→    onSuccess: () => {
   186→      qc.invalidateQueries({ queryKey: ['me'] })
   187→      toast.success('Saved')
   188→    },
   189→    onError: () => toast.error('Failed to save'),
   190→  })
   191→
   192→  const handleSave = () => {
   193→    update.mutate({ displayName, bio, avatarUrl, customStatus })
   194→  }
   195→
   196→  const handleAvatar = async (e: React.ChangeEvent<HTMLInputElement>) => {
   197→    const file = e.target.files?.[0]
   198→    if (!file) return
   199→    setUploading(true)
   200→    try {
   201→      const formData = new FormData()
   202→      formData.append('file', file)
   203→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   204→      if (!res.ok) {
   205→        const text = await res.text()
   206→        throw new Error(text.slice(0, 120) || 'Upload failed')
   207→      }
   208→      const data = await res.json()
   209→      setAvatarUrl(data.url)
   210→      toast.success('Avatar uploaded — click Save to apply')
   211→    } catch (e: any) {
   212→      toast.error(e.message || 'Upload failed')
   213→    } finally {
   214→      setUploading(false)
   215→      e.target.value = ''
   216→    }
   217→  }
   218→
   219→  if (isLoading) return <div className="flex justify-center py-12"><Loader2 className="w-6 h-6 animate-spin" /></div>
   220→
   221→  return (
   222→    <div className="space-y-6">
   223→      <GlassSurface blur={16} opacity={0.03} className="p-6">
   224→        <div className="flex flex-col md:flex-row gap-8 items-start md:items-center">
   225→          <AvatarPicker
   226→            currentAvatarUrl={avatarUrl}
   227→            userId={me?.id || 'seed'}
   228→            onPick={(url) => { setAvatarUrl(url); toast.success('Avatar selected — click Save to apply') }}
   229→            onUpload={handleAvatar}
   230→            uploading={uploading}
   231→          />
   232→          <div className="flex-1 space-y-1">
   233→            <h2 className="text-xl font-semibold"><ShinyText shimmerDuration={3}>{displayName || 'Your Profile'}</ShinyText></h2>
   234→            <p className="text-sm text-muted-foreground">Manage your public persona</p>
   235→          </div>
   236→        </div>
   237→      </GlassSurface>
   238→
   239→      <GlassSurface blur={16} opacity={0.03} className="p-6 space-y-6">
   240→        <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
   241→          <div className="space-y-2">
   242→            <Label htmlFor="displayName" className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Display name</Label>
   243→            <Input id="displayName" value={displayName} onChange={(e) => setDisplayName(e.target.value)} className="h-12 bg-black/20 border-white/10 rounded-xl px-4 focus-visible:ring-primary/50 transition-all" />
   244→          </div>
   245→
   246→          <div className="space-y-2">
   247→            <Label htmlFor="username" className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Username</Label>
   248→            <Input id="username" value={me?.username || ''} disabled className="h-12 bg-black/20 border-white/5 rounded-xl px-4 opacity-70 cursor-not-allowed" />
   249→            <p className="text-[10px] text-muted-foreground pl-1">Username cannot be changed.</p>
   250→          </div>
   251→        </div>
   252→
   253→        <div className="space-y-2">
   254→          <Label htmlFor="customStatus" className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Custom status</Label>
   255→          <Input
   256→            id="customStatus"
   257→            value={customStatus}
   258→            onChange={(e) => setCustomStatus(e.target.value)}
   259→            placeholder="🎧 Listening to lofi"
   260→            maxLength={80}
   261→            className="h-12 bg-black/20 border-white/10 rounded-xl px-4 focus-visible:ring-primary/50 transition-all"
   262→          />
   263→        </div>
   264→
   265→        <div className="space-y-2">
   266→          <Label htmlFor="bio" className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Bio</Label>
   267→          <Textarea
   268→            id="bio"
   269→            value={bio}
   270→            onChange={(e) => setBio(e.target.value)}
   271→            placeholder="Tell your friends a bit about yourself."
   272→            rows={3}
   273→            className="bg-black/20 border-white/10 rounded-xl p-4 focus-visible:ring-primary/50 transition-all resize-none"
   274→          />
   275→        </div>
   276→
   277→        <div className="pt-2 flex justify-end">
   278→          <Button onClick={handleSave} disabled={update.isPending} className="h-11 px-6 rounded-xl gradient-primary shadow-glow hover:scale-105 active:scale-95 transition-all font-semibold">
   279→            {update.isPending ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Save className="w-4 h-4 mr-2" />}
   280→            Save changes
   281→          </Button>
   282→        </div>
   283→      </GlassSurface>
   284→    </div>
   285→  )
   286→}

   290→function PrivacySection() {
   291→  const qc = useQueryClient()
   292→  const { data: me } = useQuery({
   293→    queryKey: ['me'],
   294→    queryFn: async () => {
   295→      const res = await fetch('/api/auth/me')
   296→      const data = await res.json()
   297→      return data.user
   298→    },
   299→  })
   300→
   301→  const [lastSeenVisible, setLastSeenVisible] = useState(true)
   302→  const [readReceipts, setReadReceipts] = useState(true)
   303→  const [typingIndicators, setTypingIndicators] = useState(true)
   304→
   305→  if (me && lastSeenVisible && !me.lastSeenVisible === false) {
   306→    // initial sync — only run once
   307→  }
   308→  useState(() => {
   309→    if (me) {
   310→      setLastSeenVisible(me.lastSeenVisible ?? true)
   311→      setReadReceipts(me.readReceiptsEnabled ?? true)
   312→      setTypingIndicators(me.typingIndicatorsEnabled ?? true)
   313→    }
   314→  })
   315→
   316→  const update = useMutation({
   317→    mutationFn: async (data: any) => {
   318→      const res = await fetch(`/api/users/me`, {
   319→        method: 'PATCH',
   320→        headers: { 'Content-Type': 'application/json' },
   321→        body: JSON.stringify(data),
   322→      })
   323→      if (!res.ok) throw new Error('failed')
   324→      return res.json()
   325→    },
   326→    onSuccess: () => {
   327→      qc.invalidateQueries({ queryKey: ['me'] })
   328→      toast.success('Saved')
   329→    },
   330→  })
   331→
   332→  const handleToggle = (key: 'lastSeenVisible' | 'readReceiptsEnabled' | 'typingIndicatorsEnabled', value: boolean) => {
   333→    if (key === 'lastSeenVisible') setLastSeenVisible(value)
   334→    if (key === 'readReceiptsEnabled') setReadReceipts(value)
   335→    if (key === 'typingIndicatorsEnabled') setTypingIndicators(value)
   336→    update.mutate({ [key]: value })
   337→  }
   338→
   339→  return (
   340→    <GlassSurface blur={16} opacity={0.03} className="overflow-hidden">
   341→      <div className="divide-y divide-white/5">
   342→        <PrivacyRow
   343→          title="Last seen visible"
   344→          desc="Show when you were last active"
   345→          checked={lastSeenVisible}
   346→          onChange={(v) => handleToggle('lastSeenVisible', v)}
   347→        />
   348→        <PrivacyRow
   349→          title="Read receipts"
   350→          desc="Let others know when you've read their messages"
   351→          checked={readReceipts}
   352→          onChange={(v) => handleToggle('readReceiptsEnabled', v)}
   353→        />
   354→        <PrivacyRow
   355→          title="Typing indicators"
   356→          desc="Show others when you're typing"
   357→          checked={typingIndicators}
   358→          onChange={(v) => handleToggle('typingIndicatorsEnabled', v)}
   359→        />
   360→      </div>
   361→    </GlassSurface>
   362→  )
   363→}

   392→function BotsList({ onEditBot }: { onEditBot: (bot: any) => void }) {

   396→  const [username, setUsername] = useState('')
   397→  const [description, setDescription] = useState('')
   398→
   399→  const handleCreate = async () => {
   400→    try {
   401→      await create({ name, username, description, module: 'visual' })
   402→      toast.success('Bot created! Edit it to design your flow.')
   403→      setCreateOpen(false)
   404→      setName('')
   405→      setUsername('')
   406→      setDescription('')
   407→    } catch (e: any) {
   408→      toast.error(e.message)
   409→    }
   410→  }
   411→
   412→  return (
   413→    <Card className="p-4">
   414→      <div className="flex items-center justify-between mb-3">
   415→        <h2 className="font-semibold text-sm">My Bots ({bots.length})</h2>
   416→        <Button size="sm" onClick={() => setCreateOpen(!createOpen)}>

   436→            Bots are created with the visual flow editor. After creating, click "Edit Flow" to design your bot's behavior.

   454→            Create a bot and design its behavior visually with the flow editor.
   455→          </p>
   456→        </div>

   458→        <div className="space-y-2">

   471→                  </div>
   472→                </div>
   473→                <Button variant="outline" size="sm" onClick={() => onEditBot(bot)}>
   474→                  Edit Flow
   475→                </Button>
   476→                <Button variant="ghost" size="sm" onClick={() => update({ id: bot.id, data: { enabled: !bot.enabled } })}>
   477→                  {bot.enabled ? 'Disable' : 'Enable'}

   486→            </div>
   487→          ))}
   488→        </div>
   489→      )}
   490→    </Card>
   491→  )
   492→}
