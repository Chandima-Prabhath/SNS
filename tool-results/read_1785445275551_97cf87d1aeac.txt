    48→function loadHistory(): Track[] {
    49→  try {
    50→    const raw = localStorage.getItem(HISTORY_KEY)
    51→    return raw ? JSON.parse(raw) : []
    52→  } catch { return [] }
    53→}
    54→
    55→function saveHistory(tracks: Track[]) {
    56→  try {
    57→    localStorage.setItem(HISTORY_KEY, JSON.stringify(tracks.slice(0, MAX_HISTORY)))
    58→  } catch {}
    59→}

   155→  // ── Liked songs helpers (DB-backed) ──────────────────────────────────
   156→  const toggleLike = async (track: Track) => {
   157→    const isLikedVal = liked.some((t) => t.videoId === track.videoId)
   158→    // Optimistic update — update the cache immediately
   159→    qc.setQueryData(['music-liked'], (old: any) => {
   160→      if (!old) return { songs: [track] }
   161→      const songs = isLikedVal
   162→        ? old.songs.filter((t: Track) => t.videoId !== track.videoId)
   163→        : [track, ...old.songs]
   164→      return { ...old, songs }
   165→    })

   167→      if (isLikedVal) {
   168→        await fetch(`/api/music/liked?videoId=${track.videoId}`, { method: 'DELETE' })
   169→      } else {
   170→        await fetch('/api/music/liked', {
   171→          method: 'POST',
   172→          headers: { 'Content-Type': 'application/json' },
   173→          body: JSON.stringify(track),
   174→        })
   175→      }
   176→    } catch {
   177→      // Revert on error
   178→      qc.invalidateQueries({ queryKey: ['music-liked'] })
   179→      toast.error('Failed to update liked songs')
   180→    }
   181→  }
   182→
   183→  const isLiked = (videoId: string) => liked.some((t) => t.videoId === videoId)
   184→
   185→  // ── Playlist helpers (DB-backed, with optimistic updates) ────────────
   186→  const createPlaylist = async (name: string) => {
   187→    if (!name.trim()) return
   188→    try {
   189→      const res = await fetch('/api/music/playlists', {
   190→        method: 'POST',
   191→        headers: { 'Content-Type': 'application/json' },
   192→        body: JSON.stringify({ name }),
   193→      })
   194→      if (!res.ok) throw new Error('failed')
   195→      const data = await res.json()
   196→      // Add the new playlist to the cache immediately (optimistic)
   197→      qc.setQueryData(['music-playlists'], (old: any) => {
   198→        const newPlaylist = { ...data.playlist, songs: [] }
   199→        return { playlists: [newPlaylist, ...(old?.playlists || [])] }
   200→      })
   201→      toast.success(`Playlist "${name}" created`)
   202→      setNewPlaylistName('')
   203→      setShowNewPlaylistInput(false)
   204→    } catch {
   205→      toast.error('Failed to create playlist')
   206→    }
   207→  }
   208→
   209→  const addToPlaylist = async (playlistId: string, track: Track) => {
   210→    // Optimistic update — add the song to the playlist in the cache
   211→    qc.setQueryData(['music-playlists'], (old: any) => {
   212→      if (!old) return old
   213→      return {
   214→        ...old,
   215→        playlists: old.playlists.map((pl: any) =>
   216→          pl.id === playlistId
   217→            ? {
   218→                ...pl,
   219→                songs: pl.songs.some((s: Track) => s.videoId === track.videoId)
   220→                  ? pl.songs // already in playlist — don't add duplicate
   221→                  : [...pl.songs, track],
   222→              }
   223→            : pl
   224→        ),
   225→      }
   226→    })
   227→    try {
   228→      const res = await fetch(`/api/music/playlists/${playlistId}/songs`, {
   229→        method: 'POST',
   230→        headers: { 'Content-Type': 'application/json' },
   231→        body: JSON.stringify(track),
   232→      })
   233→      if (!res.ok) throw new Error('failed')
   234→      const pl = playlists.find((p) => p.id === playlistId)
   235→      toast.success(`Added to "${pl?.name}"`)
   236→    } catch {
   237→      // Revert on error
   238→      qc.invalidateQueries({ queryKey: ['music-playlists'] })
   239→      toast.error('Failed to add to playlist')
   240→    }
   241→  }
   242→
   243→  const deletePlaylist = async (playlistId: string) => {
   244→    // Optimistic update — remove from cache immediately
   245→    qc.setQueryData(['music-playlists'], (old: any) => {
   246→      if (!old) return old
   247→      return {
   248→        ...old,
   249→        playlists: old.playlists.filter((pl: any) => pl.id !== playlistId),
   250→      }
   251→    })
   252→    try {
   253→      const res = await fetch(`/api/music/playlists/${playlistId}`, { method: 'DELETE' })
   254→      if (!res.ok) throw new Error('failed')
   255→      toast.success('Playlist deleted')
   256→    } catch {
   257→      // Revert on error
   258→      qc.invalidateQueries({ queryKey: ['music-playlists'] })
   259→      toast.error('Failed to delete playlist')
   260→    }
   261→  }
   262→
   263→  // Track when the current track changes — add to history
   264→  useEffect(() => {
   265→    if (currentTrack) {
   266→      setHistory((prev) => {
   267→        const filtered = prev.filter((t) => t.videoId !== currentTrack.videoId)
   268→        const updated = [currentTrack, ...filtered].slice(0, MAX_HISTORY)
   269→        saveHistory(updated)
   270→        return updated
   271→      })
   272→    }
   273→  }, [currentTrack])
   274→
   275→  // Fetch trending
   276→  const { data: trendingData, isLoading: trendingLoading } = useQuery({
   277→    queryKey: ['music-trending'],

   279→      const res = await fetch('/api/music/trending')

   282→    },
   283→  })
   284→
   285→  // Fetch personalized home feed ("For You" recommendations).
   286→  // The endpoint returns { sections: [{ title, tracks }] } — empty array on

   289→    queryKey: ['music-home'],
   290→    queryFn: async () => {
   291→      const res = await fetch('/api/music/home')
   292→      if (!res.ok) throw new Error('failed')
   293→      return res.json()
   294→    },
   295→  })
   296→
   297→  // Fetch rooms
   298→  const { data: roomsData } = useQuery({
   299→    queryKey: ['music-rooms'],
   300→    queryFn: async () => {
   301→      const res = await fetch('/api/music/rooms')

   303→      return res.json()

   305→    refetchInterval: 10000,
   306→  })
   307→

   311→      const res = await fetch('/api/music/rooms', {
   312→        method: 'POST',
   313→        headers: { 'Content-Type': 'application/json' },
   314→        body: JSON.stringify({ name }),

   316→      if (!res.ok) throw new Error('failed')
   317→      return res.json()

   323→      toast.success('Room created')

   325→  })

   328→  const deleteRoom = useMutation({
   329→    mutationFn: async (roomId: string) => {
   330→      const res = await fetch(`/api/music/rooms/${roomId}`, {
   331→        method: 'DELETE',
   332→      })
   333→      if (!res.ok) throw new Error('failed')
   334→    },
   335→    onSuccess: () => {
   336→      qc.invalidateQueries({ queryKey: ['music-rooms'] })
   337→      if (activeRoomId) setActiveRoomId(null)
   338→      toast.success('Room deleted')
   339→    },
   340→    onError: () => toast.error('Failed to delete room'),
   341→  })
   342→
   343→  // Debounced search with AbortController to cancel stale requests
   344→  const handleSearchChange = (value: string) => {
   345→    setSearchQuery(value)
   346→    if (searchTimeoutRef.current) clearTimeout(searchTimeoutRef.current)
   347→    // Abort any in-flight search request
   348→    if (searchAbortRef.current) {
   349→      searchAbortRef.current.abort()
   350→      searchAbortRef.current = null
   351→    }
   352→    if (!value.trim()) {
   353→      setSearchResults([])
   354→      setSearching(false)
   355→      return
   356→    }
   357→    searchTimeoutRef.current = setTimeout(async () => {
   358→      setSearching(true)
   359→      const controller = new AbortController()
   360→      searchAbortRef.current = controller
   361→      try {
   362→        const res = await fetch(`/api/music/search?q=${encodeURIComponent(value)}`, {
   363→          signal: controller.signal,
   364→        })
   365→        if (!res.ok) throw new Error('Search failed')
   366→        const data = await res.json()
   367→        // Only update if this request wasn't superseded
   368→        if (searchAbortRef.current === controller) {
   369→          setSearchResults(data.tracks || [])
   370→        }
   371→      } catch (e: any) {
   372→        if (e.name === 'AbortError') return // expected — superseded by newer search
   373→        toast.error(e.message || 'Search failed')
   374→      } finally {
   375→        if (searchAbortRef.current === controller) {
   376→          setSearching(false)
   377→          searchAbortRef.current = null
   378→        }
   379→      }
   380→    }, 500)
   381→  }
   382→
   383→  const trendingTracks = trendingData?.tracks || []
   384→  const homeSections: { title: string; tracks: Track[] }[] = homeData?.sections || []
   385→  const rooms = roomsData?.rooms || []

   412→                      ? 'gradient-primary text-primary-foreground shadow-glow'
   413→                      : 'text-muted-foreground hover:text-foreground hover:bg-white/5',
   414→                  )}
   415→                >
   416→                  <Icon className="w-3.5 h-3.5" />
   417→                  {label}
   418→                  {key === 'queue' && queue.length > 0 && (
   419→                    <span className="text-[10px] bg-primary-foreground/20 text-primary-foreground px-1.5 py-0.5 rounded-full">
   420→                      {queue.length}
   421→                    </span>
   422→                  )}
   423→                </button>
   424→              ))}
   425→            </div>
   426→          </GlassSurface>
   427→        </div>

   429→
   430→      {/* Main content area — extra bottom padding so content isn't hidden
   431→          behind the persistent global player bar. */}
   432→      <div className="flex-1 overflow-y-auto pb-40 lg:pb-28">
   433→        <div className="max-w-5xl mx-auto p-4 md:p-6">
   434→          <AnimatePresence mode="wait">
   435→            {tab === 'browse' && (
   436→              <motion.div
   437→                key="browse"
   438→                initial={{ opacity: 0, y: 10 }}
   439→                animate={{ opacity: 1, y: 0 }}
   440→                exit={{ opacity: 0, y: -10 }}
   441→                transition={{ duration: 0.2 }}
   442→                className="space-y-6"
   443→              >
   444→                {/* Search */}
   445→                <div className="relative group">
   446→                  <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-muted-foreground group-focus-within:text-primary transition-colors" />
   447→                  <Input
   448→                    value={searchQuery}
   449→                    onChange={(e) => handleSearchChange(e.target.value)}
   450→                    placeholder="Search songs, artists..."
   451→                    className="pl-12 h-14 bg-black/20 backdrop-blur-xl border-white/10 rounded-2xl text-lg focus-visible:ring-primary/50 focus-visible:border-primary/50 transition-all shadow-inner"
   452→                  />
   453→                  {searching && (
   454→                    <Loader2 className="absolute right-4 top-1/2 -translate-y-1/2 w-5 h-5 animate-spin text-muted-foreground" />
   455→                  )}
   456→                </div>
   457→
   458→                {/* Search results */}
   459→                {searchResults.length > 0 ? (
   460→                  <div className="space-y-1.5">
   461→                    <h2 className="text-sm font-bold uppercase tracking-wider mb-3">
   462→                      <ShinyText shimmerDuration={4} className="text-sm">
   463→                        Results
   464→                      </ShinyText>
   465→                    </h2>
   466→                    {searchResults.map((track) => (
   467→                      <TrackRow
   468→                        key={track.videoId}
   469→                        track={track}
   470→                        onPlay={() => playTrack(track)}
   471→                        onAddToQueue={() => playTrack(track, true)}
   472→                        isCurrent={currentTrack?.videoId === track.videoId}
   473→                        isPlaying={
   474→                          isPlaying && currentTrack?.videoId === track.videoId
   475→                        }
   476→                        isLiked={isLiked(track.videoId)}
   477→                        onToggleLike={() => toggleLike(track)}
   478→                        onAddToPlaylist={() => setAddToPlaylistTrack(track)}
   479→                      />
   480→                    ))}
   481→                  </div>
   482→                ) : (
   483→                  <>
   484→                    {/* For You — personalized recommendation sections.
   485→                        Each section is rendered as a horizontal scrolling row
   486→                        so the user can swipe through tracks without leaving
   487→                        the browse tab. Hidden entirely when the home feed is
   488→                        unavailable (empty sections array). */}
   489→                    {homeLoading && (
   490→                      <div className="space-y-3">
   491→                        <div className="h-4 w-40 bg-muted rounded animate-pulse" />
   492→                        <div className="flex gap-3 overflow-hidden">
   493→                          {[1, 2, 3, 4].map((i) => (
   494→                            <div
   495→                              key={i}
   496→                              className="shrink-0 w-40 md:w-44 space-y-2 animate-pulse"
   497→                            >
   498→                              <div className="aspect-square rounded-xl bg-muted" />
   499→                              <div className="h-3 bg-muted rounded w-3/4" />
   500→                              <div className="h-2 bg-muted rounded w-1/2" />
   501→                            </div>
   502→                          ))}
   503→                        </div>
   504→                      </div>
   505→                    )}
   506→
   507→                    {!homeLoading &&
   508→                      homeSections.map((section) => (
   509→                        <div key={section.title} className="space-y-3">
   510→                          <div className="flex items-center gap-2">
   511→                            <Sparkles className="w-4 h-4 text-primary" />
   512→                            <h2 className="text-sm font-bold uppercase tracking-wider">
   513→                              <ShinyText shimmerDuration={5} className="text-sm">
   514→                                {section.title}
   515→                              </ShinyText>
   516→                            </h2>
   517→                          </div>
   518→                          <div className="flex gap-3 overflow-x-auto pb-3 -mx-1 px-1 snap-x no-scrollbar">
   519→                            {section.tracks.map((track: Track) => (

   526→                                  onPlay={() => playTrack(track)}
   527→                                  onAddToQueue={() => playTrack(track, true)}
   528→                                  isCurrent={
   529→                                    currentTrack?.videoId === track.videoId
   530→                                  }
   531→                                  isPlaying={
   532→                                    isPlaying &&
   533→                                    currentTrack?.videoId === track.videoId
   534→                                  }
   535→                                />
   536→                              </div>
   537→                            ))}
   538→                          </div>
   539→                        </div>
   540→                      ))}
   541→
