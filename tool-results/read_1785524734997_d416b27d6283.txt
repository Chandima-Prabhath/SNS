    64→// ─── Custom Node Component ───────────────────────────────────────────────────
    65→/**
    66→ * Reads the FlowNode type from `data.type` (NOT the ReactFlow `type` prop,
    67→ * which is always 'custom' in this app). This was the root cause of every
    68→ * node rendering as a green "Send Message" node.
    69→ */
    70→function CustomNode({ data, selected }: { data: any; selected?: boolean }) {
    71→  const nodeType = (data.type as NodeType) || 'message'
    72→  const def = NODE_DEFS[nodeType] || NODE_DEFS.message
    73→  const Icon = ICONS[def.icon] || Zap
    74→
    75→  const isTrigger = nodeType === 'trigger'
    76→  const isStop = nodeType === 'stop'
    77→  const isCondition = nodeType === 'condition'
    78→  const isRandom = nodeType === 'random'
    79→  const isSwitch = nodeType === 'switch_case'
    80→
    81→  // ── Build preview content per node type ──────────────────────────────
    82→  let preview: { label: string; value: string } | null = null
    83→  switch (nodeType) {
    84→    case 'trigger': {
    85→      const tt = data.triggerType || 'any_message'
    86→      const label =
    87→        tt === 'command' ? `/${data.command || 'command'}` :
    88→        tt === 'mention' ? '@mentioned' :
    89→        'any message'
    90→      preview = { label: 'When', value: label }
    91→      break
    92→    }
    93→    case 'message':
    94→      preview = { label: 'Says', value: data.text || '(empty)' }
    95→      break
    96→    case 'typing':
    97→      preview = { label: 'For', value: `${data.seconds || 1}s` }
    98→      break
    99→    case 'input':
   100→      preview = { label: 'Asks', value: data.prompt || '(empty)' }
   101→      break
   102→    case 'wait_choice':
   103→      preview = { label: 'Buttons', value: (data.options || []).join(' · ') || '(none)' }
   104→      break
   105→    case 'condition':
   106→      preview = { label: 'If', value: `${data.variable || '∅'} ${data.operator || 'exists'} ${data.value || ''}` }
   107→      break
   108→    case 'set_var':
   109→      preview = { label: 'Sets', value: `${data.variable || '∅'} = ${data.value || '∅'}` }
   110→      break
   111→    case 'delay':
   112→      preview = { label: 'Wait', value: `${data.seconds || 1}s` }
   113→      break
   114→    case 'stop':
   115→      preview = null
   116→      break
   117→    case 'api_call':
   118→      preview = { label: 'Calls', value: `${data.method || 'GET'} ${data.url || '(no url)'}` }
   119→      break
   120→    case 'random':
   121→      preview = { label: 'Picks', value: 'a random branch' }
   122→      break
   123→    case 'ai_generate':
   124→      preview = { label: 'Asks AI', value: data.aiPrompt ? (data.aiPrompt.length > 40 ? data.aiPrompt.slice(0, 40) + '…' : data.aiPrompt) : '(no prompt)' }
   125→      break
   126→    case 'send_media':
   127→      preview = { label: 'Sends', value: data.mediaUrl || '(no url)' }
   128→      break
   129→    case 'switch_case':
   130→      preview = { label: 'Switch on', value: data.switchVariable || '(no var)' }
   131→      break
   132→    case 'counter':
   133→      preview = { label: 'Incr', value: `${data.variable || 'count'} by ${data.increment ?? 1}` }
   134→      break
   135→    case 'format_string':
   136→      preview = { label: 'Formats', value: data.text || '(no template)' }
   137→      break
   138→    case 'log':
   139→      preview = { label: 'Logs', value: data.logMessage ? (data.logMessage.length > 40 ? data.logMessage.slice(0, 40) + '…' : data.logMessage) : '(no message)' }
   140→      break
   141→    case 'tts':
   142→      preview = { label: 'Speaks', value: data.ttsText ? (data.ttsText.length > 40 ? data.ttsText.slice(0, 40) + '…' : data.ttsText) : '(no text)' }
   143→      break
   144→    case 'asr_transcribe':
   145→      preview = { label: 'Transcribes', value: data.asrAudioUrl ? (data.asrAudioUrl.length > 40 ? data.asrAudioUrl.slice(0, 40) + '…' : data.asrAudioUrl) : '{{mediaUrl}}' }
   146→      break
   147→  }
   148→
   149→  return (
   150→    <div
   151→      className="bg-[#2b2d31] rounded-xl min-w-[200px] max-w-[260px] shadow-lg border overflow-hidden transition-shadow"
   152→      style={{
   153→        borderColor: selected ? def.color : 'rgba(255,255,255,0.08)',
   154→        boxShadow: selected ? `0 0 0 2px ${def.color}40` : undefined,
   155→      }}
   156→    >
   157→      {/* Target handle (not for trigger — it has no incoming edges) */}
   158→      {!isTrigger && (
   159→        <Handle
   160→          type="target"
   161→          position={Position.Top}
   162→          className="adoo-handle"
   163→          style={{ background: def.color, width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   164→        />
   165→      )}
   166→
   167→      {/* Header */}
   168→      <div
   169→        className="flex items-center gap-2 px-3 py-2 border-b"
   170→        style={{ background: def.bg, borderColor: `${def.color}30` }}
   171→      >
   172→        <div
   173→          className="w-6 h-6 rounded-md flex items-center justify-center shrink-0"
   174→          style={{ background: def.color + '30' }}
   175→        >
   176→          <Icon className="w-3.5 h-3.5" style={{ color: def.color }} />
   177→        </div>
   178→        <span
   179→          className="text-[11px] font-bold uppercase tracking-wider flex-1"
   180→          style={{ color: def.color }}
   181→        >
   182→          {def.label}
   183→        </span>
   184→        {def.pauses && (
   185→          <span className="text-[9px] uppercase tracking-wider px-1.5 py-0.5 rounded bg-white/10 text-white/60">
   186→            pause
   187→          </span>
   188→        )}
   189→        {def.terminates && (
   190→          <span className="text-[9px] uppercase tracking-wider px-1.5 py-0.5 rounded bg-white/10 text-white/60">
   191→            end
   192→          </span>
   193→        )}
   194→      </div>
   195→
   196→      {/* Body */}
   197→      <div className="px-3 py-2 text-xs">
   198→        {preview ? (
   199→          <div className="space-y-1">
   200→            <div className="text-white/40 uppercase tracking-wider text-[10px]">{preview.label}</div>
   201→            <div className="text-white/80 break-words line-clamp-2">{preview.value}</div>
   202→            {(nodeType === 'input' || nodeType === 'wait_choice' || nodeType === 'api_call') && data.variableName && (
   203→              <div className="text-[10px] text-white/40">→ saves as <code className="text-white/60">{`{{${data.variableName}}}`}</code></div>
   204→            )}
   205→          </div>
   206→        ) : (
   207→          <div className="text-white/50 italic text-center py-1">{def.description}</div>
   208→        )}
   209→      </div>
   210→
   211→      {/* Source handles */}
   212→      {isCondition ? (
   213→        <div className="flex justify-between px-3 pb-2 pt-1">
   214→          <div className="flex flex-col items-center">
   215→            <span className="text-[10px] font-semibold text-emerald-400 mb-1">TRUE</span>
   216→            <Handle
   217→              type="source"
   218→              position={Position.Bottom}
   219→              id="true"
   220→              className="adoo-handle"
   221→              style={{ background: '#34D399', width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   222→            />
   223→          </div>
   224→          <div className="flex flex-col items-center">
   225→            <span className="text-[10px] font-semibold text-red-400 mb-1">FALSE</span>
   226→            <Handle
   227→              type="source"
   228→              position={Position.Bottom}
   229→              id="false"
   230→              className="adoo-handle"
   231→              style={{ background: '#F87171', width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   232→            />
   233→          </div>
   234→        </div>
   235→      ) : isSwitch ? (
   236→        <div className="px-3 pb-2 pt-1 space-y-1.5">
   237→          {((data.cases as string[]) || []).map((c, i) => (
   238→            <div key={i} className="flex items-center gap-2">
   239→              <span className="text-[10px] font-mono text-white/40 shrink-0 w-10">case_{i}</span>
   240→              <span className="text-[10px] text-white/70 truncate flex-1 px-1.5 py-0.5 rounded bg-white/5">{c || `(empty)`}</span>
   241→              <Handle
   242→                type="source"
   243→                position={Position.Right}
   244→                id={`case_${i}`}
   245→                className="adoo-handle"
   246→                style={{ background: '#FB7185', width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%', position: 'relative' }}
   247→              />
   248→            </div>
   249→          ))}
   250→          <div className="flex items-center gap-2 pt-1.5 border-t border-white/10">
   251→            <span className="text-[10px] font-mono text-white/40 shrink-0 w-10">default</span>
   252→            <span className="text-[10px] text-white/40 italic flex-1 px-1.5 py-0.5 rounded bg-white/5">fallback</span>
   253→            <Handle
   254→              type="source"
   255→              position={Position.Right}
   256→              id="default"
   257→              className="adoo-handle"
   258→              style={{ background: '#64748B', width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%', position: 'relative' }}
   259→            />
   260→          </div>
   261→        </div>
   262→      ) : isRandom ? (
   263→        <div className="px-3 pb-2 pt-1 text-center">
   264→          <span className="text-[10px] text-white/40 mb-1 block">All outgoing edges are random picks</span>
   265→          <Handle
   266→            type="source"
   267→            position={Position.Bottom}
   268→            className="adoo-handle"
   269→            style={{ background: def.color, width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   270→          />
   271→        </div>
   272→      ) : !isStop ? (
   273→        <Handle
   274→          type="source"
   275→          position={Position.Bottom}
   276→          className="adoo-handle"
   277→          style={{ background: def.color, width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   278→        />
   279→      ) : null}
   280→    </div>
   281→  )
   282→}

   294→export function BotBuilderEditor({ initialFlow, onSave, bot }: BotBuilderEditorProps) {
   295→  const [selectedNodeId, setSelectedNodeId] = useState<string | null>(null)
   296→  const [selectedEdgeId, setSelectedEdgeId] = useState<string | null>(null)
   297→  const [rightTab, setRightTab] = useState<'inspector' | 'issues' | 'debug'>('inspector')
   298→  const [showShortcuts, setShowShortcuts] = useState(false)
   299→  const confirm = useConfirm()
   300→
   301→  // ── Initial nodes/edges ──────────────────────────────────────────────
   302→  // FIX: previously the .map() overwrote data.type with ReactFlow's type
   303→  // ('custom'), so the inspector crashed on click. Now we use data.type
   304→  // everywhere and never let n.type leak into data.
   305→  const { initialNodes, initialEdges } = useMemo(() => {
   306→    if (initialFlow?.nodes?.length) {
   307→      return {
   308→        initialNodes: initialFlow.nodes.map((n) => ({
   309→          id: n.id,
   310→          type: 'custom',
   311→          position: n.position,
   312→          data: { ...n.data },
   313→        })) as Node[],
   314→        initialEdges: initialFlow.edges.map((e) => ({
   315→          id: e.id,
   316→          source: e.source,
   317→          target: e.target,
   318→          sourceHandle: e.sourceHandle || undefined,
   319→          label: e.label || undefined,
   320→          animated: true,
   321→          style: { stroke: '#5865F2', strokeWidth: 2 },
   322→          markerEnd: { type: MarkerType.ArrowClosed, color: '#5865F2' },
   323→        })) as Edge[],
   324→      }
   325→    }
   326→    // Brand-new bot: seed with a sensible default flow
   327→    const triggerData = defaultNodeData('trigger')
   328→    const msgData = defaultNodeData('message')
   329→    return {
   330→      initialNodes: [
   331→        { id: 'trigger-1', type: 'custom', position: { x: 320, y: 80 }, data: { ...triggerData, type: 'trigger' } },
   332→        { id: 'message-1', type: 'custom', position: { x: 320, y: 240 }, data: { ...msgData, type: 'message' } },
   333→      ] as Node[],
   334→      initialEdges: [
   335→        {
   336→          id: 'e-trigger-1-message-1',
   337→          source: 'trigger-1',
   338→          target: 'message-1',
   339→          animated: true,
   340→          style: { stroke: '#5865F2', strokeWidth: 2 },
   341→          markerEnd: { type: MarkerType.ArrowClosed, color: '#5865F2' },
   342→        },
   343→      ] as Edge[],
   344→    }
   345→  }, [initialFlow])
   346→
   347→  const [nodes, setNodes, onNodesChange] = useNodesState(initialNodes)
   348→  const [edges, setEdges, onEdgesChange] = useEdgesState(initialEdges)
   349→
   350→  const onConnect = useCallback(
   351→    (params: Connection) =>
   352→      setEdges((eds) =>
   353→        addEdge(
   354→          {
   355→            ...params,
   356→            animated: true,
   357→            style: { stroke: '#5865F2', strokeWidth: 2 },
   358→            markerEnd: { type: MarkerType.ArrowClosed, color: '#5865F2' },
   359→          },
   360→          eds
   361→        )
   362→      ),
   363→    [setEdges]
   364→  )
   365→
   366→  // ── Add node — position near the canvas center, offset by count ──────
   367→  const addNode = (type: NodeType) => {
   368→    const id = `${type}-${Date.now()}`
   369→    const data = { ...defaultNodeData(type), type }
   370→    // Place near the top-center with a slight cascade so it's always visible
   371→    const idx = nodes.length
   372→    const position = {
   373→      x: 240 + (idx % 4) * 90,
   374→      y: 80 + idx * 30,
   375→    }
   376→    setNodes((nds) => [...nds, { id, type: 'custom', position, data }])
   377→    setSelectedNodeId(id)
   378→  }
   379→
   380→  const updateNodeData = (nodeId: string, patch: Record<string, any>) =>
   381→    setNodes((nds) =>
   382→      nds.map((n) => (n.id === nodeId ? { ...n, data: { ...n.data, ...patch } } : n))
   383→    )
   384→
   385→  const deleteNode = (nodeId: string) => {
   386→    setNodes((nds) => nds.filter((n) => n.id !== nodeId))
   387→    setEdges((eds) => eds.filter((e) => e.source !== nodeId && e.target !== nodeId))
   388→    setSelectedNodeId(null)
   389→  }
   390→
   391→  // Disconnect an edge — used by the edge context menu and the Delete key.
   392→  const deleteEdge = (edgeId: string) => {
   393→    setEdges((eds) => eds.filter((e) => e.id !== edgeId))
   394→    setSelectedEdgeId(null)
   395→  }
   396→
   397→  // ── Clipboard for copy/paste ─────────────────────────────────────────
   398→  const clipboardRef = useRef<{ nodes: Node[]; edges: Edge[] } | null>(null)
   399→
   400→  // Get all currently selected nodes (supports multi-select)
   401→  const selectedNodes = useMemo(() => nodes.filter((n) => n.selected), [nodes])
   402→  const selectedEdges = useMemo(
   403→    () => edges.filter((e) => e.selected || (selectedNodeId && (e.source === selectedNodeId || e.target === selectedNodeId))),
   404→    [edges, selectedNodeId]
   405→  )
   406→
   407→  // Duplicate selected nodes (Ctrl+D)
   408→  const duplicateSelected = useCallback(() => {
   409→    if (selectedNodes.length === 0) return
   410→    const newNodes: Node[] = []
   411→    const newEdges: Edge[] = []
   412→    const idMap = new Map<string, string>()
   413→
   414→    // Create copies of each selected node, offset by 40px
   415→    for (const n of selectedNodes) {
   416→      const newId = `${(n.data as any)?.type || 'node'}-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`
   417→      idMap.set(n.id, newId)
   418→      newNodes.push({
   419→        ...n,
   420→        id: newId,
   421→        position: { x: n.position.x + 40, y: n.position.y + 40 },
   422→        selected: true,
   423→        data: { ...n.data },
   424→      })
   425→    }
   426→
   427→    // Copy edges between duplicated nodes
   428→    for (const e of edges) {
   429→      const newSource = idMap.get(e.source)
   430→      const newTarget = idMap.get(e.target)
   431→      if (newSource && newTarget) {
   432→        newEdges.push({
   433→          ...e,
   434→          id: `e-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
   435→          source: newSource,
   436→          target: newTarget,
   437→          selected: false,
   438→        })
   439→      }
   440→    }
   441→
   442→    // Deselect original nodes, add the copies
   443→    setNodes((nds) => [...nds.map((n) => ({ ...n, selected: false })), ...newNodes])
   444→    setEdges((eds) => [...eds, ...newEdges])
   445→  }, [selectedNodes, edges, setNodes, setEdges])
   446→
   447→  // Copy selected nodes to clipboard (Ctrl+C)
   448→  const copySelected = useCallback(() => {
   449→    if (selectedNodes.length === 0) return
   450→    const connectedEdges = edges.filter((e) =>
   451→      selectedNodes.some((n) => n.id === e.source) &&
   452→      selectedNodes.some((n) => n.id === e.target)
   453→    )
   454→    clipboardRef.current = { nodes: selectedNodes, edges: connectedEdges }
   455→    toast.success(`Copied ${selectedNodes.length} node${selectedNodes.length > 1 ? 's' : ''}`)
   456→  }, [selectedNodes, edges])
   457→
   458→  // Paste from clipboard (Ctrl+V)
   459→  const paste = useCallback(() => {
   460→    if (!clipboardRef.current || clipboardRef.current.nodes.length === 0) return
   461→    const { nodes: clipNodes, edges: clipEdges } = clipboardRef.current
   462→    const idMap = new Map<string, string>()
   463→    const newNodes: Node[] = []
   464→    const newEdges: Edge[] = []
   465→
   466→    for (const n of clipNodes) {
   467→      const newId = `${(n.data as any)?.type || 'node'}-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`
   468→      idMap.set(n.id, newId)
   469→      newNodes.push({
   470→        ...n,
   471→        id: newId,
   472→        position: { x: n.position.x + 60, y: n.position.y + 60 },
   473→        selected: true,
   474→        data: { ...n.data },
   475→      })
   476→    }
   477→
   478→    for (const e of clipEdges) {
   479→      const newSource = idMap.get(e.source)
   480→      const newTarget = idMap.get(e.target)
   481→      if (newSource && newTarget) {
   482→        newEdges.push({
   483→          ...e,
   484→          id: `e-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
   485→          source: newSource,
   486→          target: newTarget,
   487→          selected: false,
   488→        })
   489→      }
   490→    }
   491→
   492→    setNodes((nds) => [...nds.map((n) => ({ ...n, selected: false })), ...newNodes])
   493→    setEdges((eds) => [...eds, ...newEdges])
   494→    toast.success(`Pasted ${newNodes.length} node${newNodes.length > 1 ? 's' : ''}`)
   495→  }, [clipboardRef, setNodes, setEdges])
   496→
   497→  // Select all nodes (Ctrl+A)
   498→  const selectAll = useCallback(() => {
   499→    setNodes((nds) => nds.map((n) => ({ ...n, selected: true })))
   500→  }, [setNodes])
   501→
   502→  // Delete all selected nodes (handles multi-select Delete)
   503→  const deleteSelected = useCallback(() => {
   504→    const toDelete = nodes.filter((n) => n.selected).map((n) => n.id)
   505→    if (toDelete.length === 0) {
   506→      if (selectedNodeId) deleteNode(selectedNodeId)
   507→      else if (selectedEdgeId) deleteEdge(selectedEdgeId)
   508→      return
   509→    }
   510→    setNodes((nds) => nds.filter((n) => !n.selected))

   522→      const mod = e.metaKey || e.ctrlKey

   524→      // Ctrl/Cmd+C — copy

   533→      }
   534→
   535→      // Ctrl/Cmd+V — paste
   536→      if (mod && e.key === 'v' && !e.shiftKey) {
   537→        e.preventDefault()
   538→        paste()
   539→        return
   540→      }
   541→
   542→      // Ctrl/Cmd+D — duplicate
   543→      if (mod && e.key === 'd') {
   544→        e.preventDefault()
   545→        duplicateSelected()
   546→        return
   547→      }
   548→
   549→      // Ctrl/Cmd+A — select all
   550→      if (mod && e.key === 'a') {
   551→        e.preventDefault()
   552→        selectAll()
   553→        return
   554→      }
   555→
   556→      // Delete/Backspace — delete selected
   557→      if (e.key === 'Delete' || e.key === 'Backspace') {
   558→        e.preventDefault()
   559→        deleteSelected()
   560→        return
   561→      }
   562→
   563→      // Escape — deselect all

   565→        setNodes((nds) => nds.map((n) => ({ ...n, selected: false })))
   566→        setSelectedNodeId(null)
   567→        setSelectedEdgeId(null)
   568→        return
   569→      }
