   484→/**
   485→ * Format a timestamp for the chat list row.
   486→ *   - Today       → "HH:mm"   (e.g. "14:23")
   487→ *   - Yesterday   → "Yesterday"
   488→ *   - This year   → "MMM d"   (e.g. "Mar 5")
   489→ *   - Older       → "MMM d, yyyy"
   490→ *
   491→ * Comparison is by calendar date (via date-fns `isToday`/`isYesterday` and
   492→ * `getYear`), NOT raw epoch millis — so a message sent at 23:59 and viewed at
   493→ * 00:01 correctly shows "Yesterday" instead of "00:01 today".
   494→ */
   495→function formatListTimestamp(dateStr: string | Date): string {
   496→  const d = new Date(dateStr)
   497→  if (isNaN(d.getTime())) return ''
   498→  if (isToday(d)) return format(d, 'HH:mm')
   499→  if (isYesterday(d)) return 'Yesterday'
   500→  const now = new Date()
   501→  if (d.getFullYear() === now.getFullYear()) return format(d, 'MMM d')
   502→  return format(d, 'MMM d, yyyy')
   503→}

   505→/**
   506→ * Build the message preview text for a chat list row.
   507→ *
   508→ * Rules (matching common chat UX):
   509→ *   - Media messages show a type label ("Photo", "Video", "Voice message",
   510→ *     "File") — optionally with a caption if the message also has text.
   511→ *   - Group channels prefix with the sender's name: "Sarah: lol that was
   512→ *     wild". The current user's own messages show "You: ...".
   513→ *   - DM channels with a bot partner prefix the bot's name on bot messages.
   514→ *   - DM channels with another user show just the message body (the partner's
   515→ *     name is already in the row title).
   516→ *   - The result is truncated to ~40 chars to fit the row width.
   517→ */
   518→function getLastMessagePreview(
   519→  row: ChatRow,
   520→  myId: string | undefined
   521→): string {
   522→  const msg = row.channel.lastMessage
   523→  if (!msg) {
   524→    // Fall back to a sensible placeholder for empty channels
   525→    if (row.isGroup) return row.groupName
   526→    if (row.partner) return `@${row.partner.username}`
   527→    return 'No messages yet'
   528→  }
   529→
   530→  // Determine the media label (if any)
   531→  let mediaLabel = ''
   532→  if (msg.mediaType?.startsWith('image')) mediaLabel = 'Photo'
   533→  else if (msg.mediaType?.startsWith('video')) mediaLabel = 'Video'
   534→  else if (msg.mediaType?.startsWith('audio')) mediaLabel = 'Voice message'
   535→  else if (msg.mediaType === 'file') mediaLabel = 'File'
   536→  else if (msg.mediaUrl && !msg.body) mediaLabel = 'File' // unknown media with no text
   537→
   538→  const hasText = msg.body && msg.body.trim().length > 0
   539→  // Combine media label + text body (e.g. "Photo: check this out")
   540→  const textPart = hasText ? msg.body : ''
   541→  const combined = mediaLabel
   542→    ? hasText
   543→      ? `${mediaLabel}: ${textPart}`
   544→      : mediaLabel
   545→    : textPart
   546→
   547→  // Prefix with sender name where it adds context:
   548→  //   - Group channels: always prefix (sender could be anyone)
   549→  //   - DM with a bot partner: prefix bot name on bot messages (so the user
   550→  //     can tell bot replies apart from their own messages)
   551→  //   - DM with another user: no prefix needed (the row title is the partner)
   552→  const isMyMessage = myId && msg.senderId === myId && msg.senderType === 'user'
   553→
   554→  let prefixed: string
   555→  if (row.isGroup) {
   556→    const senderPrefix = isMyMessage ? 'You' : (msg.senderName || 'Someone')
   557→    prefixed = `${senderPrefix}: ${combined}`
   558→  } else if (msg.senderType === 'bot') {
   559→    // DM with a bot — prefix the bot's name on bot replies
   560→    prefixed = `${msg.senderName}: ${combined}`
   561→  } else if (isMyMessage) {
   562→    prefixed = `You: ${combined}`
   563→  } else {
   564→    prefixed = combined
   565→  }
   566→
   567→  // Truncate to ~40 chars for the row preview
   568→  return prefixed.length > 40 ? prefixed.slice(0, 39) + '…' : prefixed
   569→}

   584→/**
   585→ * A single text-channel row in the chat list.
   586→ *
   587→ * Extracted into its own component so each row can subscribe to its own
   588→ * typing state via `useTypingStore` without re-rendering every row on every
   589→ * typing pulse. Zustand's selector-based subscription means a row only
   590→ * re-renders when ITS channel's typing state changes.
   591→ */
   592→function ChatTextRow({
   593→  row,
   594→  active,
   595→  partnerDeleted,
   596→  presenceStatus,
   597→  unreadCount,
   598→  displayName,
   599→  myId,
   600→  onClick,
   601→  onContextMenu,
   602→  onTouchStart,
   603→}: ChatTextRowProps) {
   604→  // Subscribe to typing for THIS channel only. The selector returns a
   605→  // primitive boolean so the row only re-renders when typing starts/stops.
   606→  const isTyping = useTypingStore((s) => {
   607→    const m = s.typingByChannel[row.channel.id]
   608→    return !!m && Object.keys(m).length > 0
   609→  })
   610→
   611→  // When the channel is open, the in-chat typing indicator handles it —
   612→  // don't replace the preview here. Only show "typing..." for channels the
   613→  // user is NOT currently viewing.
   614→  const showTyping = isTyping && !active
   615→
   616→  const previewText = getLastMessagePreview(row, myId)
   617→  const timestamp = row.channel.lastMessage
   618→    ? formatListTimestamp(row.channel.lastMessage.createdAt)
   619→    : ''
   620→
   621→  return (
   622→    <button
   623→      onClick={onClick}
   624→      onContextMenu={onContextMenu}
   625→      onTouchStart={onTouchStart}
   626→      className={cn(
   627→        'w-full flex items-center gap-3.5 p-3 rounded-2xl transition-all text-left select-none relative overflow-hidden',
   628→        active
   629→          ? 'bg-primary/10 ring-1 ring-primary/20 shadow-[inset_0_1px_1px_rgba(255,255,255,0.05)]'
   630→          : 'hover:bg-white/[0.04]'
   631→      )}
   632→    >
   633→      {/* Avatar */}
   634→      {row.isGroup ? (
   635→        <div className="relative w-12 h-12 rounded-full bg-primary/15 flex items-center justify-center shrink-0">
   636→          <Hash className="w-5 h-5 text-primary" />
   637→        </div>
   638→      ) : partnerDeleted ? (
   639→        <div className="relative shrink-0">
   640→          <div className="w-12 h-12 rounded-full bg-muted flex items-center justify-center text-muted-foreground">
   641→            <UserX className="w-5 h-5" />
   642→          </div>
   643→        </div>
   644→      ) : (
   645→        <div className="relative shrink-0">
   646→          <Avatar className="w-12 h-12">
   647→            <AvatarImage src={row.partner?.avatarUrl || undefined} />
   648→            <AvatarFallback>
   649→              {row.partner?.displayName?.charAt(0) || row.channel.name.charAt(0).toUpperCase()}
   650→            </AvatarFallback>
   651→          </Avatar>
   652→          <span
   653→            className={cn(
   654→              'absolute bottom-0 right-0 w-3 h-3 rounded-full border-2 border-sidebar',
   655→              presenceStatus === 'online' && 'bg-status-online',
   656→              presenceStatus === 'idle' && 'bg-status-idle',
   657→              presenceStatus === 'dnd' && 'bg-status-dnd',
   658→              presenceStatus === 'offline' && 'bg-status-offline'
   659→            )}
   660→          />
   661→        </div>
   662→      )}
   663→
   664→      {/* Name + preview + timestamp */}
   665→      <div className="flex-1 min-w-0">
   666→        <div className="flex items-baseline justify-between gap-2">
   667→          <span
   668→            className={cn(
   669→              'truncate text-[15px]',
   670→              unreadCount > 0 ? 'font-semibold text-foreground' : 'font-medium',
   671→              partnerDeleted && 'text-muted-foreground italic'
   672→            )}
   673→          >
   674→            {displayName}
   675→          </span>
   676→          {timestamp && (
   677→            <span
   678→              className={cn(
   679→                'shrink-0 text-[11px] tabular-nums',
   680→                unreadCount > 0 ? 'text-primary font-medium' : 'text-muted-foreground'
   681→              )}
   682→            >
   683→              {timestamp}
   684→            </span>
   685→          )}
   686→        </div>
   687→        <div className="flex items-center justify-between gap-2 mt-0.5">
   688→          {showTyping ? (
   689→            <span className="text-xs truncate italic text-primary flex items-center gap-1 min-w-0">
   690→              <span className="flex gap-0.5 shrink-0">
   691→                <span className="w-1 h-1 rounded-full bg-current animate-bounce [animation-delay:0ms]" />
   692→                <span className="w-1 h-1 rounded-full bg-current animate-bounce [animation-delay:120ms]" />
   693→                <span className="w-1 h-1 rounded-full bg-current animate-bounce [animation-delay:240ms]" />
   694→              </span>
   695→              <span className="truncate">typing...</span>
   696→            </span>
   697→          ) : (
   698→            <span
   699→              className={cn(
   700→                'text-xs truncate flex-1 min-w-0',
   701→                unreadCount > 0 ? 'text-foreground/80 font-medium' : 'text-muted-foreground',
   702→                partnerDeleted && 'italic'
   703→              )}
   704→            >
   705→              {previewText}
   706→            </span>
   707→          )}
   708→          {unreadCount > 0 && !active && (
   709→            <span className="shrink-0 min-w-[20px] h-5 px-1.5 rounded-full bg-primary text-primary-foreground text-[10px] font-bold flex items-center justify-center">
   710→              {unreadCount > 99 ? '99+' : unreadCount}
   711→            </span>
   712→          )}
   713→        </div>
   714→      </div>
   715→    </button>
   716→  )

   733→function JoinGroupButton() {

   762→      <DialogTrigger asChild>
