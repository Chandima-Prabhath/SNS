   274→/**
   275→ * useVoiceRecorder — hook that manages MediaRecorder state for recording voice
   276→ * messages. Tap the mic to start, tap stop to upload + send, or cancel to
   277→ * discard. The recording is uploaded via /api/upload and sent as a message
   278→ * with mediaType 'audio/webm' (or the browser's best supported format).
   279→ */
   280→function useVoiceRecorder({
   281→  onSend,
   282→}: {
   283→  onSend: (url: string, mediaType: string) => Promise<void>
   284→}) {
   285→  const [isRecording, setIsRecording] = useState(false)
   286→  const [isUploading, setIsUploading] = useState(false)
   287→  const [seconds, setSeconds] = useState(0)
   288→  const mediaRecorderRef = useRef<MediaRecorder | null>(null)
   289→  const chunksRef = useRef<Blob[]>([])
   290→  const streamRef = useRef<MediaStream | null>(null)
   291→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   292→  const mimeTypeRef = useRef<string>('audio/webm')
   293→
   294→  // Stop the timer + release the mic on unmount
   295→  useEffect(() => {
   296→    return () => {
   297→      if (timerRef.current) clearInterval(timerRef.current)
   298→      if (streamRef.current) streamRef.current.getTracks().forEach((t) => t.stop())
   299→    }
   300→  }, [])
   301→
   302→  const pickMimeType = useCallback(() => {
   303→    const candidates = [
   304→      'audio/webm;codecs=opus',
   305→      'audio/webm',
   306→      'audio/ogg;codecs=opus',
   307→      'audio/mp4',
   308→    ]
   309→    for (const t of candidates) {
   310→      if (typeof MediaRecorder !== 'undefined' && MediaRecorder.isTypeSupported(t)) {
   311→        return t
   312→      }
   313→    }
   314→    return ''
   315→  }, [])
   316→
   317→  const stopTracks = useCallback(() => {
   318→    if (timerRef.current) {
   319→      clearInterval(timerRef.current)
   320→      timerRef.current = null
   321→    }
   322→    if (streamRef.current) {
   323→      streamRef.current.getTracks().forEach((t) => t.stop())
   324→      streamRef.current = null
   325→    }
   326→  }, [])
   327→
   328→  const start = useCallback(async () => {
   329→    try {
   330→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   331→      streamRef.current = stream
   332→      const mimeType = pickMimeType()
   333→      mimeTypeRef.current = mimeType || 'audio/webm'
   334→      const recorder = mimeType
   335→        ? new MediaRecorder(stream, { mimeType })
   336→        : new MediaRecorder(stream)
   337→      chunksRef.current = []
   338→      recorder.ondataavailable = (e) => {
   339→        if (e.data && e.data.size > 0) chunksRef.current.push(e.data)
   340→      }
   341→      recorder.start()
   342→      mediaRecorderRef.current = recorder
   343→      setIsRecording(true)
   344→      setSeconds(0)
   345→      timerRef.current = setInterval(() => {
   346→        setSeconds((s) => s + 1)
   347→      }, 1000)
   348→    } catch {
   349→      toast.error('Microphone access denied')
   350→      if (streamRef.current) {
   351→        streamRef.current.getTracks().forEach((t) => t.stop())
   352→        streamRef.current = null
   353→      }
   354→    }
   355→  }, [pickMimeType])
   356→
   357→  const cancel = useCallback(() => {
   358→    const recorder = mediaRecorderRef.current
   359→    if (recorder && recorder.state !== 'inactive') {
   360→      // Detach onstop so the pending send is discarded
   361→      recorder.onstop = null
   362→      try {
   363→        recorder.stop()
   364→      } catch {
   365→        // ignore
   366→      }
   367→    }
   368→    mediaRecorderRef.current = null
   369→    chunksRef.current = []
   370→    setIsRecording(false)
   371→    setSeconds(0)
   372→    stopTracks()
   373→  }, [stopTracks])
   374→
   375→  const stopAndSend = useCallback(async () => {
   376→    const recorder = mediaRecorderRef.current
   377→    if (!recorder || recorder.state === 'inactive') {
   378→      cancel()
   379→      return
   380→    }
   381→    const mime = mimeTypeRef.current
   382→    const blobPromise = new Promise<Blob>((resolve) => {
   383→      recorder.onstop = () => {
   384→        const blob = new Blob(chunksRef.current, { type: mime || 'audio/webm' })
   385→        resolve(blob)
   386→      }
   387→    })
   388→    try {
   389→      recorder.stop()
   390→    } catch {
   391→      cancel()
   392→      return
   393→    }
   394→    setIsRecording(false)
   395→    stopTracks()
   396→    setIsUploading(true)
   397→    try {
   398→      const blob = await blobPromise
   399→      if (blob.size === 0) {
   400→        toast.error('Recording was empty')
   401→        return
   402→      }
   403→      const ext = mime.includes('mp4') ? 'm4a' : mime.includes('ogg') ? 'ogg' : 'webm'
   404→      const file = new File([blob], `voice-${Date.now()}.${ext}`, { type: blob.type || mime })
   405→      const formData = new FormData()
   406→      formData.append('file', file)
   407→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   408→      if (!res.ok) {
   409→        const t = await res.text().catch(() => 'Upload failed')
   410→        throw new Error(t.slice(0, 120) || 'Upload failed')
   411→      }
   412→      const data = await res.json()
   413→      // ALWAYS use the MIME type we recorded with (from MediaRecorder), NOT
   414→      // data.type from the upload response. The upload route returns
   415→      // file.type which can come back as 'video/webm' (WebM is primarily a
   416→      // video container) or 'application/octet-stream' (if the type is lost
   417→      // in FormData serialization). Either of those would cause the message
   418→      // to render as a <video> element (black square) instead of our custom
   419→      // VoiceMessagePlayer. We KNOW we recorded audio, so use mime directly.
   420→      await onSend(data.url, mime || 'audio/webm')
   421→      toast.success('Voice message sent')
   422→    } catch (e: any) {
   423→      toast.error(e?.message || 'Failed to send voice message')
   424→    } finally {
   425→      setIsUploading(false)
   426→      setSeconds(0)
   427→      chunksRef.current = []
   428→      mediaRecorderRef.current = null
   429→    }
   430→  }, [cancel, onSend, stopTracks])
   431→
   432→  return { isRecording, isUploading, seconds, start, cancel, stopAndSend }
   433→}

   435→/**
   436→ * TTS Dialog — lets the user type text, pick a voice (built-in or custom),
   437→ * preview, and send. Also supports creating custom voices from audio clips.
   438→ */
   439→function TtsDialog({
   440→  open,
   441→  onOpenChange,
   442→  onSend,
   443→}: {
   444→  open: boolean
   445→  onOpenChange: (o: boolean) => void
   446→  onSend: (url: string) => Promise<void>
   447→}) {
   448→  const [tab, setTab] = useState<'generate' | 'voices'>('generate')
   449→  const [text, setText] = useState('')
   450→  const [voice, setVoice] = useState('alba')
   451→  const [selectedCustomVoiceId, setSelectedCustomVoiceId] = useState<string | null>(null)
   452→  const [generating, setGenerating] = useState(false)
   453→  const [previewUrl, setPreviewUrl] = useState<string | null>(null)
   454→  const [previewBlob, setPreviewBlob] = useState<Blob | null>(null)
   455→  const [isStreaming, setIsStreaming] = useState(false)
   456→  const [sending, setSending] = useState(false)
   457→  const qc = useQueryClient()
   458→
   459→  // Fetch built-in voices from the API (single source of truth for labels)
   460→  const { data: builtinVoicesData } = useQuery({
   461→    queryKey: ['tts-builtin-voices'],
   462→    queryFn: async () => {
   463→      const res = await fetch('/api/tts')
   464→      if (!res.ok) throw new Error('failed')
   465→      return res.json()
   466→    },
   467→    enabled: open,
   468→  })
   469→  const builtinVoices: { id: string; name: string; language: string; gender: string }[] =
   470→    builtinVoicesData?.voices || []
   471→
   472→  // Fetch custom voices
   473→  const { data: customVoicesData } = useQuery({
   474→    queryKey: ['tts-voices'],
   475→    queryFn: async () => {
   476→      const res = await fetch('/api/tts/voices')
   477→      if (!res.ok) throw new Error('failed')
   478→      return res.json()
   479→    },
   480→    enabled: open,
   481→  })
   482→  const customVoices = customVoicesData?.voices || []
   483→
   484→  const handleGenerate = async () => {
   485→    if (!text.trim() || generating) return
   486→    setGenerating(true)
   487→    // Clear any previous preview (revoke blob URL to avoid leaks)
   488→    if (previewUrl?.startsWith('blob:')) URL.revokeObjectURL(previewUrl)
   489→    setPreviewUrl(null)
   490→    setPreviewBlob(null)
   491→    setIsStreaming(true)
   492→    try {
   493→      const body: any = { text }
   494→      if (selectedCustomVoiceId) {
   495→        body.customVoiceId = selectedCustomVoiceId
   496→      } else {
   497→        body.voice = voice
   498→      }
   499→      const res = await fetch('/api/tts', {
   500→        method: 'POST',
   501→        headers: { 'Content-Type': 'application/json' },
   502→        body: JSON.stringify(body),
   503→      })
   504→      if (!res.ok) {
   505→        const err = await res.json().catch(() => ({ error: 'Generation failed' }))
   506→        throw new Error(err.error || 'Generation failed')
   507→      }
   508→
   509→      // The API route streams the TTS audio directly (no server-side save).
   510→      // We read the stream for TWO purposes simultaneously:
   511→      //   1. Instant playback via Web Audio API (chunks played as they arrive)
   512→      //   2. Collect ALL chunks into a Blob for upload when the user sends
   513→      //
   514→      // When the user clicks "Send", we upload the Blob to /api/upload which
   515→      // saves synchronously and returns a URL. This avoids the race condition
   516→      // where a background server-side save hadn't finished before the
   517→      // message URL was accessed (which caused 0-length audio in production).
   518→      const reader = res.body!.getReader()
   519→      let audioCtx: AudioContext | null = null
   520→      let headerParsed = false
   521→      let headerBuf = new Uint8Array(44)
   522→      let headerBytes = 0
   523→      let pcmBuffer = new Uint8Array(0)
   524→      let sampleRate = 24000
   525→      let nextStartTime = 0
   526→      const collectedChunks: Uint8Array[] = []
   527→
   528→      const playChunk = (data: Uint8Array) => {
   529→        if (!audioCtx) return
   530→        const samples = Math.floor(data.length / 2)
   531→        if (samples === 0) return
   532→        const audioBuffer = audioCtx.createBuffer(1, samples, sampleRate)
   533→        const int16 = new Int16Array(data.buffer, data.byteOffset, samples)
   534→        const channelData = audioBuffer.getChannelData(0)
   535→        for (let i = 0; i < samples; i++) {
   536→          channelData[i] = int16[i] / 32768
   537→        }
   538→        const source = audioCtx.createBufferSource()
   539→        source.buffer = audioBuffer
   540→        source.connect(audioCtx.destination)
   541→        const startTime = Math.max(audioCtx.currentTime, nextStartTime)
   542→        source.start(startTime)
   543→        nextStartTime = startTime + audioBuffer.duration
   544→      }
   545→
   546→      while (true) {
   547→        const { done, value } = await reader.read()
   548→        if (done) break
   549→        if (value) {
   550→          // Collect every chunk for the final Blob
   551→          collectedChunks.push(value)
   552→
   553→          if (!headerParsed) {
   554→            const needed = 44 - headerBytes
   555→            const copy = Math.min(needed, value.length)
   556→            headerBuf.set(value.slice(0, copy), headerBytes)
   557→            headerBytes += copy
   558→            if (headerBytes >= 44) {
   559→              const view = new DataView(headerBuf.buffer)
   560→              sampleRate = view.getUint32(24, true)
   561→              headerParsed = true
   562→              if (!audioCtx) {
   563→                audioCtx = new AudioContext({ latencyHint: 'playback' })
   564→              }
   565→              if (value.length > copy) {
   566→                pcmBuffer = new Uint8Array(value.slice(copy))
   567→              }
   568→            }
   569→          } else {
   570→            const merged = new Uint8Array(pcmBuffer.length + value.length)
   571→            merged.set(pcmBuffer)
   572→            merged.set(value, pcmBuffer.length)
   573→            if (merged.length >= 16384) {
   574→              playChunk(merged)
   575→              pcmBuffer = new Uint8Array(0)
   576→            } else {
   577→              pcmBuffer = merged
   578→            }
   579→          }
   580→        }
   581→      }
   582→      if (pcmBuffer.length > 0) playChunk(pcmBuffer)
   583→
   584→      // Build the final Blob from all collected chunks and create a blob URL
   585→      // for the preview player. The Blob is kept in state so handleSend can
   586→      // upload it to /api/upload.
   587→      const wavBlob = new Blob(collectedChunks as BlobPart[], { type: 'audio/wav' })
   588→      console.log(`[tts] collected ${collectedChunks.length} chunks, blob size = ${wavBlob.size} bytes`)
   589→      if (wavBlob.size === 0) {
   590→        throw new Error('TTS produced empty audio (0 bytes collected from stream)')
   591→      }
   592→      const blobUrl = URL.createObjectURL(wavBlob)
   593→      setPreviewBlob(wavBlob)
   594→      setPreviewUrl(blobUrl)
   595→      setIsStreaming(false)
   596→      toast.success('Voice generated — preview and send')
   597→    } catch (e: any) {
   598→      setIsStreaming(false)
   599→      toast.error(e.message || 'Failed to generate voice')
   600→    } finally {
   601→      setGenerating(false)
   602→    }
   603→  }
   604→
   605→  const handleSend = async () => {
   606→    if (!previewBlob || sending) return
   607→    setSending(true)
   608→    try {
   609→      // Upload the collected Blob to /api/upload.
   610→      const formData = new FormData()
   611→      formData.append('file', previewBlob, `tts-${Date.now()}.wav`)
   612→      console.log('[tts] uploading blob, size=', previewBlob.size, 'type=', previewBlob.type)
   613→      const uploadRes = await fetch('/api/upload', {
   614→        method: 'POST',
   615→        body: formData,
   616→      })
   617→      if (!uploadRes.ok) {
   618→        const errText = await uploadRes.text().catch(() => 'unknown')
   619→        console.error('[tts] upload failed:', uploadRes.status, errText)
   620→        throw new Error(`Upload failed (${uploadRes.status}): ${errText.slice(0, 100)}`)
   621→      }
   622→      const data = await uploadRes.json()
   623→      console.log('[tts] upload success, url=', data.url)
   624→
   625→      // Revoke the blob URL — the uploaded file URL is now the source of truth
   626→      if (previewUrl?.startsWith('blob:')) URL.revokeObjectURL(previewUrl)
   627→
   628→      await onSend(data.url)
   629→      setText('')
   630→      setPreviewUrl(null)
   631→      setPreviewBlob(null)
   632→    } catch (e: any) {
   633→      console.error('[tts] send failed:', e)
   634→      toast.error(e?.message || 'Failed to send voice message')
   635→    } finally {
   636→      setSending(false)
   637→    }
   638→  }
   639→
   640→  const handleClose = (o: boolean) => {

   642→      setText('')

   644→      setPreviewUrl(null)
   645→      setPreviewBlob(null)
   646→      setSelectedCustomVoiceId(null)
   647→    }
   648→    onOpenChange(o)
   649→  }

   655→  }

   679→    <Dialog open={open} onOpenChange={handleClose}>
   680→      <DialogContent className="sm:max-w-md">
   681→        <DialogHeader>
   682→          <DialogTitle className="flex items-center gap-2">
   683→            <Sparkles className="w-4 h-4 text-primary" />
   684→            AI Voice Message
   685→          </DialogTitle>
   686→          <DialogDescription>
   687→            Type a message and send it as an AI-generated voice note. Create custom voices from audio clips.
   688→          </DialogDescription>
   689→        </DialogHeader>
   690→
   691→        {/* Tabs */}
   692→        <div className="flex gap-1 p-1 bg-muted rounded-lg">

   695→            className={cn(

   697→              tab === 'generate' ? 'bg-background text-foreground' : 'text-muted-foreground'

   699→          >
   700→            Generate
   701→          </button>
   702→          <button
   703→            onClick={() => setTab('voices')}
   704→            className={cn(

   732→              </div>
   733→            </div>
   734→
   735→            <div className="space-y-2">
