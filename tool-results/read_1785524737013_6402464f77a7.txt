    65→    65→/**
    66→    66→ * Reads the FlowNode type from `data.type` (NOT the ReactFlow `type` prop,
    67→    67→ * which is always 'custom' in this app). This was the root cause of every
    68→    68→ * node rendering as a green "Send Message" node.
    69→    69→ */
    70→    70→function CustomNode({ data, selected }: { data: any; selected?: boolean }) {
    71→    71→  const nodeType = (data.type as NodeType) || 'message'
    72→    72→  const def = NODE_DEFS[nodeType] || NODE_DEFS.message
    73→    73→  const Icon = ICONS[def.icon] || Zap
    74→    74→
    75→    75→  const isTrigger = nodeType === 'trigger'
    76→    76→  const isStop = nodeType === 'stop'
    77→    77→  const isCondition = nodeType === 'condition'
    78→    78→  const isRandom = nodeType === 'random'
    79→    79→  const isSwitch = nodeType === 'switch_case'
    80→    80→
    81→    81→  // ── Build preview content per node type ──────────────────────────────
    82→    82→  let preview: { label: string; value: string } | null = null
    83→    83→  switch (nodeType) {
    84→    84→    case 'trigger': {
    85→    85→      const tt = data.triggerType || 'any_message'
    86→    86→      const label =
    87→    87→        tt === 'command' ? `/${data.command || 'command'}` :
    88→    88→        tt === 'mention' ? '@mentioned' :
    89→    89→        'any message'
    90→    90→      preview = { label: 'When', value: label }
    91→    91→      break
    92→    92→    }
    93→    93→    case 'message':
    94→    94→      preview = { label: 'Says', value: data.text || '(empty)' }
    95→    95→      break
    96→    96→    case 'typing':
    97→    97→      preview = { label: 'For', value: `${data.seconds || 1}s` }
    98→    98→      break
    99→    99→    case 'input':
   100→   100→      preview = { label: 'Asks', value: data.prompt || '(empty)' }
   101→   101→      break
   102→   102→    case 'wait_choice':
   103→   103→      preview = { label: 'Buttons', value: (data.options || []).join(' · ') || '(none)' }
   104→   104→      break
   105→   105→    case 'condition':
   106→   106→      preview = { label: 'If', value: `${data.variable || '∅'} ${data.operator || 'exists'} ${data.value || ''}` }
   107→   107→      break
   108→   108→    case 'set_var':
   109→   109→      preview = { label: 'Sets', value: `${data.variable || '∅'} = ${data.value || '∅'}` }
   110→   110→      break
   111→   111→    case 'delay':
   112→   112→      preview = { label: 'Wait', value: `${data.seconds || 1}s` }
   113→   113→      break
   114→   114→    case 'stop':
   115→   115→      preview = null
   116→   116→      break
   117→   117→    case 'api_call':
   118→   118→      preview = { label: 'Calls', value: `${data.method || 'GET'} ${data.url || '(no url)'}` }
   119→   119→      break
   120→   120→    case 'random':
   121→   121→      preview = { label: 'Picks', value: 'a random branch' }
   122→   122→      break
   123→   123→    case 'ai_generate':
   124→   124→      preview = { label: 'Asks AI', value: data.aiPrompt ? (data.aiPrompt.length > 40 ? data.aiPrompt.slice(0, 40) + '…' : data.aiPrompt) : '(no prompt)' }
   125→   125→      break
   126→   126→    case 'send_media':
   127→   127→      preview = { label: 'Sends', value: data.mediaUrl || '(no url)' }
   128→   128→      break
   129→   129→    case 'switch_case':
   130→   130→      preview = { label: 'Switch on', value: data.switchVariable || '(no var)' }
   131→   131→      break
   132→   132→    case 'counter':
   133→   133→      preview = { label: 'Incr', value: `${data.variable || 'count'} by ${data.increment ?? 1}` }
   134→   134→      break
   135→   135→    case 'format_string':
   136→   136→      preview = { label: 'Formats', value: data.text || '(no template)' }
   137→   137→      break
   138→   138→    case 'log':
   139→   139→      preview = { label: 'Logs', value: data.logMessage ? (data.logMessage.length > 40 ? data.logMessage.slice(0, 40) + '…' : data.logMessage) : '(no message)' }
   140→   140→      break
   141→   141→    case 'tts':
   142→   142→      preview = { label: 'Speaks', value: data.ttsText ? (data.ttsText.length > 40 ? data.ttsText.slice(0, 40) + '…' : data.ttsText) : '(no text)' }
   143→   143→      break
   144→   144→    case 'asr_transcribe':
   145→   145→      preview = { label: 'Transcribes', value: data.asrAudioUrl ? (data.asrAudioUrl.length > 40 ? data.asrAudioUrl.slice(0, 40) + '…' : data.asrAudioUrl) : '{{mediaUrl}}' }
   146→   146→      break
   147→   147→  }
   148→   148→
   149→   149→  return (
   150→   150→    <div
   151→   151→      className="bg-[#2b2d31] rounded-xl min-w-[200px] max-w-[260px] shadow-lg border overflow-hidden transition-shadow"
   152→   152→      style={{
   153→   153→        borderColor: selected ? def.color : 'rgba(255,255,255,0.08)',
   154→   154→        boxShadow: selected ? `0 0 0 2px ${def.color}40` : undefined,
   155→   155→      }}
   156→   156→    >
   157→   157→      {/* Target handle (not for trigger — it has no incoming edges) */}
   158→   158→      {!isTrigger && (
   159→   159→        <Handle
   160→   160→          type="target"
   161→   161→          position={Position.Top}
   162→   162→          className="adoo-handle"
   163→   163→          style={{ background: def.color, width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   164→   164→        />
   165→   165→      )}
   166→   166→
   167→   167→      {/* Header */}
   168→   168→      <div
   169→   169→        className="flex items-center gap-2 px-3 py-2 border-b"
   170→   170→        style={{ background: def.bg, borderColor: `${def.color}30` }}
   171→   171→      >
   172→   172→        <div
   173→   173→          className="w-6 h-6 rounded-md flex items-center justify-center shrink-0"
   174→   174→          style={{ background: def.color + '30' }}
   175→   175→        >
   176→   176→          <Icon className="w-3.5 h-3.5" style={{ color: def.color }} />
   177→   177→        </div>
   178→   178→        <span
   179→   179→          className="text-[11px] font-bold uppercase tracking-wider flex-1"
   180→   180→          style={{ color: def.color }}
   181→   181→        >
   182→   182→          {def.label}
   183→   183→        </span>
   184→   184→        {def.pauses && (
   185→   185→          <span className="text-[9px] uppercase tracking-wider px-1.5 py-0.5 rounded bg-white/10 text-white/60">
   186→   186→            pause
   187→   187→          </span>
   188→   188→        )}
   189→   189→        {def.terminates && (
   190→   190→          <span className="text-[9px] uppercase tracking-wider px-1.5 py-0.5 rounded bg-white/10 text-white/60">
   191→   191→            end
   192→   192→          </span>
   193→   193→        )}
   194→   194→      </div>
   195→   195→
   196→   196→      {/* Body */}
   197→   197→      <div className="px-3 py-2 text-xs">
   198→   198→        {preview ? (
   199→   199→          <div className="space-y-1">
   200→   200→            <div className="text-white/40 uppercase tracking-wider text-[10px]">{preview.label}</div>
   201→   201→            <div className="text-white/80 break-words line-clamp-2">{preview.value}</div>
   202→   202→            {(nodeType === 'input' || nodeType === 'wait_choice' || nodeType === 'api_call') && data.variableName && (
   203→   203→              <div className="text-[10px] text-white/40">→ saves as <code className="text-white/60">{`{{${data.variableName}}}`}</code></div>
   204→   204→            )}
   205→   205→          </div>
   206→   206→        ) : (
   207→   207→          <div className="text-white/50 italic text-center py-1">{def.description}</div>
   208→   208→        )}
   209→   209→      </div>
   210→   210→
   211→   211→      {/* Source handles */}
   212→   212→      {isCondition ? (
   213→   213→        <div className="flex justify-between px-3 pb-2 pt-1">
   214→   214→          <div className="flex flex-col items-center">
   215→   215→            <span className="text-[10px] font-semibold text-emerald-400 mb-1">TRUE</span>
   216→   216→            <Handle
   217→   217→              type="source"
   218→   218→              position={Position.Bottom}
   219→   219→              id="true"
   220→   220→              className="adoo-handle"
   221→   221→              style={{ background: '#34D399', width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   222→   222→            />
   223→   223→          </div>
   224→   224→          <div className="flex flex-col items-center">
   225→   225→            <span className="text-[10px] font-semibold text-red-400 mb-1">FALSE</span>
   226→   226→            <Handle
   227→   227→              type="source"
   228→   228→              position={Position.Bottom}
   229→   229→              id="false"
   230→   230→              className="adoo-handle"
   231→   231→              style={{ background: '#F87171', width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   232→   232→            />
   233→   233→          </div>
   234→   234→        </div>
   235→   235→      ) : isSwitch ? (
   236→   236→        <div className="px-3 pb-2 pt-1 space-y-1.5">
   237→   237→          {((data.cases as string[]) || []).map((c, i) => (
   238→   238→            <div key={i} className="flex items-center gap-2">
   239→   239→              <span className="text-[10px] font-mono text-white/40 shrink-0 w-10">case_{i}</span>
   240→   240→              <span className="text-[10px] text-white/70 truncate flex-1 px-1.5 py-0.5 rounded bg-white/5">{c || `(empty)`}</span>
   241→   241→              <Handle
   242→   242→                type="source"
   243→   243→                position={Position.Right}
   244→   244→                id={`case_${i}`}
   245→   245→                className="adoo-handle"
   246→   246→                style={{ background: '#FB7185', width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%', position: 'relative' }}
   247→   247→              />
   248→   248→            </div>
   249→   249→          ))}
   250→   250→          <div className="flex items-center gap-2 pt-1.5 border-t border-white/10">
   251→   251→            <span className="text-[10px] font-mono text-white/40 shrink-0 w-10">default</span>
   252→   252→            <span className="text-[10px] text-white/40 italic flex-1 px-1.5 py-0.5 rounded bg-white/5">fallback</span>
   253→   253→            <Handle
   254→   254→              type="source"
   255→   255→              position={Position.Right}
   256→   256→              id="default"
   257→   257→              className="adoo-handle"
   258→   258→              style={{ background: '#64748B', width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%', position: 'relative' }}
   259→   259→            />
   260→   260→          </div>
   261→   261→        </div>
   262→   262→      ) : isRandom ? (
   263→   263→        <div className="px-3 pb-2 pt-1 text-center">
   264→   264→          <span className="text-[10px] text-white/40 mb-1 block">All outgoing edges are random picks</span>
   265→   265→          <Handle
   266→   266→            type="source"
   267→   267→            position={Position.Bottom}
   268→   268→            className="adoo-handle"
   269→   269→            style={{ background: def.color, width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   270→   270→          />
   271→   271→        </div>
   272→   272→      ) : !isStop ? (
   273→   273→        <Handle
   274→   274→          type="source"
   275→   275→          position={Position.Bottom}
   276→   276→          className="adoo-handle"
   277→   277→          style={{ background: def.color, width: 16, height: 16, border: '3px solid #1e1f22', borderRadius: '50%' }}
   278→   278→        />
   279→   279→      ) : null}
   280→   280→    </div>
   281→   281→  )
   282→   282→}

   295→   295→  const [selectedNodeId, setSelectedNodeId] = useState<string | null>(null)
   296→   296→  const [selectedEdgeId, setSelectedEdgeId] = useState<string | null>(null)
   297→   297→  const [rightTab, setRightTab] = useState<'inspector' | 'issues' | 'debug'>('inspector')

   383→   383→    )
   384→   384→
   385→   385→  const deleteNode = (nodeId: string) => {
   386→   386→    setNodes((nds) => nds.filter((n) => n.id !== nodeId))
   387→   387→    setEdges((eds) => eds.filter((e) => e.source !== nodeId && e.target !== nodeId))
   388→   388→    setSelectedNodeId(null)
   389→   389→  }
   390→   390→
   391→   391→  // Disconnect an edge — used by the edge context menu and the Delete key.

   393→   393→    setEdges((eds) => eds.filter((e) => e.id !== edgeId))
   394→   394→    setSelectedEdgeId(null)
   395→   395→  }
   396→   396→
   397→   397→  // ── Clipboard for copy/paste ─────────────────────────────────────────

   400→   400→  // Get all currently selected nodes (supports multi-select)
   401→   401→  const selectedNodes = useMemo(() => nodes.filter((n) => n.selected), [nodes])
   402→   402→  const selectedEdges = useMemo(
   403→   403→    () => edges.filter((e) => e.selected || (selectedNodeId && (e.source === selectedNodeId || e.target === selectedNodeId))),
   404→   404→    [edges, selectedNodeId]
   405→   405→  )
   406→   406→
   407→   407→  // Duplicate selected nodes (Ctrl+D)

   409→   409→    if (selectedNodes.length === 0) return
   410→   410→    const newNodes: Node[] = []
   411→   411→    const newEdges: Edge[] = []
   412→   412→    const idMap = new Map<string, string>()
   413→   413→

   415→   415→    for (const n of selectedNodes) {
   416→   416→      const newId = `${(n.data as any)?.type || 'node'}-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`
   417→   417→      idMap.set(n.id, newId)
   418→   418→      newNodes.push({
   419→   419→        ...n,
   420→   420→        id: newId,
   421→   421→        position: { x: n.position.x + 40, y: n.position.y + 40 },
   422→   422→        selected: true,
   423→   423→        data: { ...n.data },
   424→   424→      })
   425→   425→    }
   426→   426→
   427→   427→    // Copy edges between duplicated nodes
