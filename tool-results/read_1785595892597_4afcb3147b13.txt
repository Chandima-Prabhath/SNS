     1→     1→/**
     2→     2→ * Adoo Realtime — Socket.io server setup
     3→     3→ *
     4→     4→ * Single-port architecture: this module exports a function that attaches a
     5→     5→ * Socket.io server to an existing Node.js httpServer. We use this from
     6→     6→ * `server.ts` (custom Next.js server) so that the Next.js app and the realtime
     7→     7→ * service share the SAME port. No more separate mini-service, no more
     8→     8→ * XTransformPort query param.
     9→     9→ *
    10→    10→ * Responsibilities (kept intentionally narrow — this is a dumb relay):
    11→    11→ *   1. Chat relay       — message, edit, delete, typing, read receipts
    12→    12→ *   2. Presence         — online/idle/dnd/offline broadcast
    13→    13→ *   3. Voice signaling  — WebRTC offer/answer/ICE relay only (never media)
    14→    14→ *
    15→    15→ * Auth: client sends the NextAuth session cookie via `withCredentials: true`.
    16→    16→ * The middleware reads it from `socket.handshake.headers.cookie` and decodes
    17→    17→ * the JWT directly (no HTTP roundtrip to NextAuth — single-process friendly).
    18→    18→ */
    19→    19→import type { Server as HTTPServer } from 'http'
    20→    20→import { Server as IOServer } from 'socket.io'
    21→    21→import jwt from 'next-auth/jwt'
    22→    22→import type { NextApiRequest } from 'next'
    23→    23→import { db } from '@/lib/db'
    24→    24→
    25→    25→const SOCKET_PATH = '/api/socket'
    26→    26→
    27→    27→interface AuthenticatedSocket {
    28→    28→  userId?: string
    29→    29→  username?: string
    30→    30→}
    31→    31→
    32→    32→// in-memory presence map (userId → { status, socketIds[], username })
    33→    33→// Use globalThis to share across Next.js module contexts — same fix as ioRef.
    34→    34→// Without this, API routes (which load a different module instance) see an
    35→    35→// empty presence map, so sendMusicCommand() can't find the target user's
    36→    36→// sockets and silently drops the command.
    37→    37→const globalForPresence = globalThis as unknown as {
    38→    38→  __adoo_presence?: Map<string, { status: string; socketIds: Set<string>; username: string }>
    39→    39→}
    40→    40→const presence: Map<string, { status: string; socketIds: Set<string>; username: string }> =
    41→    41→  globalForPresence.__adoo_presence || new Map()
    42→    42→globalForPresence.__adoo_presence = presence
    43→    43→
    44→    44→// Use globalThis to share the io instance across Next.js module contexts.
    45→    45→const globalForIo = globalThis as unknown as { __adoo_io?: IOServer | null }
    46→    46→
    47→    47→export function getIO(): IOServer | null {
    48→    48→  return globalForIo.__adoo_io || null
    49→    49→}
    50→    50→
    51→    51→/**
    52→    52→ * Send a music bot command to a specific user's sockets via the presence map.
    53→    53→ * Used by the bot framework's controlMusic helper — the server-side bot code
    54→    54→ * calls this to tell the user's client to play/pause/skip/stop music.
    55→    55→ *
    56→    56→ * This is the correct path for server→client music control. The old approach
    57→    57→ * (io.emit to ALL sockets) sent the wrong payload shape and broadcast to
    58→    58→ * everyone instead of just the target user.
    59→    59→ */
    60→    60→export function sendMusicCommand(
    61→    61→  targetUserId: string,
    62→    62→  command: { action: 'play' | 'pause' | 'skip' | 'queue' | 'stop'; query?: string }
    63→    63→): void {
    64→    64→  const io = getIO()
    65→    65→  if (!io) {
    66→    66→    console.warn('[sendMusicCommand] io is null — socket.io not initialized')
    67→    67→    return
    68→    68→  }
    69→    69→  const target = presence.get(targetUserId)
    70→    70→  if (!target) {
    71→    71→    console.warn(`[sendMusicCommand] target user ${targetUserId} not found in presence map (size=${presence.size})`)
    72→    72→    // Fallback: broadcast to all sockets. Not ideal for production but
    73→    73→    // ensures the command reaches the user even if presence tracking is off.
    74→    74→    io.emit('music:bot-command', command)
    75→    75→    return
    76→    76→  }
    77→    77→  console.log(`[sendMusicCommand] sending ${command.action} to ${targetUserId} (${target.socketIds.size} sockets)`)
    78→    78→  for (const sid of target.socketIds) {
    79→    79→    io.to(sid).emit('music:bot-command', command)
    80→    80→  }
    81→    81→}

    83→    83→export function attachRealtime(httpServer: HTTPServer): IOServer {
    84→    84→  if (globalForIo.__adoo_io) return globalForIo.__adoo_io
    85→    85→
    86→    86→  const io = new IOServer(httpServer, {
    87→    87→    path: SOCKET_PATH,
    88→    88→    cors: {
    89→    89→      origin: true,
    90→    90→      credentials: true,
    91→    91→      methods: ['GET', 'POST'],
    92→    92→    },
    93→    93→    pingTimeout: 60000,
    94→    94→    pingInterval: 25000,
    95→    95→    // Allow large SDP payloads (WebRTC offer/answer can be 5-10KB)
    96→    96→    maxHttpBufferSize: 1e6,
    97→    97→  })
    98→    98→  globalForIo.__adoo_io = io
    99→    99→
   100→   100→  // ─────────────────────────────────────────────────────────────────────────
   101→   101→  // Auth middleware — decode NextAuth JWT from cookie
   102→   102→  // ─────────────────────────────────────────────────────────────────────────
   103→   103→  io.use(async (socket, next) => {
   104→   104→    try {
   105→   105→      const cookieHeader = socket.handshake.headers.cookie || ''
   106→   106→      console.log('[realtime auth] cookie header present:', !!cookieHeader, 'length:', cookieHeader.length)
   107→   107→
   108→   108→      // Parse cookies into a plain object (small helper, avoids extra dep)
   109→   109→      const cookies: Record<string, string> = {}
   110→   110→      for (const part of cookieHeader.split(';')) {
   111→   111→        const idx = part.indexOf('=')
   112→   112→        if (idx === -1) continue
   113→   113→        const k = part.slice(0, idx).trim()
   114→   114→        const v = part.slice(idx + 1).trim()
   115→   115→        cookies[k] = decodeURIComponent(v)
   116→   116→      }
   117→   117→
   118→   118→      // NextAuth JWT cookie name depends on env:
   119→   119→      //   dev:      next-auth.session-token
   120→   120→      //   prod TLS: __Secure-next-auth.session-token
   121→   121→      const tokenRaw =
   122→   122→        cookies['__Secure-next-auth.session-token'] ||
   123→   123→        cookies['next-auth.session-token']
   124→   124→
   125→   125→      if (!tokenRaw) {
   126→   126→        console.log('[realtime auth] no session cookie found. Available cookies:', Object.keys(cookies))
   127→   127→        return next(new Error('unauthorized: no session cookie'))
   128→   128→      }
   129→   129→
   130→   130→      // Decode the JWT. We don't have a real Request here, so we shim one.
   131→   131→      // next-auth/jwt.decode only needs secret + token.
   132→   132→      const secret = process.env.NEXTAUTH_SECRET
   133→   133→      if (!secret) {
   134→   134→        console.error('[realtime auth] NEXTAUTH_SECRET is not set in environment!')
   135→   135→        return next(new Error('server misconfigured: NEXTAUTH_SECRET missing'))
   136→   136→      }
   137→   137→
   138→   138→      const decoded = await jwt.decode({
   139→   139→        token: tokenRaw,
   140→   140→        secret,
   141→   141→      } as any)
   142→   142→
   143→   143→      if (!decoded || !decoded.id) {
   144→   144→        console.log('[realtime auth] JWT decode returned null or missing id. decoded:', decoded ? Object.keys(decoded) : 'null')
   145→   145→        return next(new Error('unauthorized: invalid session'))
   146→   146→      }
   147→   147→
   148→   148→      ;(socket as any).userId = decoded.id
   149→   149→      ;(socket as any).username = decoded.username || decoded.email || 'user'
   150→   150→      console.log('[realtime auth] success: userId=', decoded.id, 'username=', (socket as any).username)
   151→   151→      next()
   152→   152→    } catch (e: any) {
   153→   153→      console.error('[realtime auth] error:', e?.message || e, e?.stack?.slice(0, 200))
   154→   154→      next(new Error('unauthorized'))
   155→   155→    }
   156→   156→  })
   157→   157→
   158→   158→  function broadcastPresence() {
   159→   159→    const list = Array.from(presence.entries()).map(([id, p]) => ({
   160→   160→      userId: id,
   161→   161→      username: p.username,
   162→   162→      status: p.status,
   163→   163→    }))
   164→   164→    io.emit('presence:update', list)
   165→   165→  }
   166→   166→
   167→   167→  function setPresence(userId: string, username: string, status: string, socketId: string) {
   168→   168→    const existing = presence.get(userId)
   169→   169→    if (existing) {
   170→   170→      existing.socketIds.add(socketId)
   171→   171→      existing.status = status
   172→   172→      existing.username = username
   173→   173→    } else {
   174→   174→      presence.set(userId, { status, username, socketIds: new Set([socketId]) })
   175→   175→    }
   176→   176→    broadcastPresence()
   177→   177→  }
   178→   178→
   179→   179→  function removeSocket(userId: string, socketId: string) {
   180→   180→    const p = presence.get(userId)
   181→   181→    if (!p) return
   182→   182→    p.socketIds.delete(socketId)
   183→   183→    if (p.socketIds.size === 0) {
   184→   184→      presence.delete(userId)
   185→   185→    }
   186→   186→    broadcastPresence()
   187→   187→  }
   188→   188→
   189→   189→  /**
   190→   190→   * Check if a call should end (fewer than 2 participants).
   191→   191→   * If so, notify all remaining participants and mark the call as ended.
   192→   192→   */
   193→   193→  function checkCallEnd(callId: string) {
   194→   194→    const room = io.sockets.adapter.rooms.get(`call:${callId}`)
   195→   195→    const participantCount = room ? room.size : 0
   196→   196→
   197→   197→    if (participantCount < 2) {
   198→   198→      // Notify any remaining participant (or none) that the call ended
   199→   199→      io.to(`call:${callId}`).emit('call:ended', {
   200→   200→        callId,
   201→   201→        reason: participantCount === 0 ? 'all_left' : 'insufficient_participants',
   202→   202→      })
   203→   203→
   204→   204→      // Make remaining sockets leave the room
   205→   205→      if (room) {
   206→   206→        for (const sid of room) {
   207→   207→          const s = io.sockets.sockets.get(sid)
   208→   208→          s?.leave(`call:${callId}`)
   209→   209→        }
   210→   210→      }
   211→   211→
   212→   212→      console.log(`[realtime] call ${callId} ended (${participantCount} participants remaining)`)
   213→   213→    }
   214→   214→  }
   215→   215→
   216→   216→  // ─────────────────────────────────────────────────────────────────────────
   217→   217→  // Connection lifecycle
   218→   218→  // ─────────────────────────────────────────────────────────────────────────
   219→   219→  io.on('connection', (socket) => {
   220→   220→    const auth = socket as unknown as AuthenticatedSocket & { userId: string; username: string }
   221→   221→    const { userId, username } = auth
   222→   222→    console.log(`[realtime] connect ${username} (${userId})`)
   223→   223→
   224→   224→    setPresence(userId, username, 'online', socket.id)
   225→   225→
   226→   226→    // ─── Chat channel subscriptions ──────────────────────────────────────
   227→   227→    socket.on('channel:join', async (channelId: string) => {
   228→   228→      // Verify membership before joining the socket room (C8 fix)
   229→   229→      const membership = await db.channelMember.findUnique({
   230→   230→        where: { channelId_userId: { channelId, userId } },
   231→   231→      }).catch(() => null)
   232→   232→      if (membership) {
   233→   233→        socket.join(`channel:${channelId}`)
   234→   234→      }
   235→   235→    })
   236→   236→
   237→   237→    socket.on('channel:leave', (channelId: string) => {
   238→   238→      socket.leave(`channel:${channelId}`)
   239→   239→    })
   240→   240→
   241→   241→    socket.on('channel:message', (payload: { channelId: string; message: any }) => {
   242→   242→      // C9 fix: only broadcast messages from the authenticated user
   243→   243→      if (payload.message?.senderId === userId) {
   244→   244→        socket.to(`channel:${payload.channelId}`).emit('channel:message', payload.message)
   245→   245→      }
   246→   246→    })
   247→   247→
   248→   248→    socket.on('channel:message-edit', (payload: { channelId: string; message: any }) => {
   249→   249→      // C9 fix: only broadcast edits for the user's own messages
   250→   250→      if (payload.message?.senderId === userId) {
   251→   251→        socket.to(`channel:${payload.channelId}`).emit('channel:message-edit', payload.message)
   252→   252→      }
   253→   253→    })
   254→   254→
   255→   255→    socket.on('channel:message-delete', (payload: { channelId: string; messageId: string }) => {
   256→   256→      socket.to(`channel:${payload.channelId}`).emit('channel:message-delete', {
   257→   257→        messageId: payload.messageId,
   258→   258→      })
   259→   259→    })
   260→   260→
   261→   261→    socket.on('channel:typing', async (payload: { channelId: string; isTyping: boolean }) => {
   262→   262→      const typingPayload = {
   263→   263→        userId,
   264→   264→        username,
   265→   265→        channelId: payload.channelId,
   266→   266→        isTyping: payload.isTyping,
   267→   267→      }
   268→   268→
   269→   269→      // Active viewers (sockets that have joined this channel's room) get the
   270→   270→      // event via the room broadcast — this is the existing fast path.
   271→   271→      socket.to(`channel:${payload.channelId}`).emit('channel:typing', typingPayload)
   272→   272→
   273→   273→      // Fan out to ALL channel members via the presence map so the chat list
   274→   274→      // can show "typing..." for channels the user hasn't actively joined.
   275→   275→      //
   276→   276→      // Without this, a member viewing a different channel would never see the
   277→   277→      // typing indicator in their chat list because they're not in the channel
   278→   278→      // room. We fetch the member list from the DB on each typing pulse —
   279→   279→      // typing is debounced client-side to ~1.5s, so this is at most ~0.7 QPS
   280→   280→      // per typist, which SQLite handles easily.
   281→   281→      //
   282→   282→      // We use `volatile` so a slow client connection doesn't queue up stale
   283→   283→      // typing pulses — only the most recent state matters.
   284→   284→      try {
   285→   285→        const members = await db.channelMember.findMany({
   286→   286→          where: { channelId: payload.channelId },
   287→   287→          select: { userId: true },
   288→   288→        })
   289→   289→        for (const m of members) {
   290→   290→          if (m.userId === userId) continue // skip the typist themselves
   291→   291→          const target = presence.get(m.userId)
   292→   292→          if (!target) continue
   293→   293→          for (const sid of target.socketIds) {
   294→   294→            // Skip the sender's own socket (multi-tab: don't echo back)
   295→   295→            if (sid === socket.id) continue
   296→   296→            // Skip sockets already in the channel room — they got it via the
   297→   297→            // room broadcast above and we don't want to double-deliver.
   298→   298→            const s = io.sockets.sockets.get(sid)
   299→   299→            if (s?.rooms.has(`channel:${payload.channelId}`)) continue
   300→   300→            io.to(sid).volatile.emit('channel:typing', typingPayload)
   301→   301→          }
   302→   302→        }
   303→   303→      } catch (e) {
   304→   304→        // Silent — typing is best-effort, don't crash the socket handler.
   305→   305→      }
   306→   306→    })
   307→   307→
   308→   308→    socket.on('channel:read', (payload: { channelId: string; messageId: string }) => {
   309→   309→      socket.to(`channel:${payload.channelId}`).emit('channel:read', {
   310→   310→        userId,
   311→   311→        channelId: payload.channelId,
   312→   312→        messageId: payload.messageId,
   313→   313→      })
   314→   314→    })
   315→   315→
   316→   316→    // Delivery ACK — when a client receives a message via 'channel:message',
   317→   317→    // it emits 'channel:delivered' back. We update the DB and broadcast to
   318→   318→    // the sender so their UI can show the single-checkmark "delivered" state.
   319→   319→    socket.on('channel:delivered', async (payload: { channelId: string; messageId: string }) => {
   320→   320→      try {
   321→   321→        // Only update if deliveredAt is not already set (avoid redundant writes)
   322→   322→        const msg = await db.message.findUnique({
   323→   323→          where: { id: payload.messageId },
   324→   324→          select: { deliveredAt: true, senderId: true },
   325→   325→        })
   326→   326→        if (!msg || msg.deliveredAt) return  // already delivered or not found
   327→   327→        // Don't ACK your own messages (the sender client also emits this)
   328→   328→        if (msg.senderId === userId) return
   329→   329→
   330→   330→        await db.message.update({
   331→   331→          where: { id: payload.messageId },
   332→   332→          data: { deliveredAt: new Date() },
   333→   333→        })
   334→   334→
   335→   335→        // Broadcast to the channel (the sender will pick this up and update their UI)
   336→   336→        socket.to(`channel:${payload.channelId}`).emit('channel:delivered', {
   337→   337→          messageId: payload.messageId,
   338→   338→          channelId: payload.channelId,
   339→   339→        })
   340→   340→      } catch (e) {
   341→   341→        // Best-effort — don't let delivery ACK failures break anything
   342→   342→      }
   343→   343→    })
   344→   344→
   345→   345→    // ─── Music bot control ─────────────────────────────────────────────
   346→   346→    // Bots emit this to control a user's music player. The server forwards
   347→   347→    // the command to the target user's sockets via the presence map.
   348→   348→    socket.on('music:bot-command', (payload: {
   349→   349→      targetUserId: string
   350→   350→      command: {
   351→   351→        action: 'play' | 'pause' | 'skip' | 'queue' | 'stop'
   352→   352→        query?: string
   353→   353→      }
   354→   354→    }) => {
   355→   355→      const target = presence.get(payload.targetUserId)
   356→   356→      if (!target) return
   357→   357→      for (const sid of target.socketIds) {
   358→   358→        io.to(sid).emit('music:bot-command', payload.command)
   359→   359→      }
   360→   360→    })
   361→   361→
   362→   362→    // ─── Status / Stories ────────────────────────────────────────────────
   363→   363→    socket.on('story:posted', (payload: { story: any }) => {
   364→   364→      socket.broadcast.emit('story:posted', payload.story)
   365→   365→    })
   366→   366→
   367→   367→    socket.on('story:viewed', (payload: { storyId: string; userId: string }) => {
   368→   368→      socket.broadcast.emit('story:viewed', payload)
   369→   369→    })
   370→   370→
   371→   371→    // ─── Voice call signaling (WebRTC) ───────────────────────────────────
   372→   372→    // The server tracks active calls and their participants. When a call
   373→   373→    // drops below 2 participants, it's auto-ended and everyone is notified.
   374→   374→
   375→   375→    socket.on('call:join', (callId: string) => {
   376→   376→      socket.join(`call:${callId}`)
   377→   377→
   378→   378→      // Tell everyone already in the call that we joined (so they can offer)
   379→   379→      socket.to(`call:${callId}`).emit('call:peer-joined', {
   380→   380→        peerId: socket.id,
   381→   381→        userId,
   382→   382→        username,
   383→   383→      })
   384→   384→
   385→   385→      // Tell the joiner who's already in the call
   386→   386→      const roomSocketIds = Array.from(io.sockets.adapter.rooms.get(`call:${callId}`) || []).filter(
   387→   387→        (id) => id !== socket.id
   388→   388→      )
   389→   389→      const peers = roomSocketIds.map((sid) => {
   390→   390→        const otherSocket = io.sockets.sockets.get(sid) as any
   391→   391→        return {
   392→   392→          peerId: sid,
   393→   393→          userId: otherSocket?.userId || '',
   394→   394→          username: otherSocket?.username || 'user',
   395→   395→        }
   396→   396→      })
   397→   397→      socket.emit('call:peers', { peers })
   398→   398→    })
   399→   399→
   400→   400→    socket.on('call:leave', (callId: string) => {
   401→   401→      socket.leave(`call:${callId}`)
   402→   402→      socket.to(`call:${callId}`).emit('call:peer-left', {
   403→   403→        peerId: socket.id,
   404→   404→        userId,
   405→   405→      })
   406→   406→
   407→   407→      // Check if the call should end (fewer than 2 participants remain)
   408→   408→      checkCallEnd(callId)
   409→   409→    })
   410→   410→
   411→   411→    // Also handle disconnect — user might close the tab
   412→   412→    socket.on('disconnect', () => {
   413→   413→      // Find all calls this socket was in and check if they should end
   414→   414→      for (const room of socket.rooms) {
   415→   415→        if (room.startsWith('call:')) {
   416→   416→          const callId = room.substring(5)
   417→   417→          socket.to(room).emit('call:peer-left', {
   418→   418→            peerId: socket.id,
   419→   419→            userId,
   420→   420→          })
   421→   421→          checkCallEnd(callId)
   422→   422→        }
   423→   423→      }
   424→   424→    })
   425→   425→
   426→   426→    socket.on('call:offer', (payload: { to: string; sdp: any }) => {
   427→   427→      io.to(payload.to).emit('call:offer', { from: socket.id, sdp: payload.sdp })
   428→   428→    })
   429→   429→
   430→   430→    socket.on('call:answer', (payload: { to: string; sdp: any }) => {
   431→   431→      io.to(payload.to).emit('call:answer', { from: socket.id, sdp: payload.sdp })
   432→   432→    })
   433→   433→
   434→   434→    socket.on('call:ice-candidate', (payload: { to: string; candidate: any }) => {
   435→   435→      io.to(payload.to).emit('call:ice-candidate', { from: socket.id, candidate: payload.candidate })
   436→   436→    })
   437→   437→
   438→   438→    // ─── DM call ringing (for 1:1 calls) ─────────────────────────────────
   439→   439→    // The caller rings a specific user via their socket connection(s).
   440→   440→    // We look up all sockets for that userId (via the presence map) and forward.
   441→   441→    socket.on('call:ring', (payload: {
   442→   442→      callId: string
   443→   443→      targetUserId: string
   444→   444→      from: { userId: string; username: string; displayName: string }
   445→   445→      channelId?: string
   446→   446→      dmGroupId?: string
   447→   447→      video?: boolean
   448→   448→    }) => {
   449→   449→      // Send push notification immediately (works even if app is closed)
   450→   450→      import('./push').then(m => {
   451→   451→        m.sendPushNotification(payload.targetUserId, {
   452→   452→          type: 'call',
   453→   453→          title: payload.from.displayName,
   454→   454→          body: `Incoming ${payload.video ? 'video' : 'voice'} call`,
   455→   455→          callId: payload.callId,
   456→   456→          channelId: payload.channelId,
   457→   457→          from: payload.from,
   458→   458→        })
   459→   459→      }).catch(() => {})
   460→   460→
   461→   461→      // RETRY: keep sending call:incoming every 5 seconds for up to 1 minute.
   462→   462→      // This handles the case where the app is closed — the push notification
   463→   463→      // wakes it, the socket reconnects, and the next retry delivers the event.
   464→   464→      const incomingPayload = {
   465→   465→        callId: payload.callId,
   466→   466→        from: payload.from,
   467→   467→        channelId: payload.channelId,
   468→   468→        dmGroupId: payload.dmGroupId,
   469→   469→        video: payload.video ?? false,
   470→   470→      }
   471→   471→
   472→   472→      // Send immediately
   473→   473→      const sendIncoming = () => {
   474→   474→        const target = presence.get(payload.targetUserId)
   475→   475→        if (target) {
   476→   476→          for (const sid of target.socketIds) {
   477→   477→            io.to(sid).emit('call:incoming', incomingPayload)
   478→   478→          }
   479→   479→        }
   480→   480→      }
   481→   481→      sendIncoming()
   482→   482→
   483→   483→      // Retry every 5s for 60s (12 attempts)
   484→   484→      let attempts = 0
   485→   485→      const retryInterval = setInterval(() => {
   486→   486→        attempts++
   487→   487→        if (attempts >= 12) {
   488→   488→          clearInterval(retryInterval)
   489→   489→          return
   490→   490→        }
   491→   491→        // Check if the call is still active
   492→   492→        // (the caller might have cancelled)
   493→   493→        const target = presence.get(payload.targetUserId)
   494→   494→        if (target) {
   495→   495→          // Check if any socket has acknowledged by joining the call room
   496→   496→          const callRoom = io.sockets.adapter.rooms.get(`call:${payload.callId}`)
   497→   497→          if (callRoom && callRoom.size > 0) {
   498→   498→            // Someone joined the call — stop retrying
   499→   499→            clearInterval(retryInterval)
   500→   500→            return
   501→   501→          }
   502→   502→          sendIncoming()
   503→   503→          console.log(`[call] retry ${attempts}/12 sending call:incoming to ${payload.targetUserId}`)
   504→   504→        }
   505→   505→      }, 5000)
   506→   506→
   507→   507→      // Clean up interval when caller disconnects
   508→   508→      socket.on('disconnect', () => {
   509→   509→        clearInterval(retryInterval)
   510→   510→      })
   511→   511→    })
   512→   512→
   513→   513→    socket.on('call:accept', (payload: { callId: string; byUserId: string }) => {
   514→   514→      // Notify all of caller's sockets that the call was accepted
   515→   515→      const callerSockets = presence.get(payload.byUserId)
   516→   516→      if (callerSockets) {
   517→   517→        for (const sid of callerSockets.socketIds) {
   518→   518→          io.to(sid).emit('call:accept', { callId: payload.callId, byUserId: userId })
   519→   519→        }
   520→   520→      }
   521→   521→    })
   522→   522→
   523→   523→    socket.on('call:reject', (payload: { callId: string; byUserId: string; reason?: string }) => {
   524→   524→      const callerSockets = presence.get(payload.byUserId)
   525→   525→      if (callerSockets) {
   526→   526→        for (const sid of callerSockets.socketIds) {
   527→   527→          io.to(sid).emit('call:reject', { callId: payload.callId, reason: payload.reason || 'rejected' })
   528→   528→        }
   529→   529→      }
   530→   530→    })
   531→   531→
   532→   532→    socket.on('call:cancel', (payload: { callId: string; targetUserId: string }) => {
   533→   533→      // Caller is cancelling an outgoing ring
   534→   534→      const targetSockets = presence.get(payload.targetUserId)
   535→   535→      if (targetSockets) {
   536→   536→        for (const sid of targetSockets.socketIds) {
   537→   537→          io.to(sid).emit('call:cancel', { callId: payload.callId })
   538→   538→        }
   539→   539→      }
   540→   540→    })
   541→   541→
   542→   542→    // ─── Push notifications: server pushes new messages to all of a user's ─
   543→   543→    // sessions so they get unread badges / notifications regardless of which
   544→   544→    // screen they're on. Persistent socket connection = real-time notifications.
   545→   545→    socket.on('notify:user', (payload: { userId: string; type: string; data: any }) => {
   546→   546→      const targetPresence = presence.get(payload.userId)
   547→   547→      if (!targetPresence) return
   548→   548→      for (const sid of targetPresence.socketIds) {
   549→   549→        // Don't echo back to the sender's own socket
   550→   550→        if (sid === socket.id) continue
   551→   551→        io.to(sid).emit('notify', { type: payload.type, data: payload.data })
   552→   552→      }
   553→   553→    })
   554→   554→
   555→   555→    // Broadcast a new message to all members of a channel (server-side driven,
   556→   556→    // not just relying on the sender's socket to relay)
   557→   557→    socket.on('channel:broadcast-message', (payload: { channelId: string; message: any; recipientIds: string[] }) => {
   558→   558→      for (const uid of payload.recipientIds) {
   559→   559→        const targetPresence = presence.get(uid)
   560→   560→        if (!targetPresence) continue
   561→   561→        for (const sid of targetPresence.socketIds) {
   562→   562→          if (sid === socket.id) continue // skip sender
   563→   563→          io.to(sid).emit('channel:message', payload.message)
   564→   564→          io.to(sid).emit('notify', {
   565→   565→            type: 'message',
   566→   566→            data: {
   567→   567→              channelId: payload.channelId,
   568→   568→              messageId: payload.message.id,
   569→   569→              senderId: payload.message.senderId,
   570→   570→              senderName: payload.message.sender?.displayName || payload.message.sender?.username,
   571→   571→              body: payload.message.body,
   572→   572→              senderType: payload.message.senderType,
   573→   573→            },
   574→   574→          })
   575→   575→        }
   576→   576→      }
   577→   577→    })
   578→   578→
   579→   579→    // ─── Presence management ─────────────────────────────────────────────
   580→   580→    socket.on('presence:set', (status: string) => {
   581→   581→      setPresence(userId, username, status, socket.id)
   582→   582→    })
   583→   583→
   584→   584→    socket.on('presence:request', () => {
   585→   585→      const list = Array.from(presence.entries()).map(([id, p]) => ({
   586→   586→        userId: id,
   587→   587→        username: p.username,
   588→   588→        status: p.status,
   589→   589→      }))
   590→   590→      socket.emit('presence:update', list)
   591→   591→    })
   592→   592→
   593→   593→    // ─── Music room sync ────────────────────────────────────────────────
   594→   594→    // Host-authoritative: the host broadcasts playback state changes to all
   595→   595→    // room members. Members apply the state with drift compensation.
   596→   596→    socket.on('music:join', (roomId: string) => {
   597→   597→      socket.join(`music:${roomId}`)
   598→   598→      // Notify the room that someone joined — the host should respond with
   599→   599→      // a sync broadcast so the new member gets the current state.
   600→   600→      socket.to(`music:${roomId}`).emit('music:member-joined', {
   601→   601→        roomId,
   602→   602→        userId,
   603→   603→        username,
   604→   604→      })
   605→   605→    })
   606→   606→
   607→   607→    socket.on('music:leave', (roomId: string) => {
   608→   608→      socket.leave(`music:${roomId}`)
   609→   609→    })
   610→   610→
   611→   611→    // Broadcast sync events to all room members.
   612→   612→    // The host sends: { roomId, state, position, videoId?, trackInfo? }
   613→   613→    // We include the server's receive timestamp for drift compensation.
   614→   614→    socket.on('music:sync', (payload: {
   615→   615→      roomId: string
   616→   616→      state: string
   617→   617→      position: number
   618→   618→      videoId?: string
   619→   619→      trackInfo?: any
   620→   620→      queue?: { videoId: string }[]
   621→   621→    }) => {
   622→   622→      io.to(`music:${payload.roomId}`).emit('music:sync', {
   623→   623→        ...payload,
   624→   624→        serverTimestamp: Date.now(),
   625→   625→      })
   626→   626→
   627→   627→      // ── Server-side proactive preloading ────────────────────────────
   628→   628→      // When the host broadcasts a playing state, proactively preload the
   629→   629→      // current track + the next 2 tracks in the queue. This runs on the
   630→   630→      // SERVER's event loop, completely independent of client tab throttling.
   631→   631→      // By the time the song ends, the next track is already on disk.
   632→   632→      if (payload.state === 'playing' && payload.videoId) {
   633→   633→        import('@/lib/ytdlp-download').then(({ getOrCreateDownload }) => {
   634→   634→          // Current track
   635→   635→          getOrCreateDownload(payload.videoId!).catch(() => {})
   636→   636→          // Next 2 tracks in the queue
   637→   637→          if (payload.queue && payload.queue.length > 0) {
   638→   638→            for (const track of payload.queue.slice(0, 2)) {
   639→   639→              if (track.videoId) {
   640→   640→                getOrCreateDownload(track.videoId).catch(() => {})
   641→   641→              }
   642→   642→            }
   643→   643→          }
   644→   644→        }).catch(() => {})
   645→   645→      }
   646→   646→    })
   647→   647→
   648→   648→    // Request sync — a member who just joined asks the host for the current
   649→   649→    // state. The host should respond with a music:sync broadcast.
   650→   650→    socket.on('music:request-sync', (roomId: string) => {
   651→   651→      socket.to(`music:${roomId}`).emit('music:request-sync', {
   652→   652→        roomId,
   653→   653→        fromUserId: userId,
   654→   654→      })
   655→   655→    })
   656→   656→
   657→   657→    socket.on('disconnect', () => {
   658→   658→      removeSocket(userId, socket.id)
   659→   659→      console.log(`[realtime] disconnect ${username} (${userId})`)
   660→   660→    })
   661→   661→
   662→   662→    socket.on('error', (err: any) => {
   663→   663→      console.error(`[realtime] socket error ${userId}:`, err)
   664→   664→    })
   665→   665→  })
   666→   666→
   667→   667→  console.log(`[realtime] attached, path=${SOCKET_PATH}`)
   668→   668→  return io
   669→   669→}
