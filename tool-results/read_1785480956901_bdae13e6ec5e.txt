    18→interface MessageListProps {
    19→  channelId: string

    21→
    22→export function MessageList({ channelId }: MessageListProps) {
    23→  const { messages, isLoading, send, edit, remove, sendCallback, typing, markRead, replyTo, setReplyTo } =
    24→    useChannel(channelId)
    25→  const { data: session } = useSession()
    26→  const myId = (session?.user as any)?.id
    27→  const scrollRef = useRef<HTMLDivElement>(null)
    28→  const lastMessageIdRef = useRef<string | null>(null)
    29→  const isAtBottomRef = useRef(true)
    30→
    31→  // Scroll-to-bottom button state.
    32→  //   - showScrollButton: visible when the user has scrolled >200px from the
    33→  //     bottom (matches the threshold in the task spec).
    34→  //   - unreadBelow: count of new messages that have arrived BELOW the user's
    35→  //     current scroll position. Resets to 0 when the user scrolls back to the
    36→  //     bottom (or clicks the button).
    37→  const [showScrollButton, setShowScrollButton] = useState(false)
    38→  const [unreadBelow, setUnreadBelow] = useState(0)
    39→
    40→  // Reset scroll state when switching channels — otherwise the auto-scroll
    41→  // logic thinks we already saw the last message and skips the initial scroll.
    42→  // Refs are reset here (no rerender needed); the button state is reset in
    43→  // the channel-switch derived-state check below to keep it in sync with
    44→  // React's render cycle.
    45→  useEffect(() => {
    46→    lastMessageIdRef.current = null
    47→    isAtBottomRef.current = true
    48→  }, [channelId])
    49→
    50→  // Reset the scroll-button React state when the channel changes. We use the
    51→  // "derived state" pattern (calling setState during render in a conditional)
    52→  // instead of useEffect+setState — React specifically endorses this for
    53→  // "reset state when a prop changes" because it avoids the extra render
    54→  // cycle that useEffect+setState triggers.
    55→  // https://react.dev/reference/react/useState#storing-information-from-previous-renders
    56→  const [prevChannelId, setPrevChannelId] = useState(channelId)
    57→  if (prevChannelId !== channelId) {
    58→    setPrevChannelId(channelId)
    59→    setShowScrollButton(false)
    60→    setUnreadBelow(0)
    61→  }
    62→
    63→  // Track whether the user is at the bottom of the scroll container.
    64→  // We use a ref so the scroll handler doesn't trigger re-renders, but we
    65→  // also mirror `showScrollButton` into React state so the button can
    66→  // appear/disappear. The 80px threshold for "at bottom" matches the
    67→  // existing auto-scroll logic; the 200px threshold for showing the button
    68→  // is per the task spec.
    69→  useEffect(() => {
    70→    const el = scrollRef.current
    71→    if (!el) return
    72→    const onScroll = () => {
    73→      const distanceFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight
    74→      const atBottom = distanceFromBottom < 80
    75→      isAtBottomRef.current = atBottom
    76→      // Show the floating button only when scrolled meaningfully up
    77→      setShowScrollButton(distanceFromBottom > 200)
    78→      // Reset the "new messages below" counter when the user reaches the
    79→      // bottom — they've now seen everything.
    80→      if (atBottom) {
    81→        setUnreadBelow(0)
    82→      }
    83→    }
    84→    el.addEventListener('scroll', onScroll, { passive: true })
    85→    return () => el.removeEventListener('scroll', onScroll)
    86→  }, [])
    87→
    88→  // Auto-scroll — handles BOTH the initial load (channel switch / first
    89→  // messages arriving) and subsequent new messages.
    90→  //
    91→  // • Initial load: wait 200ms so avatars/images have time to render and
    92→  //   lay out before we measure scrollHeight. Double rAF was too short for
    93→  //   channels with media-heavy history (scroll would land on "yesterday"
    94→  //   instead of the latest message).
    95→  // • New message: use requestAnimationFrame so the DOM updates before we
    96→  //   measure. Scroll smoothly so the user sees the message slide in.
    97→  // • Own message: ALWAYS scroll, even if the user scrolled up (so sending
    98→  //   a message snaps back to the bottom).
    99→  // • New message while scrolled up: DON'T scroll, but increment the
   100→  //   "unread below" counter so the scroll-to-bottom button shows a badge.
   101→  //   Multiple messages can land in a single render (e.g., a burst of bot
   102→  //   replies) — count all of them by comparing indices.
   103→  useEffect(() => {
   104→    const el = scrollRef.current
   105→    if (!el) return
   106→    const lastMsg = messages[messages.length - 1]
   107→    if (!lastMsg) return
   108→    // Already processed this exact message — nothing to do.
   109→    if (lastMessageIdRef.current === lastMsg.id) return
   110→
   111→    const isMine = lastMsg.senderId === myId
   112→    const isInitialLoad = !lastMessageIdRef.current
   113→    const shouldScroll = isInitialLoad || isMine || isAtBottomRef.current
   114→
   115→    // Count how many new messages arrived since the last one we processed.
   116→    // If the previous last-seen id is still in the array, the delta is the
   117→    // number of messages after it. Fall back to 1 if we can't find it (e.g.,
   118→    // the array was refetched and the old id is gone).
   119→    let newCount = 1
   120→    if (lastMessageIdRef.current) {
   121→      const lastSeenIdx = messages.findIndex((m) => m.id === lastMessageIdRef.current)
   122→      if (lastSeenIdx >= 0) {
   123→        newCount = Math.max(1, messages.length - 1 - lastSeenIdx)
   124→      }
   125→    }
   126→
   127→    lastMessageIdRef.current = lastMsg.id
   128→    markRead(lastMsg.id)
   129→
   130→    if (!shouldScroll) {
   131→      // User is scrolled up — count this burst as new messages below the fold.
   132→      setUnreadBelow((n) => n + newCount)
   133→      return
   134→    }
   135→
   136→    if (isInitialLoad) {
   137→      // Give the browser time to lay out all the historical messages (and
   138→      // their media) before we jump. 200ms is enough for most avatars and
   139→      // thumbnails to load without making the channel switch feel slow.
   140→      const t = window.setTimeout(() => {
   141→        const e = scrollRef.current
   142→        if (e) e.scrollTop = e.scrollHeight
   143→      }, 200)
   144→      return () => window.clearTimeout(t)
   145→    }
   146→
   147→    // New message — wait one frame so the new bubble has rendered before
   148→    // we measure scrollHeight. Otherwise we'd scroll to the OLD bottom and
   149→    // the latest message would sit just below the fold.
   150→    const raf = requestAnimationFrame(() => {
   151→      const e = scrollRef.current
   152→      if (e) {
   153→        e.scrollTo({ top: e.scrollHeight, behavior: 'smooth' })
   154→      }
   155→    })
   156→    // We scrolled to the bottom — clear the unread-below counter.
   157→    setUnreadBelow(0)
   158→    return () => cancelAnimationFrame(raf)
   159→  }, [messages, myId, markRead])
   160→
   161→  // Group consecutive messages by same sender within 5 min
   162→  const grouped = useMemo(() => {
   163→    const groups: ChannelMessage[][] = []
   164→    let current: ChannelMessage[] = []
   165→    for (const m of messages) {
   166→      if (current.length === 0) {
   167→        current.push(m)
   168→      } else {
   169→        const prev = current[current.length - 1]
   170→        const sameSender = prev.senderId === m.senderId && prev.senderType === m.senderType
   171→        const timeDiff = new Date(m.createdAt).getTime() - new Date(prev.createdAt).getTime()
   172→        if (sameSender && timeDiff < 5 * 60 * 1000) {
   173→          current.push(m)
   174→        } else {
   175→          groups.push(current)
   176→          current = [m]
   177→        }
   178→      }
   179→    }
   180→    if (current.length > 0) groups.push(current)
   181→    return groups
   182→  }, [messages])
   183→
   184→  return (
   185→    <div className="flex-1 flex flex-col min-h-0 bg-background relative overflow-hidden">
   186→      {/* Subtle cinematic glow in the background */}
   187→      <div className="absolute inset-0 bg-[radial-gradient(ellipse_80%_80%_at_50%_-20%,rgba(var(--primary),0.05),transparent)] pointer-events-none" />
   188→      <div ref={scrollRef as any} className="flex-1 min-h-0 overflow-y-auto relative z-10 scroll-smooth">
   189→        <div className="px-3 md:px-6 py-4 space-y-3 max-w-4xl mx-auto">
   190→          {isLoading && <LoadingState />}
   191→
   192→          {!isLoading && messages.length === 0 && (
   193→            <div className="text-center py-20 px-4">
   194→              <div className="w-16 h-16 mx-auto rounded-3xl bg-primary/10 flex items-center justify-center mb-3 ring-1 ring-primary/15">
   195→                <Reply className="w-7 h-7 text-primary" strokeWidth={1.5} />
   196→              </div>
   197→              <p className="text-base font-semibold">No messages yet</p>
   198→              <p className="text-sm text-muted-foreground mt-1">Be the first to say hello</p>
   199→            </div>
   200→          )}
   201→
   202→          {grouped.map((group, gi) => {
   203→            const first = group[0]
   204→            const isMine = first.senderId === myId && first.senderType === 'user'
   205→            const isBot = first.senderType === 'bot'
   206→            const senderDeleted = first.senderType === 'user' && !first.sender
   207→            const senderName =
   208→              first.senderType === 'bot'
   209→                ? first.sender?.displayName || 'Bot'
   210→                : first.sender?.displayName || (senderDeleted ? 'Deleted User' : 'Unknown')
   211→            const dayLabel = formatDay(first.createdAt)
   212→            const prevGroup = gi > 0 ? grouped[gi - 1] : null
   213→            const showDayDivider = !prevGroup || formatDay(prevGroup[0].createdAt) !== dayLabel
   214→
   215→            return (
   216→              <div key={gi}>
   217→                {showDayDivider && (
   218→                  <div className="flex items-center gap-3 my-5">
   219→                    <div className="flex-1 h-px bg-border" />
   220→                    <span className="text-xs text-muted-foreground font-medium">{dayLabel}</span>
   221→                    <div className="flex-1 h-px bg-border" />
   222→                  </div>
   223→                )}
   224→                <MessageGroup
   225→                  group={group}
   226→                  isMine={isMine}
   227→                  isBot={isBot}
   228→                  senderDeleted={senderDeleted}
   229→                  senderName={senderName}
   230→                  avatarUrl={first.sender?.avatarUrl || null}
   231→                  myId={myId}
   232→                  onReply={(m) =>
   233→                    setReplyTo({ id: m.id, body: m.body, senderName })
   234→                  }
   235→                  onDelete={(m) => remove(m.id)}
   236→                  onEditSubmit={async (messageId, text) => edit(messageId, text)}
   237→                  onCallback={async (messageId, callbackData) => sendCallback(messageId, callbackData)}
   238→                />
   239→              </div>
   240→            )
   241→          })}
   242→
   243→          {/* Typing indicator */}
   244→          <AnimatePresence>
   245→            {typing.length > 0 && (
   246→              <motion.div
   247→                initial={{ opacity: 0, y: 8 }}
   248→                animate={{ opacity: 1, y: 0 }}
   249→                exit={{ opacity: 0, y: 8 }}
   250→                className="flex items-center gap-2 text-xs text-muted-foreground italic px-2"
   251→              >
   252→                <span className="flex gap-1">
   253→                  <span className="w-1.5 h-1.5 rounded-full bg-current animate-bounce [animation-delay:0ms]" />
   254→                  <span className="w-1.5 h-1.5 rounded-full bg-current animate-bounce [animation-delay:120ms]" />
   255→                  <span className="w-1.5 h-1.5 rounded-full bg-current animate-bounce [animation-delay:240ms]" />
   256→                </span>
   257→                {typing.map((t) => t.username).join(', ')} {typing.length === 1 ? 'is' : 'are'} typing...
   258→              </motion.div>
   259→            )}
   260→          </AnimatePresence>
   261→        </div>
   262→      </div>
   263→
   264→      {/* Scroll-to-bottom button — appears when the user has scrolled >200px
   265→          up from the bottom. Shows an unread count badge for messages that
   266→          arrived below the current scroll position. Smooth-scrolls to the
   267→          bottom on click and hides itself when already at the bottom. */}
   268→      <AnimatePresence>
   269→        {showScrollButton && (
   270→          <motion.button
   271→            initial={{ opacity: 0, scale: 0.8, y: 8 }}
   272→            animate={{ opacity: 1, scale: 1, y: 0 }}
   273→            exit={{ opacity: 0, scale: 0.8, y: 8 }}
   274→            transition={{ duration: 0.15 }}
   275→            onClick={() => {
   276→              const e = scrollRef.current
   277→              if (e) {
   278→                e.scrollTo({ top: e.scrollHeight, behavior: 'smooth' })
   279→                // Optimistically clear — the scroll handler will confirm
   280→                // once the smooth scroll settles at the bottom.
   281→                setUnreadBelow(0)
   282→                setShowScrollButton(false)
   283→              }
   284→            }}
   285→            aria-label="Scroll to latest messages"
   286→            title="Scroll to latest"
   287→            className="absolute bottom-4 right-4 z-20 w-11 h-11 rounded-full bg-card border border-border/60 shadow-lg flex items-center justify-center hover:bg-accent active:scale-95 transition-all"
   288→          >
   289→            <ChevronDown className="w-5 h-5" />
   290→            {unreadBelow > 0 && (
   291→              <span className="absolute -top-1.5 -right-1.5 min-w-[20px] h-5 px-1.5 rounded-full bg-primary text-primary-foreground text-[10px] font-bold flex items-center justify-center ring-2 ring-card">
   292→                {unreadBelow > 99 ? '99+' : unreadBelow}
   293→              </span>
   294→            )}
   295→          </motion.button>
   296→        )}
   297→      </AnimatePresence>
   298→    </div>
   299→  )
   300→}
   301→
   302→function LoadingState() {
   303→  return (
   304→    <div className="space-y-3">
   305→      {[1, 2, 3].map((i) => (
   306→        <div key={i} className="flex gap-2.5 animate-pulse">
   307→          <div className="w-8 h-8 rounded-full bg-muted" />
   308→          <div className="flex-1 space-y-1.5">
   309→            <div className="h-3 w-24 bg-muted rounded" />
   310→            <div className="h-4 w-3/4 bg-muted rounded" />
   311→          </div>
   312→        </div>
   313→      ))}
   314→    </div>
   315→  )
   316→}

   318→function formatDay(dateStr: string | Date) {
   319→  const d = new Date(dateStr)
   320→  if (isToday(d)) return 'Today'
   321→  if (isYesterday(d)) return 'Yesterday'
   322→  return format(d, 'MMM d, yyyy')
   323→}

   339→function MessageGroup(props: MessageGroupProps) {
   340→  const { group, isMine, isBot, senderDeleted, senderName, avatarUrl } = props
   341→  const [editingId, setEditingId] = useState<string | null>(null)
   342→  const replyTo = useAppStore((s) => s.replyTo)
   343→  const setReplyTo = useAppStore((s) => s.setReplyTo)
   344→
   345→  return (
   346→    <div className="group/group flex gap-2.5">
   347→      {/* Avatar — only show on first message of group, hidden for own messages on mobile */}
   348→      <div className="w-8 shrink-0 pt-0.5">
   349→        {isMine ? (
   350→          <div className="w-8" />
   351→        ) : senderDeleted ? (
   352→          <div className="w-8 h-8 rounded-full bg-muted flex items-center justify-center">
   353→            <UserX className="w-4 h-4 text-muted-foreground" />
   354→          </div>
   355→        ) : (
   356→          <Avatar className="w-9 h-9 shadow-sm ring-1 ring-white/10">
   357→            <AvatarImage src={avatarUrl || undefined} />

   359→              {isBot ? <Bot className="w-[18px] h-[18px]" /> : senderName.charAt(0).toUpperCase()}
   360→            </AvatarFallback>
   361→          </Avatar>
   362→        )}
   363→      </div>
   364→
   365→      <div className="flex-1 min-w-0 space-y-0.5">
   366→        {/* Sender name + time — only on first message of group */}
   367→        <div className="flex items-baseline gap-2 mb-0.5">
   368→          <span className={cn(
   369→            'text-[13px] font-semibold',
   370→            isBot && 'text-primary',
   371→            isMine && 'text-foreground',
   372→            senderDeleted && 'text-muted-foreground italic'
   373→          )}>
   374→            {isMine ? 'You' : senderName}
   375→            {isBot && <span className="ml-1 text-[10px] uppercase font-bold text-primary/70">BOT</span>}
   376→          </span>
   377→          <span className="text-[11px] text-muted-foreground">
   378→            {format(new Date(group[0].createdAt), 'HH:mm')}

   380→        </div>
   381→
   382→        {group.map((m, i) => (
   383→          <MessageItem
   384→            key={m.id}
   385→            message={m}
   386→            isMine={isMine}
   387→            isLastOfGroup={i === group.length - 1}
   388→            isEditing={editingId === m.id}
   389→            setEditing={(v) => setEditingId(v ? m.id : null)}
   390→            onReply={() => props.onReply(m)}
   391→            onDelete={() => props.onDelete(m)}
   392→            onEditSubmit={async (text) => {
   393→              await props.onEditSubmit(m.id, text)
   394→              setEditingId(null)
   395→            }}
   396→            onCallback={async (callbackData) => {
   397→              await props.onCallback(m.id, callbackData)
   398→            }}
   399→            highlight={replyTo?.id === m.id}
   400→            clearHighlight={() => setReplyTo(null)}
   401→          />
   402→        ))}
   403→      </div>
   404→    </div>
   405→  )

   408→interface MessageItemProps {
   409→  message: ChannelMessage
   410→  isMine: boolean
   411→  isLastOfGroup: boolean
   412→  isEditing: boolean
   413→  setEditing: (v: boolean) => void
   414→  onReply: () => void
   415→  onDelete: () => void
   416→  onEditSubmit: (text: string) => Promise<void>
   417→  onCallback: (callbackData: string) => Promise<void>
   418→  highlight?: boolean
   419→  clearHighlight: () => void
   420→}

   422→function MessageItem(props: MessageItemProps) {
   423→  const { message: m, isMine, isLastOfGroup, isEditing, setEditing, onEditSubmit } = props
   424→  const [editText, setEditText] = useState(m.body)
   425→  const isDeleted = !!m.deletedAt
   426→  const ctxMenu = useContextMenu()
   427→  const { startCall } = useCall()
   428→  const setActiveRoomId = useMusicStore((s) => s.setActiveRoomId)
   429→
   430→  // Handle invitation Join button — joins a call or music room
   431→  const handleInviteJoin = async (invite: any) => {
   432→    try {
   433→      if (invite.type === 'call') {
   434→        // Join the call via the API (idempotent — joins existing or creates)
   435→        const payload = invite.dmGroupId
   436→          ? { dmGroupId: invite.dmGroupId }
   437→          : { channelId: invite.channelId }
   438→        const res = await fetch('/api/calls', {
   439→          method: 'POST',
   440→          headers: { 'Content-Type': 'application/json' },
   441→          body: JSON.stringify(payload),
   442→        })
   443→        if (!res.ok) throw new Error('Failed to join call')
   444→        const data = await res.json()
   445→        // Start our side (mic + WebRTC)
   446→        await startCall({
   447→          callId: data.call.id,
   448→          channelId: invite.channelId,
   449→          dmGroupId: invite.dmGroupId,
   450→          enableVideo: invite.isVideo ?? false,
   451→        })
   452→        toast.success('Joined the call')
   453→      } else if (invite.type === 'music') {
   454→        // Join the music room — set active room + auto-join as member
   455→        setActiveRoomId(invite.targetId)
   456→        await fetch(`/api/music/rooms/${invite.targetId}`).catch(() => {})
   457→        toast.success(`Joined music room "${invite.roomName || 'Unnamed'}"`)
   458→        // Navigate to music view
   459→        window.location.hash = 'music'
   460→      }
   461→    } catch (e: any) {
   462→      toast.error(e.message || 'Failed to join')
   463→    }
   464→  }
   465→
   466→  const showMessageContextMenu = (e: React.MouseEvent | React.TouchEvent) => {
   467→    let x: number, y: number
   468→    if ('touches' in e) {
   469→      const t = e.touches[0] || (e.changedTouches?.[0])
   470→      if (!t) return
   471→      x = t.clientX; y = t.clientY
   472→    } else {
   473→      x = e.clientX; y = e.clientY
   474→    }
   475→    if (!ctxMenu) return
   476→    const items: Array<{ label: string; icon: React.ReactNode; onClick: () => void; variant?: 'default' | 'danger' }> = [
   477→      {
   478→        label: 'Reply',
   479→        icon: <Reply className="w-4 h-4" />,
   480→        onClick: () => props.onReply(),
   481→      },
   482→      {
   483→        label: 'Copy text',
   484→        icon: <Copy className="w-4 h-4" />,
   485→        onClick: () => {
   486→          navigator.clipboard.writeText(m.body)
   487→        },
   488→      },
   489→    ]
   490→    if (isMine && !isEditing) {
   491→      items.push({
   492→        label: 'Edit',
   493→        icon: <Edit2 className="w-4 h-4" />,
   494→        onClick: () => setEditing(true),
   495→      })
   496→      items.push({
   497→        label: 'Delete',
   498→        icon: <Trash2 className="w-4 h-4" />,
   499→        variant: 'danger',
   500→        onClick: () => props.onDelete(),
   501→      })
   502→    }
   503→    ctxMenu.show(x, y, items)
   504→  }
   505→
   506→  if (isDeleted) {
   507→    return (
   508→      <div className={cn('flex', isMine ? 'justify-end' : 'justify-start')}>
   509→        <span className="inline-flex items-center gap-1.5 text-xs text-muted-foreground italic px-3 py-1.5 rounded-2xl bg-muted/50">
   510→          <Trash2 className="w-3 h-3" />
   511→          Message deleted
   512→        </span>
   513→      </div>
   514→    )
   515→  }
   516→
   517→  return (
   518→    <motion.div
   519→      initial={{ opacity: 0, y: 4 }}
   520→      animate={{ opacity: 1, y: 0 }}
   521→      transition={{ duration: 0.15 }}
   522→      className={cn(
   523→        'group/msg relative flex items-end gap-1',
   524→        isMine ? 'justify-end' : 'justify-start',
   525→        props.highlight && 'ring-2 ring-primary/40 rounded-2xl',
   526→        'transition-shadow'
   527→      )}
   528→      onClick={() => props.highlight && props.clearHighlight()}
   529→    >
   530→      {!isMine && <div className="w-8 shrink-0" />}
   531→
   532→      {/* Message bubble */}
   533→      <div
   534→        onContextMenu={(e) => { e.preventDefault(); e.stopPropagation(); showMessageContextMenu(e) }}
   535→        onTouchStart={(e) => {
   536→          // Long-press for mobile
   537→          const touch = e.touches[0]
   538→          const timer = setTimeout(() => {
   539→            showMessageContextMenu({ touches: [{ clientX: touch.clientX, clientY: touch.clientY }] } as any)
   540→            if (navigator.vibrate) navigator.vibrate(50)
   541→          }, 500)
   542→          const el = e.currentTarget
   543→          const clear = () => {
   544→            clearTimeout(timer)
   545→            el.removeEventListener('touchend', clear)
   546→            el.removeEventListener('touchmove', clear)
   547→            el.removeEventListener('touchcancel', clear)
   548→          }
   549→          el.addEventListener('touchend', clear, { once: true })
   550→          el.addEventListener('touchmove', clear, { once: true })
   551→          el.addEventListener('touchcancel', clear, { once: true })
   552→        }}
   553→        className={cn(
   554→          'relative max-w-[78%] md:max-w-[65%] px-4 py-2.5 text-[15px] leading-relaxed break-words select-none cursor-default backdrop-blur-md',
   555→          isMine
   556→            ? 'bg-gradient-to-br from-primary/90 to-primary text-primary-foreground rounded-[22px] rounded-br-sm shadow-[0_4px_12px_rgba(var(--primary),0.2)] border border-primary/20'
   557→            : 'bg-black/30 text-foreground rounded-[22px] rounded-bl-sm shadow-sm border border-white/10',
   558→          isLastOfGroup && 'mb-1.5'
   559→        )}
   560→      >
   561→        {/* Reply context */}
   562→        {m.replyTo && (
   563→          <div className={cn(
   564→            'border-l-2 pl-2 mb-1.5 text-xs opacity-80',
   565→            isMine ? 'border-bubble-mine-foreground/40' : 'border-primary/40'
   566→          )}>
   567→            <div className="font-medium">
   568→              {m.replyTo.senderType === 'bot'
   569→                ? m.replyTo.sender?.displayName || 'Bot'
   570→                : m.replyTo.sender?.displayName || 'Unknown'}
   571→            </div>
   572→            <div className="truncate opacity-70">{m.replyTo.body.slice(0, 80)}</div>
   573→          </div>
   574→        )}
   575→
   576→        {/* Invitation cards (call / music room invites) */}
   577→        {(m.mediaType === 'invite-call' || m.mediaType === 'invite-music') && (() => {
   578→          try {
   579→            const invite = JSON.parse(m.body)
   580→            const isCall = invite.type === 'call'
   581→            return (
   582→              <div className="mb-1.5 -mx-1">
   583→                <div className={cn(
   584→                  'rounded-xl border overflow-hidden',
   585→                  isCall ? 'bg-blue-500/10 border-blue-500/30' : 'bg-purple-500/10 border-purple-500/30'
   586→                )}>
   587→                  {/* Header */}
   588→                  <div className="flex items-center gap-2.5 p-3">
   589→                    <div className={cn(
   590→                      'w-10 h-10 rounded-full flex items-center justify-center shrink-0',
   591→                      isCall ? 'bg-blue-500/20' : 'bg-purple-500/20'
   592→                    )}>
   593→                      {isCall ? (
   594→                        invite.isVideo ? <Video className="w-5 h-5 text-blue-400" /> : <Phone className="w-5 h-5 text-blue-400" />
   595→                      ) : (
   596→                        <MusicIcon className="w-5 h-5 text-purple-400" />
   597→                      )}
   598→                    </div>
   599→                    <div className="flex-1 min-w-0">
   600→                      <div className="text-sm font-semibold">
   601→                        {isCall
   602→                          ? `${invite.fromName} invited you to a ${invite.isVideo ? 'video' : 'voice'} call`
   603→                          : `${invite.fromName} invited you to music room "${invite.roomName || 'Unnamed'}"`
   604→                        }
   605→                      </div>
   606→                      <div className="text-xs opacity-60">
   607→                        {isCall ? 'Tap Join to connect' : 'Tap Join to listen together'}
   608→                      </div>
   609→                    </div>
   610→                  </div>
   611→                  {/* Action buttons */}
   612→                  <div className="flex gap-1.5 p-2 pt-0">
   613→                    <button
   614→                      onClick={() => handleInviteJoin(invite)}
   615→                      className={cn(
   616→                        'flex-1 py-2 rounded-lg text-xs font-semibold transition-all active:scale-95',
   617→                        isCall
   618→                          ? 'bg-blue-500/30 text-blue-300 hover:bg-blue-500/40'
   619→                          : 'bg-purple-500/30 text-purple-300 hover:bg-purple-500/40'
   620→                      )}
   621→                    >
   622→                      Join
   623→                    </button>
   624→                  </div>
   625→                </div>
   626→              </div>
   627→            )
   628→          } catch {
   629→            return null
   630→          }
   631→        })()}
   632→
   633→        {/* Media */}
   634→        {m.mediaUrl && m.mediaType?.startsWith('image') && (
   635→          <img src={m.mediaUrl} alt="" className="rounded-lg mb-1.5 max-w-full -mx-1 w-[calc(100%+0.5rem)]" />
   636→        )}
   637→        {m.mediaUrl && m.mediaType?.startsWith('video') && (
   638→          <video src={m.mediaUrl} controls className="rounded-lg mb-1.5 max-w-full -mx-1 w-[calc(100%+0.5rem)]" />
   639→        )}
   640→        {m.mediaUrl && m.mediaType?.startsWith('audio') && (
   641→          <div className="mb-1.5 -mx-1">
   642→            <div className={cn(
   643→              'flex items-center gap-2 p-2 rounded-xl',
   644→              isMine ? 'bg-black/15' : 'bg-black/20'
   645→            )}>
   646→              <div className="w-8 h-8 rounded-full gradient-primary flex items-center justify-center shrink-0">
   647→                <AudioLines className="w-4 h-4 text-primary-foreground" />
   648→              </div>
   649→              <audio controls src={m.mediaUrl} className="flex-1 h-8 min-w-0" style={{ maxWidth: '100%' }} />
   650→            </div>
   651→          </div>
   652→        )}
   653→
   654→        {/* Telegram-style inline keyboard buttons */}
   655→        {m.keyboard && !isEditing && (() => {
   656→          try {
   657→            const keyboard = JSON.parse(m.keyboard) as KeyboardButton[][]
   658→            if (!Array.isArray(keyboard) || keyboard.length === 0) return null
   659→            return (
   660→              <div className="mt-1.5 space-y-1">
   661→                {keyboard.map((row, ri) => (
   662→                  <div key={ri} className="flex gap-1 flex-wrap">
   663→                    {row.map((btn, bi) => (
   664→                      <button
   665→                        key={bi}
   666→                        onClick={() => props.onCallback(btn.callbackData)}
   667→                        className={cn(
   668→                          'flex-1 min-w-[60px] px-3 py-1.5 rounded-lg text-xs font-medium transition-all active:scale-95',
   669→                          'backdrop-blur-md border',
   670→                          isMine
   671→                            ? 'bg-black/15 border-white/20 text-primary-foreground hover:bg-black/25'
   672→                            : 'bg-primary/15 border-primary/30 text-primary hover:bg-primary/25'
   673→                        )}
   674→                      >
   675→                        {btn.text}
   676→                      </button>
   677→                    ))}
   678→                  </div>
   679→                ))}
   680→              </div>
   681→            )
   682→          } catch {
   683→            return null
   684→          }
   685→        })()}
   686→
   687→        {/* Body or editor */}
   688→        {isEditing ? (
   689→          <div className="space-y-1.5">
   690→            <textarea
   691→              autoFocus
   692→              value={editText}
   693→              onChange={(e) => setEditText(e.target.value)}
   694→              className="w-full bg-background/20 border border-background/30 rounded p-2 text-sm resize-none text-foreground placeholder:text-foreground/60"
   695→              rows={2}
   696→              onKeyDown={(e) => {
   697→                if (e.key === 'Enter' && !e.shiftKey) {
   698→                  e.preventDefault()
   699→                  onEditSubmit(editText)
   700→                }
   701→                if (e.key === 'Escape') setEditing(false)
   702→              }}
   703→            />
   704→            <div className="flex gap-1.5">
   705→              <Button size="sm" className="h-7 text-xs" onClick={() => onEditSubmit(editText)}>
   706→                <Check className="w-3 h-3 mr-1" /> Save
   707→              </Button>
   708→              <Button size="sm" variant="ghost" className="h-7 text-xs" onClick={() => setEditing(false)}>
   709→                <X className="w-3 h-3 mr-1" /> Cancel
   710→              </Button>
   711→            </div>
   712→          </div>
   713→        ) : (
   714→          <>
   715→            {/* Don't show body text for invite messages (it's JSON) */}
   716→            {m.mediaType !== 'invite-call' && m.mediaType !== 'invite-music' && (
   717→              <div className="whitespace-pre-wrap">{m.body}</div>
   718→            )}
   719→            {m.editedAt && (
   720→              <span className={cn('text-[10px] opacity-60 ml-1')}>
   721→                edited
   722→              </span>
   723→            )}
   724→          </>
   725→        )}
   726→      </div>
   727→
   728→      {/* Hover/tap actions */}
   729→      {!isEditing && (
   730→        <div
   731→          className={cn(
   732→            'opacity-0 group-hover/msg:opacity-100 focus-within:opacity-100 transition-opacity flex items-center',
   733→            'bg-card/95 backdrop-blur-sm rounded-full shadow-sm border border-border/50',
   734→            isMine ? 'order-first' : ''
   735→          )}
   736→        >
   737→          <button onClick={props.onReply} className="p-1.5 hover:bg-accent rounded-full" title="Reply">
   738→            <Reply className="w-3.5 h-3.5" />
   739→          </button>
   740→          {isMine && (
   741→            <>
   742→              <button onClick={() => setEditing(true)} className="p-1.5 hover:bg-accent rounded-full" title="Edit">
   743→                <Edit2 className="w-3.5 h-3.5" />
   744→              </button>
   745→              <button onClick={props.onDelete} className="p-1.5 hover:bg-accent text-red-500 rounded-full" title="Delete">
   746→                <Trash2 className="w-3.5 h-3.5" />
   747→              </button>
   748→            </>
   749→          )}
   750→        </div>
   751→      )}
   752→    </motion.div>
   753→  )
   754→}
