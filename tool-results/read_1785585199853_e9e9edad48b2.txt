    24→    24→/**
    25→    25→ * GlobalMusicPlayer
    26→    26→ * ─────────────────────────────────────────────────────────────────────────
    27→    27→ * Renders the persistent <audio> element and the bottom player bar at the
    28→    28→ * app root level so that audio playback is NOT destroyed when the user
    29→    29→ * switches tabs (Chats, Status, Calls, Settings, Music).
    30→    30→ *
    31→    31→ * Architecture:
    32→    32→ *   - The Zustand store (`useMusicStore`) is the single source of truth for
    33→    33→ *     playback state (currentTrack, isPlaying, position, volume, queue,
    34→    34→ *     autoplay, shuffle, repeat, activeRoomId).
    35→    35→ *   - This component owns the <audio> element and all audio + Socket.io
    36→    36→ *     side-effects. It reads from the store and writes back to it.
    37→    37→ *   - The MusicView (browse / rooms / queue UI) reads state from the store
    38→    38→ *     and calls the action functions exposed here via `useMusicPlayer()`.
    39→    39→ *   - Socket.io sync events are listened to here. When a remote sync is
    40→    40→ *     applied, the store is updated; the audio element is then loaded /
    41→    41→ *     seeked directly. A ref flag (`applyingRemoteRef`) prevents
    42→    42→ *     `timeupdate` events from fighting with the applied position.
    43→    43→ */
    44→    44→
    45→    45→interface MusicPlayerContextValue {
    46→    46→  /** Play a track (or add it to the queue if addToQueue=true). */
    47→    47→  playTrack: (track: Track, addToQueue?: boolean) => Promise<void>
    48→    48→  /** Advance to the next track (queue → autoplay → repeat → stop). */
    49→    49→  playNext: () => Promise<void>
    50→    50→  /** Toggle play / pause. */
    51→    51→  togglePlay: () => void
    52→    52→  /** Seek to a position (seconds). */
    53→    53→  handleSeek: (pos: number) => void
    54→    54→  /** Stop playback entirely (clears current track). */
    55→    55→  stop: () => void
    56→    56→  /** Add a track to the queue (broadcasts to room). */
    57→    57→  addToQueue: (track: Track) => void
    58→    58→  /** Remove a track from the queue by index (broadcasts to room). */
    59→    59→  removeFromQueue: (index: number) => void
    60→    60→  /** Clear the entire queue (broadcasts to room). */
    61→    61→  clearQueue: () => void
    62→    62→}
    63→    63→
    64→    64→const MusicPlayerContext = createContext<MusicPlayerContextValue | null>(null)
    65→    65→
    66→    66→const NOOP_PLAYER: MusicPlayerContextValue = {
    67→    67→  playTrack: async () => {},
    68→    68→  playNext: async () => {},
    69→    69→  togglePlay: () => {},
    70→    70→  handleSeek: () => {},
    71→    71→  stop: () => {},
    72→    72→  addToQueue: () => {},
    73→    73→  removeFromQueue: () => {},
    74→    74→  clearQueue: () => {},
    75→    75→}
    76→    76→
    77→    77→/**
    78→    78→ * Access the global music player's action functions.
    79→    79→ *
    80→    80→ * Safe to call from any component rendered inside <GlobalMusicPlayer>.
    81→    81→ * Returns no-op functions if no provider is present (defensive).
    82→    82→ */
    83→    83→export function useMusicPlayer(): MusicPlayerContextValue {
    84→    84→  const ctx = useContext(MusicPlayerContext)
    85→    85→  return ctx ?? NOOP_PLAYER
    86→    86→}
    87→    87→
    88→    88→export function GlobalMusicPlayer({ children }: { children: React.ReactNode }) {
    89→    89→  const audioRef = useRef<HTMLAudioElement>(null)
    90→    90→  const { socket } = useSocket()
    91→    91→
    92→    92→  // ─── Store state (subscribe to slices) ────────────────────────────────
    93→    93→  const currentTrack = useMusicStore((s) => s.currentTrack)
    94→    94→  const isPlaying = useMusicStore((s) => s.isPlaying)
    95→    95→  const isLoading = useMusicStore((s) => s.isLoading)
    96→    96→  const position = useMusicStore((s) => s.position)
    97→    97→  const volume = useMusicStore((s) => s.volume)
    98→    98→  const queue = useMusicStore((s) => s.queue)
    99→    99→  const radioQueue = useMusicStore((s) => s.radioQueue)
   100→   100→  const shuffle = useMusicStore((s) => s.shuffle)
   101→   101→  const repeat = useMusicStore((s) => s.repeat)
   102→   102→  const activeRoomId = useMusicStore((s) => s.activeRoomId)
   103→   103→
   104→   104→  // ─── Store actions ────────────────────────────────────────────────────
   105→   105→  const setCurrentTrack = useMusicStore((s) => s.setCurrentTrack)
   106→   106→  const setIsPlaying = useMusicStore((s) => s.setIsPlaying)
   107→   107→  const setPosition = useMusicStore((s) => s.setPosition)
   108→   108→  const setVolume = useMusicStore((s) => s.setVolume)
   109→   109→  const setQueue = useMusicStore((s) => s.setQueue)
   110→   110→  const storeAddToQueue = useMusicStore((s) => s.addToQueue)
   111→   111→  const storeRemoveFromQueue = useMusicStore((s) => s.removeFromQueue)
   112→   112→  const storeClearQueue = useMusicStore((s) => s.clearQueue)
   113→   113→
   114→   114→  // ─── Refs for cross-render coordination ───────────────────────────────
   115→   115→  // The videoId currently loaded into the <audio> element. Used to detect
   116→   116→  // when currentTrack.videoId changes so we know to (re)load the stream.
   117→   117→  const loadedVideoIdRef = useRef<string | null>(null)
   118→   118→  // When applying a remote sync that changes the track, this holds the
   119→   119→  // position we should seek to once the new stream has loaded.
   120→   120→  const pendingSeekRef = useRef<number | null>(null)
   121→   121→  // True while we're applying state that arrived from a remote sync event.
   122→   122→  // Prevents local `timeupdate` from overwriting the freshly-applied
   123→   123→  // position and prevents echoing sync events back to the room.
   124→   124→  const applyingRemoteRef = useRef(false)
   125→   125→
   126→   126→  // ─── Helper: broadcast a sync event to the active room ────────────────
   127→   127→  const broadcastSync = useCallback(
   128→   128→    (payload: {
   129→   129→      state: string
   130→   130→      position: number
   131→   131→      videoId?: string
   132→   132→      trackInfo?: Partial<Track>
   133→   133→      queue?: Track[]
   134→   134→    }) => {
   135→   135→      if (!activeRoomId || !socket) return
   136→   136→      socket.emit('music:sync', {
   137→   137→        roomId: activeRoomId,
   138→   138→        ...payload,
   139→   139→      })
   140→   140→    },
   141→   141→    [activeRoomId, socket],
   142→   142→  )
   143→   143→
   144→   144→  // ─── Action: play a track ─────────────────────────────────────────────
   145→   145→  const playTrack = useCallback(
   146→   146→    async (track: Track, addToQueue = false) => {
   147→   147→      const state = useMusicStore.getState()
   148→   148→
   149→   149→      if (addToQueue && state.currentTrack) {
   150→   150→        state.addToQueue(track)
   151→   151→        toast.success(`Added to queue: ${track.title}`)
   152→   152→        // Broadcast the updated queue so members see it
   153→   153→        broadcastSync({
   154→   154→          state: state.isPlaying ? 'playing' : 'paused',
   155→   155→          position: audioRef.current?.currentTime || 0,
   156→   156→          queue: useMusicStore.getState().queue,
   157→   157→        })
   158→   158→        return
   159→   159→      }
   160→   160→
   161→   161→      // Set state first so the UI updates instantly
   162→   162→      setCurrentTrack(track)
   163→   163→      setIsPlaying(true)
   164→   164→      setPosition(0)
   165→   165→      useMusicStore.getState().setIsLoading(true)
   166→   166→
   167→   167→      // Broadcast to room (include the full queue so members sync)
   168→   168→      broadcastSync({
   169→   169→        state: 'playing',
   170→   170→        position: 0,
   171→   171→        videoId: track.videoId,
   172→   172→        trackInfo: {
   173→   173→          title: track.title,
   174→   174→          artist: track.artist,
   175→   175→          thumbnail: track.thumbnail,
   176→   176→          durationSeconds: track.durationSeconds,
   177→   177→        },
   178→   178→        queue: useMusicStore.getState().queue,
   179→   179→      })
   180→   180→
   181→   181→      // Load + play the audio
   182→   182→      if (audioRef.current) {
   183→   183→        audioRef.current.src = `/api/music/stream/${track.videoId}`
   184→   184→        audioRef.current.volume = useMusicStore.getState().volume
   185→   185→        loadedVideoIdRef.current = track.videoId
   186→   186→        pendingSeekRef.current = null
   187→   187→        try {
   188→   188→          await audioRef.current.play()
   189→   189→          useMusicStore.getState().setIsLoading(false)
   190→   190→        } catch (e: any) {
   191→   191→          useMusicStore.getState().setIsLoading(false)
   192→   192→          if (e?.name === 'NotSupportedError' || e?.name === 'MediaError') {
   193→   193→            try {
   194→   194→              const res = await fetch(`/api/music/stream/${track.videoId}`, {
   195→   195→                method: 'HEAD',
   196→   196→              })
   197→   197→              if (!res.ok) {
   198→   198→                const data = await res.json().catch(() => ({}))
   199→   199→                toast.error(data.error || 'Could not download this track.')
   200→   200→                setCurrentTrack(null)
   201→   201→                setIsPlaying(false)
   202→   202→              }
   203→   203→            } catch {
   204→   204→              toast.error(
   205→   205→                'The server is downloading this track — try again in a moment.',
   206→   206→              )
   207→   207→            }
   208→   208→          }
   209→   209→        }
   210→   210→      }
   211→   211→    },
   212→   212→    [broadcastSync, setCurrentTrack, setIsPlaying, setPosition],
   213→   213→  )
   214→   214→
   215→   215→  // ─── Action: play next track ──────────────────────────────────────────
   216→   216→  const playNext = useCallback(async () => {
   217→   217→    const state = useMusicStore.getState()
   218→   218→    let nextTrack: Track | null = state.popNextFromQueue()
   219→   219→
   220→   220→    // If main queue is empty, try the radio queue (prefetched autoplay tracks)
   221→   221→    if (!nextTrack && state.autoplay && state.currentTrack) {
   222→   222→      nextTrack = state.popFromRadioQueue()
   223→   223→
   224→   224→      // If radio queue is also empty, fetch related tracks (fallback)
   225→   225→      if (!nextTrack) {
   226→   226→        try {
   227→   227→          console.log('[player] radio queue empty, fetching related tracks...')
   228→   228→          const res = await fetch(
   229→   229→            `/api/music/related/${state.currentTrack.videoId}`,
   230→   230→          )
   231→   231→          if (res.ok) {
   232→   232→            const data = await res.json()
   233→   233→            if (data.tracks?.length > 0) {
   234→   234→              // Store all but the first in the radio queue
   235→   235→              state.setRadioQueue(data.tracks.slice(1, 6))
   236→   236→              nextTrack = data.tracks[0] as Track
   237→   237→            }
   238→   238→          }
   239→   239→        } catch {
   240→   240→          console.error('[player] autoplay fetch failed')
   241→   241→        }
   242→   242→      }
   243→   243→    }
   244→   244→
   245→   245→    if (nextTrack) {
   246→   246→      await playTrack(nextTrack)
   247→   247→    } else if (state.repeat && state.currentTrack) {
   248→   248→      await playTrack(state.currentTrack)
   249→   249→    } else {
   250→   250→      setIsPlaying(false)
   251→   251→      setPosition(0)
   252→   252→    }
   253→   253→  }, [playTrack, setIsPlaying, setPosition])
   254→   254→
   255→   255→  // ─── Action: toggle play / pause ──────────────────────────────────────
   256→   256→  const togglePlay = useCallback(() => {
   257→   257→    const audio = audioRef.current
   258→   258→    const state = useMusicStore.getState()
   259→   259→    if (!audio || !state.currentTrack) return
   260→   260→
   261→   261→    if (state.isPlaying) {
   262→   262→      audio.pause()
   263→   263→      setIsPlaying(false)
   264→   264→      broadcastSync({
   265→   265→        state: 'paused',
   266→   266→        position: audio.currentTime,
   267→   267→        queue: state.queue,
   268→   268→      })
   269→   269→    } else {
   270→   270→      audio.play().catch(() => {})
   271→   271→      setIsPlaying(true)
   272→   272→      broadcastSync({
   273→   273→        state: 'playing',
   274→   274→        position: audio.currentTime,
   275→   275→        queue: state.queue,
   276→   276→      })
   277→   277→    }
   278→   278→  }, [broadcastSync, setIsPlaying])
   279→   279→
   280→   280→  // ─── Action: seek ─────────────────────────────────────────────────────
   281→   281→  const handleSeek = useCallback(
   282→   282→    (pos: number) => {
   283→   283→      const audio = audioRef.current
   284→   284→      const state = useMusicStore.getState()
   285→   285→      if (!audio) return
   286→   286→      audio.currentTime = pos
   287→   287→      setPosition(pos)
   288→   288→      broadcastSync({
   289→   289→        state: state.isPlaying ? 'playing' : 'paused',
   290→   290→        position: pos,
   291→   291→        queue: state.queue,
   292→   292→      })
   293→   293→    },
   294→   294→    [broadcastSync, setPosition],
   295→   295→  )
   296→   296→
   297→   297→  // ─── Action: stop playback entirely ───────────────────────────────────
   298→   298→  const stop = useCallback(() => {
   299→   299→    setCurrentTrack(null)
   300→   300→    setIsPlaying(false)
   301→   301→    setPosition(0)
   302→   302→    if (audioRef.current) {
   303→   303→      audioRef.current.pause()
   304→   304→      audioRef.current.src = ''
   305→   305→    }
   306→   306→    loadedVideoIdRef.current = null
   307→   307→  }, [setCurrentTrack, setIsPlaying, setPosition])
   308→   308→
   309→   309→  // ─── Action: add to queue (broadcasts) ────────────────────────────────
   310→   310→  const addToQueue = useCallback(
   311→   311→    (track: Track) => {
   312→   312→      const state = useMusicStore.getState()
   313→   313→      storeAddToQueue(track)
   314→   314→      broadcastSync({
   315→   315→        state: state.isPlaying ? 'playing' : 'paused',
   316→   316→        position: audioRef.current?.currentTime || 0,
   317→   317→        queue: useMusicStore.getState().queue,
   318→   318→      })
   319→   319→    },
   320→   320→    [broadcastSync, storeAddToQueue],
   321→   321→  )
   322→   322→
   323→   323→  // ─── Action: remove from queue (broadcasts) ───────────────────────────
   324→   324→  const removeFromQueue = useCallback(
   325→   325→    (index: number) => {
   326→   326→      const state = useMusicStore.getState()
   327→   327→      storeRemoveFromQueue(index)
   328→   328→      broadcastSync({
   329→   329→        state: state.isPlaying ? 'playing' : 'paused',
   330→   330→        position: audioRef.current?.currentTime || 0,
   331→   331→        queue: useMusicStore.getState().queue,
   332→   332→      })
   333→   333→    },
   334→   334→    [broadcastSync, storeRemoveFromQueue],
   335→   335→  )
   336→   336→
   337→   337→  // ─── Action: clear queue (broadcasts) ─────────────────────────────────
   338→   338→  const clearQueue = useCallback(() => {
   339→   339→    const state = useMusicStore.getState()
   340→   340→    storeClearQueue()
   341→   341→    broadcastSync({
   342→   342→      state: state.isPlaying ? 'playing' : 'paused',
   343→   343→      position: audioRef.current?.currentTime || 0,
   344→   344→      queue: [],
   345→   345→    })
   346→   346→  }, [broadcastSync, storeClearQueue])
   347→   347→
   348→   348→  // ─── Effect: keep <audio> volume in sync with the store ───────────────
   349→   349→  useEffect(() => {
   350→   350→    if (audioRef.current) audioRef.current.volume = volume
   351→   351→  }, [volume])
   352→   352→
   353→   353→  // ─── Effect: wire up <audio> event listeners ──────────────────────────
   354→   354→  // Re-attaches whenever playNext changes (which it does when its deps
   355→   355→  // change). The listeners read fresh state via `useMusicStore.getState()`
   356→   356→  // so they never see stale data.
   357→   357→  useEffect(() => {
   358→   358→    const audio = audioRef.current
   359→   359→    if (!audio) return
   360→   360→
   361→   361→    const onTimeUpdate = () => {
   362→   362→      // Don't fight with a freshly-applied remote position
   363→   363→      if (applyingRemoteRef.current) return
   364→   364→      setPosition(audio.currentTime)
   365→   365→    }
   366→   366→    const onEnded = () => {
   367→   367→      void playNext()
   368→   368→    }
   369→   369→    const onError = () => {
   370→   370→      const state = useMusicStore.getState()
   371→   371→      state.setIsLoading(false)
   372→   372→      if (state.currentTrack) {
   373→   373→        toast.error('Could not play this track — skipping...')
   374→   374→        void playNext()
   375→   375→      }
   376→   376→    }
   377→   377→    const onWaiting = () => {
   378→   378→      useMusicStore.getState().setIsLoading(true)
   379→   379→    }
   380→   380→    const onPlaying = () => {
   381→   381→      useMusicStore.getState().setIsLoading(false)
   382→   382→    }
   383→   383→    const onCanPlay = () => {
   384→   384→      useMusicStore.getState().setIsLoading(false)
   385→   385→    }
   386→   386→
   387→   387→    audio.addEventListener('timeupdate', onTimeUpdate)
   388→   388→    audio.addEventListener('ended', onEnded)
   389→   389→    audio.addEventListener('error', onError)
   390→   390→    audio.addEventListener('waiting', onWaiting)
   391→   391→    audio.addEventListener('playing', onPlaying)
   392→   392→    audio.addEventListener('canplay', onCanPlay)
   393→   393→    return () => {
   394→   394→      audio.removeEventListener('timeupdate', onTimeUpdate)
   395→   395→      audio.removeEventListener('ended', onEnded)
   396→   396→      audio.removeEventListener('error', onError)
   397→   397→      audio.removeEventListener('waiting', onWaiting)
   398→   398→      audio.removeEventListener('playing', onPlaying)
   399→   399→      audio.removeEventListener('canplay', onCanPlay)
   400→   400→    }
   401→   401→  }, [playNext, setPosition])
   402→   402→
   403→   403→  // ─── Effect: Pre-download upcoming queue tracks ───────────────────────
   404→   404→  // When a track starts playing, pre-download the next 2 tracks in the queue
   405→   405→  // so they're ready to play instantly when the user gets to them.
   406→   406→  //
   407→   407→  // The predownload API is fire-and-forget (returns 202 immediately and
   408→   408→  // downloads in the background). We don't await the response — just fire
   409→   409→  // the request and let the server handle it.
   410→   410→  useEffect(() => {
   411→   411→    if (!currentTrack || queue.length === 0) return
   412→   412→    // Pre-download the next 2 tracks (or all if queue has fewer)
   413→   413→    const toPreDownload = queue.slice(0, 2)
   414→   414→    for (const track of toPreDownload) {
   415→   415→      console.log(`[predownload] requesting ${track.videoId} (${track.title})`)
   416→   416→      // Fire-and-forget — the server returns 202 immediately and downloads
   417→   417→      // in the background. We use keepalive so the request survives page
   418→   418→      // navigation, and catch() to prevent unhandled rejection errors.
   419→   419→      fetch(`/api/music/predownload/${track.videoId}`, { method: 'POST', keepalive: true })
   420→   420→        .then((res) => {
   421→   421→          console.log(`[predownload] ${track.videoId} → HTTP ${res.status} (${res.ok ? 'cached/downloading' : 'error'})`)
   422→   422→        })
   423→   423→        .catch((e) => console.error(`[predownload] ${track.videoId} network error:`, e))
   424→   424→    }
   425→   425→  }, [currentTrack, queue])
   426→   426→
   427→   427→  // ─── Effect: Prefetch radio queue + predownload for autoplay ──────────
   428→   428→  // When a track starts playing and the main queue is empty, fetch related
   429→   429→  // tracks ahead of time and store them in the radio queue. Also predownload
   430→   430→  // the first radio queue track so it's ready to play instantly when the
   431→   431→  // current song ends — even if the app is in the background on mobile.
   432→   432→  useEffect(() => {
   433→   433→    if (!currentTrack) return
   434→   434→    const state = useMusicStore.getState()
   435→   435→    if (!state.autoplay) return
   436→   436→    if (queue.length > 0) return // don't prefetch if main queue has items
   437→   437→    if (state.radioQueue.length > 0) {
   438→   438→      // Radio queue already has items — just predownload the first one
   439→   439→      const nextRadio = state.radioQueue[0]
   440→   440→      console.log(`[radio] predownloading next: ${nextRadio.videoId} (${nextRadio.title})`)
   441→   441→      fetch(`/api/music/predownload/${nextRadio.videoId}`, { method: 'POST', keepalive: true })
   442→   442→        .then((res) => console.log(`[radio] predownload ${nextRadio.videoId} → ${res.status}`))
   443→   443→        .catch(() => {})
   444→   444→      return
   445→   445→    }
   446→   446→    // Radio queue is empty — fetch related tracks
   447→   447→    console.log(`[radio] fetching related for ${currentTrack.videoId}...`)
   448→   448→    fetch(`/api/music/related/${currentTrack.videoId}`)
   449→   449→      .then((res) => res.json())
   450→   450→      .then((data) => {
   451→   451→        if (data.tracks?.length > 0) {
   452→   452→          const tracks = data.tracks.slice(0, 5)
   453→   453→          state.setRadioQueue(tracks)
   454→   454→          console.log(`[radio] got ${tracks.length} tracks, predownloading first: ${tracks[0].title}`)
   455→   455→          // Predownload the first track
   456→   456→          fetch(`/api/music/predownload/${tracks[0].videoId}`, { method: 'POST', keepalive: true })
   457→   457→            .then((res) => console.log(`[radio] predownload ${tracks[0].videoId} → ${res.status}`))
   458→   458→            .catch(() => {})
   459→   459→        }
   460→   460→      })
   461→   461→      .catch((e) => console.error('[radio] fetch failed:', e))
   462→   462→  }, [currentTrack, queue.length])
   463→   463→
   464→   464→  // ─── Effect: Socket.io room sync ──────────────────────────────────────
   465→   465→  // Joins the music room's socket channel, requests the host's current
   466→   466→  // state, and listens for incoming sync events. Re-subscribes whenever
   467→   467→  // the active room changes.
   468→   468→  useEffect(() => {
   469→   469→    if (!socket || !activeRoomId) return
   470→   470→
   471→   471→    socket.emit('music:join', activeRoomId)
   472→   472→
   473→   473→    // Ask the host for the current state shortly after joining. The small
   474→   474→    // delay gives the join room propagation time to complete.
   475→   475→    const requestTimer = setTimeout(() => {
   476→   476→      socket.emit('music:request-sync', activeRoomId)
   477→   477→    }, 500)
   478→   478→
   479→   479→    // ── Incoming sync from the host (or another member's broadcast) ──
   480→   480→    const onSync = (data: {
   481→   481→      roomId: string
   482→   482→      state: string
   483→   483→      position: number
   484→   484→      videoId?: string
   485→   485→      trackInfo?: Track
   486→   486→      queue?: Track[]
   487→   487→      serverTimestamp: number
   488→   488→    }) => {
   489→   489→      if (data.roomId !== activeRoomId) return
   490→   490→
   491→   491→      applyingRemoteRef.current = true
   492→   492→
   493→   493→      const networkDelay = (Date.now() - data.serverTimestamp) / 1000
   494→   494→      const adjustedPosition =
   495→   495→        data.position + (data.state === 'playing' ? networkDelay : 0)
   496→   496→
   497→   497→      const state = useMusicStore.getState()
   498→   498→      const audio = audioRef.current
   499→   499→
   500→   500→      // Track changed → load the new stream and seek to host's position
   501→   501→      if (
   502→   502→        data.videoId &&
   503→   503→        (!state.currentTrack || state.currentTrack.videoId !== data.videoId)
   504→   504→      ) {
   505→   505→        const track: Track = data.trackInfo || {
   506→   506→          videoId: data.videoId,
   507→   507→          title: 'Now Playing',
   508→   508→          artist: '',
   509→   509→          thumbnail: null,
   510→   510→          durationSeconds: null,
   511→   511→        }
   512→   512→        pendingSeekRef.current = adjustedPosition
   513→   513→        setCurrentTrack(track)
   514→   514→        setIsPlaying(data.state === 'playing')
   515→   515→        setPosition(adjustedPosition)
   516→   516→
   517→   517→        if (audio) {
   518→   518→          audio.src = `/api/music/stream/${data.videoId}`
   519→   519→          audio.volume = state.volume
   520→   520→          loadedVideoIdRef.current = data.videoId
   521→   521→          // Wait a tick for the new src to be accepted before seeking
   522→   522→          setTimeout(() => {
   523→   523→            if (audioRef.current) {
   524→   524→              try {
   525→   525→                audioRef.current.currentTime = adjustedPosition
   526→   526→              } catch {
   527→   527→                // Stream not seekable yet — ignore
   528→   528→              }
   529→   529→              if (data.state === 'playing') {
   530→   530→                audioRef.current.play().catch(() => {})
   531→   531→              }
   532→   532→            }
   533→   533→            applyingRemoteRef.current = false
   534→   534→          }, 200)
   535→   535→        } else {
   536→   536→          applyingRemoteRef.current = false
   537→   537→        }
   538→   538→      } else if (audio && state.currentTrack) {
   539→   539→        // Same track — sync position + play/pause state only
   540→   540→        const drift = Math.abs(audio.currentTime - adjustedPosition)
   541→   541→        if (drift > 1.5) {
   542→   542→          try {
   543→   543→            audio.currentTime = adjustedPosition
   544→   544→          } catch {
   545→   545→            // ignore
   546→   546→          }
   547→   547→          setPosition(adjustedPosition)
   548→   548→        }
   549→   549→        if (data.state === 'playing' && !state.isPlaying) {
   550→   550→          audio.play().catch(() => {})
   551→   551→          setIsPlaying(true)
   552→   552→        } else if (data.state === 'paused' && state.isPlaying) {
   553→   553→          audio.pause()
   554→   554→          setIsPlaying(false)
   555→   555→        }
   556→   556→        applyingRemoteRef.current = false
   557→   557→      } else {
   558→   558→        applyingRemoteRef.current = false
   559→   559→      }
   560→   560→
   561→   561→      // Apply queue update from remote (broadcast by host)
   562→   562→      if (Array.isArray(data.queue)) {
   563→   563→        setQueue(data.queue)
   564→   564→      }
   565→   565→    }
   566→   566→
   567→   567→    // ── A member just joined and is requesting our current state ──
   568→   568→    // Only the host (or anyone with a current track) responds.
   569→   569→    const onRequestSync = (data: {
   570→   570→      roomId: string
   571→   571→      fromUserId: string
   572→   572→    }) => {
   573→   573→      if (data.roomId !== activeRoomId) return
   574→   574→      const state = useMusicStore.getState()
   575→   575→      if (!state.currentTrack || !socket) return
   576→   576→
   577→   577→      socket.emit('music:sync', {
   578→   578→        roomId: activeRoomId,
   579→   579→        state: state.isPlaying ? 'playing' : 'paused',
   580→   580→        position: audioRef.current?.currentTime || 0,
   581→   581→        videoId: state.currentTrack.videoId,
   582→   582→        trackInfo: {
   583→   583→          title: state.currentTrack.title,
   584→   584→          artist: state.currentTrack.artist,
   585→   585→          thumbnail: state.currentTrack.thumbnail,
   586→   586→          durationSeconds: state.currentTrack.durationSeconds,
   587→   587→        },
   588→   588→        queue: state.queue,
   589→   589→      })
   590→   590→    }
   591→   591→
   592→   592→    socket.on('music:sync', onSync)
   593→   593→    socket.on('music:request-sync', onRequestSync)
   594→   594→
   595→   595→    return () => {
   596→   596→      clearTimeout(requestTimer)
   597→   597→      socket.off('music:sync', onSync)
   598→   598→      socket.off('music:request-sync', onRequestSync)
   599→   599→      socket.emit('music:leave', activeRoomId)
   600→   600→      applyingRemoteRef.current = false
   601→   601→    }
   602→   602→  }, [
   603→   603→    socket,
   604→   604→    activeRoomId,
   605→   605→    setCurrentTrack,
   606→   606→    setIsPlaying,
   607→   607→    setPosition,
   608→   608→    setQueue,
   609→   609→  ])
   610→   610→
   611→   611→  // ─── Bot music control listener ───────────────────────────────────────
   612→   612→  // Listens for 'music:bot-command' socket events emitted by the server when
   613→   613→  // a bot's music_play / music_pause / music_skip / music_queue_add / music_stop
   614→   614→  // node fires. The server forwards the command to the target user's sockets.
   615→   615→  useEffect(() => {
   616→   616→    if (!socket) return
   617→   617→
   618→   618→    const onBotCommand = async (cmd: {
   619→   619→      action: 'play' | 'pause' | 'skip' | 'queue' | 'stop'
   620→   620→      query?: string
   621→   621→    }) => {
   622→   622→      const state = useMusicStore.getState()
   623→   623→
   624→   624→      try {
   625→   625→        switch (cmd.action) {
   626→   626→          case 'play': {
   627→   627→            if (!cmd.query) return
   628→   628→            // Search for the track via the music search API
   629→   629→            const res = await fetch(`/api/music/search?q=${encodeURIComponent(cmd.query)}`)
   630→   630→            if (!res.ok) return
   631→   631→            const data = await res.json()
   632→   632→            const track = data.tracks?.[0]
   633→   633→            if (track) {
   634→   634→              await playTrack(track)
   635→   635→              toast.success(`🎵 Playing: ${track.title}`)
   636→   636→            } else {
   637→   637→              toast.error(`No results for "${cmd.query}"`)
   638→   638→            }
   639→   639→            break
   640→   640→          }
   641→   641→          case 'pause': {
   642→   642→            state.setIsPlaying(false)
   643→   643→            toast.info('⏸️ Music paused')
   644→   644→            break
   645→   645→          }
   646→   646→          case 'skip': {
   647→   647→            const next = state.popNextFromQueue()
   648→   648→            if (next) {
   649→   649→              await playTrack(next)
   650→   650→              toast.success('⏭️ Skipped to next song')
   651→   651→            } else {
   652→   652→              state.stop()
   653→   653→              toast.info('⏭️ Queue is empty')
   654→   654→            }
   655→   655→            break
   656→   656→          }
   657→   657→          case 'queue': {
   658→   658→            if (!cmd.query) return
   659→   659→            const res = await fetch(`/api/music/search?q=${encodeURIComponent(cmd.query)}`)
   660→   660→            if (!res.ok) return
   661→   661→            const data = await res.json()
   662→   662→            const track = data.tracks?.[0]
   663→   663→            if (track) {
   664→   664→              state.addToQueue(track)
   665→   665→              toast.success(`📋 Added to queue: ${track.title}`)
   666→   666→            }
   667→   667→            break
   668→   668→          }
   669→   669→          case 'stop': {
   670→   670→            state.stop()
   671→   671→            toast.info('⏹️ Music stopped')
   672→   672→            break
   673→   673→          }
   674→   674→        }
   675→   675→      } catch (e) {
   676→   676→        console.error('[music:bot-command] failed:', e)
   677→   677→      }
   678→   678→    }
   679→   679→
   680→   680→    socket.on('music:bot-command', onBotCommand)
   681→   681→    return () => {
   682→   682→      socket.off('music:bot-command', onBotCommand)
   683→   683→    }
   684→   684→  }, [socket, playTrack])
   685→   685→
   686→   686→  // ─── Memoized context value ───────────────────────────────────────────
   687→   687→  const contextValue = useMemo<MusicPlayerContextValue>(
   688→   688→    () => ({
   689→   689→      playTrack,
   690→   690→      playNext,
   691→   691→      togglePlay,
   692→   692→      handleSeek,
   693→   693→      stop,
   694→   694→      addToQueue,
   695→   695→      removeFromQueue,
   696→   696→      clearQueue,
   697→   697→    }),
   698→   698→    [
   699→   699→      playTrack,
   700→   700→      playNext,
   701→   701→      togglePlay,
   702→   702→      handleSeek,
   703→   703→      stop,
   704→   704→      addToQueue,
   705→   705→      removeFromQueue,
   706→   706→      clearQueue,
   707→   707→    ],
   708→   708→  )
   709→   709→
   710→   710→  return (
   711→   711→    <MusicPlayerContext.Provider value={contextValue}>
   712→   712→      {children}
   713→   713→      {/* The hidden audio element — persists across all tabs */}
   714→   714→      <audio ref={audioRef} />
   715→   715→      {/* The persistent floating player — collapses to a mini FAB */}
   716→   716→      <AnimatePresence>
   717→   717→        {currentTrack && (
   718→   718→          <PlayerBar
   719→   719→            currentTrack={currentTrack}
   720→   720→            isPlaying={isPlaying}
   721→   721→            isLoading={isLoading}
   722→   722→            position={position}
   723→   723→            volume={volume}
   724→   724→            queue={queue}
   725→   725→            radioQueue={radioQueue}
   726→   726→            shuffle={shuffle}
   727→   727→            repeat={repeat}
   728→   728→            onTogglePlay={togglePlay}
   729→   729→            onSeek={handleSeek}
   730→   730→            onNext={playNext}
   731→   731→            onStop={stop}
   732→   732→            onVolumeChange={setVolume}
   733→   733→            onShuffleToggle={() =>
   734→   734→              useMusicStore.getState().setShuffle(!shuffle)
   735→   735→            }
   736→   736→            onRepeatToggle={() =>
   737→   737→              useMusicStore.getState().setRepeat(!repeat)
   738→   738→            }
   739→   739→            onPlayTrack={playTrack}
   740→   740→          />
   741→   741→        )}
   742→   742→      </AnimatePresence>
   743→   743→    </MusicPlayerContext.Provider>
   744→   744→  )
   745→   745→}
   746→   746→
   747→   747→// ─── Player Bar — Collapsible Floating Mini-Player ───────────────────────
   748→   748→// When collapsed: a small floating circle (FAB) with album art + play/pause.
   749→   749→// When expanded: a full player bar with all controls.
   750→   750→// The player collapses automatically when the user switches away from the
   751→   751→// Music tab, so it doesn't block content on other pages.
   752→   752→
   753→   753→function PlayerBar({

   761→   761→  shuffle,
   762→   762→  repeat,

   764→   764→  onSeek,
   765→   765→  onNext,
   766→   766→  onStop,

   769→   769→  onRepeatToggle,
   770→   770→  onPlayTrack,
   771→   771→}: {

   774→   774→  isLoading: boolean

   789→   789→}) {
   790→   790→  const [expanded, setExpanded] = useState(false)

   909→   909→            >
