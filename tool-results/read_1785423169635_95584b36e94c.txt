    68→    68→  const selectedGroupId = useAppStore((s) => s.selectedGroupId)

    70→    70→  const setFilter = useAppStore((s) => s.setChatFilter)
    71→    71→  const presence = usePresence()
    72→    72→  const { data: unreadData } = useUnreadCounts()
    73→    73→  const { data: session } = useSession()
    74→    74→  const myId = (session?.user as any)?.id
    75→    75→
    76→    76→  const [search, setSearch] = useState('')
    77→    77→
    78→    78→  const { data: groups } = useQuery({
    79→    79→    queryKey: ['channels'],
    80→    80→    queryFn: async () => {
    81→    81→      const res = await fetch('/api/channels')

    85→    85→  })
    86→    86→
    87→    87→  // The currently selected group object (or null for DMs)
    88→    88→  const selectedGroup = groups?.find((g) => g.id === selectedGroupId) || null
    89→    89→  const isViewingDms = selectedGroupId === 'dm'
    90→    90→  const setServerRailOpen = useAppStore((s) => s.setServerRailOpen)
    91→    91→  const ctxMenu = useContextMenu()

    93→    93→
    94→    94→  // Show a context menu for a chat row — supports right-click (desktop) and
    95→    95→  // long-press (mobile, via the custom touch handler).

    97→    97→    let x: number, y: number
    98→    98→    if ('touches' in e) {
    99→    99→      const t = e.touches[0] || (e.changedTouches?.[0])
   100→   100→      if (!t) return
   101→   101→      x = t.clientX; y = t.clientY
   102→   102→    } else {
   103→   103→      x = e.clientX; y = e.clientY
   104→   104→    }
   105→   105→    if (!ctxMenu) return
   106→   106→    ctxMenu.show(x, y, [
   107→   107→      {
   108→   108→        label: 'Open',
   109→   109→        icon: <MessageCircle className="w-4 h-4" />,
   110→   110→        onClick: () => setActiveChannel(row.channel.id),
   111→   111→      },
   112→   112→      {
   113→   113→        label: 'Mark as read',
   114→   114→        icon: <Check className="w-4 h-4" />,
   115→   115→        onClick: () => {
   116→   116→          fetch(`/api/channels/${row.channel.id}/read`, { method: 'POST' }).then(() => {
   117→   117→            qc.invalidateQueries({ queryKey: ['unread-counts'] })
   118→   118→            qc.invalidateQueries({ queryKey: ['channels'] })
   119→   119→            toast.success('Marked as read')
   120→   120→          })
   121→   121→        },
   122→   122→      },
   123→   123→      {
   124→   124→        label: row.isDm ? 'Delete conversation' : 'Leave channel',
   125→   125→        icon: <Trash2 className="w-4 h-4" />,
   126→   126→        variant: 'danger',
   127→   127→        onClick: () => {
   128→   128→          if (row.isDm) {
   129→   129→            if (confirm('Delete this conversation? Messages will be removed for you.')) {
   130→   130→              // For DMs, we leave the channel membership
   131→   131→              fetch(`/api/channels/${row.channel.id}/members`, {
   132→   132→                method: 'DELETE',
   133→   133→              }).then(() => {
   134→   134→                qc.invalidateQueries({ queryKey: ['channels'] })
   135→   135→                if (activeChannelId === row.channel.id) setActiveChannel(null)
   136→   136→                toast.success('Conversation deleted')
   137→   137→              }).catch(() => toast.error('Failed to delete'))
   138→   138→            }
   139→   139→          } else {
   140→   140→            if (confirm(`Leave #${row.channel.name}?`)) {
   141→   141→              fetch(`/api/channels/${row.channel.id}/members`, {
   142→   142→                method: 'DELETE',
   143→   143→              }).then(() => {
   144→   144→                qc.invalidateQueries({ queryKey: ['channels'] })
   145→   145→                if (activeChannelId === row.channel.id) setActiveChannel(null)
   146→   146→                toast.success('Left channel')
   147→   147→              }).catch(() => toast.error('Failed to leave'))
   148→   148→            }
   149→   149→          }
   150→   150→        },
   151→   151→      },
   152→   152→    ])
   153→   153→  }
   154→   154→
   155→   155→  // Join a voice/video channel — creates a call and switches to the voice view

   157→   157→  const { startCall } = useCall()

   159→   159→    mutationFn: async (channelId: string) => {
   160→   160→      const res = await fetch('/api/calls', {
   161→   161→        method: 'POST',
   162→   162→        headers: { 'Content-Type': 'application/json' },
   163→   163→        body: JSON.stringify({ channelId }),
   164→   164→      })
   165→   165→      if (!res.ok) throw new Error('failed')
   166→   166→      return res.json()
   167→   167→    },
   168→   168→    onSuccess: async (data) => {
   169→   169→      await startCall({ callId: data.call.id, channelId: data.call.channelId })
   170→   170→      unlockAudio()
   171→   171→      setView('voice')
   172→   172→      toast.success('Joined channel')
   173→   173→    },
   174→   174→    onError: () => toast.error('Failed to join channel'),
   175→   175→  })
   176→   176→
   177→   177→  // Auto-select first channel on desktop only (mobile shows the list first)
   178→   178→  useEffect(() => {
   179→   179→    if (!activeChannelId && groups && groups.length > 0 && typeof window !== 'undefined' && window.innerWidth >= 1024) {
   180→   180→      const first = groups[0].channels.find((c: any) => c.type === 'text')
   181→   181→      if (first) setActiveChannel(first.id)
   182→   182→    }
   183→   183→  }, [activeChannelId, groups, setActiveChannel])
   184→   184→
   185→   185→  // Flatten channels into a unified list of "chat rows"
   186→   186→  // When a server (non-DM group) is selected in the rail, show ALL channel
   187→   187→  // types (text, voice, video) for that group. When 'dm' is selected, only
   188→   188→  // show DM text channels.
   189→   189→  const allChats: ChatRow[] = useMemo(() => {
   190→   190→    if (!groups) return []
   191→   191→    const rows: ChatRow[] = []
   192→   192→    for (const g of groups) {
   193→   193→      // Filter by selected group from the server rail
   194→   194→      if (selectedGroupId === 'dm' && !g.isDm) continue
   195→   195→      if (selectedGroupId && selectedGroupId !== 'dm' && g.id !== selectedGroupId) continue
   196→   196→      for (const ch of g.channels) {
   197→   197→        // DMs only have text channels. Groups show all types.
   198→   198→        if (g.isDm && ch.type !== 'text') continue
   199→   199→        rows.push({
   200→   200→          channel: ch,
   201→   201→          isDm: g.isDm,
   202→   202→          isGroup: !g.isDm,
   203→   203→          isBot: false,
   204→   204→          partner: g.partner, // populated by the API for DM groups
   205→   205→          groupName: g.name,
   206→   206→          groupIcon: g.iconUrl,
   207→   207→        })
   208→   208→      }
   209→   209→    }
   210→   210→    return rows
   211→   211→  }, [groups, selectedGroupId])
   212→   212→
   213→   213→  // Apply filter
   214→   214→  const filteredChats = useMemo(() => {
   215→   215→    let list = allChats
   216→   216→    if (filter === 'dms') list = list.filter((c) => c.isDm)
   217→   217→    else if (filter === 'groups') list = list.filter((c) => c.isGroup)
   218→   218→    else if (filter === 'bots') list = list.filter((c) => c.isBot)
   219→   219→
   220→   220→    if (search.trim()) {
   221→   221→      const q = search.toLowerCase()
   222→   222→      list = list.filter(
   223→   223→        (c) =>
   224→   224→          c.channel.name.toLowerCase().includes(q) ||
   225→   225→          c.groupName.toLowerCase().includes(q) ||
   226→   226→          (c.partner?.displayName.toLowerCase().includes(q) ?? false) ||
   227→   227→          (c.partner?.username.toLowerCase().includes(q) ?? false)
   228→   228→      )
   229→   229→    }
   230→   230→    return list
   231→   231→  }, [allChats, filter, search])
   232→   232→
   233→   233→  return (
   234→   234→    <div className="flex flex-col h-full w-full bg-sidebar/50 backdrop-blur-2xl">
   235→   235→      {/* Header — shows the selected group name or "Direct Messages" */}
   236→   236→      <div className="px-4 pt-4 pb-3 space-y-4 border-b border-white/5 bg-background/30 backdrop-blur-3xl shadow-sm">
   237→   237→        <div className="flex items-center justify-between">
   238→   238→          <div className="flex items-center gap-2.5 min-w-0">
   239→   239→            {/* Mobile: hamburger to open server rail drawer */}
   240→   240→            <button
   241→   241→              onClick={() => setServerRailOpen(true)}
   242→   242→              className="md:hidden w-9 h-9 rounded-xl bg-sidebar-accent flex items-center justify-center shrink-0 hover:bg-accent transition-colors active:scale-95"
   243→   243→              title="Groups & servers"
   244→   244→            >
   245→   245→              <Menu className="w-4 h-4" />
   246→   246→            </button>
   247→   247→            {isViewingDms ? (
   248→   248→              <>
   249→   249→                <div className="w-9 h-9 rounded-xl bg-primary/15 flex items-center justify-center shrink-0">
   250→   250→                  <MessageCircle className="w-4 h-4 text-primary" />
   251→   251→                </div>
   252→   252→                <div className="min-w-0">
   253→   253→                  <h1 className="text-lg font-semibold tracking-tight truncate">Direct Messages</h1>
   254→   254→                  <p className="text-[11px] text-muted-foreground">Private conversations</p>
   255→   255→                </div>
   256→   256→              </>
   257→   257→            ) : selectedGroup ? (
   258→   258→              <>
   259→   259→                <div className="w-9 h-9 rounded-xl bg-primary/15 flex items-center justify-center shrink-0 overflow-hidden">
   260→   260→                  {selectedGroup.iconUrl ? (
   261→   261→                    <img src={selectedGroup.iconUrl} alt="" className="w-full h-full object-cover" />
   262→   262→                  ) : (
   263→   263→                    <Users className="w-4 h-4 text-primary" />
   264→   264→                  )}
   265→   265→                </div>
   266→   266→                <div className="min-w-0">
   267→   267→                  <h1 className="text-lg font-semibold tracking-tight truncate">{selectedGroup.name}</h1>
   268→   268→                  {selectedGroup.description && (
   269→   269→                    <p className="text-[11px] text-muted-foreground truncate">{selectedGroup.description}</p>
   270→   270→                  )}
   271→   271→                </div>
   272→   272→              </>
   273→   273→            ) : (
   274→   274→              <h1 className="text-2xl font-semibold tracking-tight">Chats</h1>
   275→   275→            )}
   276→   276→          </div>
   277→   277→          <div className="flex gap-1 shrink-0">
   278→   278→            {isViewingDms && <NewDmButton />}
   279→   279→            {!isViewingDms && selectedGroup && <GroupSettingsButton group={selectedGroup} />}
   280→   280→          </div>
   281→   281→        </div>
   282→   282→
   283→   283→        {/* Search */}
   284→   284→        <div className="relative group">
   285→   285→          <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground group-focus-within:text-primary transition-colors" />
   286→   286→          <Input
   287→   287→            value={search}
   288→   288→            onChange={(e) => setSearch(e.target.value)}
   289→   289→            placeholder={isViewingDms ? "Search DMs..." : "Search channels..."}
   290→   290→            className="pl-10 h-10 bg-black/20 border-white/10 focus-visible:ring-primary/30 rounded-xl shadow-inner transition-all"
   291→   291→          />
   292→   292→        </div>
   293→   293→
   294→   294→        {/* Filter chips — only show for DMs */}
   295→   295→        {isViewingDms && (
   296→   296→          <div className="flex gap-2 overflow-x-auto no-scrollbar -mx-4 px-4">
   297→   297→            {([
   298→   298→              ['all', 'All'],
   299→   299→              ['unread', 'Unread'],
   300→   300→            ] as const).map(([key, label]) => (
   301→   301→              <button
   302→   302→                key={key}
   303→   303→                onClick={() => setFilter(key)}
   304→   304→                className={cn(
   305→   305→                  'px-3.5 py-1.5 rounded-full text-xs font-medium whitespace-nowrap transition-colors',
   306→   306→                  filter === key
   307→   307→                    ? 'bg-primary text-primary-foreground'
   308→   308→                    : 'bg-muted text-muted-foreground hover:bg-accent hover:text-foreground'
   309→   309→                )}
   310→   310→              >
   311→   311→                {label}
   312→   312→              </button>
   313→   313→            ))}
   314→   314→          </div>
   315→   315→        )}
   316→   316→      </div>
   317→   317→
   318→   318→      {/* Chat list */}
   319→   319→      <ScrollArea className="flex-1">
   320→   320→        <div className="px-2 pb-4">
   321→   321→          {filteredChats.length === 0 ? (
   322→   322→            <EmptyChatList />
   323→   323→          ) : (
   324→   324→            <>
   325→   325→              <div className="space-y-0.5">
   326→   326→                {filteredChats.map((row) => {
   327→   327→                  const active = row.channel.id === activeChannelId
   328→   328→                  // A DM partner may have been deleted — detect and show a
   329→   329→                  // graceful "Deleted User" label instead of @user fallback.
   330→   330→                  const partnerDeleted = !row.isGroup && !row.partner
   331→   331→                  const presenceInfo = row.partner ? presence[row.partner.id] : null
   332→   332→                  const presenceStatus = presenceInfo?.status || 'offline'
   333→   333→                  const unreadCount = unreadData?.unread?.[row.channel.id] || 0
   334→   334→                  const displayName = row.isGroup
   335→   335→                    ? row.channel.name
   336→   336→                    : row.partner?.displayName || 'Deleted User'
   337→   337→                  const handleText = row.isGroup
   338→   338→                    ? row.groupName
   339→   339→                    : row.partner ? `@${row.partner.username}` : 'account no longer exists'
   340→   340→
   341→   341→                  // Voice/video channels are rendered as "join" rows, not chat rows
   342→   342→                  const isCallChannel = row.channel.type === 'voice' || row.channel.type === 'video'
   343→   343→                  const CallIcon = row.channel.type === 'video' ? Video : Volume2
   344→   344→
   345→   345→                  if (isCallChannel) {
   346→   346→                    return (
   347→   347→                      <button
   348→   348→                        key={row.channel.id}
   349→   349→                        onClick={() => joinCallMutation.mutate(row.channel.id)}
   350→   350→                        disabled={joinCallMutation.isPending}
   351→   351→                        className={cn(
   352→   352→                          'w-full flex items-center gap-3 p-2.5 rounded-xl transition-colors text-left group/call',
   353→   353→                          'hover:bg-accent/50 disabled:opacity-50'
   354→   354→                        )}
   355→   355→                      >
   356→   356→                        {/* Icon */}
   357→   357→                        <div className={cn(
   358→   358→                          'relative w-12 h-12 rounded-full flex items-center justify-center shrink-0',
   359→   359→                          row.channel.type === 'video'
   360→   360→                            ? 'bg-emerald-500/15 text-emerald-400'
   361→   361→                            : 'bg-primary/15 text-primary'
   362→   362→                        )}>
   363→   363→                          <CallIcon className="w-5 h-5" />
   364→   364→                        </div>
   365→   365→
   366→   366→                        {/* Name + status */}
   367→   367→                        <div className="flex-1 min-w-0">
   368→   368→                          <div className="flex items-baseline justify-between gap-2">
   369→   369→                            <span className="truncate text-[15px] font-medium">
   370→   370→                              {row.channel.name}
   371→   371→                            </span>
   372→   372→                          </div>
   373→   373→                          <div className="text-xs text-muted-foreground truncate flex items-center gap-1.5">
   374→   374→                            <span className="capitalize">{row.channel.type} channel</span>
   375→   375→                            <span>·</span>
   376→   376→                            <span className="text-primary flex items-center gap-0.5">
   377→   377→                              <Phone className="w-3 h-3" />
   378→   378→                              Tap to join
   379→   379→                            </span>
   380→   380→                          </div>
   381→   381→                        </div>
   382→   382→
   383→   383→                        {/* Join button */}
   384→   384→                        <div className="w-9 h-9 rounded-full gradient-primary flex items-center justify-center shrink-0 group-hover/call:scale-110 transition-transform shadow-glow">
   385→   385→                          <Phone className="w-4 h-4 text-primary-foreground" />
   386→   386→                        </div>
   387→   387→                      </button>
   388→   388→                    )
   389→   389→                  }
   390→   390→
   391→   391→                  return (
   392→   392→                    <button
   393→   393→                      key={row.channel.id}
   394→   394→                      onClick={() => setActiveChannel(row.channel.id)}
   395→   395→                      onContextMenu={(e) => { e.preventDefault(); showChatContextMenu(e, row) }}
   396→   396→                      onTouchStart={(e) => {
   397→   397→                        // Long-press detection for mobile
   398→   398→                        const touch = e.touches[0]
   399→   399→                        const timer = setTimeout(() => {
   400→   400→                          showChatContextMenu({
   401→   401→                            touches: [{ clientX: touch.clientX, clientY: touch.clientY }],
   402→   402→                          } as any, row)
   403→   403→                          // Trigger haptic feedback if available
   404→   404→                          if (navigator.vibrate) navigator.vibrate(50)
   405→   405→                        }, 500)
   406→   406→                        const cancel = () => clearTimeout(timer)
   407→   407→                        ;(e.currentTarget as HTMLElement).dataset.longPressTimer = String(timer)
   408→   408→                        // One-time listeners for cancellation
   409→   409→                        const el = e.currentTarget
   410→   410→                        const clear = () => {
   411→   411→                          cancel()
   412→   412→                          el.removeEventListener('touchend', clear)
   413→   413→                          el.removeEventListener('touchmove', clear)
   414→   414→                          el.removeEventListener('touchcancel', clear)
   415→   415→                        }
   416→   416→                        el.addEventListener('touchend', clear, { once: true })
   417→   417→                        el.addEventListener('touchmove', clear, { once: true })
   418→   418→                        el.addEventListener('touchcancel', clear, { once: true })
   419→   419→                      }}
   420→   420→                      className={cn(
   421→   421→                        'w-full flex items-center gap-3.5 p-3 rounded-2xl transition-all text-left select-none relative overflow-hidden',
   422→   422→                        active 
   423→   423→                          ? 'bg-primary/10 ring-1 ring-primary/20 shadow-[inset_0_1px_1px_rgba(255,255,255,0.05)]' 
   424→   424→                          : 'hover:bg-white/[0.04]'
   425→   425→                      )}
   426→   426→                    >
   427→   427→                      {/* Avatar */}
   428→   428→                      {row.isGroup ? (
   429→   429→                        <div className="relative w-12 h-12 rounded-full bg-primary/15 flex items-center justify-center shrink-0">
   430→   430→                          <Hash className="w-5 h-5 text-primary" />
   431→   431→                        </div>
   432→   432→                      ) : partnerDeleted ? (
   433→   433→                        <div className="relative shrink-0">
   434→   434→                          <div className="w-12 h-12 rounded-full bg-muted flex items-center justify-center text-muted-foreground">
   435→   435→                            <UserX className="w-5 h-5" />
   436→   436→                          </div>
   437→   437→                        </div>
   438→   438→                      ) : (
