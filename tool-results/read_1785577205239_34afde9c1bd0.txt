     1→/**
     2→ * Visual Bot Builder — Types & Execution Engine (v2)
     3→ *
     4→ * Architecture
     5→ * ────────────
     6→ * A bot flow is a directed graph stored as JSON in `Bot.flow`:
     7→ *   { nodes: FlowNode[], edges: FlowEdge[] }
     8→ *
     9→ * The engine walks the graph starting from a TRIGGER node. Crucially, it
    10→ * supports **pause/resume** for INPUT and CHOICE nodes — when the engine
    11→ * reaches one, it stops, persists `{ currentNodeId, variables }` to
    12→ * ConversationSession, and resumes on the user's next message.
    13→ *
    14→ * Node categories (used by the UI for palette grouping):
    15→ *   - trigger  : starts the flow (1 type, multiple subtypes)
    16→ *   - output   : bot sends something (message, choice, typing)
    17→ *   - input    : wait for user reply (text input, choice input)
    18→ *   - logic    : control flow (condition, set_var, delay, stop)
    19→ *   - advanced : power features (api_call, random)
    20→ */
    21→

    78→  // ── trigger ──

    81→

    85→  // ── wait_choice ──
    86→  prompt?: string
    87→  options?: string[] // quick-reply choices

    89→
    90→  // ── typing ──
    91→  seconds?: number
    92→
    93→  // ── input ──
    94→  // (uses prompt + variableName from above)
    95→
    96→  // ── condition ──
    97→  variable?: string
    98→  operator?: 'equals' | 'not_equals' | 'contains' | 'starts_with' | 'ends_with' | 'exists' | 'not_exists' | 'regex_match' | 'regex_not_match' | 'greater_than' | 'less_than' | 'in_array' | 'not_in_array'
    99→  value?: string
   100→
   101→  // ── set_var ──
   102→  // (uses variable + value from above)
   103→
   104→  // ── api_call ──
   105→  url?: string
   106→  method?: 'GET' | 'POST'
   107→  headers?: string // JSON string
   108→  body?: string
   109→  // (also uses variableName to store response)
   110→
   111→  // ── random ──
   112→  // (no data — uses N outgoing edges; engine picks one)
   113→
   114→  // ── ai_generate ──
   115→  /** The prompt to send to the LLM. Supports {{var}} interpolation. */
   116→  aiPrompt?: string
   117→  /** Optional system prompt to set the LLM's persona/behavior. */
   118→  aiSystemPrompt?: string
   119→  /** Ollama model name, e.g. 'gemma3:270m'. */
   120→  aiModel?: string
   121→  /** Sampling temperature (0 = deterministic, 1 = creative). */
   122→  aiTemperature?: number
   123→  /** Max tokens to generate. */
   124→  aiMaxTokens?: number
   125→  // (also uses variableName to store the generated response)
   126→
   127→  // ── switch_case ──
   128→  /** Variable to switch on. */
   129→  switchVariable?: string
   130→  /** Case values — each gets its own outgoing handle (case_0, case_1, ...). */
   131→  cases?: string[]
   132→  // (uses 'default' handle for unmatched values)
   133→
   134→  // ── counter ──
   135→  // (uses `variable` to hold the counter name)
   136→  /** Amount to increment by (negative = decrement). */
   137→  increment?: number
   138→  /** Starting value when the variable doesn't exist yet. */
   139→  startValue?: number
   140→
   141→  // ── format_string ──
   142→  // (uses `text` as the template with {{var}} placeholders)
   143→  // (uses `variableName` to store the result)
   144→
   145→  // ── send_media ──
   146→  /** URL of the media to send (image/video/audio). */
   147→  mediaUrl?: string
   148→  /** Caption text. Supports {{var}} interpolation. */
   149→  caption?: string
   150→  /** Media type: image, video, audio. */
   151→  mediaType?: 'image' | 'video' | 'audio'
   152→
   153→  // ── log ──
   154→  /** Message to log. Supports {{var}} interpolation. */
   155→  logMessage?: string
   156→  /** Log level. */
   157→  logLevel?: 'info' | 'warn' | 'error'
   158→
   159→  // ── tts ──
   160→  /** Text to speak. Supports {{var}} interpolation. */
   161→  ttsText?: string
   162→  /** Voice name (alba, charles, etc.) or custom voice ID. */
   163→  ttsVoice?: string
   164→  // (sends the audio as a voice message — no variable needed)
   165→
   166→  // ── asr_transcribe ──
   167→  /** URL of the audio file to transcribe. Supports {{var}} interpolation.
   168→   *  Typically set to {{mediaUrl}} which the bot framework populates with
   169→   *  the incoming message's mediaUrl when the user sends a voice message. */
   170→  asrAudioUrl?: string
   171→  /** Language hint (currently ignored — Moonshine v1 is English-only). */
   172→  asrLanguage?: string
   173→  /** Whether to also send the transcript as a reply message. If false,
   174→   *  the transcript is only stored in the variable. */
   175→  asrReply?: boolean
   176→  // (uses `variableName` to store the transcript text)
   177→
   178→  // ── regex_extract ──
   179→  /** The regex pattern to match against the input. Uses JavaScript RegExp syntax. */
   180→  regexPattern?: string
   181→  /** Flags for the regex (e.g. 'i' for case-insensitive, 'g' for global). */
   182→  regexFlags?: string
   183→  /** The input text to match against. Supports {{var}} interpolation. */
   184→  regexInput?: string
   185→  // (uses `variableName` to store the first match, or matched group 1 if present)
   186→
   187→  // ── json_parse ──
   188→  /** The JSON string to parse. Supports {{var}} interpolation — typically {{apiResult}}. */
   189→  jsonInput?: string
   190→  /** JSON path to extract (dot-notation, e.g. 'data.user.name' or 'choices[0].message.content'). */
   191→  jsonPath?: string
   192→  // (uses `variableName` to store the extracted value as a string)
   193→
   194→  // ── comment ──
   195→  /** Comment/note text — shown on the canvas, ignored by the engine. */
   196→  commentText?: string
   197→  /** Comment color (for visual categorization). */
   198→  commentColor?: 'yellow' | 'green' | 'blue' | 'pink' | 'gray'
   199→
   200→  // ── UI metadata (not used by engine) ──

   203→   *  n.type to 'custom' for all custom nodes. */
   204→  type?: NodeType
   205→  label?: string

   264→  /** Show typing indicator — best-effort, no-op if unsupported. */

   291→export type TraceEvent =
   292→  | { type: 'flow_start'; timestamp: number; triggerNodeId?: string; input: { body: string; sender: string; command?: string } }
   293→  | { type: 'node_enter'; timestamp: number; nodeId: string; nodeType: NodeType; nodeLabel: string }
   294→  | { type: 'node_exit'; timestamp: number; nodeId: string; durationMs: number }
   295→  | { type: 'message_sent'; timestamp: number; nodeId: string; text: string }
   296→  | { type: 'variable_set'; timestamp: number; nodeId: string; variable: string; value: string }
   297→  | { type: 'condition_eval'; timestamp: number; nodeId: string; variable: string; value: string; operator: string; compareTo: string; result: boolean }
   298→  | { type: 'branch_taken'; timestamp: number; nodeId: string; handle: string; targetNodeId: string }
   299→  | { type: 'ai_call'; timestamp: number; nodeId: string; model: string; prompt: string; responseLength: number; durationMs: number }
   300→  | { type: 'api_call'; timestamp: number; nodeId: string; url: string; method: string; status: number; durationMs: number }
   301→  | { type: 'asr_call'; timestamp: number; nodeId: string; audioUrl: string; transcriptLength: number; durationMs: number; success: boolean }
   302→  | { type: 'paused'; timestamp: number; nodeId: string; variableName: string }
   303→  | { type: 'resumed'; timestamp: number; nodeId: string; inputText: string }
   304→  | { type: 'error'; timestamp: number; nodeId?: string; message: string }
   305→  | { type: 'flow_end'; timestamp: number; reason: 'stop' | 'no_edge' | 'max_steps' | 'error'; sentCount: number }
   306→  | { type: 'log'; timestamp: number; nodeId: string; level: 'info' | 'warn' | 'error'; message: string }

   312→const MAX_STEPS = 50
   313→
   314→/**
   315→ * Resume descriptor — when set, the engine resumes from `nodeId` instead of
   316→ * searching for a trigger. The caller passes this when a paused session
   317→ * receives its awaited reply.
   318→ */
   319→export interface ResumeDescriptor {
   320→  nodeId: string
   321→  /** The text to assign to the input variable (usually the user's reply). */
   322→  inputText: string
   323→  /** The variable name to assign inputText to. */
   324→  inputVariable: string
   325→}

   327→/**
   328→ * Replace {{var}} placeholders in a string.
   329→ */
   330→function interpolate(text: string, ctx: BotExecutionContext): string {
   331→  let out = text
   332→  for (const [k, v] of Object.entries(ctx.variables)) {
   333→    out = out.replace(new RegExp(`\\{\\{${k}\\}\\}`, 'g'), v)
   334→  }
   335→  out = out.replace(/\{\{sender\}\}/g, ctx.senderName)
   336→  out = out.replace(/\{\{args\}\}/g, ctx.args.join(' '))
   337→  out = out.replace(/\{\{body\}\}/g, ctx.body)
   338→  // Convenience: expose the incoming message's media URL/type and any
   339→  // pre-existing ASR transcript as {{mediaUrl}}, {{mediaType}}, {{transcript}}.
   340→  // These are also available as ctx.variables (set by visual.ts at flow start)
   341→  // but the explicit fallbacks here make them work even in debug mode.
   342→  out = out.replace(/\{\{mediaUrl\}\}/g, ctx.variables.__mediaUrl || '')
   343→  out = out.replace(/\{\{mediaType\}\}/g, ctx.variables.__mediaType || '')
   344→  // IMPORTANT: prefer the user-set 'transcript' variable (from an explicit
   345→  // asr_transcribe node) over the auto-set '__transcript' (from the incoming
   346→  // message's auto-transcription). The auto-transcript may be stale or empty,
   347→  // and the user's explicit ASR node should take precedence.
   348→  out = out.replace(/\{\{transcript\}\}/g, ctx.variables.transcript || ctx.variables.__transcript || '')
   349→  // Also expose the detected message type (set by the message_type node)
   350→  out = out.replace(/\{\{messageType\}\}/g, ctx.variables.__messageType || '')
   351→  return out
   352→}

   354→/**
   355→ * Execute a bot flow. Returns when:
   356→ *   - the flow reaches a `stop` node,
   357→ *   - the flow ends naturally (no outgoing edge),
   358→ *   - the flow pauses at an INPUT/WAIT_CHOICE node,
   359→ *   - MAX_STEPS is exceeded.
   360→ *
   361→ * Emits trace events for debugging — each node entry/exit, variable change,
   362→ * message sent, condition evaluation, branch taken, and error is recorded.
   363→ * The caller can pass these to the UI for a test-run visualization.
   364→ */
   365→export async function executeBotFlow(
   366→  flow: BotFlow,
   367→  ctx: BotExecutionContext,
   368→  resume?: ResumeDescriptor
   369→): Promise<BotExecutionResult> {
   370→  let sentCount = 0
   371→  let steps = 0
   372→  let currentNode: FlowNode | undefined
   373→  const trace: TraceEvent[] = []
   374→  const now = () => Date.now()
   375→
   376→  // ── Helpers (declared early so the resume block can use them) ─────────
   377→  const edgesFrom = (nodeId: string, handle?: string | null) =>
   378→    flow.edges.filter((e) => {
   379→      if (e.source !== nodeId) return false
   380→      if (handle === undefined) return true
   381→      if (handle === null) return !e.sourceHandle
   382→      return e.sourceHandle === handle
   383→    })
   384→
   385→  const followEdge = (nodeId: string, handle?: string | null): FlowNode | undefined => {
   386→    const edges = edgesFrom(nodeId, handle)
   387→    if (edges.length === 0) return undefined
   388→    return flow.nodes.find((n) => n.id === edges[0].target)
   389→  }
   390→
   391→  // ── Resolve starting node ─────────────────────────────────────────────
   392→  if (resume) {
   393→    const inputNode = flow.nodes.find((n) => n.id === resume.nodeId)
   394→    if (!inputNode) {
   395→      trace.push({ type: 'error', timestamp: now(), message: 'resume target not found' })
   396→      return { sentCount: 0, paused: false, variables: ctx.variables, trace, error: 'resume target not found' }
   397→    }
   398→    ctx.variables[resume.inputVariable] = resume.inputText
   399→    trace.push({ type: 'resumed', timestamp: now(), nodeId: inputNode.id, inputText: resume.inputText })
   400→    trace.push({ type: 'variable_set', timestamp: now(), nodeId: inputNode.id, variable: resume.inputVariable, value: resume.inputText })
   401→    currentNode = followEdge(inputNode.id, null)
   402→    if (!currentNode) {
   403→      trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   404→      return { sentCount, paused: false, variables: ctx.variables, trace }
   405→    }
   406→  } else {
   407→    const trigger = flow.nodes.find((n) => n.type === 'trigger')
   408→    if (!trigger) {
   409→      trace.push({ type: 'error', timestamp: now(), message: 'No trigger node found' })
   410→      return { sentCount: 0, paused: false, variables: ctx.variables, trace, error: 'No trigger node found' }
   411→    }
   412→
   413→    trace.push({
   414→      type: 'flow_start',
   415→      timestamp: now(),
   416→      triggerNodeId: trigger.id,
   417→      input: { body: ctx.body, sender: ctx.senderName, command: ctx.command },
   418→    })
   419→
   420→    // Trigger matching
   421→    const tt = trigger.data.triggerType || 'any_message'
   422→    if (tt === 'command' && trigger.data.command) {
   423→      if (!ctx.command || ctx.command !== trigger.data.command.replace(/^\//, '')) {
   424→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   425→        return { sentCount: 0, paused: false, variables: ctx.variables, trace }
   426→      }
   427→    } else if (tt === 'mention') {
   428→      if (!ctx.isMention) {
   429→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   430→        return { sentCount: 0, paused: false, variables: ctx.variables, trace }
   431→      }
   432→    }
   433→
   434→    currentNode = trigger
   435→  }
   436→
   437→  // ── Walk the graph ────────────────────────────────────────────────────
   438→  while (currentNode && steps < MAX_STEPS) {
   439→    steps++
   440→    const nodeStartTime = now()
   441→    const nodeLabel = currentNode.data?.label || currentNode.type
   442→
   443→    trace.push({
   444→      type: 'node_enter',
   445→      timestamp: nodeStartTime,
   446→      nodeId: currentNode.id,
   447→      nodeType: currentNode.type,
   448→      nodeLabel,
   449→    })
   450→
   451→    try {
   452→      switch (currentNode.type) {
   453→        // ── TRIGGER ────────────────────────────────────────────────────────
   454→        case 'trigger':
   455→          currentNode = followEdge(currentNode.id, null)
   456→          break
   457→
   458→      // ── OUTPUT: message ────────────────────────────────────────────────
   459→      case 'message': {
   460→        const text = interpolate(currentNode.data.text || '', ctx)
   461→        if (text.trim()) {
   462→          await ctx.reply(text)
   463→          sentCount++
   464→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text })
   465→        }
   466→        currentNode = followEdge(currentNode.id, null)
   467→        break
   468→      }
   469→
   470→      // ── OUTPUT: send_media ─────────────────────────────────────────────
   471→      case 'send_media': {
   472→        const url = interpolate(currentNode.data.mediaUrl || '', ctx)
   473→        const caption = interpolate(currentNode.data.caption || '', ctx)
   474→        if (url.trim()) {
   475→          // ctx.reply is text-only; we log it but the actual media send
   476→          // would need an extended reply API. For now, send the URL as text.
   477→          const text = caption ? `${caption}\n${url}` : url
   478→          await ctx.reply(text)
   479→          sentCount++
   480→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text })
   481→        }
   482→        currentNode = followEdge(currentNode.id, null)
   483→        break
   484→      }
   485→
   486→      // ── OUTPUT: typing ─────────────────────────────────────────────────
   487→      case 'typing': {
   488→        const seconds = currentNode.data.seconds || 1
   489→        if (ctx.setTyping) await ctx.setTyping(seconds)
   490→        else await new Promise((r) => setTimeout(r, Math.min(seconds, 2) * 1000))
   491→        currentNode = followEdge(currentNode.id, null)
   492→        break
   493→      }
   494→
   495→      // ── INPUT: pause and wait for next message ────────────────────────
   496→      case 'input': {
   497→        const prompt = interpolate(currentNode.data.prompt || '', ctx)
   498→        if (prompt.trim()) {
   499→          await ctx.reply(prompt)
   500→          sentCount++
   501→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text: prompt })
   502→        }
   503→        trace.push({ type: 'paused', timestamp: now(), nodeId: currentNode.id, variableName: currentNode.data.variableName || 'reply' })
   504→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   505→        return {
   506→          sentCount,
   507→          paused: true,
   508→          pausedAtNodeId: currentNode.id,
   509→          variables: ctx.variables,
   510→          trace,
   511→        }
   512→      }
   513→
   514→      // ── INPUT: wait_choice ────────────────────────────────────────────
   515→      case 'wait_choice': {
   516→        const prompt = interpolate(currentNode.data.prompt || '', ctx)
   517→        const options = currentNode.data.options || []
   518→        // Build a Telegram-style inline keyboard — each option is a button.
   519→        // The button's callbackData is the option text itself, so the resume
   520→        // path in visual.ts can match it directly.
   521→        const keyboard: BotKeyboardButton[][] = options.map((opt) => [{
   522→          text: opt,
   523→          callbackData: opt,
   524→        }])
   525→
   526→        // Telegram-style behavior: if we have an existing keyboard message
   527→        // (from a previous pause at this node, i.e. a loop-back), EDIT it
   528→        // in-place instead of sending a new message. This prevents the bot
   529→        // from spamming new keyboard messages every time the user clicks a
   530→        // button that loops back to this menu.
   531→        //
   532→        // The keyboardMessageId is set in the session state by visual.ts
   533→        // when it persists the pause. We read it from ctx.variables under
   534→        // a reserved key.
   535→        const existingKeyboardMsgId = ctx.variables['__keyboardMsgId']
   536→
   537→        if (prompt.trim()) {
   538→          if (existingKeyboardMsgId && ctx.editMessage) {
   539→            // Edit-in-place: update the existing keyboard message
   540→            await ctx.editMessage(existingKeyboardMsgId, prompt, keyboard)
   541→            trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text: `[edited] ${prompt}` })
   542→          } else {
   543→            // Fresh keyboard: send a new message
   544→            const msgId = await ctx.reply(prompt, keyboard)
   545→            sentCount++
   546→            // Store the message ID so future re-pauses can edit it
   547→            ctx.variables['__keyboardMsgId'] = msgId
   548→            trace.push({ type: 'message_sent', timestamp: now(), nodeId: currentNode.id, text: prompt })
   549→          }
   550→        }
   551→        trace.push({ type: 'paused', timestamp: now(), nodeId: currentNode.id, variableName: currentNode.data.variableName || 'choice' })
   552→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   553→        return {
   554→          sentCount,
   555→          paused: true,
   556→          pausedAtNodeId: currentNode.id,
   557→          variables: ctx.variables,
   558→          trace,
   559→        }
   560→      }
   561→
   562→      // ── LOGIC: condition ──────────────────────────────────────────────
   563→      case 'condition': {
   564→        const varName = currentNode.data.variable || ''
   565→        const varValue = ctx.variables[varName] || ''
   566→        const op = currentNode.data.operator || 'exists'
   567→        // Interpolate the comparison value so {{sender}}, {{body}}, {{mediaType}},
   568→        // {{transcript}} etc. work in condition comparisons.
   569→        const cmp = interpolate(currentNode.data.value || '', ctx)
   570→
   571→        let met = false
   572→        switch (op) {
   573→          case 'equals':       met = varValue === cmp; break
   574→          case 'not_equals':   met = varValue !== cmp; break
   575→          case 'contains':     met = varValue.includes(cmp); break
   576→          case 'starts_with':  met = varValue.startsWith(cmp); break
   577→          case 'ends_with':    met = varValue.endsWith(cmp); break
   578→          case 'exists':       met = !!varValue; break
   579→          case 'not_exists':   met = !varValue; break
   580→          case 'regex_match': {
   581→            try { met = new RegExp(cmp).test(varValue) } catch { met = false }
   582→            break
   583→          }
   584→          case 'regex_not_match': {
   585→            try { met = !new RegExp(cmp).test(varValue) } catch { met = false }
   586→            break
   587→          }
   588→          case 'greater_than': {
   589→            const a = parseFloat(varValue), b = parseFloat(cmp)
   590→            met = !isNaN(a) && !isNaN(b) && a > b
   591→            break
   592→          }
   593→          case 'less_than': {
   594→            const a = parseFloat(varValue), b = parseFloat(cmp)
   595→            met = !isNaN(a) && !isNaN(b) && a < b
   596→            break
   597→          }
   598→          case 'in_array': {
   599→            // cmp is a comma-separated list, e.g. "yes,yep,ok,confirm"
   600→            const items = cmp.split(',').map((s) => s.trim()).filter(Boolean)
   601→            met = items.includes(varValue)
   602→            break
   603→          }
   604→          case 'not_in_array': {
   605→            const items = cmp.split(',').map((s) => s.trim()).filter(Boolean)
   606→            met = !items.includes(varValue)
   607→            break
   608→          }
   609→        }
   610→
   611→        trace.push({
   612→          type: 'condition_eval',
   613→          timestamp: now(),
   614→          nodeId: currentNode.id,
   615→          variable: varName,
   616→          value: varValue,
   617→          operator: op,
   618→          compareTo: cmp,
   619→          result: met,
   620→        })
   621→
   622→        const handle = met ? 'true' : 'false'
   623→        const next = followEdge(currentNode.id, handle)
   624→        if (next) {
   625→          trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle, targetNodeId: next.id })
   626→          currentNode = next
   627→        } else {
   628→          // No matching branch — fall through to default edge if any
   629→          const fallback = followEdge(currentNode.id, null)
   630→          if (fallback) {
   631→            trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: 'default', targetNodeId: fallback.id })
   632→            currentNode = fallback
   633→          } else {
   634→            trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   635→            currentNode = undefined
   636→          }
   637→        }
   638→        break
   639→      }
   640→
   641→      // ── LOGIC: message_type (route by message type) ──────────────────
   642→      case 'message_type': {
   643→        // Determine the message type from mediaType + mediaUrl + body
   644→        const mediaType = ctx.variables.__mediaType || ''
   645→        const mediaUrl = ctx.variables.__mediaUrl || ''
   646→        const body = ctx.body || ''
   647→
   648→        let detectedType: string
   649→        if (mediaType.startsWith('audio')) {
   650→          // Distinguish voice recordings from audio file uploads:
   651→          // - voice recordings: mediaType is 'audio/webm', 'audio/ogg', 'audio/mp4' (from MediaRecorder)
   652→          // - audio files: mediaType is 'audio' (TTS-generated) or 'audio/mpeg', 'audio/flac' (uploaded)
   653→          // Heuristic: 'audio/webm', 'audio/ogg', 'audio/mp4' → voice; 'audio' or 'audio/mpeg' → audio
   654→          if (mediaType === 'audio/webm' || mediaType === 'audio/ogg' || mediaType === 'audio/mp4' || mediaType === 'audio/m4a') {
   655→            detectedType = 'voice'
   656→          } else {
   657→            detectedType = 'audio'
   658→          }
   659→        } else if (mediaType.startsWith('image')) {
   660→          detectedType = 'image'
   661→        } else if (mediaType.startsWith('video')) {
   662→          detectedType = 'video'
   663→        } else if (mediaUrl && !body) {
   664→          // Has a media URL but no text body and not audio/image/video → treat as file
   665→          detectedType = 'file'
   666→        } else if (body && !mediaUrl) {
   667→          detectedType = 'text'
   668→        } else if (mediaUrl && body) {
   669→          // Text + media → treat as text (the body is the caption)
   670→          detectedType = 'text'
   671→        } else {
   672→          detectedType = 'other'
   673→        }
   674→
   675→        // Expose the detected type as a variable for downstream nodes
   676→        ctx.variables.__messageType = detectedType
   677→
   678→        trace.push({
   679→          type: 'branch_taken',
   680→          timestamp: now(),
   681→          nodeId: currentNode.id,
   682→          handle: detectedType,
   683→          targetNodeId: followEdge(currentNode.id, detectedType)?.id || '',
   684→        })
   685→
   686→        const next = followEdge(currentNode.id, detectedType)
   687→        if (next) {
   688→          currentNode = next
   689→        } else {
   690→          // No edge for this type — fall through to 'other' or null
   691→          const fallback = followEdge(currentNode.id, 'other') || followEdge(currentNode.id, null)
   692→          if (fallback) {
   693→            trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: 'fallback', targetNodeId: fallback.id })
   694→            currentNode = fallback
   695→          } else {
   696→            trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   697→            currentNode = undefined
   698→          }
   699→        }
   700→        break
   701→      }
   702→
   703→      // ── LOGIC: switch_case (multi-branch) ─────────────────────────────
   704→      case 'switch_case': {
   705→        const varName = currentNode.data.switchVariable || ''
   706→        const varValue = ctx.variables[varName] || ''
   707→        // Interpolate each case value so {{var}} placeholders work
   708→        const cases = (currentNode.data.cases || []).map((c) => interpolate(c, ctx))
   709→
   710→        const matchIdx = cases.findIndex((c) => c === varValue)
   711→        const handle = matchIdx >= 0 ? `case_${matchIdx}` : 'default'
   712→
   713→        trace.push({
   714→          type: 'condition_eval',
   715→          timestamp: now(),
   716→          nodeId: currentNode.id,
   717→          variable: varName,
   718→          value: varValue,
   719→          operator: 'switch',
   720→          compareTo: matchIdx >= 0 ? cases[matchIdx] : '(default)',
   721→          result: matchIdx >= 0,
   722→        })
   723→
   724→        const next = followEdge(currentNode.id, handle)
   725→        if (next) {
   726→          trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle, targetNodeId: next.id })
   727→          currentNode = next
   728→        } else {
   729→          const fallback = followEdge(currentNode.id, null)
   730→          if (fallback) {
   731→            trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: 'default', targetNodeId: fallback.id })
   732→            currentNode = fallback
   733→          } else {
   734→            trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
   735→            currentNode = undefined
   736→          }
   737→        }
   738→        break
   739→      }
   740→
   741→      // ── LOGIC: set_var ────────────────────────────────────────────────
   742→      case 'set_var': {
   743→        const name = currentNode.data.variable || ''
   744→        const val = interpolate(currentNode.data.value || '', ctx)
   745→        if (name) {
   746→          ctx.variables[name] = val
   747→          trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: name, value: val })
   748→        }
   749→        currentNode = followEdge(currentNode.id, null)
   750→        break
   751→      }
   752→
   753→      // ── LOGIC: counter (increment/decrement a numeric variable) ───────
   754→      case 'counter': {
   755→        const name = currentNode.data.variable || ''
   756→        const inc = currentNode.data.increment ?? 1
   757→        const startVal = currentNode.data.startValue ?? 0
   758→        const current = parseInt(ctx.variables[name] || '', 10)
   759→        const newVal = isNaN(current) ? startVal + inc : current + inc
   760→        ctx.variables[name] = String(newVal)
   761→        trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: name, value: String(newVal) })
   762→        currentNode = followEdge(currentNode.id, null)
   763→        break
   764→      }
   765→
   766→      // ── LOGIC: format_string (template a string with variables) ──────
   767→      case 'format_string': {
   768→        const template = currentNode.data.text || ''
   769→        const result = interpolate(template, ctx)
   770→        const vname = currentNode.data.variableName
   771→        if (vname) {
   772→          ctx.variables[vname] = result
   773→          trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: result })
   774→        }
   775→        currentNode = followEdge(currentNode.id, null)
   776→        break
   777→      }
   778→
   779→      // ── LOGIC: regex_extract ──────────────────────────────────────────
   780→      case 'regex_extract': {
   781→        const pattern = currentNode.data.regexPattern || ''
   782→        const flags = currentNode.data.regexFlags || 'i'
   783→        const input = interpolate(currentNode.data.regexInput || '{{body}}', ctx)
   784→        const vname = currentNode.data.variableName || 'match'
   785→
   786→        if (!pattern) {
   787→          trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: 'Regex node has empty pattern' })
   788→          ctx.variables[vname] = ''
   789→          currentNode = followEdge(currentNode.id, null)
   790→          break
   791→        }
   792→
   793→        let matched = ''
   794→        try {
   795→          const re = new RegExp(pattern, flags)
   796→          const m = re.exec(input)
   797→          if (m) {
   798→            // If the regex has capture groups, return group 1; otherwise return the full match
   799→            matched = m[1] !== undefined ? m[1] : m[0]
   800→          }
   801→        } catch (e: any) {
   802→          trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `Invalid regex: ${e?.message || e}` })
   803→        }
   804→
   805→        ctx.variables[vname] = matched
   806→        trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: matched.slice(0, 100) })
   807→        currentNode = followEdge(currentNode.id, null)
   808→        break
   809→      }
   810→
   811→      // ── LOGIC: json_parse ─────────────────────────────────────────────
   812→      case 'json_parse': {
   813→        const input = interpolate(currentNode.data.jsonInput || '{{apiResult}}', ctx)
   814→        const path = currentNode.data.jsonPath || ''
   815→        const vname = currentNode.data.variableName || 'jsonValue'
   816→
   817→        if (!input.trim()) {
   818→          trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: 'JSON node has empty input' })
   819→          ctx.variables[vname] = ''
   820→          currentNode = followEdge(currentNode.id, null)
   821→          break
   822→        }
   823→
   824→        let result = ''
   825→        try {
   826→          const parsed = JSON.parse(input)
   827→          if (path) {
   828→            // Navigate dot-notation path with array index support
   829→            // e.g. 'data.user.name' or 'choices[0].message.content'
   830→            const parts = path.split(/\.|\[(\d+)\]/).filter(Boolean)
   831→            let cur: any = parsed
   832→            for (const part of parts) {
   833→              if (cur == null) break
   834→              cur = cur[part]
   835→            }
   836→            result = cur == null ? '' : (typeof cur === 'object' ? JSON.stringify(cur) : String(cur))
   837→          } else {
   838→            result = typeof parsed === 'object' ? JSON.stringify(parsed) : String(parsed)
   839→          }
   840→        } catch (e: any) {
   841→          trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `JSON parse failed: ${e?.message || e}` })
   842→        }
   843→
   844→        ctx.variables[vname] = result
   845→        trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: result.slice(0, 100) })
   846→        currentNode = followEdge(currentNode.id, null)
   847→        break
   848→      }
   849→
   850→      // ── LOGIC: comment (visual note, no execution) ───────────────────
   851→      case 'comment': {
   852→        // Comments are purely visual — they don't do anything when the flow
   853→        // runs. Just pass through to the next node.
   854→        currentNode = followEdge(currentNode.id, null)
   855→        break
   856→      }
   857→
   858→      // ── LOGIC: delay ──────────────────────────────────────────────────
   859→      case 'delay': {
   860→        const s = Math.max(0, Math.min(60, currentNode.data.seconds || 1))
   861→        await new Promise((r) => setTimeout(r, s * 1000))
   862→        currentNode = followEdge(currentNode.id, null)
   863→        break
   864→      }
   865→
   866→      // ── LOGIC: log (debug logging — shows in trace, no user output) ───
   867→      case 'log': {
   868→        const msg = interpolate(currentNode.data.logMessage || '', ctx)
   869→        const level = currentNode.data.logLevel || 'info'
   870→        trace.push({ type: 'log', timestamp: now(), nodeId: currentNode.id, level, message: msg })
   871→        currentNode = followEdge(currentNode.id, null)
   872→        break
   873→      }
   874→
   875→      // ── LOGIC: stop ───────────────────────────────────────────────────
   876→      case 'stop':
   877→        trace.push({ type: 'flow_end', timestamp: now(), reason: 'stop', sentCount })
   878→        return { sentCount, paused: false, variables: ctx.variables, trace }
   879→
   880→      // ── OUTPUT: tts (Pocket TTS voice message) ────────────────────────
   881→      case 'tts': {
   882→        const ttsStart = now()
   883→        const ttsNode = currentNode
   884→        const text = interpolate(ttsNode.data.ttsText || '', ctx)
   885→        const voice = ttsNode.data.ttsVoice || 'alba'
   886→
   887→        // Show typing indicator while TTS generates (can take 2-10s)
   888→        if (ctx.setTyping) await ctx.setTyping(10)
   889→
   890→        if (!text.trim()) {
   891→          trace.push({ type: 'error', timestamp: now(), nodeId: ttsNode.id, message: 'TTS node has empty text' })
   892→          currentNode = followEdge(ttsNode.id, null)
   893→          break
   894→        }
   895→
   896→        trace.push({ type: 'log', timestamp: now(), nodeId: ttsNode.id, level: 'info', message: `TTS: generating "${text.slice(0, 60)}…" with voice=${voice}` })
   897→
   898→        let mediaUrl: string | null = null
   899→        if (ctx.generateTTS) {
   900→          try {
   901→            mediaUrl = await ctx.generateTTS(text, voice)
   902→          } catch (e: any) {
   903→            trace.push({ type: 'error', timestamp: now(), nodeId: ttsNode.id, message: `TTS failed: ${e?.message || e}` })
   904→          }
   905→        } else {
   906→          // Debug mode — no actual TTS available
   907→          trace.push({ type: 'error', timestamp: now(), nodeId: ttsNode.id, message: 'TTS not available in debug mode' })
   908→        }
   909→
   910→        if (mediaUrl) {
   911→          // Send as a voice message
   912→          if (ctx.replyWithMedia) {
   913→            await ctx.replyWithMedia(mediaUrl, 'audio')
   914→          } else {
   915→            await ctx.reply(`🔊 ${mediaUrl}`)
   916→          }
   917→          sentCount++
   918→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: ttsNode.id, text: `[voice message: ${text.slice(0, 60)}…]` })
   919→          trace.push({ type: 'ai_call', timestamp: now(), nodeId: ttsNode.id, model: `tts:${voice}`, prompt: text.slice(0, 100), responseLength: 0, durationMs: now() - ttsStart })
   920→        } else {
   921→          // Fallback: send the text as a plain message
   922→          await ctx.reply(`🔊 ${text}`)
   923→          sentCount++
   924→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: ttsNode.id, text: `[TTS fallback] ${text.slice(0, 80)}` })
   925→        }
   926→        currentNode = followEdge(ttsNode.id, null)
   927→        break
   928→      }
   929→
   930→      // ── ADVANCED: asr_transcribe (Moonshine ASR voice-to-text) ───────
   931→      case 'asr_transcribe': {
   932→        const asrStart = now()
   933→        const asrNode = currentNode
   934→        const audioUrl = interpolate(asrNode.data.asrAudioUrl || '{{mediaUrl}}', ctx)
   935→        const language = asrNode.data.asrLanguage || 'en'
   936→        const shouldReply = asrNode.data.asrReply !== false // default true
   937→        const vname = asrNode.data.variableName || 'transcript'
   938→
   939→        // Show typing indicator while ASR transcribes (can take 1-15s)
   940→        if (ctx.setTyping) await ctx.setTyping(15)
   941→
   942→        if (!audioUrl || !audioUrl.trim()) {
   943→          trace.push({ type: 'error', timestamp: now(), nodeId: asrNode.id, message: 'ASR node has no audio URL (set asrAudioUrl or send a voice message)' })
   944→          currentNode = followEdge(asrNode.id, null)
   945→          break
   946→        }
   947→
   948→        trace.push({ type: 'log', timestamp: now(), nodeId: asrNode.id, level: 'info', message: `ASR: transcribing "${audioUrl.slice(0, 60)}…" (lang=${language})` })
   949→
   950→        let transcript: string | null = null
   951→        if (ctx.transcribeAudio) {
   952→          try {
   953→            transcript = await ctx.transcribeAudio(audioUrl, language)
   954→          } catch (e: any) {
   955→            trace.push({ type: 'error', timestamp: now(), nodeId: asrNode.id, message: `ASR failed: ${e?.message || e}` })
   956→          }
   957→        } else {
   958→          // Debug mode — no actual ASR available
   959→          trace.push({ type: 'error', timestamp: now(), nodeId: asrNode.id, message: 'ASR not available in debug mode' })
   960→        }
   961→
   962→        const transcriptText = transcript?.trim() || ''
   963→        const success = !!transcriptText
   964→
   965→        // Store in variable (even if empty — downstream condition nodes can check `exists`)
   966→        ctx.variables[vname] = transcriptText
   967→        trace.push({ type: 'variable_set', timestamp: now(), nodeId: asrNode.id, variable: vname, value: transcriptText.slice(0, 100) })
   968→        trace.push({
   969→          type: 'asr_call',
   970→          timestamp: now(),
   971→          nodeId: asrNode.id,
   972→          audioUrl: audioUrl.slice(0, 100),
   973→          transcriptLength: transcriptText.length,
   974→          durationMs: now() - asrStart,
   975→          success,
   976→        })
   977→
   978→        // Optionally send the transcript as a reply (default behavior)
   979→        if (shouldReply && transcriptText) {
   980→          await ctx.reply(`📝 ${transcriptText}`)
   981→          sentCount++
   982→          trace.push({ type: 'message_sent', timestamp: now(), nodeId: asrNode.id, text: `[transcript] ${transcriptText.slice(0, 80)}` })
   983→        } else if (shouldReply && !transcriptText) {
   984→          await ctx.reply('📝 (transcription unavailable — no speech detected or ASR server down)')
   985→          sentCount++
   986→        }
   987→
   988→        currentNode = followEdge(asrNode.id, null)
   989→        break
   990→      }
   991→
   992→      // ── ADVANCED: api_call ────────────────────────────────────────────
   993→      case 'api_call': {
   994→        const callStart = now()
   995→        // Show typing indicator while the API call runs (can take 1-30s)
   996→        if (ctx.setTyping) await ctx.setTyping(15)
   997→        try {
   998→          const url = interpolate(currentNode.data.url || '', ctx)
   999→          const method = currentNode.data.method || 'GET'
  1000→          const headers = currentNode.data.headers ? JSON.parse(currentNode.data.headers) : {}
  1001→          const body = currentNode.data.body ? interpolate(currentNode.data.body, ctx) : undefined
  1002→          const res = await fetch(url, { method, headers, body })
  1003→          const text = await res.text()
  1004→          const vname = currentNode.data.variableName
  1005→          if (vname) {
  1006→            ctx.variables[vname] = text.slice(0, 4000)
  1007→            trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: text.slice(0, 200) + (text.length > 200 ? '…' : '') })
  1008→          }
  1009→          trace.push({ type: 'api_call', timestamp: now(), nodeId: currentNode.id, url, method, status: res.status, durationMs: now() - callStart })
  1010→        } catch (e: any) {
  1011→          trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `API call failed: ${e?.message || e}` })
  1012→        }
  1013→        currentNode = followEdge(currentNode.id, null)
  1014→        break
  1015→      }
  1016→
  1017→      // ── ADVANCED: random ──────────────────────────────────────────────
  1018→      case 'random': {
  1019→        const allEdges = edgesFrom(currentNode.id, undefined)
  1020→        if (allEdges.length === 0) {
  1021→          trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
  1022→          currentNode = undefined
  1023→          break
  1024→        }
  1025→        const pick = allEdges[Math.floor(Math.random() * allEdges.length)]
  1026→        const target = flow.nodes.find((n) => n.id === pick.target)
  1027→        if (target) {
  1028→          trace.push({ type: 'branch_taken', timestamp: now(), nodeId: currentNode.id, handle: pick.sourceHandle || 'random', targetNodeId: target.id })
  1029→          currentNode = target
  1030→        } else {
  1031→          currentNode = undefined
  1032→        }
  1033→        break
  1034→      }
  1035→
  1036→      // ── ADVANCED: ai_generate ──────────────────────────────────────────
  1037→      case 'ai_generate': {
  1038→        const vname = currentNode.data.variableName || 'aiResponse'
  1039→        const callStart = now()
  1040→        // Show typing indicator while the LLM generates (can take 5-30s)
  1041→        if (ctx.setTyping) await ctx.setTyping(30)
  1042→        try {
  1043→          const model = currentNode.data.aiModel || 'gemma3:270m'
  1044→          const prompt = interpolate(currentNode.data.aiPrompt || '', ctx)
  1045→          const systemPrompt = currentNode.data.aiSystemPrompt
  1046→            ? interpolate(currentNode.data.aiSystemPrompt, ctx)
  1047→            : undefined
  1048→          const temperature = currentNode.data.aiTemperature ?? 0.7
  1049→          const maxTokens = currentNode.data.aiMaxTokens ?? 256
  1050→
  1051→          const ollamaUrl = process.env.OLLAMA_URL || 'http://localhost:11434'
  1052→          const res = await fetch(`${ollamaUrl}/api/generate`, {
  1053→            method: 'POST',
  1054→            headers: { 'Content-Type': 'application/json' },
  1055→            body: JSON.stringify({
  1056→              model, prompt, system: systemPrompt, stream: false,
  1057→              options: { temperature, num_predict: maxTokens },
  1058→            }),
  1059→          })
  1060→
  1061→          if (!res.ok) {
  1062→            const errText = await res.text().catch(() => 'unknown error')
  1063→            const errMsg = `Ollama error ${res.status}: ${errText.slice(0, 100)}`
  1064→            ctx.variables[vname] = `[AI error: ${res.status}]`
  1065→            trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: errMsg })
  1066→            trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: ctx.variables[vname] })
  1067→            trace.push({ type: 'ai_call', timestamp: now(), nodeId: currentNode.id, model, prompt: prompt.slice(0, 100), responseLength: 0, durationMs: now() - callStart })
  1068→          } else {
  1069→            const data = await res.json()
  1070→            const generated: string = data.response || ''
  1071→            ctx.variables[vname] = generated.slice(0, 4000)
  1072→            trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: generated.slice(0, 200) + (generated.length > 200 ? '…' : '') })
  1073→            trace.push({ type: 'ai_call', timestamp: now(), nodeId: currentNode.id, model, prompt: prompt.slice(0, 100), responseLength: generated.length, durationMs: now() - callStart })
  1074→          }
  1075→        } catch (e: any) {
  1076→          ctx.variables[vname] = `[AI error: ${e?.message || 'unavailable'}]`
  1077→          trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `AI generate failed: ${e?.message || e}` })
  1078→          trace.push({ type: 'variable_set', timestamp: now(), nodeId: currentNode.id, variable: vname, value: ctx.variables[vname] })
  1079→        }
  1080→        currentNode = followEdge(currentNode.id, null)
  1081→        break
  1082→      }
  1083→
  1084→      default:
  1085→        trace.push({ type: 'error', timestamp: now(), nodeId: currentNode.id, message: `unknown node type: ${currentNode.type}` })
  1086→        return { sentCount, paused: false, variables: ctx.variables, trace, error: `unknown node type: ${currentNode.type}` }
  1087→      }
  1088→    } catch (e: any) {
  1089→      // Top-level catch — any node throwing would otherwise abort the entire
  1090→      // flow silently. Log the error, try to reply to the user so they know
  1091→      // something went wrong, and stop the flow gracefully.
  1092→      const errMsg = e?.message || String(e)
  1093→      const errNodeId = currentNode?.id || ''
  1094→      trace.push({ type: 'error', timestamp: now(), nodeId: errNodeId, message: `Node "${nodeLabel}" threw: ${errMsg}` })
  1095→      try {
  1096→        await ctx.reply(`⚠️ Bot error in "${nodeLabel}": ${errMsg.slice(0, 200)}`)
  1097→        sentCount++
  1098→      } catch {
  1099→        // reply also failed — nothing more we can do
  1100→      }
  1101→      trace.push({ type: 'flow_end', timestamp: now(), reason: 'error', sentCount })
  1102→      return { sentCount, paused: false, variables: ctx.variables, trace, error: `Node ${errNodeId} threw: ${errMsg}` }
  1103→    }
  1104→
  1105→    // Record node exit
  1106→    if (currentNode) {
  1107→      trace.push({ type: 'node_exit', timestamp: now(), nodeId: currentNode.id, durationMs: 0 })
  1108→    }
  1109→  }
  1110→
  1111→  if (steps >= MAX_STEPS) {
  1112→    await ctx.reply('⚠️ Bot flow exceeded maximum steps (possible infinite loop).')
  1113→    trace.push({ type: 'error', timestamp: now(), message: 'max steps exceeded' })
  1114→    trace.push({ type: 'flow_end', timestamp: now(), reason: 'max_steps', sentCount })
  1115→    return { sentCount, paused: false, variables: ctx.variables, trace, error: 'max steps' }
  1116→  }
  1117→
  1118→  trace.push({ type: 'flow_end', timestamp: now(), reason: 'no_edge', sentCount })
  1119→  return { sentCount, paused: false, variables: ctx.variables, trace }
  1120→}
