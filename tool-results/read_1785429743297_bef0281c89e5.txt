    48→    48→// ─── Custom Node Component ───────────────────────────────────────────────────
    49→    49→/**
    50→    50→ * Reads the FlowNode type from `data.type` (NOT the ReactFlow `type` prop,
    51→    51→ * which is always 'custom' in this app). This was the root cause of every
    52→    52→ * node rendering as a green "Send Message" node.
    53→    53→ */
    54→    54→function CustomNode({ data, selected }: { data: any; selected?: boolean }) {
    55→    55→  const nodeType = (data.type as NodeType) || 'message'
    56→    56→  const def = NODE_DEFS[nodeType] || NODE_DEFS.message
    57→    57→  const Icon = ICONS[def.icon] || Zap
    58→    58→
    59→    59→  const isTrigger = nodeType === 'trigger'
    60→    60→  const isStop = nodeType === 'stop'
    61→    61→  const isCondition = nodeType === 'condition'
    62→    62→  const isRandom = nodeType === 'random'
    63→    63→
    64→    64→  // ── Build preview content per node type ──────────────────────────────
    65→    65→  let preview: { label: string; value: string } | null = null
    66→    66→  switch (nodeType) {
    67→    67→    case 'trigger': {
    68→    68→      const tt = data.triggerType || 'any_message'
    69→    69→      const label =
    70→    70→        tt === 'command' ? `/${data.command || 'command'}` :
    71→    71→        tt === 'mention' ? '@mentioned' :
    72→    72→        'any message'
    73→    73→      preview = { label: 'When', value: label }
    74→    74→      break
    75→    75→    }
    76→    76→    case 'message':
    77→    77→      preview = { label: 'Says', value: data.text || '(empty)' }
    78→    78→      break
    79→    79→    case 'typing':
    80→    80→      preview = { label: 'For', value: `${data.seconds || 1}s` }
    81→    81→      break
    82→    82→    case 'input':
    83→    83→      preview = { label: 'Asks', value: data.prompt || '(empty)' }
    84→    84→      break
    85→    85→    case 'wait_choice':
    86→    86→      preview = { label: 'Picks from', value: (data.options || []).join(' · ') || '(none)' }
    87→    87→      break
    88→    88→    case 'condition':
    89→    89→      preview = { label: 'If', value: `${data.variable || '∅'} ${data.operator || 'exists'} ${data.value || ''}` }
    90→    90→      break
    91→    91→    case 'set_var':
    92→    92→      preview = { label: 'Sets', value: `${data.variable || '∅'} = ${data.value || '∅'}` }
    93→    93→      break
    94→    94→    case 'delay':
    95→    95→      preview = { label: 'Wait', value: `${data.seconds || 1}s` }
    96→    96→      break
    97→    97→    case 'stop':
    98→    98→      preview = null
    99→    99→      break
   100→   100→    case 'api_call':
   101→   101→      preview = { label: 'Calls', value: `${data.method || 'GET'} ${data.url || '(no url)'}` }
   102→   102→      break
   103→   103→    case 'random':
   104→   104→      preview = { label: 'Picks', value: 'a random branch' }
   105→   105→      break
   106→   106→  }
   107→   107→
   108→   108→  return (
   109→   109→    <div
   110→   110→      className="bg-[#2b2d31] rounded-xl min-w-[200px] max-w-[260px] shadow-lg border overflow-hidden transition-shadow"
   111→   111→      style={{
   112→   112→        borderColor: selected ? def.color : 'rgba(255,255,255,0.08)',
   113→   113→        boxShadow: selected ? `0 0 0 2px ${def.color}40` : undefined,
   114→   114→      }}
   115→   115→    >
   116→   116→      {/* Target handle (not for trigger — it has no incoming edges) */}
   117→   117→      {!isTrigger && (
   118→   118→        <Handle
   119→   119→          type="target"
   120→   120→          position={Position.Top}
   121→   121→          style={{ background: def.color, width: 10, height: 10, border: 'none' }}
   122→   122→        />
   123→   123→      )}
   124→   124→
   125→   125→      {/* Header */}
   126→   126→      <div
   127→   127→        className="flex items-center gap-2 px-3 py-2 border-b"
   128→   128→        style={{ background: def.bg, borderColor: `${def.color}30` }}
   129→   129→      >
   130→   130→        <div
   131→   131→          className="w-6 h-6 rounded-md flex items-center justify-center shrink-0"
   132→   132→          style={{ background: def.color + '30' }}
   133→   133→        >
   134→   134→          <Icon className="w-3.5 h-3.5" style={{ color: def.color }} />
   135→   135→        </div>
   136→   136→        <span
   137→   137→          className="text-[11px] font-bold uppercase tracking-wider flex-1"
   138→   138→          style={{ color: def.color }}
   139→   139→        >
   140→   140→          {def.label}
   141→   141→        </span>
   142→   142→        {def.pauses && (
   143→   143→          <span className="text-[9px] uppercase tracking-wider px-1.5 py-0.5 rounded bg-white/10 text-white/60">
   144→   144→            pause
   145→   145→          </span>
   146→   146→        )}
   147→   147→        {def.terminates && (
   148→   148→          <span className="text-[9px] uppercase tracking-wider px-1.5 py-0.5 rounded bg-white/10 text-white/60">
   149→   149→            end
   150→   150→          </span>
   151→   151→        )}
   152→   152→      </div>
   153→   153→
   154→   154→      {/* Body */}
   155→   155→      <div className="px-3 py-2 text-xs">
   156→   156→        {preview ? (
   157→   157→          <div className="space-y-1">
   158→   158→            <div className="text-white/40 uppercase tracking-wider text-[10px]">{preview.label}</div>
   159→   159→            <div className="text-white/80 break-words line-clamp-2">{preview.value}</div>
   160→   160→            {(nodeType === 'input' || nodeType === 'wait_choice' || nodeType === 'api_call') && data.variableName && (
   161→   161→              <div className="text-[10px] text-white/40">→ saves as <code className="text-white/60">{`{{${data.variableName}}}`}</code></div>
   162→   162→            )}
   163→   163→          </div>
   164→   164→        ) : (
   165→   165→          <div className="text-white/50 italic text-center py-1">{def.description}</div>
   166→   166→        )}
   167→   167→      </div>
   168→   168→
   169→   169→      {/* Source handles */}
   170→   170→      {isCondition ? (
   171→   171→        <div className="flex justify-between px-3 pb-2 pt-1">
   172→   172→          <div className="flex flex-col items-center">
   173→   173→            <span className="text-[10px] font-semibold text-emerald-400 mb-1">TRUE</span>
   174→   174→            <Handle
   175→   175→              type="source"
   176→   176→              position={Position.Bottom}
   177→   177→              id="true"
   178→   178→              style={{ background: '#34D399', width: 10, height: 10, border: 'none' }}
   179→   179→            />
   180→   180→          </div>
   181→   181→          <div className="flex flex-col items-center">
   182→   182→            <span className="text-[10px] font-semibold text-red-400 mb-1">FALSE</span>
   183→   183→            <Handle
   184→   184→              type="source"
   185→   185→              position={Position.Bottom}
   186→   186→              id="false"
   187→   187→              style={{ background: '#F87171', width: 10, height: 10, border: 'none' }}
   188→   188→            />
   189→   189→          </div>
   190→   190→        </div>
   191→   191→      ) : isRandom ? (
   192→   192→        <div className="px-3 pb-2 pt-1 text-center">
   193→   193→          <span className="text-[10px] text-white/40 mb-1 block">All outgoing edges are random picks</span>
   194→   194→          <Handle
   195→   195→            type="source"
   196→   196→            position={Position.Bottom}
   197→   197→            style={{ background: def.color, width: 10, height: 10, border: 'none' }}
   198→   198→          />
   199→   199→        </div>
   200→   200→      ) : !isStop ? (
   201→   201→        <Handle
   202→   202→          type="source"
   203→   203→          position={Position.Bottom}
   204→   204→          style={{ background: def.color, width: 10, height: 10, border: 'none' }}
   205→   205→        />
   206→   206→      ) : null}
   207→   207→    </div>
   208→   208→  )
   209→   209→}

   213→   213→// ─── Bot Builder Editor ──────────────────────────────────────────────────────
   214→   214→
   215→   215→interface BotBuilderEditorProps {
   216→   216→  initialFlow?: BotFlow
   217→   217→  onSave: (flow: BotFlow) => void
   218→   218→}
   219→   219→
   220→   220→export function BotBuilderEditor({ initialFlow, onSave }: BotBuilderEditorProps) {
   221→   221→  const [selectedNodeId, setSelectedNodeId] = useState<string | null>(null)
   222→   222→  const [selectedEdgeId, setSelectedEdgeId] = useState<string | null>(null)
   223→   223→
   224→   224→  // ── Initial nodes/edges ──────────────────────────────────────────────
   225→   225→  // FIX: previously the .map() overwrote data.type with ReactFlow's type
   226→   226→  // ('custom'), so the inspector crashed on click. Now we use data.type
   227→   227→  // everywhere and never let n.type leak into data.
   228→   228→  const { initialNodes, initialEdges } = useMemo(() => {
   229→   229→    if (initialFlow?.nodes?.length) {
   230→   230→      return {
   231→   231→        initialNodes: initialFlow.nodes.map((n) => ({
   232→   232→          id: n.id,
   233→   233→          type: 'custom',
   234→   234→          position: n.position,
   235→   235→          data: { ...n.data },
   236→   236→        })) as Node[],
   237→   237→        initialEdges: initialFlow.edges.map((e) => ({
   238→   238→          id: e.id,
   239→   239→          source: e.source,
   240→   240→          target: e.target,
   241→   241→          sourceHandle: e.sourceHandle || undefined,
   242→   242→          label: e.label || undefined,
   243→   243→          animated: true,
   244→   244→          style: { stroke: '#5865F2', strokeWidth: 2 },
   245→   245→          markerEnd: { type: MarkerType.ArrowClosed, color: '#5865F2' },
   246→   246→        })) as Edge[],
   247→   247→      }
   248→   248→    }
   249→   249→    // Brand-new bot: seed with a sensible default flow
   250→   250→    const triggerData = defaultNodeData('trigger')
   251→   251→    const msgData = defaultNodeData('message')
   252→   252→    return {
   253→   253→      initialNodes: [
   254→   254→        { id: 'trigger-1', type: 'custom', position: { x: 320, y: 80 }, data: { ...triggerData, type: 'trigger' } },
   255→   255→        { id: 'message-1', type: 'custom', position: { x: 320, y: 240 }, data: { ...msgData, type: 'message' } },
   256→   256→      ] as Node[],
   257→   257→      initialEdges: [
   258→   258→        {
   259→   259→          id: 'e-trigger-1-message-1',
   260→   260→          source: 'trigger-1',
   261→   261→          target: 'message-1',
   262→   262→          animated: true,
   263→   263→          style: { stroke: '#5865F2', strokeWidth: 2 },
   264→   264→          markerEnd: { type: MarkerType.ArrowClosed, color: '#5865F2' },
   265→   265→        },
   266→   266→      ] as Edge[],
   267→   267→    }
   268→   268→  }, [initialFlow])
   269→   269→
   270→   270→  const [nodes, setNodes, onNodesChange] = useNodesState(initialNodes)
   271→   271→  const [edges, setEdges, onEdgesChange] = useEdgesState(initialEdges)
   272→   272→
   273→   273→  const onConnect = useCallback(
   274→   274→    (params: Connection) =>
   275→   275→      setEdges((eds) =>
   276→   276→        addEdge(
   277→   277→          {
   278→   278→            ...params,
   279→   279→            animated: true,
   280→   280→            style: { stroke: '#5865F2', strokeWidth: 2 },
   281→   281→            markerEnd: { type: MarkerType.ArrowClosed, color: '#5865F2' },
   282→   282→          },
   283→   283→          eds
   284→   284→        )
   285→   285→      ),
   286→   286→    [setEdges]
   287→   287→  )
   288→   288→
   289→   289→  // ── Add node — position near the canvas center, offset by count ──────
   290→   290→  const addNode = (type: NodeType) => {
   291→   291→    const id = `${type}-${Date.now()}`
   292→   292→    const data = { ...defaultNodeData(type), type }
   293→   293→    // Place near the top-center with a slight cascade so it's always visible
   294→   294→    const idx = nodes.length
   295→   295→    const position = {
   296→   296→      x: 240 + (idx % 4) * 90,
   297→   297→      y: 80 + idx * 30,
   298→   298→    }
   299→   299→    setNodes((nds) => [...nds, { id, type: 'custom', position, data }])
   300→   300→    setSelectedNodeId(id)
   301→   301→  }
   302→   302→
   303→   303→  const updateNodeData = (nodeId: string, patch: Record<string, any>) =>
   304→   304→    setNodes((nds) =>
   305→   305→      nds.map((n) => (n.id === nodeId ? { ...n, data: { ...n.data, ...patch } } : n))
   306→   306→    )
   307→   307→
   308→   308→  const deleteNode = (nodeId: string) => {
   309→   309→    setNodes((nds) => nds.filter((n) => n.id !== nodeId))
   310→   310→    setEdges((eds) => eds.filter((e) => e.source !== nodeId && e.target !== nodeId))
   311→   311→    setSelectedNodeId(null)
   312→   312→  }
   313→   313→
   314→   314→  // Disconnect an edge — used by the edge context menu and the Delete key.
   315→   315→  const deleteEdge = (edgeId: string) => {
   316→   316→    setEdges((eds) => eds.filter((e) => e.id !== edgeId))
   317→   317→    setSelectedEdgeId(null)
   318→   318→  }
   319→   319→
   320→   320→  // Keyboard shortcuts: Delete/Backspace deletes the selected node or edge.
   321→   321→  // ReactFlow's built-in `deleteKeyCode` only handles nodes, so we wire up
   322→   322→  // edges manually here.
   323→   323→  const onKeyDown = useCallback(
   324→   324→    (e: React.KeyboardEvent) => {
   325→   325→      if (e.key !== 'Delete' && e.key !== 'Backspace') return
   326→   326→      // Don't intercept if the user is typing in an input/textarea
   327→   327→      const target = e.target as HTMLElement
   328→   328→      if (target.tagName === 'INPUT' || target.tagName === 'TEXTAREA' || target.isContentEditable) return
   329→   329→      if (selectedNodeId) {
   330→   330→        e.preventDefault()
   331→   331→        deleteNode(selectedNodeId)
   332→   332→      } else if (selectedEdgeId) {
   333→   333→        e.preventDefault()
   334→   334→        deleteEdge(selectedEdgeId)
   335→   335→      }
   336→   336→    },
   337→   337→    [selectedNodeId, selectedEdgeId] // eslint-disable-line react-hooks/exhaustive-deps
   338→   338→  )
   339→   339→
   340→   340→  const clearAll = () => {
   341→   341→    if (!confirm('Clear the entire flow? This cannot be undone.')) return
   342→   342→    setNodes([])
   343→   343→    setEdges([])
   344→   344→    setSelectedNodeId(null)
   345→   345→  }
   346→   346→
   347→   347→  const handleSave = () => {
   348→   348→    const flow: BotFlow = {
   349→   349→      nodes: nodes.map((n) => ({
   350→   350→        id: n.id,
   351→   351→        type: n.data.type as NodeType,
   352→   352→        position: n.position,
   353→   353→        data: { ...n.data },
   354→   354→      })) as FlowNode[],
   355→   355→      edges: edges.map((e) => ({
   356→   356→        id: e.id,
   357→   357→        source: e.source,
   358→   358→        target: e.target,
   359→   359→        sourceHandle: e.sourceHandle || null,
   360→   360→        label: typeof e.label === 'string' ? e.label : undefined,
   361→   361→      })),
   362→   362→    }
   363→   363→    onSave(flow)
   364→   364→  }
   365→   365→
   366→   366→  const selectedNode = nodes.find((n) => n.id === selectedNodeId) || null
   367→   367→
   368→   368→  // ── Validation warnings ──────────────────────────────────────────────
   369→   369→  const warnings = useMemo(() => {
   370→   370→    const w: string[] = []
   371→   371→    const triggerCount = nodes.filter((n) => n.data.type === 'trigger').length
   372→   372→    if (triggerCount === 0) w.push('No trigger node — the bot will never start.')
   373→   373→    if (triggerCount > 1) w.push('Multiple trigger nodes — only the first one will be used.')
   374→   374→
   375→   375→    // Orphaned nodes (no incoming or outgoing edges, except trigger)
   376→   376→    const connected = new Set<string>()
   377→   377→    edges.forEach((e) => { connected.add(e.source); connected.add(e.target) })
   378→   378→    const orphans = nodes.filter((n) => n.data.type !== 'trigger' && !connected.has(n.id))
   379→   379→    if (orphans.length > 0) w.push(`${orphans.length} node(s) are not connected to the flow.`)
   380→   380→
   381→   381→    return w
   382→   382→  }, [nodes, edges])
   383→   383→
   384→   384→  return (
   385→   385→    <div className="flex h-full bg-[#1e1f22]">
   386→   386→      {/* ─── LEFT: node palette ────────────────────────────────────────── */}
   387→   387→      <div className="w-56 shrink-0 bg-[#2b2d31] border-r border-white/5 flex flex-col">
   388→   388→        <div className="p-3 border-b border-white/5">
   389→   389→          <div className="flex items-center gap-2">
   390→   390→            <Sparkles className="w-4 h-4 text-primary" />
   391→   391→            <h3 className="text-sm font-semibold text-white/80">Flow Builder</h3>
   392→   392→          </div>
   393→   393→          <p className="text-[11px] text-white/40 mt-1">Drag or click a node to add it to the canvas.</p>
   394→   394→        </div>
   395→   395→
   396→   396→        <div className="flex-1 overflow-y-auto p-3 space-y-4">
   397→   397→          {CATEGORY_ORDER.map((cat) => {
   398→   398→            const nodesInCat = Object.values(NODE_DEFS).filter((d) => d.category === cat)
   399→   399→            if (nodesInCat.length === 0) return null
   400→   400→            return (
   401→   401→              <div key={cat}>
   402→   402→                <h4 className="text-[10px] font-bold uppercase tracking-wider text-white/40 mb-2 px-1">
   403→   403→                  {CATEGORY_LABELS[cat]}
   404→   404→                </h4>
   405→   405→                <div className="space-y-1">
   406→   406→                  {nodesInCat.map((def) => {
   407→   407→                    const Icon = ICONS[def.icon] || Zap
   408→   408→                    return (
   409→   409→                      <button
   410→   410→                        key={def.type}
   411→   411→                        onClick={() => addNode(def.type)}
   412→   412→                        className="w-full flex items-center gap-2.5 p-2 rounded-lg hover:bg-white/5 transition-colors text-left group"
   413→   413→                      >
   414→   414→                        <div
   415→   415→                          className="w-7 h-7 rounded-lg flex items-center justify-center shrink-0 transition-transform group-hover:scale-110"
   416→   416→                          style={{ background: def.bg }}
   417→   417→                        >
   418→   418→                          <Icon className="w-3.5 h-3.5" style={{ color: def.color }} />
   419→   419→                        </div>
   420→   420→                        <div className="flex-1 min-w-0">
   421→   421→                          <div className="text-xs text-white/80 font-medium">{def.label}</div>
   422→   422→                          <div className="text-[10px] text-white/30 truncate">{def.description}</div>
   423→   423→                        </div>
   424→   424→                      </button>
   425→   425→                    )
   426→   426→                  })}
   427→   427→                </div>
   428→   428→              </div>
   429→   429→            )
   430→   430→          })}
   431→   431→        </div>
   432→   432→
   433→   433→        <div className="p-3 border-t border-white/5 space-y-2">
   434→   434→          <Button onClick={handleSave} className="w-full" size="sm">
   435→   435→            <Save className="w-4 h-4 mr-1.5" /> Save Flow
   436→   436→          </Button>
   437→   437→          <Button onClick={clearAll} variant="ghost" className="w-full text-red-400 hover:text-red-300" size="sm">
   438→   438→            <Trash2 className="w-3.5 h-3.5 mr-1.5" /> Clear All
   439→   439→          </Button>
   440→   440→        </div>
   441→   441→      </div>
   442→   442→
   443→   443→      {/* ─── CENTER: canvas ────────────────────────────────────────────── */}
   444→   444→      <div className="flex-1 relative flex flex-col">
   445→   445→        {/* Toolbar */}

   875→   875→  if (nodeType === 'api_call') {

   928→   928→

   977→   977→function OptionsEditor({
   978→   978→  options,
   979→   979→  onChange,
   980→   980→}: {
   981→   981→  options: string[]
   982→   982→  onChange: (options: string[]) => void
   983→   983→}) {
   984→   984→  const [newOption, setNewOption] = useState('')
   985→   985→
   986→   986→  const add = () => {
   987→   987→    const v = newOption.trim()
   988→   988→    if (!v) return
   989→   989→    onChange([...options, v])
   990→   990→    setNewOption('')
   991→   991→  }
   992→   992→
   993→   993→  return (
   994→   994→    <div className="space-y-2">
   995→   995→      <Label className="text-white/60 text-xs">Options ({options.length})</Label>
   996→   996→      <div className="space-y-1.5">
   997→   997→        {options.map((opt, i) => (
   998→   998→          <div key={i} className="flex items-center gap-2">
   999→   999→            <span className="text-[10px] text-white/30 w-4">{i + 1}.</span>
  1000→  1000→            <Input
  1001→  1001→              value={opt}
  1002→  1002→              onChange={(e) => onChange(options.map((o, j) => (j === i ? e.target.value : o)))}
  1003→  1003→              className="bg-[#1e1f22] border-white/10 text-white text-xs h-8"
  1004→  1004→            />
  1005→  1005→            <Button
  1006→  1006→              variant="ghost"
  1007→  1007→              size="icon"
  1008→  1008→              className="h-8 w-8 text-red-400 hover:text-red-300 shrink-0"
  1009→  1009→              onClick={() => onChange(options.filter((_, j) => j !== i))}
  1010→  1010→            >
  1011→  1011→              <X className="w-3 h-3" />
  1012→  1012→            </Button>
  1013→  1013→          </div>
  1014→  1014→        ))}
  1015→  1015→      </div>
  1016→  1016→      <div className="flex gap-2">
  1017→  1017→        <Input
  1018→  1018→          value={newOption}
  1019→  1019→          onChange={(e) => setNewOption(e.target.value)}
  1020→  1020→          onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); add() } }}
  1021→  1021→          placeholder="Add option…"
  1022→  1022→          className="bg-[#1e1f22] border-white/10 text-white text-xs h-8"
  1023→  1023→        />
  1024→  1024→        <Button size="sm" variant="outline" onClick={add} className="shrink-0 h-8">
  1025→  1025→          <Plus className="w-3 h-3" />
  1026→  1026→        </Button>
  1027→  1027→      </div>
  1028→  1028→    </div>
  1029→  1029→  )
  1030→  1030→}
